beard_suppression_absolute = {
	usage = game
	selection_behavior = weighted_random
	priority = 100

	default_beard = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = beards
				template = no_beard
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 100000
				has_trait = eunuch
			}
			modifier = {
				add = 100000
				age_in_years < 18
			}
		}
	}
}

beard_suppression = {
	usage = game
	selection_behavior = weighted_random
	priority = 3

	default_beard = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = beards
				template = no_beard
				range = { 0 1 } # For the randomness to work correctly
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 100000
				owner ?= { modifier:no_beards = yes }
			}
			modifier = {
				add = 100000
				portrait_clothing_trigger = { CULTURE_FLAG = indochina_gfx }
			}
			modifier = {
				add = 100000
				AND = {
					age_in_years < 50
					portrait_clothing_trigger = { CULTURE_FLAG = american_gfx }
				}
			}
			modifier = {
				add = 400
				age_in_years < 21
			}
			modifier = {
				add = -100
				OR = {
					portrait_clothing_trigger = { CULTURE_FLAG = middle_east_gfx }
					portrait_clothing_trigger = { CULTURE_FLAG = indian_gfx }
					portrait_clothing_trigger = { CULTURE_FLAG = east_mediterranean_gfx }
					portrait_clothing_trigger = { CULTURE_FLAG = eastern_christian_gfx }
				}
			}
		}
	}
}

beards = {
	usage = game
	selection_behavior = weighted_random
	fallback = default_beard

	default_beard = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = beards

			}
		}
		outfit_tags = { no_clothes }
		weight = {

				OR = {

					portrait_clothing_trigger = { CULTURE_FLAG = east_mediterranean_gfx }
					portrait_clothing_trigger = { CULTURE_FLAG = eastern_christian_gfx }
				}
			}

			modifier = {
				add = 50000
				OR = {
					AND = {
						gfx_culture_applicable = military_order_gfx
						OR = {
							always = no #birth_age = age:age_5_absolutism #P:A Removed vanilla reference
							always = no #birth_age = age:age_6_revolutions #P:A Removed vanilla reference
						}
					}
					AND = {
						OR = {
							gfx_culture_applicable = benedictine_gfx
							gfx_culture_applicable = cistercian_gfx
							gfx_culture_applicable = augustine_gfx
						}
						OR = {
							always = no #birth_age = age:age_4_reformation #P:A Removed vanilla reference
							always = no #birth_age = age:age_5_absolutism #P:A Removed vanilla reference
							always = no #birth_age = age:age_6_revolutions #P:A Removed vanilla reference

						gfx_culture_applicable = franciscan_gfx
						always = no #birth_age = age:age_6_revolutions #P:A Removed vanilla reference
					}
				}
			}
		}
	}
