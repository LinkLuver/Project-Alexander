keep_as_territory = {
	is_available = { always = yes }

	default_priority = {
		value = 0
		# Default only when province is NOT overseas
		if = {
			limit = {
				scope:target = {
					NOT = { any_location_in_province_definition = { would_be_overseas_for = scope:owner } }
				}
			}
			add = 150
		}
	}

	ai_will_select = {
		value = -1000
		# AI never keeps overseas territory directly
		if = {
			limit = { scope:target = { any_location_in_province_definition = { would_be_overseas_for = scope:owner } } }
			add = -9000
		}
		else = {
			add = 2000
		}
	}
}
