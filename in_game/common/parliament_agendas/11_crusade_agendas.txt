pa_join_crusade_nobles = {
	type = country
	estate = nobles_estate

	importance = 2

	potential = {
		exists = international_organization:crusade
		NOT = { is_member_of_international_organization = international_organization:crusade }
		religion = religion:catholic
		country_can_join_international_organization = international_organization:crusade
	}

	on_accept = {
		international_organization:crusade = {
			add_country_to_international_organization = root
		}
		if = {
			limit = {
				international_organization:crusade = {
					var:phase = 2
					exists = var:active_war
				}
			}
			join_war_as_attacker = { war = international_organization:crusade.var:active_war }
		}
	}

	ai_will_do = {
		add = 5
		if = {
			limit = { is_rival_of = international_organization:crusade.international_organization_target }
			add = 25
		}
		if = {
			limit = { at_war = yes }
			subtract = 15
		}
	}
	chance = 15
}
