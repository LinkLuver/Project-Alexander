# Estate-driven culture-tier movement agendas
# Estates lobby to move cultures up or down the ladder based on their dominant culture's cultural_view of them.
# *_enemy and *_kindred variants are higher-importance, higher-frequency versions for extreme opinions.

# Shared estate set (mirrors pa_accept_estate_culture)
# nobles, clergy, burghers, peasants, tribes, cossacks, dhimmi

# ============================================================================
# DEMOTION AGENDAS (move cultures down the ladder)
# ============================================================================

pa_add_discriminated_culture = {
	type = country
	estate = nobles_estate
	estate = clergy_estate
	estate = burghers_estate
	estate = peasants_estate
	estate = tribes_estate
	estate = cossacks_estate
	estate = dhimmi_estate

	importance = 1.0

	potential = {
		any_present_culture_in_country = {
			NOR = {
				is_discriminated_in = root
				this = root.culture
				is_accepted_in = root
			}
			save_temporary_scope_as = candidate_culture
			root = {
				"estate(scope:target)" ?= {
					dominant_culture ?= {
						cultural_view = {
							target = scope:candidate_culture
							value = negative
						}
					}
				}
			}
		}
	}

	on_accept = {
		random_present_culture_in_country = {
			limit = {
				NOR = {
					is_discriminated_in = root
					this = root.culture
					is_accepted_in = root
				}
				save_temporary_scope_as = candidate_culture
				root = {
					"estate(scope:target)" ?= {
						dominant_culture ?= {
							cultural_view = {
								target = scope:candidate_culture
								value = negative
							}
						}
					}
				}
			}
			save_scope_as = chosen_culture
		}
		add_discriminated_culture = scope:chosen_culture
		add_estate_satisfaction = { type = scope:target value = 0.05 }
	}

	ai_will_do = {
		if = {
			limit = {
				political_influence <= 90
			}

			add = {
				value = -100
				desc = "low political influence"
			}
		}
	}

	can_bribe = { can_bribe_default_trigger = yes }
	on_bribe = { on_bribe_default_effect = yes }

	chance = 1
}
