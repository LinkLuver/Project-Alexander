pa_remove_free_village = {
	estate = nobles_estate
	potential = {
		#total_effective_building_levels:free_village > 0 #P:A Removed vanilla reference
		always = no #P:A Removed vanilla reference
	}
	
	on_accept = {
		ordered_owned_location = {
			limit = {
				#has_building_with_at_least_one_level = free_village #P:A Removed vanilla reference
				always = no #P:A Removed vanilla reference
			}
			order_by = population
			max = {
				value = 1
				add = {
					value = num_provinces
					divide = 10
				}
				#max = root.total_effective_building_levels:free_village #P:A Removed vanilla reference
			}
			check_range_bounds = no
			#change_building_level_in_location = { #P:A Removed vanilla reference
				#building = building_type:free_village
				#value = -1
			#}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate
			value = estate_satisfaction_mild_penalty
		}
		
	}
	chance = 20
}
