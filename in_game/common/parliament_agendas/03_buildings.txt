#Nobles

pa_remove_free_village = {
	estate = nobles_estate
	potential = {
		total_effective_building_levels:free_village > 0
	}
	
	on_accept = {
		ordered_owned_location = {
			limit = {
				has_building_with_at_least_one_level = free_village
			}
			order_by = population
			max = {
				value = 1
				add = {
					value = num_provinces
					divide = 10
				}
				max = root.total_effective_building_levels:free_village
			}
			check_range_bounds = no
			change_building_level_in_location = {
				building = building_type:free_village
				value = -1
			}
		}
		add_estate_satisfaction = { type = estate_type:peasants_estate
			value = estate_satisfaction_mild_penalty
		}
		
	}
	chance = 20
}

#construct buildings for estate
#Clergy
pa_construct_monastery = {
	estate = clergy_estate
	
	potential = {
		num_locations > 0
		OR = {
			AND = {
				modifier:can_have_monasteries = yes
				any_owned_location = {
					is_city = no
					NOT = { has_building = building_type:monastery }
				}
			}
			AND = {
				religion = religion:orthodox
				any_owned_location = {
					is_city = no
					NOT = { has_building = building_type:orthodox_monastery }
				}
			}
			AND = {
				religion = religion:miaphysite
				any_owned_location = {
					is_city = no
					NOT = { has_building = building_type:miaphysite_monastery }
				}
			}
		}
	}
	
	on_accept = {
		random_owned_location = {
			limit = {
				is_city = no
				NOT = { has_building = building_type:monastery }
			}
			if = {
				limit = {
					ROOT.religion = religion:miaphysite
					is_city = no
					NOT = { has_building = building_type:miaphysite_monastery }
				}
				construct_building = {
					building_type = building_type:miaphysite_monastery
					cost_multiplier = 0.5
					cost_multiplier_reason = "game_concept_parliament_agenda"
					instant = yes
				}
			}
			else_if = {
				limit = {
					ROOT.religion = religion:orthodox
					is_city = no
