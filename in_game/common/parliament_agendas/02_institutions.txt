pa_resist_feudalism = {
	estate = burghers_estate
	estate = peasants_estate

	potential = {
		#knows_about_institution = institution:feudalism #P:A Removed vanilla reference
		#NOT = { #P:A Removed vanilla reference
			#has_embraced_institution = institution:feudalism #P:A Removed vanilla reference
		#}
		any_owned_location = {
			institution_progress:legalism > 0
			institution_progress:legalism < 99
		}
		always = no #P:A Removed vanilla reference
	}

	on_accept = {
		ordered_owned_location = {
			limit = {
				#institution_progress:feudalism > 0 #P:A Removed vanilla reference
				#institution_progress:feudalism < 99 #P:A Removed vanilla reference
				always = no #P:A Removed vanilla reference
			}
			order_by = population
			max = {
				value = 1
				add = {
					value = num_provinces
					divide = 20
				}
			}
			#change_institution_progress = { #P:A Removed vanilla reference
				#type = institution:feudalism
				#value = institution_progress_mild_penalty
			#}
		}
	}
	chance = 10
}
