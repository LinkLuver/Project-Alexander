adu_proclaim_crown_al_andalus = {
	type = situation

	ai_tick = monthly
	ai_tick_frequency = 1

	automation_tick = never
	automation_tick_frequency = 12

	price = price:gc_actions_price
	price_modifier = {
		add = 1
		add = {
			desc = "gc_actions_cost"
			value = scope:actor.modifier:gc_actions_cost
		}
	}

	potential = {
		scope:actor = { OR = { has_or_had_tag = ADU has_or_had_tag = MOR } }
	}

	allow = {
		trigger_if = {
			limit = { situation:golden_century = { has_variable = gc_european_dominance } }
			situation:golden_century.var:gc_european_dominance >= 100
		}
		trigger_else = { always = no }

        scope:actor = {
            NOT = { has_reform = government_reform:crown_of_al_andalus }
        }
	}

	cooldown = {
		type = adu_proclaim_crown_al_andalus
		years = 10
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			this = situation:golden_century
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			unlock_government_reform_effect = { type = crown_of_al_andalus }
            add_reform = government_reform:crown_of_al_andalus
		}
	}

	ai_will_do = {
		add = 1000
	}
}

invoke_ummayad_legacy = {
	type = situation
	
	potential = {
		scope:actor = {
			OR = { has_or_had_tag = ADU has_or_had_tag = MOR }
		}
	}

	allow = {
		trigger_if = {
			limit = { situation:golden_century = { has_variable = gc_european_dominance } }
			situation:golden_century.var:gc_european_dominance >= 25
		}
		trigger_else = { always = no }
		scope:actor ?= {
			at_war = no
			has_raised_levies = no
			NOT = { receiving_scripted_relation_of_type = relation_type:rein_in_junior_diplomacy }
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	cooldown = {
		type = invoke_ummayad_legacy
		years = 5
	}

	price = price:gc_actions_price
	price_modifier = {
		add = 1
		add = {
			desc = "gc_actions_cost"
			value = scope:actor.modifier:gc_actions_cost
		}
	}

	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:golden_century = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			this = situation:golden_century
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scripted_geography:umayyad_caliphate_geography = {
				every_country_with_capital_in_geography = {
					limit = {
						NOT = { this = scope:actor }
							NOT = { country_type = building }
							NOT = { is_allied_with = { target = scope:actor } }
					}
					add_to_list = source
				}
			}
		}
		target_flag = target_country
		name = "invoke_ummayad_legacy_select_country"
		none_available_msg_key = "invoke_ummayad_legacy_no_country_available"
		default_sort = will
		column = { data = name }
		column = { data = army_size }
		column = { data = country_tax_base }
        enabled = {
			NOT = { has_truce_with = scope:actor }
			scope:actor = { 
				NOT = { has_truce_with = root }
			}
		}
	}

	effect = { 
		if = {
			limit = { exists = scope:target_country }
			if = {
				limit = {
					scope:actor = {
						is_human = no
						num_of_diplomats >= 1
						can_declare_war_on = scope:target_country
					}
				}
				scope:actor = {
					declare_war_with_cb = {
						target = scope:target_country
						type = casus_belli:cb_restore_ummayad_caliphate
						apply_declaration_costs = yes
					}
				}
			}
			else = { 
				scope:actor = {
					add_casus_belli = {
						target = scope:target_country
						type = casus_belli:cb_restore_ummayad_caliphate
					}
				}
			}
		}
		else = { custom_tooltip = selected_country_ummayad_cb_tt }
	}

	ai_will_do = {
		add = {
			value = 35
			desc = BASE_VALUE
		}
		if = {
			limit = { exists = scope:target_country }
			add = {
				desc = "RELATIVE_STRENGTH"
				value = 100
				multiply = "scope:actor.relative_strength(scope:target_country)"
				divide = 5
			}
			scope:target_country = {
				subtract = {
					desc = "their defensive alliance"
					value = this.defensive_alliance_strength
				}
			}
			add = {
				desc = "our offensive alliance"
				value = scope:actor.offensive_alliance_strength
			}
			add = {
				desc = "CURRENT_MANPOWER_TOOLTIP"
				value = scope:actor.manpower_percentage
				multiply = 25
			}
			if = {
				limit = { scope:actor = { is_during_bankruptcy = yes } }
				subtract = {
					desc = "[bankruptcy|e]"
					value = 1000
				}
			}
			if = {
				limit = { scope:actor = { war_exhaustion > 0 } }
				subtract = {
					desc = "war_exhaustion"
					value = scope:actor.war_exhaustion
					multiply = 2
				}
			}
			add = {
				desc = "aggressiveness modifier minus carefulness modifier"
				value = scope:actor.modifier:aggressiveness_modifier
				subtract = scope:actor.modifier:carefulness_modifier
				multiply = 50
			}
			if = {
				limit = { scope:actor = { wants_to_attack = scope:target_country } }
				add = {
					desc = "We want to attack them"
					value = 50
				}
			}
			else = {
				subtract = {
					desc = "We dont want to attack them"
					value = 10
				}
			}
		}
	}
}

	allow = {

			this = situation:golden_century
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = area
		source = actor
		target_flag = target
		name = "spa_enforce_conformity_select_area"
		none_available_msg_key = "spa_enforce_conformity_no_areas"
		column = { data = name }
		column = { data = area_primary_culture_percentage }
		visible = {
			trigger_if = {
				limit = {
					situation:golden_century = { has_variable = gc_european_dominance }
					situation:golden_century.var:gc_european_dominance >= 75
				}
				continent = continent:europe
			}
			trigger_else = {
				NOT = { continent = continent:europe }
			}
			any_location_in_area = {
				owner ?= { this = scope:actor }
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:actor = {
				trigger_event_non_silently = golden_century_events.4
			}
		}
		else = { custom_tooltip = explain_enforce_conformity_tt }
	}

	ai_will_do = {
		add = {
			value = 50
			desc = BASE_VALUE
		}

		if = {
			limit = { exists = scope:target }
			add = {
				desc = area_population
				value = "scope:target.population_in_area(scope:actor)"
				multiply = 0.01
			}
			scope:target = {
				every_province_in_area = {
					limit = {
						province_capital = {
							exists = dominant_culture
							dominant_culture = {
								NOT = { is_primary_or_accepted_in = scope:actor }
							}
						}
					}
					add = {
						value = 100
						desc = "Province Capital with incompatible religion"
					}
				}
				every_province_in_area = {
					limit = {
						province_capital = {
							exists = dominant_religion
							NOT = { dominant_religion.group = scope:actor.religion.group }
						}
					}
					add = {
						value = 100
						desc = "Province Capital with incompatible religion"
					}
				}
			}
		}
	}
}
