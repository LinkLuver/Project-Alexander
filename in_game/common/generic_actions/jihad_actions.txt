# Jihad-specific IO member actions.

claim_jihad_leadership = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	allow = {
		scope:actor = {
			is_subject = no
		}
	}
	cooldown = {
		type = claim_jihad_leadership
		years = 1
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_holy_war_for_leadership"
		column = {
			data = name
		}
		visible = {
			international_organization_type = international_organization_type:jihad
			leader_country ?= { religion.group ?= religion_group:muslim }
			scope:actor = {
				is_member_of_international_organization = prev
				NOT = { is_leader_of_international_organization = prev }
			}
		}
		enabled = {
			leader_country = {
				great_power_score < scope:actor.great_power_score
			}
		}
	}

	effect = {
		scope:recipient.leader_country = { save_scope_as = target }
		scope:recipient = {
			set_leader_country = scope:actor
		}
		if = {
			limit = {
				scope:recipient = {
					trigger_if = {
						limit = { has_variable = phase }
						var:phase = 2
					}
					trigger_else = {
						always = no
					}
				}
				scope:recipient = { exists = var:active_war }
			}
			scope:recipient.var:active_war = { save_scope_as = hw_war }
			scope:actor = {
				set_war_leader = { war = scope:hw_war }
			}
		}
	}

	ai_will_do = {
		add = -100
		if = {
			limit = {
				scope:recipient.leader_country = {
					great_power_score < {
						value = scope:actor.great_power_score
						multiply = 0.66
					}
					military_strength < scope:actor.military_strength
				}
			}
			add = 150
		}
	}
}

offer_recognition_for_jihad = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	price = price:offer_recognition_for_jihad_price

	allow = {
	}
	cooldown = {
		type = offer_recognition_for_jihad
		years = 1
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_holy_war_for_recruitment"
		column = {
			data = name
		}
		visible = {
			trigger_if = {
				limit = {
					international_organization_type = international_organization_type:jihad
					has_variable = phase
				}
				var:phase = 1
			}
			trigger_else = {
				always = no
			}
			leader_country ?= scope:actor
			leader_country ?= { religion.group ?= religion_group:muslim }
			has_enabled_currency = gold
			gold > 0
		}
	}
	select_trigger = {
		looking_for_a = country
		ai_interaction_source_list = {
			scope:actor = {
				ordered_country_in_diplomatic_range = {
					limit = {
						this != scope:actor
						religion.group ?= religion_group:muslim
						is_rebel_country = no
					}
					order_by = military_strength
					max = 10
					check_range_bounds = no
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "choose_country"
		none_available_msg_key = "no_valid_countries"
		column = {
			data = name
		}
		visible = {
			root != scope:actor
			religion.group ?= religion_group:muslim
			is_rebel_country = no
			knows_country = scope:recipient.international_organization_target
			NOT = { is_member_of_international_organization = scope:recipient }
			country_can_join_international_organization = scope:recipient
			NOT = { has_variable = recently_invited_to_jihad }
		}
	}

	effect = {
		scope:actor = { save_scope_as = recognition_offerer }
		scope:recipient = { save_scope_as = recognition_io }
		if = {
			limit = { exists = scope:target }
			scope:target = {
				set_variable = {
					name = recently_invited_to_jihad
					value = 1
					months = 12
				}
				trigger_event_non_silently = { id = holy_war.10 }
			}
		}
		else = {
			custom_tooltip = offer_recognition_for_jihad_invite_tt
		}
	}

	ai_will_do = {
		add = -25
		if = {
			limit = { scope:recipient = { holy_war_overwhelms_target = yes } }
			add = -1000
		}
		add = {
			desc = "HOLY_WAR_AGE_DAMPENER"
			value = crusade_jihad_age_dampener
		}
		if = {
			limit = { scope:recipient = { total_members < 5 } }
			add = 40
		}
		if = {
			limit = { scope:recipient = { total_members > 10 } }
			add = -1000
		}
		if = {
			limit = {
				exists = scope:target
				scope:target = { great_power_score > scope:actor.great_power_score }
			}
			add = 25
		}
	}
}

		}
	}
