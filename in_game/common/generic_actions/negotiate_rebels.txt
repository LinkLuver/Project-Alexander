negotiate_rebels_buy_off = {
	type = owncountry

	allow = {
		scope:actor = {
			any_rebel = {
				rebel_progress > 0.8
			}
			NOT = {
				has_cooldown = negotiate_rebels_buy_off_cooldown
			}
		}
	}

	price = price:negotiate_rebels_buy_off_price
	price_modifier = {
		value = negotiate_rebels_cost
	}
	automation_tick = never
	automation_tick_frequency = 12

	select_trigger = {
		looking_for_a = rebels
		source = actor
		target_flag = target
		column = {
			data = name
		}
		column = {
			data = rebel_progress
		}
		column = {
			data = monthly_progress
		}
		enabled = {
			rebel_progress < 1
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
			}
			scope:target = {
				add_rebel_progress = -0.4
			}
			scope:actor = {
				add_cooldown = {
					type = negotiate_rebels_buy_off_cooldown
					years = 2
				}
			}
		}
		else = {
			custom_tooltip = negotiate_rebels_buy_off_no_target_tt
		}
	}

	ai_will_do = {
		add = {
			value = -100
			desc = "base"
		}

		if = {
			limit = {
				scope:target = {
					rebel_last_months_progress <= 0
				}
			}
			add = {
				value = -10000
				desc = "Rebel faction is not growing"
			}
		}

		if = {
			limit = {
				scope:target = {
					rebel_progress >= 0.8
				}
			}
			add = {
				value = 150
				desc = "Rebellion nearly ready to revolt"
			}
		}

		if = {
			limit = {
				scope:actor = {
					at_war = yes
				}
			}
			add = {
				value = -500
				desc = "At war"
			}
		}

		if = {
			limit = {
				scope:actor = {
					gold < 200
				}
			}
			add = {
				value = -500
				desc = "Low treasury"
			}
		}
	}
}

negotiate_rebels_accept_demands = {
	type = owncountry

	allow = {
		scope:actor = {
			any_rebel = {
				rebel_progress > 0.8
			}
		}
	}

	automation_tick = never
	automation_tick_frequency = 12

	select_trigger = {
		looking_for_a = rebels
		source = actor
		target_flag = target
		column = {
			data = name
		}
		column = {
			data = rebel_progress
		}
		column = {
			data = monthly_progress
		}
		enabled = {
			rebel_progress < 1
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
			}
			scope:actor = {
				apply_rebel_demands = scope:target
				add_legitimacy = -25
				destroy_rebel = scope:target
			}
		}
		else = {
			custom_tooltip = negotiate_rebels_accept_demands_no_target_tt
		}
	}

	ai_will_do = {
		add = {
			value = -100
			desc = "base"
		}

		if = {
			limit = {
				scope:target = {
					rebel_last_months_progress <= 0
				}
			}
			add = {
				value = -10000
				desc = "Rebel faction is not growing"
			}
		}

		if = {
			limit = {
				scope:target = {
					rebel_category = pretender
				}
			}
			add = {
				value = -10000
				desc = "Never concede to pretenders"
			}
		}

		if = {
			limit = {
				scope:target = {
					rebel_category = religious
				}
				scope:actor = {
					government_type = government_type:theocracy
					is_subject = yes
					overlord ?= { government_type = government_type:theocracy }
				}
			}
			add = {
				value = -10000
				desc = "Never abandon faith as a theocratic subject"
			}
		}

		if = {
			limit = {
				scope:actor = {
					legitimacy < 30
				}
			}
			add = {
				value = -500
				desc = "Already low legitimacy"
			}
		}

		if = {
			limit = {
				scope:target = {
					rebel_progress >= 0.9
				}
				scope:actor = {
					stability < -30
				}
			}
			add = {
				value = 150
				desc = "Desperate situation"
			}
		}
	}
}
