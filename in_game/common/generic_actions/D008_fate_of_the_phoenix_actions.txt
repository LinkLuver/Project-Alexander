accept_catholic_delegation = {
	type = disaster

	potential = { 
		scope:actor = { religion != religion:catholic }
	}

	allow = {
		scope:actor = { 
			at_war = no
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
	price = price:fate_of_phoenix_actions_price

	cooldown = {
		type = accept_catholic_delegation
		years = 10
	}
	
	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:fate_of_the_phoenix
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = { religion != religion:catholic }
				custom_tooltip = {
					text = pops_in_capital_province_convert_catholic_tt
	
					capital.province = {
						every_location_in_province = {
							every_pop = {
								limit = {
									religion = scope:actor.religion
									owner = scope:actor
								}
								split_pop = {
									fraction = { 0.05 0.1 }
									religion = religion:catholic
								}
							}
						}
					}
				}
			}
			change_societal_value = { 
				type = latinization_vs_hellenization 
				value = societal_value_minor_move_to_left
			}
			if = {
				limit = { 
					religion != religion:catholic 
					country_has_estate = estate_type:clergy_estate
				}

				add_estate_satisfaction = { 
					type = estate_type:clergy_estate 
					value = estate_satisfaction_weak_penalty 
				}
			}

			else_if = {
				limit = {
					religion = religion:catholic 
					country_has_estate = estate_type:clergy_estate
				}

				add_estate_satisfaction = { 
					type = estate_type:clergy_estate 
					value = estate_satisfaction_weak_bonus
				}
			}
		}

		religion:catholic = {
			ordered_country_in_religion = {
				limit = {
					is_subject = no
					NOR = {
						is_enemy_of = scope:actor
						is_rival_of = scope:actor
					}
					capital = { "border_distance_to(scope:actor)" < 500 }
				}

				order_by = total_population

				max = 5

				add_opinion = {
					target = scope:actor
					modifier = opinion_catholic_delegation
				}
			}
		}

		if = {
			limit = {
				country_exists = c:PAP
				scope:actor = {
					NOT = { is_enemy_of = c:PAP }
					is_rival_of = c:PAP
				}
			}
			c:PAP = {
				add_opinion = {
					target = scope:actor
					modifier = opinion_catholic_delegation
				}
			}
		}
	}

	ai_will_do = {
		if = {
			limit = { 
				scope:actor.government_power > 90
			}

			add = {
				value = 5
				desc = "high government power"
			}
		}
		else = {
			subtract = {
				value = 1
				desc = "low government power"
			}
		}
	}
}
