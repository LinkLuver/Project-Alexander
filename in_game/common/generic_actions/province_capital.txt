set_province_capital = {
	type = owncountry

	ai_tick = never

	potential = {
		scope:actor = {
			is_human = yes
		}
	}

	allow = {
		scope:actor = {
			at_war = no
		}
	}

	price = price:set_province_capital

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target.province = {
				set_province_capital = scope:target
			}
		}
	}
}
