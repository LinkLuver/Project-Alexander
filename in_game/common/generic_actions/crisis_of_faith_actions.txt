preach_renewal = {
	type = disaster
	show_message = no

	ai_tick = daily
	ai_tick_frequency = 40

	automation_tick = never
	automation_tick_frequency = 12

	price = price:crisis_of_faith_disaster_actions_price

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = schismatic_clergy_active_tt
				exists = var:schismatic_clergy_rebel
			}
		}
	}

	cooldown = {
		type = preach_renewal
		months = 6
	}

	select_trigger = {
		looking_for_a = disaster
		name = "choose_disaster"
		target_flag = recipient
		column = {
			data = name
		}
		visible = {
			disaster_type = disaster_type:crisis_of_faith
			disaster_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			trigger_event_non_silently = crisis_of_faith.10
		}
	}

	ai_will_do = {
		add = 100
	}
}
