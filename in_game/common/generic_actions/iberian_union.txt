proclaim_spanish_monarchy = {
	type = internationalorganization
	show_message = no

	automation_tick = never
	automation_tick_frequency = 12

	potential = {
		has_dlc = "d009_across_the_pillars"
	}

	allow = {
		scope:actor = {
			has_special_status_in_international_organization = {
				international_organization = scope:recipient
				type = special_status:senior_partner
			}
		}
		custom_tooltip = {
			text = proclaim_spanish_monarchy_integration_tt
			scope:recipient = {
				modifier:union_integration_level >= 5
			}
		}
		NOT = {
			country_exists = c:SPA
		}
		NOT = { is_situation_active = situation:battle_for_the_strait }
		scope:actor = {
			modifier:blocked_from_forming_countries = no
		}
		region:iberia_region = {
			any_location_in_region = {
				percent >= 0.5
				owner ?= {
					is_member_of_international_organization = scope:recipient
				}
			}
		}
		trigger_if = {
			limit = {
				scope:actor = { is_ai = yes }
			}
			NOT = {
				scope:recipient = {
					any_international_organization_member = {
						is_ai = no
					}
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		none_available_msg_key = "no_valid_international_organizations"
		column = {
			data = name
		}
		visible = {
			has_variable = iberian_union_active
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:recipient
			}
			scope:actor = {
				trigger_event_non_silently = d009_flavor_iberian_union.1
			}
		}
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				exists = scope:recipient
			}
			add = 10
		}
	}
}
