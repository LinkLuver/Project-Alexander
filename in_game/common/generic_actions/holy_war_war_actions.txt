# Shared Crusade / Jihad war-stage (phase 2) actions.

convert_occupied_territory = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	price = price:convert_occupied_territory_price

	potential = {
		scope:actor = {
			any_controlled_location = {
				owner = { is_at_war_with = scope:actor }
				religion_percentage = {
					religion = scope:actor.religion
					value < 0.25
				}
			}
		}
	}
	allow = {
	}
	cooldown = {
		type = convert_occupied_territory
		years = 1
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_holy_war"
		column = {
			data = name
		}
		visible = {
			trigger_if = {
				limit = {
					OR = {
						international_organization_type = international_organization_type:crusade
						international_organization_type = international_organization_type:jihad
					}
					has_variable = phase
				}
				var:phase = 2
			}
			trigger_else = {
				always = no
			}
			scope:actor = {
				is_member_of_international_organization = prev
			}
		}
	}

	effect = {
		scope:recipient = {
			change_variable = { name = religious_zeal add = -10 }
			clamp_variable = { name = religious_zeal max = 100 min = 0 }
		}
		if = {
			limit = { exists = scope:recipient.var:war_target }
			scope:recipient.var:war_target = { save_scope_as = hw_convert_target }
			if = {
				limit = { scope:recipient = { this = international_organization:crusade } }
				scope:actor = {
					every_controlled_location = {
						limit = {
							owner = scope:hw_convert_target
							religion_percentage = {
								religion = religion:catholic
								value < 0.25
							}
						}
						every_pop = {
							limit = { religion.group != religion_group:christian }
							split_pop = {
								fraction = 0.1
								religion = religion:catholic
							}
						}
						owner = {
							if = {
								limit = { NOT = { is_in_list = hw_convert_owners } }
								add_to_list = hw_convert_owners
							}
						}
					}
				}
			}
			else = {
				scope:recipient.leader_country = { save_scope_as = hw_convert_faith_holder }
				scope:actor = {
					every_controlled_location = {
						limit = {
							owner = scope:hw_convert_target
							religion_percentage = {
								religion = scope:hw_convert_faith_holder.religion
								value < 0.25
							}
						}
						every_pop = {
							limit = { NOT = { religion = scope:hw_convert_faith_holder.religion } }
							split_pop = {
								fraction = 0.1
								religion = scope:hw_convert_faith_holder.religion
							}
						}
						owner = {
							if = {
								limit = { NOT = { is_in_list = hw_convert_owners } }
								add_to_list = hw_convert_owners
							}
						}
					}
				}
			}
			every_in_list = {
				list = hw_convert_owners
				trigger_event_non_silently = { id = holy_war.3 }
			}
		}
	}

	ai_will_do = {
		add = -10
		if = {
			limit = {
				exists = scope:recipient.var:war_target
				scope:actor = {
					any_controlled_location = { owner = scope:recipient.var:war_target }
				}
			}
			add = 35
		}
	}
}

discipline_the_faithless = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	allow = {
		scope:actor = {
			has_meaningful_holy_war_participation = yes
		}
	}
	cooldown = {
		type = discipline_the_faithless
		years = 1
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_holy_war"
		column = {
			data = name
		}
		visible = {
			trigger_if = {
				limit = {
					OR = {
						international_organization_type = international_organization_type:crusade
						international_organization_type = international_organization_type:jihad
					}
					has_variable = phase
				}
				var:phase = 2
			}
			trigger_else = {
				always = no
			}
			leader_country ?= scope:actor
		}
	}
	select_trigger = {
		looking_for_a = country
		source = recipient
		target_flag = target
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			root != scope:actor
			has_variable = holy_war_participation
			OR = {
				var:holy_war_participation <= scope:recipient.var:min_holy_war_participation
				var:holy_war_participation <= 0.02
			}
		}
	}

	effect = {
		scope:recipient = { save_scope_as = discipline_io }
		scope:actor = { save_scope_as = discipline_leader }
		if = {
			limit = { exists = scope:target }
			scope:target = {
				save_scope_as = discipline_target
				trigger_event_non_silently = { id = holy_war.4 }
			}
		}
	}

	ai_will_do = {
		add = -50
		if = {
			limit = {
				exists = scope:target
				scope:actor = {
					opinion = { target = scope:target value < 0 }
				}
			}
			add = 60
		}
	}
}

			limit = { scope:recipient = { gold > 3000 } }
			add = 25
		}
	}

hasten_holy_war_preparations = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never

	price = price:hasten_holy_war_preparations_price

	allow = {
	}
	cooldown = {
		type = hasten_holy_war_preparations
		months = 6
	}
	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_holy_war_for_preparations"
		column = {
			data = name
		}
		visible = {
			trigger_if = {
				limit = {
					OR = {
						international_organization_type = international_organization_type:crusade
						international_organization_type = international_organization_type:jihad
					}
					has_variable = phase
				}
				var:phase = 1
			}
			trigger_else = {
				always = no
			}
		}
	}
	effect = {
		scope:recipient = {
			custom_tooltip = {
				text = hasten_holy_war_preparations_tt
				change_variable = { name = preparation add = 10 }
				clamp_variable = { name = preparation max = 100 min = 0 }
				change_variable = { name = religious_zeal add = -5 }
				clamp_variable = { name = religious_zeal max = 100 min = 0 }
			}
		}
	}
	ai_will_do = {
		add = -25
		if = {
			limit = { scope:recipient = { total_members >= 3 } }
			add = 30
		}
		if = {
			limit = { scope:recipient = { var:preparation >= 50 } }
			add = 25
		}
		if = {
			limit = { scope:recipient = { var:religious_zeal < 20 } }
			add = -50
		}
	}
}
