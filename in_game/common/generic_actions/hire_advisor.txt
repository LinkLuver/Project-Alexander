hire_advisor = {
	type = owncountry
	

	price = price:hire_advisor
	

	ai_tick = daily
	ai_tick_frequency = 180

	automation_tick = monthly
	automation_tick_frequency = 1
	player_automated_category = governmentreforms


	ai_prerequisite = {
		num_open_cabinet_positions > 0
	}

	potential = {
	}

	allow = {
	}

	ai_will_do = {
		scope:actor = {
			#first, how many do we need?
			value = {
				#how many should we have?
				value = {
					add = {
						desc = "Size of government"
						value = num_open_cabinet_positions
					}
				}
			}
		}
	}

	effect = {
		scope:actor = {
			random_estate = {
				limit = {
					estate_type = {
						eligible_for_cabinet = {
							target = scope:actor
							value = yes
						}
					}
					not = { estate_type = estate_type:crown_estate }
				}
				estate_type = {
					save_scope_as = target_estate_type
				}
			}
			if = {
				limit = {
					has_bureaucracy_of_type = bureaucracy_type:court_eunuchs_bureaucracy
					modifier:may_hire_eunuch_advisors = yes
				}
				custom_tooltip = {
					text = hire_advisor_eunuch_chance_tt
					random = {
						chance = hire_advisor_eunuch_chance
						save_scope_as = spawn_eunuch_trigger_scope
					}
				}
			}
			custom_tooltip = {
				text = hire_advisor_effect
				create_character = {
					important_character_counts = advisor
					skip_update_inheritance = yes
					skip_on_created_action = yes
					use_random_core_location = yes
					trait_category = cabinet
					estate = scope:target_estate_type
					adm = {
						integer_range = { min = 0 max = 100 }
						if = {
							limit = { exists = scope:spawn_eunuch_trigger_scope }
							add = 10
						}
					}
					dip = {
						integer_range = { min = 0 max = 100 }
						if = {
							limit = { exists = scope:spawn_eunuch_trigger_scope }
							add = 10
						}
					}
					mil = {
						integer_range = { min = 0 max = 100 }
						if = {
							limit = { exists = scope:spawn_eunuch_trigger_scope }
							add = 10
						}
					}
					random_estate_type_trigger = {
						scope:actor = {
							estate_type_allowed_in_cabinet = {
								estate_type = prev
							}					
						}
					}
					if = {
						limit = { exists = scope:spawn_eunuch_trigger_scope }
						add_trait = trait:eunuch
					}
					save_scope_as = new_character
				}
			}
		}
	}
}
