summon_volunteers_of_the_faith = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 60
	automation_tick = never

	price = price:summon_volunteers_of_the_faith_price

		has_dlc = "d009_across_the_pillars"

			religion.group = religion_group:muslim

			any_international_organizations_member_of = {
				international_organization_type = international_organization_type:jihad
				leader_country ?= scope:actor

		}

proclaim_the_ghazw = {
	type = internationalorganization
	ai_tick = monthly
	ai_tick_frequency = 12
	automation_tick = never

	potential = {
		has_dlc = "d009_across_the_pillars"
		scope:actor = {
			religion.group = religion_group:muslim
			capital = {
				OR = {
					region = region:iberia_region
					region = region:maghreb_region
				}
			}
			any_international_organizations_member_of = {
				trigger_if = {
					limit = {
						international_organization_type = international_organization_type:jihad
						has_variable = phase
					}
					var:phase = 1
				}
				trigger_else = {
					always = no
				}
				leader_country ?= { religion.group ?= religion_group:muslim }
				international_organization_target ?= {
					religion = religion:catholic
					capital = { region = region:iberia_region }
				}
			}
		}
	}
	allow = {
		scope:actor = {
			any_international_organizations_member_of = {
				trigger_if = {
					limit = {
						international_organization_type = international_organization_type:jihad
						has_variable = phase
					}
					var:phase = 1
				}
				trigger_else = {
					always = no
				}
				leader_country ?= { religion.group ?= religion_group:muslim }
				international_organization_target ?= {
					religion = religion:catholic
					capital = { region = region:iberia_region }
					any_neighbor_country = { this = scope:actor }
				}
			}
		}
	}
	cooldown = {
		type = proclaim_the_ghazw
		years = 1
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_holy_war"
		column = {
			data = name
		}
		visible = {
			trigger_if = {
				limit = {
					international_organization_type = international_organization_type:jihad
					has_variable = phase
				}
				var:phase = 1
			}
			trigger_else = {
				always = no
			}
			leader_country ?= { religion.group ?= religion_group:muslim }
			international_organization_target ?= {
				religion = religion:catholic
				capital = { region = region:iberia_region }
				any_neighbor_country = { this = scope:actor }
			}
		}
	}
	select_trigger = {
		looking_for_a = province
		target_flag = target_province
		name = "choose_province_for_ghazw"
		none_available_msg_key = "proclaim_the_ghazw_no_provinces"
		column = {
			data = name
		}
		interaction_source_list = {
			scope:recipient.international_organization_target = {
				every_province = {
					add_to_list = source
				}
			}
		}
		visible = {
			any_location_in_province = {
				has_fort = no
				NOT = { any_unit_in_location = { always = yes } }
			}
			any_location_in_province = { is_neighbor_of = scope:actor }
		}
		map_color = {
			if = {
				limit = {
					province = { any_location_in_province = { is_neighbor_of = scope:actor } }
				}
				if = {
					limit = {
						has_fort = no
						NOT = { any_unit_in_location = { always = yes } }
					}
					value = define:NMapColors|MAP_COLOR_HIGH
				}
				else = {
					value = define:NMapColors|MAP_COLOR_LOW
				}
			}
		}
	}

	effect = {
		scope:actor = { save_scope_as = ghazw_raider }
		scope:actor = {
			add_manpower = {
				value = monthly_manpower
				multiply = -12
			}
		}
		custom_tooltip = {
			text = proclaim_the_ghazw_tt
			scope:recipient = {
				change_variable = { name = religious_zeal add = -10 }
				clamp_variable = { name = religious_zeal max = 100 min = 0 }
				change_variable = { name = preparation add = -10 }
				clamp_variable = { name = preparation max = 100 min = 0 }
			}
		}
		custom_tooltip = {
			text = proclaim_the_ghazw_loot_tt
			if = {
				limit = { exists = scope:target_province }
				scope:target_province = {
					every_location_in_province = {
						limit = {
							has_fort = no
							NOT = { any_unit_in_location = { always = yes } }
						}
						scope:actor = { loot_location = prev }
						change_development = development_mild_penalty
					}
				}
			}
		}
		scope:recipient.international_organization_target = {
			trigger_event_non_silently = { id = holy_war.15 }
		}
		scope:actor = {
			trigger_event_non_silently = {
				id = d009_holy_war.5
				months = { 1 2 }
			}
		}
	}

	ai_will_do = {
		add = 40
		if = {
			limit = { scope:actor = { gold < 100 } }
			add = 20
		}
		if = {
			limit = {
				scope:actor = {
					any_international_organizations_member_of = {
						international_organization_type = international_organization_type:jihad
						leader_country ?= { religion.group ?= religion_group:muslim }
						var:preparation > 90
					}
				}
			}
			multiply = 0
		}
	}
}
