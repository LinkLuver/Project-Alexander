# Vanilla generic actions restored as disabled stubs. Their vanilla files are emptied
# (colonial_charters.txt, conquistadors.txt), but the vanilla Geopolitics panel
# (gui/expansion_lateralview.gui) still binds buttons to them, and opening the panel
# crashed the game (EXCEPTION_ACCESS_VIOLATION, 2026-10-18). Found through error.log
# ("pdx_db_ptr.h:46 Failed to cache an item for GenericAction").
# potential/allow are always false, so the actions are never shown, run or automated.

	potential = {
		always = no
	}

		always = no
	}

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick_frequency = 1

		column = {
			data = colonial_charter
		}

			always = no
		}
	}

		add = -1000
	}
