
claim_argead_empire = {
	type = internationalorganization
	price = 100
	potential = {
		exists = international_organization:achaemenid_argead
		scope:actor = { is_member_of_international_organization = international_organization:achaemenid_argead }
	}
	allow = {
		international_organization:achaemenid_argead = {
			has_land_ownership_rule = yes
		}
		scope:actor = {
			not = {
				country_rank_level = country_rank:rank_empire
			}
			is_member_of_international_organization = international_organization:achaemenid_argead
			at_war = no
			total_population_in_international_organization_percentage ={
				international_organization = international_organization:achaemenid_argead
				value > 0.50
			}
		}

	}
	effect = {
		# form empire
		scope:actor = {
			form_country = formable_country:IRE_f # TODO Change this to the real Empire tag 
			if = {
				limit = { country_rank_level < 3 }
				set_country_rank_effect = { rank = country_rank:rank_empire }
			}
		}
	}
}

arrange_marriage_with_argead_noble = {
	type = internationalorganization
	price = 100
	potential = {
		exists = international_organization:achaemenid_argead
		scope:actor = { is_member_of_international_organization = international_organization:achaemenid_argead }
	}
	allow = {
		scope:actor = {
			is_member_of_international_organization = international_organization:achaemenid_argead
		}
	}
	select_trigger = {
		top_widget = marry_noble_character_top
		looking_for_a = character
		source = actor
		target_flag = target
		name = "marry_argead_noble_select_title"
		none_available_msg_key = "marry_argead_noble_no_characters"
		column = {
			data = character_info
		}
		visible = {
			OR = {
				scope:actor = {
					modifier:can_marry_only_overlord_dynasty_characters = no
				}
				AND = {
					scope:actor = {
						modifier:can_marry_only_overlord_dynasty_characters = yes
					}
					has_dynasty = yes
					OR = {
						owner = {
							any_subject = {
								this = scope:actor
							}
						}
						dynasty = {
							any_character_in_dynasty = {
								owner = {
									any_subject = {
										this = scope:actor
									}
								}
							}
						}
					}
				}
			}
			NOT = {
				is_same_gender = scope:recipient
			}
			NOT = {
				is_close_relative = scope:recipient
			}
			character_can_marry_trigger = yes
			not = {
				is_spouse_of = scope:recipient
			}
			modifier:blocked_from_character_interactions = no
		}
	}
	effect = {
		if = {
			limit = {
				exists = scope:recipient
				exists = scope:target
			}
			scope:recipient = {
				marry_character = scope:target
			}
		}
	}
}

imperial_reconquest_claim = {

	type = internationalorganization
	price = 100
	potential = {
		exists = international_organization:achaemenid_argead
		scope:actor = { is_member_of_international_organization = international_organization:achaemenid_argead }
	}
	allow = {
		scope:actor = {
			is_member_of_international_organization = international_organization:achaemenid_argead
			at_war = no
		}
	}

    select_trigger = {
        looking_for_a = international_organization
        source = actor
        target_flag = recipient
        name = "choose_international_organization"
        column = { data = name }
        visible = { international_organization_type = international_organization_type:achaemenid_argead }
    }

    select_trigger = {
        looking_for_a = country
        target_flag = target
        name = "choose_country"
        column = { data = name }
        visible = {
			NOT = { this = scope:actor }
			is_member_of_international_organization = international_organization:achaemenid_argead
        }

		enabled = {
			#is_neighbor_of = scope:actor
		}
    }

    effect = {
        if = {
            limit = { exists = scope:target }
            scope:actor = {
                add_casus_belli = {
                    type   = casus_belli:attack_threat
                    target = scope:target
                    years  = 5
                }
            }
        }
        else = { custom_tooltip = lordship_of_ireland_casus_belli }
    }

}
