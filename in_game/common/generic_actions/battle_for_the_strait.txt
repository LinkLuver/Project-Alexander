gra_reduce_castilian_influence = {
	type = situation
	show_message_if = {
		is_battle_for_the_strait_principal = yes
	}

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	price = price:bfts_actions_price
	price_modifier = {
		add = 1
		add = {
			desc = "bfts_actions_cost"
			value = scope:actor.modifier:bfts_actions_cost
		}
	}

	potential = {
		country_exists = c:CAS
		country_exists = c:GRA
		has_dlc = "d009_across_the_pillars"
		is_situation_active = situation:battle_for_the_strait
		scope:actor = c:GRA
	}

	allow = {
		situation:battle_for_the_strait.var:influence_over_granada < 0
	}

	cooldown = {
		type = gra_reduce_castilian_influence
		years = 3
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:battle_for_the_strait = this
			situation_is_active = yes
		}
	}

	effect = {
		change_influence_over_granada_effect = { value = bfts_pull_toward_mor_mid }
		scope:actor = {
			if = {
				limit = {
					is_subject_of = c:CAS
				}
				add_liberty_desire = liberty_desire_mild_plus
			}
		}
	}

	ai_will_do = {
		if = {
			limit = {
				situation:battle_for_the_strait = { has_variable = influence_over_granada }
				situation:battle_for_the_strait.var:influence_over_granada < 25
			}
			add = {
				desc = "BFTS_INFLUENCE_AXIS_POSITION"
				value    = situation:battle_for_the_strait.var:influence_over_granada
				multiply = -100
			}
		}
	}
}

gra_enforce_sovereignty = {
	type = situation
	show_message_if = {
		is_battle_for_the_strait_principal = yes
	}
	show_in_gui_list = no

	ai_tick = monthly
	ai_tick_frequency = 6

	automation_tick = never
	automation_tick_frequency = 12

	price = price:bfts_actions_price
	price_modifier = {
		add = 1
		add = {
			desc = "bfts_actions_cost"
			value = scope:actor.modifier:bfts_actions_cost
		}
	}

	potential = {
		country_exists = c:GRA
		has_dlc = "d009_across_the_pillars"
		is_situation_active = situation:battle_for_the_strait
		
	}

	allow = {
		custom_tooltip = {
			text = BFTS_ACTION_REQUIRES_GRANADA
			scope:actor = c:GRA
		}
	}

	cooldown = {
		type = gra_enforce_sovereignty
		years = 3
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:battle_for_the_strait = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = { NOT = { has_country_modifier = enforcing_sovereignty_modifier } }
				add_country_modifier = {
					modifier = enforcing_sovereignty_modifier
					years = -1
					mode = replace
					desc = until_we_cease_our_projection_efforts_tt
				}
				set_variable = {
					name = duration_of_enforcing_variable
					value = {
						add = current_year
						add = 3
					}
				}
			}
			else = {
				remove_country_modifier = enforcing_sovereignty_modifier

				hidden_effect = { remove_variable = duration_of_enforcing_variable }
			}
		}
	}

	ai_will_do = {
		if = {
			limit = { 
				scope:actor = {
					NOT = { has_country_modifier = enforcing_sovereignty_modifier }
				}
			}
			add = {
				desc = "Low [granadan_sovereignty|e] calls for projection"
				value = 100
				subtract = situation:battle_for_the_strait.var:granadan_sovereignty
			}
			add = {
				desc = "Do we have any loans? -10/loan"
				value = {
					add = scope:actor.num_loans
					multiply = -10
				}
			}
		}
		else = {
			add = {
				desc = "Low [granadan_sovereignty|e] calls for projection"
				value = situation:battle_for_the_strait.var:granadan_sovereignty
				subtract = 50
			}
		}

		if = {
			limit = { scope:actor = { has_variable = duration_of_enforcing_variable } }
			add = {
				desc = "How long have we had this active for?"
				value = {
					add = scope:actor.var:duration_of_enforcing_variable
					subtract = current_year
					multiply = -5
				}
			}
		}
		else = {
			add = {
				value = 1
				desc = "BASE_VALUE"
			}
			add = {
				desc = "How healthy is our monthly balance?"
				value = {
					add = scope:actor.monthly_balance
					multiply = 0.1
				}
			}
			add = {
				desc = "How big is our treasury?"
				value = scope:actor.gold
				multiply = 0.01
			}
		}
		
	}
}

	}

	ai_tick_frequency = 12

	automation_tick = never

	price_modifier = {
		add = 1
		add = {
			desc = "bfts_actions_cost"
			value = scope:actor.modifier:bfts_actions_cost
		}
		if = {
			limit = { exists = scope:target }
			add = {
				desc = "cas_demand_muslim_converts_price_abilities"
				value = scope:target.total_abilities
				multiply = 0.001
			}
		}
	}

	potential = {
		country_exists = c:CAS
		country_exists = c:GRA
		has_dlc = "d009_across_the_pillars"
		is_situation_active = situation:battle_for_the_strait
	}

	allow = {
		custom_tooltip = {
			text = BFTS_ACTION_REQUIRES_CASTILE
			c:CAS ?= scope:actor
		}
		scope:actor = {
			at_war = no
			NOT = { religion.group = religion_group:muslim }
		}
		c:GRA ?= { at_war = no }
	}

	cooldown = { type = cas_demand_muslim_converts years = 10 }

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:battle_for_the_strait = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = character
		interaction_source_list = {
			c:GRA ?= {
				every_character = {
					limit = {
						is_alive = yes
						is_adult = yes
						religion.group = religion_group:muslim
						is_ruler = no
						is_heir = no
						is_consort = no
						is_regent = no
						in_cabinet = no
						is_general = no
						is_admiral = no
						is_explorer = no
						modifier:blocked_from_character_interactions = no
					}
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "cas_demand_muslim_converts_select_title"
		none_available_msg_key = "cas_demand_muslim_converts_no_characters"
		column = { data = character_info }
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			scope:target = {
				save_scope_as = converted_character
				move_country = scope:actor
				change_character_religion = scope:actor.religion
				estate_type ?= { save_scope_as = converted_character_estate }
				add_trait = trait:court_favorite
			}
			scope:actor = {
				if = {
					limit = { exists = scope:converted_character_estate }

claim_andalusi_inheritance = {
	type = situation
	show_message_if = {
		OR = {
			is_battle_for_the_strait_principal = yes
			AND = {
				exists = scope:target_country
				this = scope:target_country
			}
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 6
	automation_tick = never
	automation_tick_frequency = 12

	price = price:bfts_actions_price
	price_modifier = {
		add = 1
		add = {
			desc = "bfts_actions_cost"
			value = scope:actor.modifier:bfts_actions_cost
		}
	}

	potential = {
		has_dlc = "d009_across_the_pillars"
		is_situation_active = situation:battle_for_the_strait
	}

	allow = {
		custom_tooltip = {
			text = BFTS_ACTION_REQUIRES_GRANADA_OR_MOROCCO
			OR = {
				scope:actor = c:GRA
				c:MOR ?= scope:actor
			}
		}
		scope:actor = {
			at_war = no
			has_raised_levies = no
		}
	}

	cooldown = { type = claim_andalusi_inheritance years = 5 }

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = { data = name }
		visible = {
			situation:battle_for_the_strait = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = target_country
		name = "choose_target_country"
		none_available_msg_key = "no_countries_available"
		column = { data = name }
		interaction_source_list = {
			region:iberia_region = {
				every_country_with_capital_in_geography = {
					limit = {
						NOT = { this = scope:actor }
						num_locations > 0
						NOT = { has_truce_with = scope:actor }
						scope:actor = { can_declare_war_on = prev }
					}
					add_to_list = source
				}
			}
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target_country }
			scope:actor = {
				add_casus_belli = {
					type = casus_belli:cb_reclaim_andalucia
					target = scope:target_country
					years = 5
				}
				if = {
					limit = {
						is_ai = yes
						num_of_diplomats >= 1
						can_declare_war_on = scope:target_country
					}
					declare_war_with_cb = {
						target = scope:target_country
						type = casus_belli:cb_reclaim_andalucia
						apply_declaration_costs = yes
					}
				}
			}
		}
		else = {
			custom_tooltip = claim_andalusi_inheritance_no_target_tt
		}
	}

	ai_will_do = {
		if = {
			limit = { exists = scope:target_country }

			if = {
				limit = {
					scope:target_country = {
						is_subject = yes
						top_overlord ?= scope:actor
					}
				}
				add = { value = -1000 desc = "our subject" }
			}
			if = {
				limit = { scope:actor = { is_during_bankruptcy = yes } }
				subtract = { desc = "[bankruptcy|e]" value = 1000 }
			}
			if = {
				limit = { scope:actor = { num_loans > 0 } }
				subtract = { desc = "loans" value = 1000 }
			}
			if = {
				limit = { scope:actor = { stability < 0 } }
				subtract = { desc = "low stability" value = 1000 }
			}
			if = {
				limit = { scope:actor = { has_any_active_disaster = yes } }
				subtract = { desc = "ongoing disaster" value = 1000 }
			}
			if = {
				limit = { scope:actor = { has_truce_with = scope:target_country } }
				add = { value = -1000 desc = "[truce|e]" }
			}
			if = {
				limit = { scope:actor = { has_casus_belli_on = scope:target_country } }
				add = { value = -10000 desc = "we already have a CB on them" }
			}
			if = {
				limit = { scope:actor = { is_allied_with = { target = scope:target_country } } }
				add = { value = -1000 desc = "our ally" }
			}
			# MOR needs a fleet to cross the strait.
			if = {
				limit = {
					scope:actor = {
						NOT = { capital.region = region:iberia_region }
						navy_size < 10
					}
				}
				subtract = { value = 2500 desc = "insufficient navy to cross the strait" }
			}

			if = {
				limit = { scope:actor = { wants_to_attack = scope:target_country } }
				add = { value = 2000 desc = "we want to attack them" }
			}
			else = {
				add = { value = -2000 desc = "we do not want to attack them" }
			}

			add = {
				desc = "RELATIVE_STRENGTH"
				value = "scope:actor.relative_defensive_alliance_strength(scope:target_country)"
				multiply = 150
				subtract = 60
			}
			add = {
				desc = "we want their land"
				value = "scope:actor.conquer_desire(scope:target_country)"
				divide = 5
			}
			add = {
				desc = "CURRENT_MANPOWER_TOOLTIP"
				value = scope:actor.manpower_percentage
				multiply = 10
			}
			add = {
				desc = "navy size"
				value = scope:actor.navy_size
				divide = 5
			}
			add = {
				value = scope:actor.modifier:aggressiveness_modifier
				subtract = scope:actor.modifier:carefulness_modifier
				multiply = 50
				desc = "aggressiveness modifier minus carefulness modifier"
			}
			if = {
				limit = { scope:target_country = { num_locations < 5 } }
				divide = { value = 3 desc = "very small enemy" }
			}
		}
		else = {
			add = { value = 25 desc = BASE_VALUE }
		}
	}
}
