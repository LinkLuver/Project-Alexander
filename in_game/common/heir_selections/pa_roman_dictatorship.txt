pa_roman_dictator_succession = {
	use_election = yes
	term_duration = 0 # Reigns until death; a successor is then selected.
	max_possible_candidates = 3
	allowed_estates = { nobles_estate burghers_estate }
	heir_is_allowed = { is_female = no age_in_years >= 30 }
	potential = { has_policy = pa_roman_dictatorship_policy }
	locked = { has_policy = pa_roman_dictatorship_policy }
	calc = {
		if = { limit = { exists = root } add = { desc = "HEIR_SELECTION_MIL_DESC" value = root.mil } add = { desc = "HEIR_SELECTION_ADM_DESC" value = root.adm } }
		if = { limit = { NOT = { exists = root } } add = { desc = "HEIR_SELECTION_MIL_DESC" value = 1 } }
	}
}
