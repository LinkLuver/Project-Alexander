iberian_union_succession = {
	include_ruler_siblings = yes
	through_female = yes
	allow_female = yes
	allow_children = yes
	ignore_ruler = yes
	allowed_estates = { nobles_estate }
	custom_tags = { cannot_be_transferred }
	heir_is_allowed = {
	}
	traverse_family_tree = yes
	depth_first = yes
	calc = {
		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_IS_RULER"
				if = {
					limit = { is_ruler = yes }
					value = 100
				}
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
			}
			add = {
				desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
				if = { limit = { modifier:blocked_from_being_ruler = yes } value = -1000 }
			}
		}
	}
	sibling_score = {
		if = {
			limit = {
				exists = root
			}
			add = {
				desc = "HEIR_SELECTION_MALE_DESC"
				if = {
					limit = {
						OR = {
							not = { exists = root }
							is_female = no
						}
					}
					value = 100
				}
			}
			add = {
				desc = "HEIR_SELECTION_LEGALLY_MALE_DESC"
				if = {
					limit = {
						is_a_praefecta = yes
					}
					value = 100
				}
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = character_age
			}
			add = {
				desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
				if = { limit = { modifier:blocked_from_being_ruler = yes } value = -1000 }
			}
		}
		else = {
			add = {
				desc = "HEIR_SELECTION_MALE_DESC"
				value = 100
			}
			add = {
				desc = "HEIR_SELECTION_LEGALLY_MALE_DESC"
				value = 100
			}
			add = {
				desc = "HEIR_SELECTION_OLDER_HEIR_DESC"
				value = 1
			}
			add = {
				desc = "HEIR_SELECTION_IS_UNSUITED_TO_RULE_DESC"
				value = -1000
			}
		}
	}

	locked = {
		succession_law = heir_selection:iberian_union_succession
	}
	setup_fallback = union_of_crowns_succession
	potential = {
		has_dlc = "d009_across_the_pillars"
		succession_law = heir_selection:iberian_union_succession
	}
	succession_effect = {
		custom_tooltip = iberian_union_succession_tt
	}
}
