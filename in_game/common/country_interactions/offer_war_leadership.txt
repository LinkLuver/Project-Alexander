offer_war_leadership = {
	type = diplomacy
	category = CATEGORY_INFLUENCE_ACTIONS

	diplomatic_cost = offer_war_leadership_cost
	diplomatic_cost_modifier = {
		if = {
			limit = {
				scope:recipient = {
					is_subject_of = scope:actor
				}
			}
			value = 0
		}
		else = {
			value = 1
		}
	}

	cooldown = {
		type = offer_war_leadership
		months = 6
		target = recipient
	}

	potential = {
		scope:actor = {
			at_war = yes
			any_current_war = {
				or = {
					attacker_leader = scope:actor
					defender_leader = scope:actor
				}
				not = { is_civil_war_for = scope:actor }
			}
		}
	}

	allow = {
	}

	select_trigger = {
		looking_for_a = war
		target_flag = war
		interaction_source_list = {
			scope:actor = {
				every_current_war = {
					limit = {
						or = {
							attacker_leader = scope:actor
							defender_leader = scope:actor
						}
						not = { is_civil_war_for = scope:actor }
					}
					add_to_list = source
				}
			}
		}
		cache_interaction_source_list = yes
		name = "choose_war_to_transfer_leadership"
		column = { data = name }
		visible = { }
		enabled = {
			trigger_if = {
				limit = { exists = scope:recipient }
				trigger_if = {
					limit = { scope:war = { attacker_leader = scope:actor } }
					scope:recipient = { not = { has_truce_with = scope:war.defender_leader } }
				}
				trigger_else_if = {
					limit = { scope:war = { defender_leader = scope:actor } }
					scope:recipient = { not = { has_truce_with = scope:war.attacker_leader } }
				}
				trigger_else = {
					always = yes
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		interaction_source_list = {
			scope:actor = {
				every_country_that_can_be_called_offensively = {
					add_to_list = source
				}
				every_country_that_can_be_called_defensively = {
					add_to_list = source
				}
			}
		}
		name = "choose_country"
		show_why_not_enabled = yes
		none_available_msg_key = "no_valid_war_leader_candidates"
		column = { data = name }
		visible = { }
		enabled = {
			scope:actor = {
				great_power_score < {
					value = root.great_power_score
				}
			}
			# scope:war is only bound once a war has been chosen in the first
			# stage; the diplomacy actions list pre-checks this recipient stage
			# with scope:war unset, so all war-dependent checks must be guarded
			# behind exists = scope:war to avoid silently greying out the action.
			trigger_if = {
				limit = { exists = scope:war }
				scope:war = {
					not = { is_in_war = root }
				}
				trigger_if = {
					limit = { scope:war = { attacker_leader = scope:actor } }
					root = { can_join_offensive_war_with = scope:actor }
					not = {
						scope:war = { defender_leader = { is_subject_or_below_of = root } }
					}
					# Recipient would join the attacker side; block if they hold a
					# truce with the defending leader so accepting cannot bypass the
					# trucebreak penalty gained by a prior separate peace (CAESAR-69033).
					root = { not = { has_truce_with = scope:war.defender_leader } }
				}
				trigger_else_if = {
					limit = { scope:war = { defender_leader = scope:actor } }
					root = { can_join_defensive_war_with = scope:actor }
					not = {
						scope:war = { attacker_leader = { is_subject_or_below_of = root } }
					}
					# Recipient would join the defender side; block if they hold a
					# truce with the attacking leader (see CAESAR-69033).
					root = { not = { has_truce_with = scope:war.attacker_leader } }
				}
				trigger_else = {
					# Actor leads neither side of the chosen war, so no
					# join-side restriction applies.
					always = yes
				}
			}
		}
	}

	ai_will_do = {
		add = -1
	}

	diplo_chance = {
		base = 60
		levy_availability = -100
		low_manpower = -20
		negative_stability = -0.2
		war_exhaustion = -2
		in_debt = -5
		recipient_civil_war = -100
		recipient_at_war = -100
		opinion = 0.3
		lacks_border = -10
		border_distance = -0.1
	}

	accept = {
		add = {
			desc = "DIPLOREASON_GAINING_WAR_LEADERSHIP"
			value = 20
		}
		if = {
			limit = { exists = scope:war }
			if = {
				limit = { scope:war = { attacker_leader = scope:actor } }
				if = {
					limit = { exists = scope:war.defender_leader }
					subtract = {
						desc = "DIPLOREASON_TARGET_OPINION"
						value = "opinion(scope:war.defender_leader)"
						multiply = 0.5
					}
					subtract = {
						desc = "DIPLOREASON_BORDER_DISTANCE"
						value = "border_distance_to(scope:war.defender_leader)"
						multiply = 0.1
					}
				}
			}
			else = {
				if = {
					limit = { exists = scope:war.attacker_leader }
					subtract = {
						desc = "DIPLOREASON_TARGET_OPINION"
						value = "opinion(scope:war.attacker_leader)"
						multiply = 0.5
					}
					subtract = {
						desc = "DIPLOREASON_BORDER_DISTANCE"
						value = "border_distance_to(scope:war.attacker_leader)"
						multiply = 0.1
					}
				}
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:war
				exists = scope:recipient
			}
			scope:war = {
				if = {
					limit = { attacker_leader = scope:actor }
					scope:recipient = {
						join_war_as_attacker = {
							war = prev
							call_in_subjects = yes
						}
					}
				}
				else_if = {
					limit = { defender_leader = scope:actor }
					scope:recipient = {
						join_war_as_defender = {
							war = prev
							call_in_subjects = yes
						}
					}
				}
			}
			scope:recipient = {
				set_war_leader = { war = scope:war }
			}
		}
	}

	reject_effect = {
	}
}
