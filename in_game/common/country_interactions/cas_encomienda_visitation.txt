cas_encomienda_visitation = {
	type = subject
	category = CATEGORY_SUBJECT_ACTIONS

	potential = {

			has_dlc = "d009_across_the_pillars"
			OR = {
				has_or_had_tag = CAS
				has_or_had_tag = SPA
			}
			has_policy = encomienda_system_policy
		}
	}

	allow = {
	}

	cooldown = {
		type = cas_encomienda_visitation
		years = 10
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			scope:actor = {
				every_subject = {
					add_to_list = source
				}
			}
		}
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_subject
		}
		visible = {
			is_subject_type = colonial_nation
		}
		enabled = {
			subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
			liberty_desire < MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION
			NOT = { has_country_modifier = cas_encomienda_visitation_subject_modifier }
		}
	}

	effect = {
		if = {
			limit = { exists = scope:recipient }
			scope:actor = {
				add_country_modifier = {
					modifier = cas_encomienda_visitation_modifier
					years = 50
					mode = add_and_extend
				}
			}
			scope:recipient = {
				add_country_modifier = {
					modifier = cas_encomienda_visitation_subject_modifier
					years = 50
					mode = add_and_extend
				}
			}
		}
	}

	ai_will_do = {
		add = {
			value = 10
		}
	}
