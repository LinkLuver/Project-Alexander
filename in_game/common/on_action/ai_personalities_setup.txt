# AI Personality on_action hooks.
# Historical assignments are in setup/countries/26_ai_personalities.txt.
# Countries not listed there default to ai_cautious (set in C++ constructor).
# This file handles:
#   - Random mode: override all AI countries at game start
#   - Random per age: re-roll at each new age
#   - Random per ruler: re-roll whenever a new ruler takes the throne

on_game_start = {
	on_actions = { on_ai_personalities_game_start }
}

on_ai_personalities_game_start = {
	effect = {
		every_country = {
			limit = { is_ai = yes }

			if = {
				limit = { has_game_rule = ai_personalities_random }
				# Random mode: ignore setup file assignments, pick randomly
				random_list = {
					1 = { set_personality = ai_personality:ai_aggressive }
					1 = { set_personality = ai_personality:ai_expansionist }
					1 = { set_personality = ai_personality:ai_defensive }
					1 = { set_personality = ai_personality:ai_cautious }
					1 = { set_personality = ai_personality:ai_opportunistic }
					1 = { set_personality = ai_personality:ai_isolationist }
					1 = { set_personality = ai_personality:ai_friendly }
				}
			}
		}
	}
}
