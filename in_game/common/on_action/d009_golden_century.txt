on_winning_war = {
	on_actions = { d009_gc_winning_war }
}

d009_gc_winning_war = {
	effect = {
		if = {
			limit = {
				has_dlc = "d009_across_the_pillars"
				is_situation_active = situation:golden_century
				has_global_variable = battle_for_the_strait_winner_country
				this = global_var:battle_for_the_strait_winner_country
				NOT = { has_variable = gc_recently_humbled_grand_alliance }
				io_exists = international_organization:grand_alliance
				scope:war = {
					OR = {
						AND = {
							attacker_leader = root
							defender_leader = { is_member_of_international_organization = international_organization:grand_alliance }
						}
						AND = {
							defender_leader = root
							attacker_leader = { is_member_of_international_organization = international_organization:grand_alliance }
						}
					}
				}
			}
			set_variable = {
				name = gc_recently_humbled_grand_alliance
				days = 730
			}
			gc_change_european_dominance = { value = gc_bump_won_war }
			scope:war = {
				if = {
					limit = { attacker_leader = root }
					every_defender = {
						limit = { is_member_of_international_organization = international_organization:grand_alliance }
						international_organization:grand_alliance = {
							remove_country_from_international_organization = prev
						}
					}
				}
				else = {
					every_attacker = {
						limit = { is_member_of_international_organization = international_organization:grand_alliance }
						international_organization:grand_alliance = {
							remove_country_from_international_organization = prev
						}
					}
				}
			}
			trigger_event_non_silently = golden_century_events.54
		}
	}
}
