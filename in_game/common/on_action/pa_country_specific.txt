on_diadochi_occupy_location_horde = {
	trigger = {
		scope:winner ?= {
			# is diadochi 
		 	has_reform = government_reform:diadochi_gov   #Diadochi gov

			modifier:auto_conquer_at_war = yes
			always = no 
		}
		scope:loser ?= {
			# is diadochi 
		 	has_reform = government_reform:diadochi_gov   #Diadochi gov

		}
	}
	effect = {
		root = { save_scope_as = target_location }
		scope:loser = { save_scope_as = previous_owner }
		scope:winner = {
			on_diadochi_occupy_location_effect = yes
		}
	}
}
on_diadochi_occupy_location = {
	trigger = { 
		has_reform = government_reform:diadochi_gov 
		scope:target ?= {
			is_in_scripted_geography = scripted_geography:alexanders_empire = yes
		}
	}
	effect = {
		scope:target = { save_scope_as = target_location }
		scope:target.owner = { save_scope_as = previous_owner }
		on_diadochi_occupy_location_effect = yes
	}
}
