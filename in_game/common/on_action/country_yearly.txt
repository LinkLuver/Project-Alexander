# Each Catholic country accumulates antagonism toward a non-Christian holder of Rome every year,
# but only while that owner has not yet cored Rome. Coring Rome stops the accrual and lets the
# antagonism decay away, giving the holder a path to legitimize the conquest.
yearly_infidel_holds_rome_pulse = {
	trigger = {
		exists = religion
		religion = religion:catholic
		location:rome = {
			owner = {
				NOT = { religion.group = religion_group:christian }
				is_real_country = yes
				NOT = { has_core = prev }
			}
		}
	}
	effect = {
		add_antagonism = {
			target = location:rome.owner
			modifier = antagonism_infidel_holds_rome
		}
	}
}
