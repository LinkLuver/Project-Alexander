# Removes the placeholder pops in motu_maha, see main_menu/setup/start/06_pops.txt
on_game_start = {
	on_actions = { on_game_start_remove_placeholder_pops }
}

on_game_start_remove_placeholder_pops = {
	effect = {
		# The event only needs some country to fire for; it works on location:motu_maha itself.
		# Most populous country, so it still exists when the event fires.
		# 7 days late: deleting them on day 0 brought the "Culture has no pops" errors back.
		ordered_country = {
			order_by = total_population
			trigger_event_silently = { id = motu_maha.1 days = 7 }
		}
	}
}
