on_culture_group_merged = {
	effect = {
		scope:merged_culture_group = {
			save_scope_as = merged_group
		}
		culture = {
			save_scope_as = new_merged_culture
		}

		add_accepted_culture = scope:old_culture

		assimilate_group_into_merged_culture = yes

		every_country = {
			limit = {
				NOT = { this = root }
				exists = culture
				culture = {
					any_culture_group = {
						this = scope:merged_group
					}
				}
				NOT = { culture = { is_merged_culture_group = yes } }
			}
			set_variable = { name = culture_unification_inviter value = root }
			set_variable = { name = culture_unification_group value = scope:merged_group }
			set_variable = { name = culture_unification_culture value = root.culture }
			trigger_event_non_silently = cultural_event.15
		}
	}
}
