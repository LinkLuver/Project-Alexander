on_four_yearly_check_for_script = {
	#it is on a 4 year check on purpose as it is a heavy check and should only touch on countries with the union of crowns inheritance
	effect = {
		if = {
			limit = {
				OR = {
					has_country_modifier = union_of_crowns_pact_modifier
					succession_law ?= heir_selection:union_of_crowns_succession
				}
				NOT = { has_mutual_scripted_relation_of_type = relation_type:union_of_crowns_pact }
			}
			remove_country_modifier = union_of_crowns_pact_modifier
			change_heir_selection = heir_selection:cognatic_primogeniture
			if = {
				limit = { has_variable = union_of_crowns_partner }
				var:union_of_crowns_partner = {
					remove_trust_equilibrium = {
						modifier = trust_union_of_crowns_pact
						target = root
					}
				}
				remove_variable = union_of_crowns_partner
			}
		}
	}
}

on_albanian_migration = {
	effect = {
		# Optimization - the chance to move further into greece depend on the albanian population in the "previous area"
		# Otherwise, we would need to calculate the same culture percentages per each location
		set_local_variable = {
			name = weight_for_attica
			value = {
				value = "province_definition:thessaly_province.culture_percentage(culture:albanian)"
				add = "province_definition:south_epirus_province.culture_percentage(culture:albanian)"
				add = "province_definition:north_epirus_province.culture_percentage(culture:albanian)"
				add = "province_definition:aetolia_province.culture_percentage(culture:albanian)"
				divide = 4
				max = 0.1
				multiply = 10
			}
		}
		set_local_variable = {
			name = weight_for_morea
			value = {
				value = "province_definition:attica_province.culture_percentage(culture:albanian)"
				add = "province_definition:neopatras_province.culture_percentage(culture:albanian)"
				add = "province_definition:south_epirus_province.culture_percentage(culture:albanian)"
				add = "province_definition:north_epirus_province.culture_percentage(culture:albanian)"
				add = "province_definition:aetolia_province.culture_percentage(culture:albanian)"
				divide = 5
				max = 0.1
				multiply = 10
			}
		}
		set_local_variable = {
			name = weight_for_italy
			value = {
				value = "area:morea_area.culture_percentage(culture:albanian)"
				add = "area:northern_greece_area.culture_percentage(culture:albanian)"
				divide = 2
				max = 0.1
				multiply = 10
			}
		}

		add_to_local_variable_map = {
			name = previous_weights
			key = province_definition:attica_province
			value = local_var:weight_for_attica
		}
		add_to_local_variable_map = {
			name = previous_weights
			key = province_definition:neopatras_province
			value = local_var:weight_for_attica
		}
		area:basilicata_area = {
			every_province_definition_in_area = {
				add_to_local_variable_map = {
					name = previous_weights
					key = this
					value = local_var:weight_for_italy
				}
			}
		}
		area:apulia_area = {
			every_province_definition_in_area = {
				add_to_local_variable_map = {
					name = previous_weights
					key = this
					value = local_var:weight_for_italy
				}
			}
		}
		area:morea_area = {
			every_province_definition_in_area = {
				add_to_local_variable_map = {
					name = previous_weights
					key = this
					value = local_var:weight_for_morea
				}
			}
		}

		# Try to source a location
		scripted_geography:albanian_migration_source_geography = {
			random_location_in_scripted_geography = {
				limit = {
					has_owner = yes
				}
				weight = {
					factor = 0
					modifier = {
						add = albanian_emigration_chance
					}
				}
				save_scope_as = source_location
			}
		}
		
		if = {
			limit = {
				exists = scope:source_location
			}
			scripted_geography:albanian_migration_destination_geography = {
				random_location_in_scripted_geography = {
					limit = {
						has_owner = yes
					}
					weight = {
						factor = 0
						modifier = {
							add = albanian_immigration_chance
						}
					}

					save_scope_as = destination_location
				}
			}
			if = {
				limit = {
					exists = scope:destination_location
				}
				set_local_variable = {
					name = amount_of_migration
					value = {
						value = scope:source_location.population
						multiply = { 0.01 0.05 }
						multiply = {
							value = 1
							add = {
								value = scope:source_location.albanian_immigration_chance
								divide = 100
							}
						}
					}
				}

				scope:source_location.owner = {
					#trigger_event_non_silently = { #P:A TODO Associated event was removed
						#id = flavor_byz.58 #P:A TODO Associated event was removed
					#} #P:A TODO Associated event was removed
				}
				scope:destination_location.owner = {
					#trigger_event_non_silently = { #P:A TODO Associated event was removed
						#id = flavor_byz.59 #P:A TODO Associated event was removed
					#} #P:A TODO Associated event was removed
				}
			}
		}
	}
