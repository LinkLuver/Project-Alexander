# Monthly gold contribution from all members to the Jihad treasury
jihad_treasury_contribution = {
	get_payer_list = {
		every_international_organization_member = {
			limit = { root = { var:phase = 1 } }
			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		add_to_local_variable_list = {
			name = payees
			target = root
		}
	}
	price = jihad_treasury_contribution_price
	uses_maintenance = yes
	min_slider_value = 0
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|jihad_treasury_contribution)" < 0.5 }
		scale = {
			value = 1
			subtract = {
				value = "payment_maintenance(scope:recipient|jihad_treasury_contribution)"
				multiply = 2
			}
		}
		monthly_prestige = -0.2
	}
	maintenance_modifier = {
		potential_trigger = { "payment_maintenance(scope:recipient|jihad_treasury_contribution)" > 0.5 }
		scale = {
			value = "payment_maintenance(scope:recipient|jihad_treasury_contribution)"
			subtract = 0.5
			multiply = 2
		}
		monthly_prestige = 0.2
	}
	price_multiplier = {
		every_international_organization_member = {
			add = monthly_income_trade_and_tax
		}
		multiply = {
			desc = "IO_PAYMENT_CONTRIBUTION_RATE_LABEL"
			value = 0.3
		}
	}
	proportion_for_payer = {
		add = {
			desc = "game_concept_monthly_income_trade_and_tax"
			value = monthly_income_trade_and_tax
		}
	}
	proportion_for_payee = {
		add = {
			desc = "jihad_treasury_contribution"
			value = 1
		}
	}
}
