italian_league_sponsor_gold_contribution = {
	get_payer_list = {
		every_country_with_special_status_of_type = {
			type = italian_league_sponsor
			add_to_local_variable_list = {
				name = payers
				target = this
			}
		}
	}
	get_payee_list = {
		leader_country ?= {
			add_to_local_variable_list = {
				name = payees
				target = this
			}
		}
	}
	price = italian_league_sponsor_gold_price
	uses_maintenance = yes
	min_slider_value = 0.1
	maintenance_modifier = {
		potential_trigger = {
			"payment_maintenance(scope:recipient|italian_league_sponsor_gold_contribution)" < 0.5
		}
		scale = {
			value = 1
			subtract = {
				value = "payment_maintenance(scope:recipient|italian_league_sponsor_gold_contribution)"
				multiply = 2
			}
		}
		research_speed_modifier = -0.1
	}
	maintenance_modifier = {
		potential_trigger = {
			"payment_maintenance(scope:recipient|italian_league_sponsor_gold_contribution)" > 0.5
		}
		scale = {
			value = "payment_maintenance(scope:recipient|italian_league_sponsor_gold_contribution)"
			subtract = 0.5
			multiply = 2
		}
		research_speed_modifier = 0.2
	}
	price_multiplier = {
		every_country_with_special_status_of_type = {
			type = italian_league_sponsor
			add = {
				desc = "ITALIAN_LEAGUE_SPONSOR_GOLD_MULTIPLIER"
				value = monthly_income_trade_and_tax
				multiply = 0.15
			}
		}
	}
	proportion_for_payer = {
		add = {
			desc = "game_concept_monthly_income_trade_and_tax"
			value = monthly_income_trade_and_tax
		}
	}
	proportion_for_payee = {
		add = {
			desc = "italian_league_sponsor_gold_contribution"
			value = 1
		}
	}
}
