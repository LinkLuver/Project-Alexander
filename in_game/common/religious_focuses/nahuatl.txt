adopt_ometeotl = {	
	monthly_progress = {
		# Default Base
		add = {
			desc = "DIPLOREASON_BASE"
			value = 0.05
		}

		# 3 Basic Trends
		add = {
			desc = "[ShowModifierTypeName('monthly_nahualt_reform_progress')]"
			value = modifier:monthly_nahualt_reform_progress
		}
		add = {
			desc = "[crown_power|e]"
			value = "estate_power(estate_type:crown_estate)"
			multiply = 0.1
		}
		if = {
			limit = { exists = ruler_or_regent }
			add = {
				desc = "ruler_or_regent_total_abilities"
				value = ruler_or_regent.total_abilities
				multiply = 0.0015	#~0.5 if our ruler or regent is a 300-stat god
			}
		}

		# 1-2 Unique Drifts

		add = {
			desc = "[true_faith_tolerance|e]"
			value = {
				add = modifier:tolerance_own
				multiply = 0.25
			}
		}
	}

	allow = {
		has_completed_religious_focus = elevate_patron_god
		has_completed_religious_focus = establish_cihuacoatl
		has_completed_religious_focus = expand_cuahpipiltin
		has_completed_religious_focus = increase_militarism
		has_completed_religious_focus = institute_flower_wars
		has_completed_religious_focus = legal_reforms
		has_completed_religious_focus = raise_sacrifice_rate
	}
	
	modifier_while_progressing = {
		clergy_estate_max_tax = -0.1
	}

	modifier_on_completion = {
		country_child_education = 0.1
		monthly_doom = -0.1
		global_estate_target_satisfaction = small_permanent_target_satisfaction
	}

	ai_will_do = { add = 1 }
}
