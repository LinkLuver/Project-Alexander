crown_heretic_ruler_discontent = {
	potential_trigger = {
		has_ruler = yes
		modifier:tolerance_heretic < 0
		"estate(estate_type:crown_estate)" = {
			dominant_religion ?= {
				this != root.ruler.religion
				group = root.ruler.religion.group
			}
		}
	}

	scales_with = {
		value = modifier:tolerance_heretic
		multiply = -1
		min = 0
		max = 4
	}

	crown_estate_target_satisfaction = -0.01
}
