proclaim_pase_foral = {
	ability = adm
	allow_multiple = no

	potential = {
		has_dlc = "d009_across_the_pillars"
		culture = culture:basque
		OR = {
			is_subject = yes
			is_junior_partner = yes
		}
	}

	allow = {
		has_estate_privilege = estate_privilege:basque_pase_foral
		NOT = {
			any_cabinet = {
				has_cabinet_action = yes
				cabinet_action = cabinet_action:frustrate_annexation
			}
		}
	}

	country_modifier = {
		hostile_diplomatic_annexation_efficiency = -0.2
		loyalty_to_overlord = -0.2
	}

	ai_will_do = {
		add = 10
		scope:actor = {
			if = {
				limit = {
					is_being_annexed = yes
				}
				add = { value = 20 desc = BEING_ANNEXED }
			}
		}
	}
}
