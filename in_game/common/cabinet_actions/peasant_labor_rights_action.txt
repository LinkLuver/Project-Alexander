peasant_labor_rights_action = {
	ability = adm

	potential = {
		has_reform = government_reform:tyranny_of_the_capital_reform
		num_locations > 0
	}

	allow = {
		any_province = {
			any_location_in_province = {
				integration_level = core
				count = all
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		source = actor
		target_flag = target
		name = "peasant_labor_rights_action_select_province"
		none_available_msg_key = "peasant_labor_rights_action_no_provinces"
		column = {
			data = name
		}
		column = {
			data = development
		}
		column = {
			data = population
		}
		visible = {
			NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = peasant_labor_rights_action interaction_target = target } }
		}
		enabled = {
			any_location_in_province = {
				integration_level = core
				count = all
			}
		}
		map_color = {
			lerp = {
				min_color = define:NMapColors|MAP_COLOR_LOW
				max_color = define:NMapColors|MAP_COLOR_HIGH
				factor = prosperity
			}
		}
	}

	location_modifier = {
		local_monthly_prosperity = 0.001
		local_monthly_literacy = 0.1
	}

	progress = {
		scope:target = {
			value = province_prosperity
		}
	}

	map_marker = {
		show_for_owner = yes
	}
	

	ai_will_do = {
		add = {
			if = {
				limit = {
					scope:actor = {
						is_subject = yes
					}
				}
				value = -1000
			}
		}
	}

}
