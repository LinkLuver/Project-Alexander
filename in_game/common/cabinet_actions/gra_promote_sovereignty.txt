	}

	country_modifier = {
		monthly_granadan_sovereignty = 0.2
	}

	ai_will_do = {

				desc = "BFTS_GRANADAN_SOVEREIGNTY_POSITION"

				multiply = -5
			}
		}

			limit = { scope:actor = { is_subject = yes } }
			add = {
				scope:actor.overlord = {
					scope:actor = {
						value = this_opinion_of_prev
						multiply = -0.25
					}
				}
				desc = "our opinion of overlord"
			}
		}
	}
