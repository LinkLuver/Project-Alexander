# AUSTRIA
austria_check = {
	year = 1835
	end_year = 1836
	fail_on_end_year = yes
	success = {
		exists = c:HAB
	}
	success_child = {
		desc = "Austria is Emperor"
		trigger = {
			exists = international_organization:hre
			c:HAB = {
				is_leader_of_international_organization = international_organization:hre
			}
		}
	}
	success_child = {
		desc = "Austria has Hungary in a Union"
		trigger = {
			exists = c:HUN
			c:HAB = { in_union_with = c:HUN }
		}
	}
	success_child = {
		desc = "Austria has Bohemia in a Union"
		trigger = {
			exists = c:BOH
			c:HAB = { in_union_with = c:BOH }
		}
	}
	success_child = {
		desc = "Austria owns the core land"
		trigger = {
			area:styria_area = {
				any_location_in_area = { OR = { is_ownable = no is_owned_or_owned_by_subjects_of = c:HAB } count = all }
			}
			area:upper_austria_area = {
				any_location_in_area = { OR = { is_ownable = no is_owned_or_owned_by_subjects_of = c:HAB } count = all }
			}
			area:austria_area = {
				any_location_in_area = { OR = { is_ownable = no is_owned_or_owned_by_subjects_of = c:HAB } count = all }
			}
			area:salzburg_area = {
				any_location_in_area = { OR = { is_ownable = no is_owned_or_owned_by_subjects_of = c:HAB } count = all }
			}
			area:carinthia_area = {
				any_location_in_area = { OR = { is_ownable = no is_owned_or_owned_by_subjects_of = c:HAB } count = all }
			}
			area:tirol_area = {
				any_location_in_area = { OR = { is_ownable = no is_owned_or_owned_by_subjects_of = c:HAB } count = all }
			}
		}
	}
}
