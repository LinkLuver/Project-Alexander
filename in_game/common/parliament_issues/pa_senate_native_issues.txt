# Country-parliament issues adapted to Rome's international-organization Senate.
pa_senate_support_ruler_issue = {
	type = international_organization
	special_status = pa_senate_roman_citizens
	potential = { international_organization_type = international_organization_type:pa_roman_senate }
	allow = { leader_country ?= { has_ruler = yes is_rulers_main_country = yes } }
	selectable_for = { is_leader_of_international_organization = scope:recipient }
	chance = 10
	on_debate_start = { leader_country ?= { add_country_modifier = { modifier = pa_senate_support_ruler_debate years = 2 mode = replace } } }
	on_debate_passed = {
		leader_country ?= {
			remove_country_modifier = pa_senate_support_ruler_debate
			if = { limit = { has_ruler = yes } ruler = { add_adm = 3 add_dip = 3 add_mil = 3 } }
			else_if = { limit = { has_regent = yes } regent = { add_adm = 3 add_dip = 3 add_mil = 3 } }
			pa_senate_motion_passed_effect = yes
		}
	}
	on_debate_failed = { leader_country ?= { remove_country_modifier = pa_senate_support_ruler_debate pa_senate_motion_failed_effect = yes } }
}

pa_senate_bolster_republic_issue = {
	type = international_organization
	special_status = pa_senate_roman_citizens
	potential = { international_organization_type = international_organization_type:pa_roman_senate }
	allow = { leader_country ?= { government_type = government_type:republic at_war = no "estate_power(estate_type:crown_estate)" < 0.5 NOT = { has_country_modifier = bolstered_state_authority } } }
	selectable_for = { is_leader_of_international_organization = scope:recipient }
	chance = 10
	on_debate_start = { leader_country ?= { add_country_modifier = { modifier = pa_senate_bolster_republic_debate years = 2 mode = replace } } }
	on_debate_passed = { leader_country ?= { remove_country_modifier = pa_senate_bolster_republic_debate add_country_modifier = { modifier = bolstered_state_authority years = 10 mode = add_and_extend } pa_senate_motion_passed_effect = yes } }
	on_debate_failed = { leader_country ?= { remove_country_modifier = pa_senate_bolster_republic_debate pa_senate_motion_failed_effect = yes } }
}

pa_senate_census_issue = {
	type = international_organization
	special_status = pa_senate_roman_citizens
	potential = { international_organization_type = international_organization_type:pa_roman_senate }
	allow = { always = yes }
	selectable_for = { is_leader_of_international_organization = scope:recipient }
	chance = 10
	on_debate_start = { leader_country ?= { add_country_modifier = { modifier = pa_senate_census_debate years = 2 mode = replace } } }
	on_debate_passed = { leader_country ?= { remove_country_modifier = pa_senate_census_debate add_stability = stability_mild_bonus pa_senate_motion_passed_effect = yes } }
	on_debate_failed = { leader_country ?= { remove_country_modifier = pa_senate_census_debate pa_senate_motion_failed_effect = yes } }
}
