# A country Parliament issue: choose the law and policy before convening.
pa_senate_country_law_debate = {
	estate = crown_estate
	potential = { tag = ROM government_type = government_type:republic has_variable = pa_senate_initialized }
	allow = { has_ruler = yes }
	chance = { add = 1 }
	on_debate_passed = {
		save_scope_as = pa_senate_law_country
		if = {
			limit = {
				var:pa_senate_country_proposed_law ?= { NOT = { is_locked_for = scope:pa_senate_law_country } }
				var:pa_senate_country_proposed_policy ?= { policy_possible_for_country = scope:pa_senate_law_country }
			}
			add_policy = var:pa_senate_country_proposed_policy
		}
		remove_variable = pa_senate_country_proposed_law
		remove_variable = pa_senate_country_proposed_policy
	}
	on_debate_failed = {
		remove_variable = pa_senate_country_proposed_law
		remove_variable = pa_senate_country_proposed_policy
	}
}
