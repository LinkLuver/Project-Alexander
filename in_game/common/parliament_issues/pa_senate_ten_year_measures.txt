pa_senate_ten_year_debate = {
	type = international_organization
	special_status = pa_senate_roman_citizens
	potential = { international_organization_type = international_organization_type:pa_roman_senate }
	allow = { always = yes }
	chance = 0
	on_debate_passed = {
		leader_country ?= {
			if = { limit = { var:pa_senate_ten_year_proposal = 1 } add_country_modifier = { modifier = pa_senate_ten_year_tax years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 2 } add_country_modifier = { modifier = pa_senate_ten_year_manpower years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 3 } add_country_modifier = { modifier = pa_senate_ten_year_grain years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 4 } add_country_modifier = { modifier = pa_senate_ten_year_farms years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 5 } add_country_modifier = { modifier = pa_senate_ten_year_works years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 6 } add_country_modifier = { modifier = pa_senate_ten_year_industry years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 7 } add_country_modifier = { modifier = pa_senate_ten_year_aristocracy years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 8 } add_country_modifier = { modifier = pa_senate_ten_year_levies years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 9 } add_country_modifier = { modifier = pa_senate_ten_year_registers years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 10 } add_country_modifier = { modifier = pa_senate_ten_year_construction years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 11 } add_country_modifier = { modifier = pa_senate_ten_year_distribution years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 12 } add_country_modifier = { modifier = pa_senate_ten_year_foodworks years = 10 mode = replace } }
			else_if = { limit = { var:pa_senate_ten_year_proposal = 13 gold >= 25 } add_gold = -25 save_scope_as = pa_senate_ten_year_holdings_rome capital = { add_pop = { culture = scope:pa_senate_ten_year_holdings_rome.culture religion = scope:pa_senate_ten_year_holdings_rome.religion type = pop_type:nobles size = 0.1 } } add_country_modifier = { modifier = pa_senate_ten_year_holdings years = 10 mode = replace } }
			if = { limit = { var:pa_senate_ten_year_proposal = 13 gold < 25 } pa_senate_motion_failed_effect = yes }
			else = { pa_senate_motion_passed_effect = yes }
			remove_variable = pa_senate_ten_year_proposal
		}
	}
	on_debate_failed = { leader_country ?= { pa_senate_motion_failed_effect = yes remove_variable = pa_senate_ten_year_proposal } }
}
