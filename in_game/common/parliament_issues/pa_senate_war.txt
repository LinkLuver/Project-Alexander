pa_senate_war_debate = {
	type = international_organization
	special_status = pa_senate_roman_citizens
	potential = { international_organization_type = international_organization_type:pa_roman_senate }
	allow = { always = yes }
	chance = 0
	on_debate_passed = {
		leader_country ?= {
			set_variable = { name = pa_senate_war_mandate value = yes days = 365 }
			if = { limit = { NOT = { has_variable = pa_senate_war_vote_defied } } change_variable = { name = pa_senate_authority add = 8 max = 100 } }
			remove_variable = pa_senate_war_vote_defied
			remove_variable = pa_senate_vote_defied
			remove_variable = pa_senate_last_motion_waiting
			set_variable = pa_senate_last_motion_passed
		}
	}
	on_debate_failed = {
		leader_country ?= {
			remove_variable = pa_senate_war_vote_defied
			remove_variable = pa_senate_vote_defied
			change_variable = { name = pa_senate_authority subtract = 5 min = 0 }
			remove_variable = pa_senate_last_motion_waiting
			set_variable = pa_senate_last_motion_rejected
		}
	}
}
