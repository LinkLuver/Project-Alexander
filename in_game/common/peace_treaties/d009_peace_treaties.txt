peace_partition = {

	blocks_full_annexation = yes

	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 25
		}
		add = {
			desc = [country_strength|e]
			value = 100
			multiply = "scope:loser.relative_military_strength(scope:winner)"
		}
	}

	potential = {
		scope:war = {
			is_no_cb_war = no
			casus_belli ?= casus_belli:cb_partition
		}
	}

	allow = {
		scope:loser = {
			any_other_core_country = {
				count > 0
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = peace_partition_tt
			scope:loser = { 
				if = {
					limit = {
						any_owned_location = {
							NOT = { is_core_of = scope:loser }
						}
					}
					release_non_cores = {}
				}
				every_owned_location = {
					every_core_in_location = {
						limit = {
							NOT = { country_exists = this }
							NOT = { is_in_list = partition_targets }
						}
						add_to_temporary_list = partition_targets
					}
				}
				every_in_list = {
					list = partition_targets
					save_temporary_scope_as = partition_target
					scope:loser = {
						create_country_from_cores_in_our_locations = scope:partition_target
					}
				}
			}
		}
	}

	ai_desire = {
		value = 1000
	}
}

peace_andalusi_submission = {

	blocks_full_annexation = yes

	category = country

	cost = {
		add = {
			desc  = "DIPLOREASON_BASE"
			value = 20
		}
		add = {
			desc     = [country_strength|e]
			value    = 40
			multiply = "scope:loser.relative_military_strength(scope:winner)"
			max      = 40
			min      = 0
		}
	}

	base_antagonism = 20
	antagonism_type = antagonism_force_tributary

	potential = {
		scope:war = {
			is_no_cb_war = no
			casus_belli ?= casus_belli:cb_demand_andalusi_submission
		}
		c:GRA ?= scope:loser
		scope:loser = {
			OR = {
				can_make_subject_of = {
					target = scope:winner
					type = subject_type:taifa
					ignore_war_limitation = yes
				}
				can_make_subject_of = {
					target = scope:winner
					type = subject_type:tributary
					ignore_war_limitation = yes
				}
			}
		}
	}

	allow = {
	}

	effect = {
		scope:loser = {
			if = {
				limit = {
					scope:winner = { has_or_had_tag = MOR }
					can_make_subject_of = {
						target = scope:winner
						type = subject_type:taifa
						ignore_war_limitation = yes
					}
				}
				make_subject_of = {
					target = scope:winner
					type = subject_type:taifa
					war = scope:war
				}
			}
			else = {
				make_subject_of = {
					target = scope:winner
					type = subject_type:tributary
					war = scope:war
				}
			}
		}
	}

	ai_desire = {
		value = 1000
	}
	ai_force_add = yes
}
