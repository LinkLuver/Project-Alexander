break_from_tatar_yoke = {
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 50
		}
	}
	
	potential = {
		scope:war = {
			is_no_cb_war = no
			casus_belli ?= casus_belli:cb_break_free_tatar_yoke
		}
		scope:winner = {
			is_member_of_international_organization = international_organization:tatar_yoke
		}
		scope:loser = {
			OR = {
				is_leader_of_international_organization = international_organization:tatar_yoke
				any_subject_or_below = {
					is_leader_of_international_organization = international_organization:tatar_yoke	
				}
			}
		}
	}
	allow = {
	}

	effect = {
		international_organization:tatar_yoke = {
			remove_country_from_international_organization = {
				country = scope:winner
				reason = WonFreedom
			}
		}
		scope:winner = {
			every_subject_or_below = {
				international_organization:tatar_yoke = {
					remove_country_from_international_organization = {
						country = prev
						reason = WonFreedom
					}
				}
			}
			every_neighbor_country = {
				limit = {
					scope:winner = {
						can_create_casus_belli_of_type_on = {
							type = casus_belli:cb_expansion_into_the_steppes
							target = prev
						}
					}
				}
				scope:winner = {
					add_casus_belli = {
						target = prev
						type = casus_belli:cb_expansion_into_the_steppes
					}
				}
			}
		}

	}

	ai_force_add = yes

	ai_desire = {
		value = 1000
	}
}
