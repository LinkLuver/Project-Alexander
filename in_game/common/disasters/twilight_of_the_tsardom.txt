twilight_of_the_tsardom = {

	image = "gfx/interface/illustrations/disaster/twilight_of_the_tsardom.dds"
	
	monthly_spawn_chance = monthly_spawn_chance_unique

	can_start = {
		tag = BUL
		has_any_active_disaster = no
		current_year < 1338
	}
	
	fire_only_once = yes
	
	can_end = {
		twilight_of_the_tsardom_end_trigger = yes
	}
	
	modifier = {
		global_estate_target_satisfaction = big_permanent_target_satisfaction_penalty
		monthly_rebel_growth = 0.003
		monthly_complacency = 0.2
		global_monthly_control = -0.0005
		has_complacency_effects = yes
	}
	
	on_start = {
		trigger_event_non_silently = twilight_of_the_tsardom.1
	}
	
	on_end = {
		if = {
			limit = {
				owns = location:constantinople
				is_great_power = yes
			}
			trigger_event_non_silently = twilight_of_the_tsardom.4
		}
		else_if = {
			limit = {
				owns = location:constantinople
			}
			trigger_event_non_silently = twilight_of_the_tsardom.3
		}	
		else = {
			trigger_event_non_silently = twilight_of_the_tsardom.2
		}
	}
	
	on_monthly = {
		random_list = {
			10 = { custom_tooltip = an_event_occurs_tt }
			90 = { custom_tooltip = no_event_occurs_tt }
		}
		hidden_effect = {
			random_list = {
				2 = { trigger_event_silently = twilight_of_the_tsardom.5 }
				2 = { trigger_event_silently = twilight_of_the_tsardom.6 }
				1 = { trigger_event_silently = twilight_of_the_tsardom.7 }
				2 = { trigger_event_silently = twilight_of_the_tsardom.8 }
				1 = { trigger_event_silently = twilight_of_the_tsardom.9 }
				1 = { trigger_event_silently = twilight_of_the_tsardom.10 }
				1 = { trigger_event_silently = twilight_of_the_tsardom.11 }
				90 = { }
				#More non-generic events related to the disaster are located in events/flavor_BUL
			}
		}
	}
}
