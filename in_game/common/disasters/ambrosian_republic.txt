ambrosian_republic_disaster = {

	image = "gfx/interface/illustrations/disaster/ambrosian_republic_disaster.dds"
	
	monthly_spawn_chance = monthly_spawn_chance_very_high

	fire_only_once = yes

	can_start = {
		tag = MLO
		government_type = government_type:monarchy

		OR = {
			has_regent = yes
			AND = {
				has_heir = no 
				ruler ?= { age_in_years >= 40 }
			}
		}
		legitimacy < 80

		has_any_active_disaster = no
	}
	

	can_end = {
		ambrosian_republic_disaster_end_trigger = yes
	}
	modifier = {
		global_estate_target_satisfaction = large_permanent_target_satisfaction_penalty
		stability_cost_efficiency = -0.25
	}
	
	on_start = {
		trigger_event_non_silently = ambrosian_republic_disaster.10
	}
	on_end = {
		trigger_event_non_silently = ambrosian_republic_disaster.1
	}
	
	on_monthly = {
		
		random_list = {
			5 = {custom_tooltip = an_event_occurs_tt}
			95 = {custom_tooltip = no_event_occurs_tt}
		}
		hidden_effect = {
			random_list = {
				2 = { trigger_event_silently =  ambrosian_republic_disaster.20 }
				2 = { trigger_event_silently =  ambrosian_republic_disaster.31 }
				2 = { trigger_event_silently =  ambrosian_republic_disaster.5 }
				95 = { }
			}
		}
	}
}
