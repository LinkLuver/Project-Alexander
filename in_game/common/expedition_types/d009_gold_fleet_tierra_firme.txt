gold_fleet_tierra_firme_expedition = {

	unique = yes
	returns_home = yes
	travel_speed = 1
	origin = none
	category = commercial
	icon = category_commercial
	travel_mode = sea
	dynamic_first_waypoint = yes

	potential = {
		has_dlc = "d009_across_the_pillars"
		OR = {
			tag = CAS
			tag = SPA
		}
		has_advance = colonial_charters
		has_presence_in = region:caribbean_region
		any_owned_location = {
			is_port = yes
			is_core_of = root
		}
		scripted_geography:island_of_cuba_geography = {
			any_location_in_scripted_geography = {
				is_port = yes
				top_owner ?= root
			}
		}
		region:colombia_region = {
			any_location_in_region = {
				is_port = yes
				top_owner ?= root
				NOT = { is_on_pacific_coast_americas = yes }
			}
		}
	}

	can_start = {
		gold >= gold_fleet_departure_cost
		has_non_rural_port = yes
		any_owned_location = {
			is_port = yes
			is_core_of = root
			continent = continent:europe
		}
		scripted_geography:island_of_cuba_geography = {
			any_location_in_scripted_geography = {
				is_port = yes
				top_owner ?= root
			}
		}
		region:colombia_region = {
			any_location_in_region = {
				is_port = yes
				top_owner ?= root
				NOT = { is_on_pacific_coast_americas = yes }
			}
		}
		location:northern_fuerteventura_coast = { is_discovered_by = root }
		location:atlantic_north_equatorial_current10 = { is_discovered_by = root }
		location:atlantic_north_equatorial_current25 = { is_discovered_by = root }
		location:florida_keys = { is_discovered_by = root }
		location:bermuda_high_sea2 = { is_discovered_by = root }
		location:flores_island = { is_discovered_by = root }
	}

	leader = {
		is_admiral = yes
	}

	variables = {
		gold_fleet_cargo = {
			icon = "gfx/interface/icons/flat_icons/gold_fleet_cargo.dds"
			start = 0
			min = 0
		}
		gold_fleet_danger = {
			icon = "gfx/interface/icons/flat_icons/gold_fleet_danger.dds"
			start = 1
			min = 1
		}
	}

	on_start = {
		ordered_owned_location = {
			limit = {
				is_port = yes
				is_core_of = root
				continent = continent:europe
			}
			order_by = {
				value = modifier:harbor_suitability
				multiply = -1
			}
			max = 1
			save_scope_as = departure_port_scope
		}
		scripted_geography:island_of_cuba_geography = {
			ordered_location_in_scripted_geography = {
				limit = {
					is_port = yes
					top_owner ?= root
				}
				order_by = {
					value = modifier:harbor_suitability
					multiply = -1
				}
				max = 1
				save_scope_as = best_cuba_port
			}
		}
		region:colombia_region = {
			ordered_location_in_region = {
				limit = {
					is_port = yes
					top_owner ?= root
					NOT = { is_on_pacific_coast_americas = yes }
				}
				order_by = {
					value = modifier:harbor_suitability
					multiply = -1
				}
				max = 1
				save_scope_as = primary_loot_port_scope
			}
		}
		region:central_america_region = {
			ordered_location_in_region = {
				limit = {
					is_port = yes
					top_owner ?= root
					NOT = { is_on_pacific_coast_americas = yes }
					NOT = { this = scope:best_cuba_port }
					NOT = { this = scope:primary_loot_port_scope }
				}
				order_by = {
					value = modifier:harbor_suitability
					multiply = -1
				}
				max = 1
				save_scope_as = secondary_loot_port_scope
			}
		}
		region:colombia_region = {
			ordered_location_in_region = {
				limit = {
					top_owner ?= root
					raw_material = goods:silver
				}
				order_by = {
					value = population
					multiply = -1
				}
				max = 1
				save_scope_as = silver_source_scope
			}
		}
		region:colombia_region = {
			ordered_location_in_region = {
				limit = {
					top_owner ?= root
					raw_material = goods:gems
				}
				order_by = {
					value = population
					multiply = -1
				}
				max = 1
				save_scope_as = gem_source_scope
			}
		}
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			set_variable = { name = cuba_rendezvous value = scope:best_cuba_port }
			set_variable = { name = primary_loot_port value = scope:primary_loot_port_scope }
			set_variable = { name = departure_port value = scope:departure_port_scope }
			pause_expedition = 5
			add_new_waypoint = scope:departure_port_scope
			add_new_waypoint = location:northern_fuerteventura_coast
			add_new_waypoint = location:atlantic_north_equatorial_current10
			add_new_waypoint = location:atlantic_north_equatorial_current25
			add_new_waypoint = scope:primary_loot_port_scope
			add_new_waypoint = scope:best_cuba_port
			add_new_waypoint = location:florida_keys
			add_new_waypoint = location:bermuda_high_sea2
			add_new_waypoint = location:flores_island
		}
		if = {
			limit = { exists = scope:secondary_loot_port_scope }
			scope:expedition = { set_variable = { name = secondary_loot_port value = scope:secondary_loot_port_scope } }
		}
		if = {
			limit = { exists = scope:silver_source_scope }
			scope:expedition = { set_variable = { name = silver_source value = scope:silver_source_scope } }
		}
		if = {
			limit = { exists = scope:gem_source_scope }
			scope:expedition = { set_variable = { name = gem_source value = scope:gem_source_scope } }
		}
		set_variable = gold_fleet_tf_active
		trigger_event_non_silently = gold_fleet_tierra_firme_events.1
	}

	on_monthly = {
		if = {
			limit = { scope:expedition = { has_variable = gf_awaiting_rendezvous } }
			if = {
				limit = { OR = { has_variable = gold_fleet_ne_at_havana has_variable = gold_fleet_rendezvous } }
				scope:expedition = {
					remove_variable = gf_awaiting_rendezvous
					set_variable = { name = gf_havana_state value = 2 }
					change_variable = { name = gold_fleet_danger add = -1 }
					clamp_variable = { name = gold_fleet_danger min = 1 }
					set_variable = gf_show_together_tt
				}
				if = {
					limit = { NOT = { has_variable = gold_fleet_rendezvous } }
					set_variable = gold_fleet_rendezvous
					add_prestige = prestige_mild_bonus
				}
				trigger_event_non_silently = gold_fleet_tierra_firme_events.3
			}
			else_if = {
				limit = { NOT = { has_variable = gold_fleet_ne_active } }
				scope:expedition = {
					remove_variable = gf_awaiting_rendezvous
					set_variable = { name = gf_havana_state value = 0 }
				}
				trigger_event_non_silently = gold_fleet_tierra_firme_events.3
			}
			else = {
				scope:expedition = {
					change_variable = { name = gf_havana_wait_months add = 1 }
					pause_expedition = 32
				}
				if = {
					limit = { scope:expedition = { var:gf_havana_wait_months >= 3 } }
					scope:expedition = { set_variable = { name = gf_havana_wait_months value = 0 } }
					trigger_event_non_silently = gold_fleet_tierra_firme_events.9
				}
			}
		}
		if = {
			limit = {
				scope:expedition = {
					expedition_is_traveling = yes
					NOT = { has_variable = gf_storm_done }
				}
				scope:location.province_definition = {
					any_location_in_province_definition = {
						any_weather_system_in_location = { count > 0 }
					}
				}
			}
			trigger_event_non_silently = gold_fleet_shared_events.5
		}
		if = {
			limit = {
				scope:expedition = {
					expedition_is_traveling = yes
					NOT = { has_variable = gf_sickness_done }
				}
			}
			random = {
				chance = 8
				trigger_event_non_silently = gold_fleet_shared_events.6
			}
		}
		if = {
			limit = {
				scope:expedition = {
					NOT = { has_variable = gf_havana_done }
					NOT = { has_variable = gf_leak_done }
				}
				scope:location = { region = region:caribbean_region }
			}
			random = {
				chance = 8
				trigger_event_non_silently = gold_fleet_shared_events.7
			}
		}
		if = {
			limit = {
				scope:expedition = {
					var:gold_fleet_cargo > 0
					has_variable = gem_source
					NOT = { has_variable = gf_havana_done }
					NOT = { has_variable = gf_emeralds_done }
				}
				scope:location.region = {
					any_location_in_region = {
						top_owner ?= root
						raw_material = goods:gems
					}
				}
			}
			random = {
				chance = 20
				trigger_event_non_silently = gold_fleet_tierra_firme_events.14
			}
		}
		if = {
			limit = {
				scope:expedition = {
					var:gold_fleet_cargo > 0
					NOT = { has_variable = gf_havana_done }
					NOT = { has_variable = gf_mule_done }
				}
				scope:location = { region = region:central_america_region }
			}
			random = {
				chance = 15
				trigger_event_non_silently = gold_fleet_tierra_firme_events.15
			}
		}
		if = {
			limit = {
				scope:expedition = {
					var:gold_fleet_cargo > 0
					NOT = { has_variable = gf_havana_done }
					NOT = { has_variable = gf_pearls_done }
				}
				scope:location.region = {
					any_location_in_region = {
						top_owner ?= root
						raw_material = goods:pearls
					}
				}
			}
			random = {
				chance = 15
				trigger_event_non_silently = gold_fleet_tierra_firme_events.16
			}
		}
		if = {
			limit = {
				scope:expedition = {
					NOT = { has_variable = gf_havana_done }
					NOT = { has_variable = gf_passengers_done }
				}
				scope:location.region = {
					any_location_in_region = {
						top_owner ?= root
					}
				}
			}
			random = {
				chance = 10
				trigger_event_non_silently = gold_fleet_tierra_firme_events.17
			}
		}
	}

	on_arrive_to_new_location = {
		discover_area = scope:location.area
		if = {
			limit = {
				scope:expedition = {
					var:gold_fleet_cargo > 0
					NOT = { has_variable = gf_pirates_done }
				}
				OR = {
					AND = {
						scope:expedition = { NOT = { has_variable = gf_havana_done } }
						scope:location = { region = region:caribbean_region }
					}
					scope:location.area = area:bahamas_sea_area
					scope:location = location:sargasso_sea
				}
				scope:location.area = {
					any_privateer_in_area = {
						owner = { is_enemy_of = root }
					}
				}
			}
			random = {
				chance = {
					value = 15
					multiply = scope:expedition.var:gold_fleet_danger
				}
				trigger_event_non_silently = gold_fleet_shared_events.1
			}
		}
		if = {
			limit = {
				scope:expedition = {
					var:gold_fleet_cargo > 0
					NOT = { has_variable = gf_havana_done }
					NOT = { has_variable = gf_enemy_done }
				}
				scope:location = {
					region = region:caribbean_region
				}
				scope:location.region = {
					any_location_in_region = {
						any_unit_in_location = {
							owner = { is_enemy_of = root }
						}
					}
				}
			}
			random = {
				chance = {
					value = 20
					multiply = scope:expedition.var:gold_fleet_danger
				}
				trigger_event_non_silently = gold_fleet_shared_events.2
			}
		}
	}

	on_arrive_to_waypoint = {
		if = {
			limit = {
				scope:location = scope:expedition.var:primary_loot_port
				scope:expedition = { NOT = { has_variable = gf_cargo_loaded } }
			}
			scope:expedition = { set_variable = gf_cargo_loaded }
			trigger_event_non_silently = gold_fleet_tierra_firme_events.2
		}
		else_if = {
			limit = { scope:location = scope:expedition.var:cuba_rendezvous }
			if = {
				limit = { scope:expedition = { NOT = { has_variable = gf_havana_done } } }
				scope:expedition = { set_variable = gf_havana_done }
				set_variable = gold_fleet_tf_at_havana
				if = {
					limit = { OR = { has_variable = gold_fleet_ne_at_havana has_variable = gold_fleet_rendezvous } }
					scope:expedition = {
						change_variable = { name = gold_fleet_danger add = -1 }
						clamp_variable = { name = gold_fleet_danger min = 1 }
						set_variable = gf_show_together_tt
					}
					if = {
						limit = { NOT = { has_variable = gold_fleet_rendezvous } }
						set_variable = gold_fleet_rendezvous
						add_prestige = prestige_mild_bonus
					}
					scope:expedition = { set_variable = { name = gf_havana_state value = 2 } }
					trigger_event_non_silently = gold_fleet_tierra_firme_events.3
				}
				else_if = {
					limit = { has_variable = gold_fleet_ne_active }
					scope:expedition = {
						set_variable = gf_awaiting_rendezvous
						set_variable = { name = gf_havana_wait_months value = 0 }
						pause_expedition = 30
					}
					trigger_event_non_silently = gold_fleet_tierra_firme_events.8
				}
				else = {
					scope:expedition = { set_variable = { name = gf_havana_state value = 0 } }
					trigger_event_non_silently = gold_fleet_tierra_firme_events.3
				}
			}
		}
		else_if = {
			limit = { scope:location = location:florida_keys }
			if = {
				limit = { has_variable = gold_fleet_tf_at_havana }
				remove_variable = gold_fleet_tf_at_havana
			}
			if = {
				limit = {
					NOT = { has_variable = gold_fleet_ne_at_havana }
					NOT = { has_variable = gold_fleet_tf_at_havana }
				}
				remove_variable = gold_fleet_rendezvous
			}
		}
	}

	on_end = {
		if = {
			limit = {
				scope:expedition = {
					var:gold_fleet_cargo > 0
					has_variable = gf_cargo_damaged
				}
			}
			custom_tooltip = gold_fleet_shared_events.cargo_damaged_on_return_tt
		}
		if = {
			limit = { scope:expedition = { var:gold_fleet_cargo > 0 } }
			add_gold = { value = scope:expedition.var:gold_fleet_cargo }
		}
		add_prestige = prestige_mild_bonus
		remove_variable = gold_fleet_tf_active
		if = {
			limit = { has_variable = gold_fleet_tf_at_havana }
			remove_variable = gold_fleet_tf_at_havana
		}
		if = {
			limit = {
				NOT = { has_variable = gold_fleet_ne_at_havana }
				NOT = { has_variable = gold_fleet_tf_at_havana }
			}
			remove_variable = gold_fleet_rendezvous
		}
		scope:expedition = {
			remove_variable = gold_fleet_cargo
			remove_variable = gf_awaiting_rendezvous
			remove_variable = gf_havana_state
			remove_variable = gf_havana_wait_months
			remove_variable = gf_havana_done
			remove_variable = gf_cargo_loaded
			remove_variable = cuba_rendezvous
			remove_variable = primary_loot_port
			remove_variable = departure_port
			remove_variable = secondary_loot_port
			remove_variable = silver_source
			remove_variable = gem_source
			remove_variable = gold_fleet_danger
			remove_variable = gf_cargo_damaged
			remove_variable = gf_fleet_divided
			remove_variable = gf_pending_port_reunion
			remove_variable = gf_pending_sea_reunion
			remove_variable = gf_show_together_tt
			remove_variable = gf_storm_done
			remove_variable = gf_pirates_done
			remove_variable = gf_enemy_done
			remove_variable = gf_sickness_done
			remove_variable = gf_leak_done
			remove_variable = gf_emeralds_done
			remove_variable = gf_mule_done
			remove_variable = gf_pearls_done
			remove_variable = gf_passengers_done
		}
	}

	on_fail = {
		remove_variable = gold_fleet_tf_active
		if = {
			limit = { has_variable = gold_fleet_tf_at_havana }
			remove_variable = gold_fleet_tf_at_havana
		}
		if = {
			limit = {
				NOT = { has_variable = gold_fleet_ne_at_havana }
				NOT = { has_variable = gold_fleet_tf_at_havana }
			}
			remove_variable = gold_fleet_rendezvous
		}
		scope:expedition = {
			remove_variable = gold_fleet_cargo
			remove_variable = gf_awaiting_rendezvous
			remove_variable = gf_havana_state
			remove_variable = gf_havana_wait_months
			remove_variable = gf_havana_done
			remove_variable = gf_cargo_loaded
			remove_variable = cuba_rendezvous
			remove_variable = primary_loot_port
			remove_variable = departure_port
			remove_variable = secondary_loot_port
			remove_variable = silver_source
			remove_variable = gem_source
			remove_variable = gold_fleet_danger
			remove_variable = gf_cargo_damaged
			remove_variable = gf_fleet_divided
			remove_variable = gf_pending_port_reunion
			remove_variable = gf_pending_sea_reunion
			remove_variable = gf_show_together_tt
			remove_variable = gf_storm_done
			remove_variable = gf_pirates_done
			remove_variable = gf_enemy_done
			remove_variable = gf_sickness_done
			remove_variable = gf_leak_done
			remove_variable = gf_emeralds_done
			remove_variable = gf_mule_done
			remove_variable = gf_pearls_done
			remove_variable = gf_passengers_done
		}
	}

	ai_chance_to_check = {
		value = define:NAI|EXPEDITION_DEFAULT_CHANCE_TO_CHECK
		if = {
			limit = { any_character = { is_admiral = yes } }
			multiply = 2
		}
	}

	utility = {
		value = 8
	}

	leader_utility = {
		value = 1
		if = {
			limit = { is_admiral = yes }
			add = 3
		}
	}
}
