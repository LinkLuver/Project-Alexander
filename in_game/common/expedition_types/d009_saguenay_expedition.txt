saguenay_expedition = {

	unique = yes
	repeatable = no
	should_end_at_last_waypoint = no
	travel_speed = 0.75
	origin = closest_non_rural_port

	waypoints = {
		mouth_of_saint_lawrence
	}

	potential = {
		has_dlc = "d009_across_the_pillars"
		capital = { sub_continent = sub_continent:western_europe }
		has_discovered_area = area:east_coast_sea_zones_area
		NOT = { has_global_variable = saguenay_found }
		NOT = { has_variable = saguenay_expedition_attempted }
	}

	can_start = {
		gold >= 80
		has_non_rural_port = yes
		at_war = no
	}

	leader = {
		is_explorer = yes
	}

	on_start = {
		scope:expedition = {
			set_variable = { name = target_city_id value = 8 }
			set_variable = { name = leads_found value = 0 }
			set_variable = { name = saguenay_fresh_stores value = 3 }
			expedition_leader = { save_scope_as = target_leader }
			pause_expedition = 5
		}
		add_country_modifier = {
			modifier = searching_for_saguenay
			days = -1
		}
		trigger_event_non_silently = seven_cities_saguenay.10
	}

	on_monthly = {
		scope:expedition = {
			if = {
				limit = {
					has_variable = saguenay_fresh_stores
					var:saguenay_fresh_stores > 0
				}
				change_variable = { name = saguenay_fresh_stores subtract = 1 }
			}
		}
		if = {
			limit = {
				scope:expedition = {
					expedition_is_traveling = yes
					NOT = { expedition_is_paused = yes }
					has_variable = saguenay_fresh_stores
					NOT = { var:saguenay_fresh_stores > 0 }
					NOT = { has_variable = saguenay_scurvy_struck }
					NOT = { has_variable = city_revealed }
				}
			}
			scope:expedition = { set_variable = { name = saguenay_scurvy_struck value = 1 } }
			trigger_event_non_silently = seven_cities_saguenay.11
		}
	}

	on_arrive_to_new_location = {
		scope:location = { discover_location = root }
		if = {
			limit = {
				scope:location = {
					is_land = yes
					sub_continent = sub_continent:north_america
				}
			}
			random_list = {
				75 = { }
				25 = {
					random_list = {
						10 = { trigger_event_non_silently = seven_cities.10 }
						10 = { trigger_event_non_silently = seven_cities.11 }
						10 = { trigger_event_non_silently = seven_cities.12 }
						1 = { trigger_event_non_silently = seven_cities.13 }
						10 = { trigger_event_non_silently = seven_cities.14 }
						10 = { trigger_event_non_silently = seven_cities.15 }
						10 = { trigger_event_non_silently = seven_cities.16 }
						10 = {
							trigger = { scope:location = { is_port = yes } }
							trigger_event_non_silently = seven_cities.17
						}
						10 = {
							trigger = { scope:location = { is_port = yes } }
							trigger_event_non_silently = seven_cities.18
						}
					}
				}
			}
			random_list = {
				90 = { }
				4 = { trigger_event_non_silently = seven_cities.20 }
				3 = { trigger_event_non_silently = seven_cities.21 }
				3 = { trigger_event_non_silently = seven_cities.22 }
			}
		}
	}

	on_arrive_to_waypoint = {
		if = {
			limit = {
				scope:location = location:mouth_of_saint_lawrence
				scope:expedition = { NOT = { has_variable = city_revealed } }
			}
			scope:expedition = { set_variable = { name = city_revealed value = 1 } }
			if = {
				limit = { scope:expedition = { var:leads_found >= 1 } }
				scope:expedition = {
					add_new_waypoint = location:saguenay_fjord
					add_new_waypoint = root.capital
				}
				trigger_event_non_silently = seven_cities_saguenay.12
			}
			else = {
				trigger_event_non_silently = seven_cities.30
			}
		}
		else_if = {
			limit = {
				scope:location = location:saguenay_fjord
				scope:expedition = { has_variable = city_revealed }
			}
			if = {
				limit = { scope:expedition = { var:leads_found >= 3 } }
				trigger_event_non_silently = seven_cities_saguenay.3
			}
			else_if = {
				limit = { scope:expedition = { var:leads_found >= 2 } }
				trigger_event_non_silently = seven_cities_saguenay.2
			}
			else = {
				trigger_event_non_silently = seven_cities_saguenay.1
			}
		}
	}

	on_end = {
		remove_country_modifier = searching_for_saguenay
		set_global_variable = { name = saguenay_found value = 1 }
		set_variable = { name = saguenay_expedition_attempted value = 1 }

		scope:expedition = {
			if = {
				limit = {
					has_variable = saguenay_chief_character_var
					var:saguenay_chief_character_var ?= { is_alive = yes }
				}
				kill_character_silently = var:saguenay_chief_character_var
			}
		}
	}

	on_fail = {
		remove_country_modifier = searching_for_saguenay
		set_variable = { name = saguenay_expedition_attempted value = 1 }

		scope:expedition = {
			if = {
				limit = {
					has_variable = saguenay_chief_character_var
					var:saguenay_chief_character_var ?= { is_alive = yes }
				}
				kill_character_silently = var:saguenay_chief_character_var
			}
		}
	}

	utility = {
		value = 8
		if = {
			limit = { any_character = { has_trait = treasure_hunter } }
			add = 4
		}
		if = { limit = { has_global_variable = cibola_found } add = -2 }
		if = { limit = { has_global_variable = quivira_found } add = -2 }
		if = { limit = { has_global_variable = el_dorado_found } add = -2 }
		if = { limit = { has_global_variable = fountain_of_youth_found } add = -2 }
		if = { limit = { has_global_variable = city_of_caesars_found } add = -2 }
		if = { limit = { has_global_variable = sierra_de_la_plata_found } add = -2 }
		if = { limit = { has_global_variable = norumbega_found } add = -2 }
	}
}
