#P:A Imported from Invictus: neighbor_cities_svalue (common/script_values/00_city_civ.txt)
# Location scope: number of neighbouring cities (EU5: non-rural locations).
pa_imp_neighbor_cities_svalue = {
	value = 0
	every_neighbor_location = {
		limit = { NOT = { location_rank = location_rank:rural_settlement } } # Imperator has_city_status = yes
		add = 1
	}
}

#P:A Imported from Imperator: cities_in_province_svalue (common/script_values/00_event_values.txt)
# Location scope (Imperator: province): 10 per city, or city being founded, in the location's province (Imperator: state).
pa_imp_cities_in_province_svalue = {
	value = 0
	if = {
		limit = {
			province = {
				any_location_in_province = { NOT = { location_rank = location_rank:rural_settlement } }
			}
		}
		province = {
			every_location_in_province = {
				limit = {
					OR = {
						NOT = { location_rank = location_rank:rural_settlement }
						has_location_modifier = founding_city_in_progress
					}
				}
				add = 10
			}
		}
	}
	min = 0
}
