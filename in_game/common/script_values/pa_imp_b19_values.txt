# Mercenary bribing (warfare.3/4/5) and the Invictus AI mercenary helpers (common/script_values/ai_war_values.txt,
# 00_character_interactions_inv.txt). In Imperator a mercenary was a character (the leader) employed by the country,
# commanding army units made of cohorts. In EU5 a mercenary is a mercenary company (mercenary scope: monthly_cost,
# customer, every_mercenary_sub_unit; sub_units have mercenary_hire_cost / mercenary_maintenance_cost).
# These helpers take the EU5 mercenary scope wherever Imperator used the mercenary leader:
# scope:target / scope:mercenary_to_bribe / scope:mercenary_to_recruit are EU5 mercenary companies.
############################################################################################################

#P:A Imported from Invictus: army_size_merc_estimation (common/script_values/00_character_interactions_inv.txt). Reads scope:target.
pa_imp_army_size_merc_estimation = {
	value = 0
	if = {
		limit = {
			exists = scope:target
		}
		scope:target = {
			add = pa_imp_army_size_merc_estimation_calculation
		}
	}
}

#P:A Imported from Invictus: army_size_merc_estimation_calculation (common/script_values/00_character_interactions_inv.txt). Mercenary scope (Imperator: mercenary leader).

	value = 0
	every_mercenary_sub_unit = {
		add = mercenary_hire_cost #P:A TODO Approximate: Imperator summed subunits_base_recruitment_cost (fixed cost per cohort type) over the leader's units; EU5 sums the hire cost of the company's sub-units
	}
	multiply = 2
	min = 200
}

		}

			value = monthly_balance
			multiply = 6
		}
	}

#P:A Imported from Invictus: ai_max_mercenary_maintenance_without_bribed (common/script_values/ai_war_values.txt). Country scope; reads scope:mercenary_to_bribe.
pa_imp_ai_max_mercenary_maintenance_without_bribed = {
	if = {
		limit = {
			gold >= pa_imp_recruit_mercenary_cost_svalue
		}

		# Use treasury to offset temporary deficit, starting with wartime fund
		value = pa_imp_ai_treasury_for_mercenaries
		subtract = pa_imp_recruit_mercenary_cost_svalue
		divide = 42 # ai_expected_months_to_keep_mercs

		# Current treasury monthly balance
		add = monthly_balance

		# Our monthly balance will be higher without maintenance of the bribed mercs
		scope:mercenary_to_bribe = {
			add = pa_imp_mercenary_adjusted_maintenance # Imperator: unit_adjusted_maintenance of every unit of the bribed mercenary
		}

		# If navy is not at default economic policy, add delta between current policy and default
		add = pa_imp_navy_adjusted_maintenance_delta

		# Mercenary maintenance nerf for subjects, absolute goes before multiplier to soften its impact
		if = {
			limit = {
				is_subject = yes
				# All types that have joins_overlord_in_war = yes
				OR = {
					is_subject_type = feudatory
					is_subject_type = satrapy
					is_subject_type = client_state
					is_subject_type = subject_mercenary_city_state
					is_subject_type = kushite_client
					is_subject_type = indian_client
					is_subject_type = fiefdom
				}
			}
			subtract = pa_imp_ai_max_mercenary_maintenance_subject_absolute
			multiply = 0.75 # ai_max_mercenary_maintenance_subject_multiplier
		}

		min = 0
	}
	else = {
		value = 0
	}
}

#P:A Imported from Invictus: ai_treasury_for_mercenaries (common/script_values/ai_war_values.txt). Country scope.
pa_imp_ai_treasury_for_mercenaries = {
	# Imperator first read a cached local variable (ai_treasury_for_mercenaries_cached) set by its AI scripts; nothing sets it in EU5
	# Start with allocating wartime fund
	value = gold
	max = pa_imp_ai_wartime_fund_desired
	# Everything above it can be used only partially
	if = {
		limit = {
			gold > pa_imp_ai_wartime_fund_desired
		}
		# Add part of the treasury at a decent rate, but with a rather harsh cap
		add = {
			value = gold
			subtract = pa_imp_ai_wartime_fund_desired
			min = 0
			if = {
				limit = {
					pa_imp_ai_is_facing_existential_threat = yes
				}
				multiply = 0.40
			}
			else = {
				multiply = 0.20
			}
			# No more than 175% or 250% of wartime fund is allowed in sum depending on threat
			max = {
				value = pa_imp_ai_wartime_fund_desired
				if = {
					limit = {
						pa_imp_ai_is_facing_existential_threat = yes
					}
					multiply = 1.50
				}
				else = {
					multiply = 0.75
				}
			}
		}
		# Add part of the treasury at a low rate, but without a cap
		add = {
			value = gold
			subtract = pa_imp_ai_wartime_fund_desired
			min = 0
			if = {
				limit = {
					pa_imp_ai_is_facing_existential_threat = yes
				}
				multiply = 0.10
			}
			else = {
				multiply = 0.05
			}
		}
	}
}
