pa_senate_optimates_approving_seats = {
	value = var:pa_senate_optimates_seats
	multiply = var:pa_senate_optimates_approval
	divide = 100
}

pa_senate_boni_approving_seats = {
	value = var:pa_senate_boni_seats
	multiply = var:pa_senate_boni_approval
	divide = 100
}

pa_senate_populares_approving_seats = {
	value = var:pa_senate_populares_seats
	multiply = var:pa_senate_populares_approval
	divide = 100
}

pa_senate_occupied_seats = {
	value = var:pa_senate_optimates_seats
	add = var:pa_senate_boni_seats
	add = var:pa_senate_populares_seats
}

pa_senate_approval_share = {
	value = 0
	if = {
		limit = { has_variable = pa_senate_initialized pa_senate_occupied_seats > 0 }
		add = pa_senate_optimates_approving_seats
		add = pa_senate_boni_approving_seats
		add = pa_senate_populares_approving_seats
		divide = pa_senate_occupied_seats
	}
}
