#P:A Imported from Invictus: dahae_belligerency_svalue3 (common/script_values/00_mission_parthia.txt). Any scope (dhe_parthians.2).
#P:A Unhappy Dahae pops in the Imperator Parthia and Sakia regions under non-Scythian owners raise the Dahae Belligerency.
#P:A TODO Approximate: Imperator added 1 per Derbiccan pop with happiness <= 35%; EU5 adds the size (thousands) of every dahae-culture pop with pop_satisfaction <= 0.35. EU5 pop satisfaction is not Imperator pop happiness, and the regions are the pa_imp_in_region_* approximations (checked on every location in the world)
pa_imp_dahae_belligerency_svalue3 = {
	value = 0
	every_location_in_the_world = {
		limit = {
			exists = owner
			OR = {
				pa_imp_in_region_parthia_region = yes
				pa_imp_in_region_sakia_region = yes
			}
			owner = {
				NOT = {
					culture = { has_culture_group = culture_group:saka_group } # Imperator country_culture_group = scythia
				}
			}
		}
		every_pop = {
			limit = {
				culture = culture:dahae # Imperator pop_culture = derbiccan
				pop_satisfaction <= 0.35 # Imperator pop_hapiness <= 0.35
			}
			add = pop_size
		}
	}
}

			exists = owner
			OR = {

				pa_imp_in_region_sakia_region = yes
