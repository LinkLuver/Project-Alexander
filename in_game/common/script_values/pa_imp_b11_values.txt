# Character scope: how important the character is to their country.
pa_imp_importance_value = {
	value = 0
	add = { #P:A TODO Approximate: Imperator character power base; EU5 has none, (adm + dip + mil) / 10 used
		value = adm
		add = dip
		add = mil
		divide = 10
	}
	if = {
		limit = { #P:A TODO Approximate: Imperator is_previous_ruler (has ever ruled); EU5 only links the immediately preceding ruler
			exists = employer.previous_ruler
			this = employer.previous_ruler
		}
		add = 50
	}
	if = {
		limit = {
			is_ruler = yes
		}
		add = 50
	}
	if = {
		limit = {
			is_dynasty_head = yes # is_head_of_family
		}
		add = 25
	}
	if = {
		limit = { #P:A TODO Approximate: Imperator has_any_office; EU5 has no offices, cabinet membership used
			in_cabinet = yes
		}
		add = 10
	}
	if = {
		limit = {
			is_general = yes
		}
		add = 20
	}
	if = {
		limit = {
			is_admiral = yes
		}
		add = 20
	}
	if = {
		limit = { #P:A TODO Approximate: Imperator from_ruler_family (same noble family as the ruler); EU5 same dynasty as the employer's ruler
			dynasty ?= employer.ruler.dynasty
		}
		add = 50
	}
}
