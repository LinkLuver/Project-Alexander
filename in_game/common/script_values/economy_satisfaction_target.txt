# Auto minting (zero inflation) toggle. 1 = on, 0 = off. Used by economy balance tab checkbox.
auto_coin_minting_enabled = {
	scope:country = {
		if = { limit = { exists = var:auto_coin_minting }
			if = { limit = { var:auto_coin_minting > 0 } value = 1 }
			else = { value = 0 }
		} else = { value = 0 }
	}
}
