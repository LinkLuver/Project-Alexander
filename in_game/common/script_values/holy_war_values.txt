crusade_jihad_age_dampener = {
	value = 0
	if = {
		limit = { current_age = age_2_renaissance }
		value = -10
	}
	else_if = {
		limit = { current_age = age_3_discovery }
		value = -20
	}
	else_if = {
		limit = { current_age = age_4_reformation }
		value = -35
	}
	else_if = {
		limit = { current_age = age_5_absolutism }
		value = -50
	}
	else_if = {
		limit = { current_age = age_6_revolutions }
		value = -70
	}
}

holy_war_members_strength = {
	value = 0
	every_international_organization_member = {
		add = country_strength
	}
}

holy_war_target_side_strength = {
	value = 0
	international_organization_target ?= {
		add = country_strength
		every_related_country = {
			type = alliance
			add = country_strength
		}
	}
}

jihad_plunder_estimated_total = {
	value = 0
	if = {
		limit = { exists = scope:recipient.var:war_target }
		every_controlled_location = {
			limit = { owner = scope:recipient.var:war_target }
			add = estimated_loot_income
		}
	}
	multiply = {
		value = 1
		add = modifier:amount_looted_modifier
	}
}
