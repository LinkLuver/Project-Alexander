# Vote-weight components for the high_kingship_election resolution.
# Evaluated with root = the voting country and scope:recipient = the High Kingship IO.

# +1 per location the voter holds that belongs to the High Kingship.
high_kingship_voter_io_locations = {
	every_owned_location = {
		limit = { is_owned_by_international_organization = scope:recipient }
		add = 1
	}
}

# +1 per fort level across the voter's High Kingship lands.
high_kingship_voter_fort_levels = {
	every_owned_location = {
		limit = { is_owned_by_international_organization = scope:recipient }
		add = modifier:fort_level
	}
}

# +1 per subject (and their subjects, recursively) that is also a High Kingship member.
high_kingship_voter_io_subjects = {
	every_subject_or_below = {
		limit = { is_member_of_international_organization_of_type = { type = high_kingship } }
		add = 1
	}
}
