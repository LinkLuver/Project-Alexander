jap_clan_spy_objective = {
    days_between_checks = 365
    actor_trigger = { 
        exists = situation:yayoijidai
        is_member_of_international_organization = international_organization:japan_organization
    }

    recipient_trigger = {
        scope:recipient = {
            is_member_of_international_organization = international_organization:japan_organization
        }

        scope:recipient = {
            num_locations < 1
        }
        "scope:actor.relative_military_strength(scope:recipient)" > 0.5
    }

    recipient_priority = {
        add = {
            value = 50
        }
        if = {
            limit = {
                not = { scope:actor.capital = scope:recipient.capital }
            }
            divide = "scope:actor.capital.distance_to(scope:recipient.capital)"
        }
    }

    pause_trigger = { 
        always = no
    }

    cancel_trigger = {
        "scope:actor.relative_military_strength(scope:recipient)" < 0.5
    }

    country_interactions = {
       scout_enemy_holdings = yes
    }

    destroy = yes
    spy_network_target = 70
    fabricate_cb = "cb_japan_be_real"

    max_allowed = 2
}
