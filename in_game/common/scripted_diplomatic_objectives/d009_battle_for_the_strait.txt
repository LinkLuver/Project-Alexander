bfts_court_granada = {
	days_between_checks = 180

	actor_trigger = {
		OR = {
			tag = CAS
			tag = MOR
		}
		is_situation_active = situation:battle_for_the_strait
		country_exists = c:GRA
		NOT = { is_at_war_with = c:GRA }
		NOT = { is_rival_of = c:GRA }
		NOT = { wants_to_attack = c:GRA }
		c:GRA = {
			is_subject = no
			NOT = { is_rival_of = root }
			opinion = {
				target = root
				value < 150
			}
		}
	}

	recipient_trigger = {
		scope:recipient = { this = c:GRA }
	}

	recipient_priority = {
		add = 100
		if = {
			limit = {
				scope:actor = { tag = CAS }
				situation:battle_for_the_strait.var:influence_over_granada > 0
			}
			add = situation:battle_for_the_strait.var:influence_over_granada
		}
		else_if = {
			limit = {
				scope:actor = { tag = MOR }
				situation:battle_for_the_strait.var:influence_over_granada < 0
			}
			subtract = situation:battle_for_the_strait.var:influence_over_granada
		}
	}

	pause_trigger = {
		OR = {
			scope:actor = { is_at_war_with = c:GRA }
			scope:actor = { is_rival_of = c:GRA }
			scope:actor = { wants_to_attack = c:GRA }
			c:GRA ?= { is_subject = yes }
			AND = {
				scope:actor = { tag = CAS }
				situation:battle_for_the_strait.var:influence_over_granada >= 50
			}
			AND = {
				scope:actor = { tag = MOR }
				situation:battle_for_the_strait.var:influence_over_granada <= -50
			}
		}
	}

	cancel_trigger = {
		OR = {
			NOT = { country_exists = c:GRA }
			NOT = { scope:actor = { is_situation_active = situation:battle_for_the_strait } }
		}
	}

	improve_relation = yes
	improve_relation_limit = 150
	max_allowed = 1
}
