support_elective_heir = {
	days_between_checks = 90

	actor_trigger = {
		has_for_elective_heir_selection = no
		ruler_or_heir_if_regent ?= { has_dynasty = yes }
		num_relations_above_limit <= 0
	}

	recipient_trigger = {
		scope:recipient = {
			has_for_elective_heir_selection = yes
			has_regent = yes
			has_heir = no
			NOT = { is_rival_of = scope:actor }
			NOT = { is_at_war_with = scope:actor }
		}
	}

	recipient_priority = 40

	cancel_trigger = {
		scope:recipient = {
			OR = {
				has_regent = no
				has_heir = yes
			}
		}
	}

	country_relations = {
		support_heir = yes
	}

	max_allowed = 1
}
