cas_court_aragon = {
	days_between_checks = 360

	actor_trigger = {
		tag = CAS
		country_exists = c:ARA
		NOT = { is_at_war_with = c:ARA }
		NOT = { is_rival_of = c:ARA }
		NOT = { wants_to_attack = c:ARA }
		c:ARA = {
			is_subject = no
			NOT = { is_rival_of = root }
			opinion = {
				target = root
				value < 100
			}
		}
	}

	recipient_trigger = {
		scope:recipient = { this = c:ARA }
	}

	recipient_priority = 50

	pause_trigger = {
		OR = {
			scope:actor = { is_at_war_with = c:ARA }
			scope:actor = { is_rival_of = c:ARA }
			scope:actor = { wants_to_attack = c:ARA }
			c:ARA ?= { is_subject = yes }
		}
	}

	cancel_trigger = {
		NOT = { country_exists = c:ARA }
	}

	improve_relation = yes
	improve_relation_limit = 100
	max_allowed = 1
}
