join_foreign_order_of_chivalry = {
	is_consort_action = no
	on_own_nation = yes
	message = yes
	context_menu_click_mode = click # Disable confirm

	potential = {
		scope:actor = {
			modifier:is_pope = no
			religion.group = religion_group:christian
			any_country_with_chivalric_order = {
				has_different_order_of_chivalry = scope:actor
			}
		}
	}

	ai_tick = daily
	ai_tick_frequency = 180

	allow = {
		scope:actor = {
			custom_tooltip = {
				text = order_of_chivalry_country_cooldown
				not = { has_variable = order_of_chivalry_country_cooldown }
			}
		}
	}

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
			is_alive = yes
			owner = scope:actor
			is_artist = no
		}
		enabled = {
			is_in_chivalric_order = no
			religion.group = religion_group:christian
			OR = {
				has_estate = estate_type:nobles_estate
				has_estate = estate_type:crown_estate
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			every_country_with_chivalric_order = {
				add_to_list = source
			}
		}
		target_flag = target
		name = "join_foreign_order_of_chivalry_select_title"
		none_available_msg_key = "join_foreign_order_of_chivalry_no_countries"
		column = { data = name }
		visible = {
			NOT = { root = scope:actor }
			scope:recipient = {
				custom_tooltip = {
					text = character_can_join_order_of_chivalry_tt
					is_adult = yes
					character_is_eligible_for_active_order = root
				}
			}
			not = { is_at_war_with = scope:actor }
			has_different_order_of_chivalry = scope:actor
			within_diplomatic_range = scope:actor
		}
	}
	ai_will_do = {
		add = 0
		# Never join a foreign order if we have our own
		if = {
			limit = {
				scope:actor = { has_chivalric_order = yes }
			}
			add = -1000
		}
		# Only worthwhile if the target order is diplomatically desirable
		if = {
			limit = {
				scope:target = { has_desirable_order_of_chivalry_for_target_country = { country = scope:actor } }
			}
			add = 10
		}
		# Strongly prefer rulers and heirs for the diplomatic prestige
		if = {
			limit = {
				scope:recipient = {
					OR = {
						is_ruler = yes
						is_heir = yes
					}
				}
			}
			add = 15
		}
		# Moderately prefer high-skill characters
		if = {
			limit = {
				scope:recipient = {
					OR = {
						mil >= 5
						adm >= 5
						dip >= 5
					}
				}
			}
			add = 5
		}
	}
	effect = {
		hidden_effect = {
			scope:actor = {
				set_variable = { name = order_of_chivalry_country_cooldown value = yes years = 10 }
			}
		}
		scope:recipient = {
			join_active_chivalric_order_of = scope:target
		}
		if = {
			limit = { exists = scope:target }
			scope:target = {
				add_opinion = {
					target = scope:actor
					modifier = opinion_joined_our_order_of_chivalry
				}
			}
		}
	}
}

invite_to_order_of_chivalry = {
	is_consort_action = no
	on_own_nation = yes
	message = yes

	potential = {
		scope:actor = {
			has_chivalric_order = yes
		}
	}

	allow = {
		scope:actor = {

			custom_tooltip = {
				text = order_of_chivalry_country_cooldown
				not = { has_variable = order_of_chivalry_country_cooldown }
			}
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 90

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
			is_alive = yes
			custom_tooltip = {
				text = character_can_join_order_of_chivalry_tt
				is_adult = yes
				character_is_eligible_for_active_order = scope:actor
			}
		}
		enabled = {
			is_in_chivalric_order = no
			religion.group = religion_group:christian
			owner = { within_diplomatic_range = scope:actor }
			OR = {
				has_estate = estate_type:nobles_estate
				has_estate = estate_type:crown_estate
			}
		}
	}

	ai_will_do = {
		add = 0
		# Strongly prefer rulers and heirs
		if = {
			limit = {
				scope:recipient = {
					OR = {
						is_ruler = yes
						is_heir = yes
					}
				}
			}
			add = 25
		}
		# Moderately prefer high-skill characters
		if = {
			limit = {
				scope:recipient = {
					OR = {
						mil >= 5
						adm >= 5
						dip >= 5
					}
				}
			}
			add = 10
		}
		# Base weight for other eligible nobles — low to avoid spamming
		if = {
			limit = {
				scope:recipient = {
					NOT = { is_ruler = yes }
					NOT = { is_heir = yes }
				}
			}
			add = 2
		}
	}
	effect = {
		hidden_effect = {
			scope:actor = {
				set_variable = { name = order_of_chivalry_country_cooldown value = yes years = 10 }
			}
		}
		scope:recipient = {
			join_active_chivalric_order_of = scope:actor
		}
	}
}
