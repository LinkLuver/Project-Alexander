seppuku = { 

	on_own_nation = yes
	
	message = yes
	
	potential = {
		scope:actor = {
			religion = {
				has_purity = yes
			}
		}
	}

	ai_tick = daily
	ai_tick_frequency = 180

	allow = {
		scope:actor = { has_ruler = yes }
	}
	
	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
		}
		enabled = {
			is_ruler = yes
		}
	}

	effect = {
		kill_character = {
			target = scope:recipient
			reason = seppuku
		}
		scope:actor = {
			add_purity = purity_extreme_bonus
			add_stability = stability_ultimate_penalty
		}
	}

	ai_will_do = {
		add = {
			value = 20
			subtract = scope:actor.purity
		}
		if = {
			limit = {
				scope:actor = { stability < 0 }
			}
			subtract = 25
		}
		if = {
			limit = {
				scope:actor = { legitimacy < 50 }
			}
			subtract = 10
		}
		if = {
			limit = {
				scope:actor.ruler = { age_in_years < 25 }
			}
			subtract = 25
		}
	}
	
}
