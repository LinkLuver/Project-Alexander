pa_imprison_character = {
	on_own_nation = yes
	message = yes
	ai_tick = never
	potential = {
		scope:actor = { tag = ROM government_type = government_type:republic }
	}
	allow = { scope:actor = { always = yes } }
	select_trigger = {
		looking_for_a = character
		source = world
		target_flag = recipient
		name = "choose_character"
		column = { data = name }
		visible = { is_alive = yes }
		enabled = { NOT = { this = scope:actor.ruler } NOT = { has_character_modifier = imprisoned } }
	}
	effect = {
		scope:recipient = { imprison_character_in_country_effect = { target = scope:actor years = -1 } }
		scope:actor = {
			if = { limit = { NOT = { has_variable = pa_senate_tyranny } } set_variable = { name = pa_senate_tyranny value = 0 } }
			change_variable = { name = pa_senate_tyranny add = 5 max = 100 }
		}
	}
	ai_will_do = { add = 0 }
}
