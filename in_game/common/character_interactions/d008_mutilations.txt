castrate_character = {
	message = yes
	on_own_nation = yes

	potential = {
		has_dlc = "d008_fate_of_the_phoenix"
		scope:actor = {
			succession_law = heir_selection:byzantine_succession
		}
	}

	allow = {
		scope:actor = {
			has_ruler = yes
		}
	}

	ai_tick = never

	price = price:castrate_character_price

	select_trigger = {
		looking_for_a = character
		source = actor
		target_flag = recipient
		name = "choose_character"
		column = {
			data = name
		}
		visible = {
			is_alive = yes
			is_female = no
			NOR = {
				has_trait = castrated
				has_trait = eunuch
			}
		}
		enabled = {
			is_adult = yes
			is_ruler = no
			is_regent = no
			NOT = { is_spouse_of = scope:actor.ruler }
			NOT = { modifier:is_head_of_cabinet = yes }
		}
	}

	effect = {
		scope:actor = {
			add_prestige = 5
			if = {
				limit = { scope:recipient.estate_type = estate_type:crown_estate }
				add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_extreme_penalty }
			}
			else = {
				add_estate_satisfaction = { type = scope:recipient.estate_type value = estate_satisfaction_extreme_penalty }
			}
		}
		scope:recipient = {
			add_trait = trait:castrated
		}
		if = {
			limit = { scope:recipient = { has_variable = scope_pretender_var } }

			custom_tooltip = blind_pretender_removed_tt

			scope:actor = {
				ordered_character = {
					limit = {
						NOT = { this = scope:recipient }
						owner = scope:actor
						is_alive = yes
						is_adult = yes
						is_female = no
						is_ruler = no
						is_regent = no
						has_character_modifier = plotting_rebellion_modifier
						NOR = {
							has_trait = blind
							has_trait = castrated
							has_trait = eunuch
						}
					}
					order_by = total_abilities
					max = 1
					check_range_bounds = no
					save_scope_as = new_pretender_scope
					set_variable = {
						name = scope_pretender_var
						value = yes
					}
					custom_tooltip = blind_new_pretender_emerges_tt

					scope:actor = {
						clear_gui_variable = { type = pretender_rebel_character }
						create_gui_variable = {
							type = pretender_rebel_character
							target = prev
						}

						every_character = {
							limit = {
								NOT = { this = scope:new_pretender_scope }
								owner = scope:actor
								is_alive = yes
								is_adult = yes
								is_female = no
								is_ruler = no
								is_regent = no
								NOT = { has_character_modifier = plotting_rebellion_modifier }
								OR = {
									is_close_relative = scope:new_pretender_scope
									is_spouse_of = scope:new_pretender_scope
								}
							}
							change_character_allegiance = scope:actor.var:succession_crisis_pretender_variable
							hidden_effect = {
								add_character_modifier = {
									modifier = plotting_rebellion_modifier
									years = -1
									mode = extend
								}
							}
						}
					}
				}
			}

			hidden_effect = {
				scope:recipient = {
					remove_variable = scope_pretender_var
					remove_character_allegiance = yes
					if = {
						limit = { has_character_modifier = plotting_rebellion_modifier }
						remove_character_modifier = plotting_rebellion_modifier
					}
				}
			}
		}
	}

	ai_will_do = {}
}
