nobles_estate = {
	color = pop_nobles
	tax_per_pop = 100
	rival = -0.01
	alliance = 0.01

	

	characters_have_dynasty = always # new characters always spawn with a dynasty
	can_generate_mercenary_leaders = yes
	bank = yes 	#can and will loan money

	revolt_court_language = court_language
	

	#multiplied with (satisfaction - LOW_SATISFACTION_THRESHOLD)
	satisfaction = {
		monthly_prestige = 0.2
		monthly_legitimacy = 0.2
	}
	
	#multiplied with (relative_power - LOW_POWER_THRESHOLD) if > LOW_POWER_THRESHOLD
	high_power = {
		nobles_estate_max_tax = -1.0
		levy_combat_efficiency_modifier = 1.0
		fort_maintenance_efficiency = 1.0
		monthly_legitimacy = high_power_estate_government_power_penalty
		monthly_republican_tradition = high_power_estate_government_power_penalty
		monthly_devotion = high_power_estate_government_power_penalty
		monthly_horde_unity = high_power_estate_government_power_penalty
		monthly_tribal_cohesion = high_power_estate_government_power_penalty
	}
	
	
	#multiplied with -(relative_power - LOW_POWER_THRESHOLD) if < LOW_POWER_THRESHOLD
	low_power = {
		nobles_estate_max_tax = 0.5
	}
	
	opinion = {
		if = {
			limit = { country_exists = scope:target }
			add = {
				desc = "ESTATE_OPINION_BASE"
				value = "opinion(scope:target)"
				multiply = 0.05
			}

			if = {
				limit = {
					scope:target = {
						has_discriminated_culture = root.culture
					}
				}
				add = {
					desc = "ESTATE_OPINION_TARGET_DISCRIMINATES_OUR_CULTURE"
					value = -50
				}
			}

			add = {
				desc = "ESTATE_OPINION_PRESTIGE"
				value = prestige
				subtract = scope:target.prestige
				multiply = 0.05
			}
		
			add = {
				desc = "ESTATE_OPINION_POWER_PROJECTION"
				value = power_projection
				subtract = scope:target.power_projection
				multiply = 0.1
			}
		
			add = {
				desc = "aristocracy_vs_plutocracy"
				value = {
					value = scope:target.societal_value:aristocracy_vs_plutocracy
					subtract = societal_value:aristocracy_vs_plutocracy
					multiply = -1
					divide = 10
				}
			}
		
			add = {
				desc = "serfdom_vs_free_subjects"
				value = {
					value = scope:target.societal_value:serfdom_vs_free_subjects
					subtract = societal_value:serfdom_vs_free_subjects
					multiply = -1
					divide = 10
				}
			}

			if = {
			
				limit = {
					government_type = government_type:monarchy
					scope:target = { government_type = government_type:monarchy }
				}	
				add = {
					desc = "ESTATE_OPINION_BOTH_MONARCHY"
					value = 5
				}
			}
		
			add = {
				desc = "ESTATE_OPINION_TARGET_N_ESTATE_POWER"
				value = "scope:target.estate_power(estate_type:nobles_estate)"
				multiply = 5
			}
		
			add = {
				desc = "ESTATE_OPINION_TARGET_DIPLOMATIC_REPUTATION"
				value = scope:target.modifier:diplomatic_reputation
				multiply = 0.25
			}
		
			if = {
				limit = {
					culture.cultural_influence < scope:target.culture.cultural_influence
				}	
			
				add = {
					desc = "ESTATE_OPINION_HIGHER_CULTURE_INF"
					value = 5
				}
			}


			if = {
				limit = {
					expected_army_size < scope:target.expected_army_size
				}	
				add = {
					desc = "ESTATE_OPINION_HIGHER_MIL_POWER"
					value = -10
				}
			}


			if = {
				limit = {
					country_art_quality > scope:target.country_art_quality
				}	


				add = {
					desc = "ESTATE_OPINION_LOWER_ART"
					value = -10
				}
			}

		
		
			if = {
				limit = {
					is_neighbor_of = scope:target
				}	
			
				add = {
					desc = "ESTATE_OPINION_NEIGHBOR"
					value = -5
				}
			}

			if = {
				limit = {
					court_language = scope:target.court_language
				}
				add = {
					desc = "ESTATE_OPINION_SAME_COURT_LANGUAGE"
					value = 5
				}
			}

			if = {
				limit = {
					language = scope:target.court_language
				}
				add = {
					desc = "ESTATE_OPINION_COURT_LANGUAGE_IS_OUR_COMMON_LANGUAGE"
					value = 5
				}
			}
		
		
			if = {
				limit = {
					is_subject_of = scope:target
				}
				if = {
					limit = { is_disloyal_subject = yes }
					add = {
						desc = "ESTATE_OPINION_SUBJECT_DISLOYAL"
						value = -10
					}
				}
				else = {
					add = {
						desc = "ESTATE_OPINION_SUBJECT_LOYAL"
						value = 10
					}
				}
			}
			else_if = {
				limit = { any_subject = { this = scope:target } }
				if = {
					limit = {
						scope:target = { is_disloyal_subject = yes }
					}
					add = {
						desc = "ESTATE_OPINION_SUBJECT_DISLOYAL"
						value = -10
					}
				}
				else = {
					add = {
						desc = "ESTATE_OPINION_SUBJECT_LOYAL"
						value = 10
					}
				}
			}

			if = {
				limit = {
					"estate(estate_type:nobles_estate)" = {
						dominant_culture ?= { this = scope:target.culture }
					}
				}
				add = { desc = "ESTATE_OPINION_SAME_ESTATE_CULTURE" value = 20 }
			}
			else_if = {
				limit = {
					"estate(estate_type:nobles_estate)" = {
						scope:target = { has_accepted_culture = prev.dominant_culture }
					}
				}
				add = { desc = "ESTATE_OPINION_ACCEPTED_ESTATE_CULTURE" value = 10 }
			}
			else_if = {
				limit = {
					"estate(estate_type:nobles_estate)" = {

					}
				}
				add = { desc = "ESTATE_OPINION_TOLERATED_ESTATE_CULTURE" value = 5 }
			}

			if = {
				limit = {
					"estate(estate_type:nobles_estate)" = {
						dominant_religion ?= { this = scope:target.religion }
					}
				}
				add = { desc = "ESTATE_OPINION_SAME_ESTATE_RELIGION" value = 20 }
			}

		}
	}

	private_army_per_pop = 0.1
