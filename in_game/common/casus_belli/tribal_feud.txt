#scope:target is the target 

# Tribal vs Tribal Feud
cb_tribal_feud = {
	create_visible = {
		government_type = government_type:tribe
		scope:target = {
			government_type = government_type:tribe
		}
	}

	declare_enabled = {
		government_type = government_type:tribe
		scope:target = {
			government_type = government_type:tribe
		}
		NOT = {
			is_member_of_international_organization_of_type = { type = high_kingship }
		}
	}

	create_enabled = {
		scope:target = {
			government_type = government_type:tribe
		}
		is_neighbor_of = scope:target
		NOT = {
			is_member_of_international_organization_of_type = { type = high_kingship }
		}
	}

	province = {
		any_location_in_province = {
			is_neighbor_of = scope:actor
		}
	}

	war_goal_type = take_province_tribal_feud
}
