	ai_will_do = {

			limit = {

					has_variable = influence_over_granada 
					var:influence_over_granada < 0
				}
			}
			add = {
				desc     = "BFTS_INFLUENCE_AXIS_POSITION"
				value    = situation:battle_for_the_strait.var:influence_over_granada

				max      = 50
			}
		}
		else_if = {
			limit = {
				situation:battle_for_the_strait = { 
					has_variable = influence_over_granada 
					var:influence_over_granada > 0
				}
			}
			add = {
				desc     = "BFTS_INFLUENCE_AXIS_POSITION"
