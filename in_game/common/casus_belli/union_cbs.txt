cb_union_independence = {
	create_visible = {
        exists = union
        union ?= {
			has_active_senior_partner = yes
        }
        junior_union_with = { target = scope:target }
	}

	declare_enabled = {
        exists = union
        union ?= {
			has_active_senior_partner = yes
        }
        junior_union_with = { target = scope:target }
	}

	create_enabled = {
		NOT = { has_truce_with = scope:target }
	}

	war_goal_type = defend_capital
}

cb_union_war_for_seniority = {
	create_visible = {
        exists = union
        union ?= {
			has_active_senior_partner = yes
        }
        junior_union_with = { target = scope:target }
	}

	declare_enabled = {
        exists = union
        union ?= {
			has_active_senior_partner = yes
        }
        junior_union_with = { target = scope:target }
	}

	create_enabled = {
        country_economical_base >= {
			value = scope:target.country_economical_base
			divide = 2
		}
		NOT = { has_truce_with = scope:target }
	}

	war_goal_type = take_capital
}
