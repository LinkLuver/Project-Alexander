# Restore Roman Borders
cb_restore_roman_borders = {
	create_visible = {
		has_unlocked_casus_belli_trigger = { type = cb_restore_roman_borders }
		scope:target = {
			is_subject = no
			owns_or_has_subject_in = scripted_geography:roman_borders_geography
		}
	}
	declare_enabled = {
		scope:target = {
			is_subject = no
			owns_or_has_subject_in = scripted_geography:roman_borders_geography
		}
	}
	create_enabled = {
		OR = {
			is_neighbor_of_country_or_across_one_seazone = scope:target
			scope:target = {
				any_subject = {
					is_neighbor_of_country_or_across_one_seazone = root
				}
			}
			any_subject = {
				is_neighbor_of_country_or_across_one_seazone = scope:target
			}
		}
	}
	speed = 10
	war_goal_type = superiority_restore_roman_borders
}
