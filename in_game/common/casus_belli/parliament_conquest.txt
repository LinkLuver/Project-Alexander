#root is country
#scope:target is the target 


cb_parliament_conquer_province = {
	create_visible = {
		has_casus_belli_of_type_on = {
			target = scope:target
			type = casus_belli:cb_parliament_conquer_province
		}
	}
	create_enabled = {
		scope:target = {
			any_owned_location = {
				is_core_of = root
			}
		}
	}
	declare_enabled = {
		scope:target_province ?= {
			any_location_in_province = {
				owner = scope:target
			}
		}
	}

	#scopes are actor/recipient for countries, while root = province checked.
	province = {
		any_location_in_province = {
			is_core_of = scope:actor
		}
	}

	war_goal_type = parliament_conquer_province
}
