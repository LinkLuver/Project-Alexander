		}
	}
}

# Location scope: despite its name, true when the location does NOT produce a mined good (stone, marble, metals, gems ...).
#P:A TODO Approximate: EU5 mined goods with no Imperator counterpart (silver, tin, lead, coal, saltpeter, alum ...) are not listed, so a location producing them counts as "not mined" here
pa_imp_has_unminable_trade_good_trigger = {
	NOR = {
		raw_material = goods:marble
		raw_material = goods:stone
		raw_material = goods:goods_gold #P:A TODO Approximate: Imperator trade good precious_metals -> EU5 gold
		raw_material = goods:copper #P:A TODO Approximate: Imperator trade good base_metals -> EU5 copper
		raw_material = goods:iron
		raw_material = goods:mercury #P:A TODO Approximate: Imperator trade good cinnabar -> EU5 mercury
		raw_material = goods:gems # Imperator gems; also lapis and jade (approximated to gems)
	}
}
