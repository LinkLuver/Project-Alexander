# Imperator scripted triggers for batch b02 of the Invictus/Indomita event import (2026-09-30).
# Imperator republic parties have no EU5 equivalent. They are approximated by the character's estate, the same
# mapping everywhere in this batch (change it here if a better one is chosen):
#   populares / democratic party   -> burghers_estate  (the citizen commons)
#   optimates / oligarchic party   -> nobles_estate    (the wealthy elite families)
#   boni / traditionalist party    -> clergy_estate    (the conservative, religious faction)

	}
}

#P:A Imported from Invictus: character_populares_or_oligarchic_trigger (common/scripted_triggers)
# Character scope. Imperator: populares party (Rome) or oligarchic party (other republics).
pa_imp_character_populares_or_oligarchic_trigger = {
	OR = { #P:A TODO Approximate: Imperator populares (Rome) or oligarchic party; EU5 has no parties, estates stand in
		AND = {
			employer ?= { tag = ROM }
			has_estate = estate_type:burghers_estate
		}
		AND = {
			NOT = { employer ?= { tag = ROM } }
			has_estate = estate_type:nobles_estate
		}
	}
}

#P:A Imported from Invictus: has_non_empty_global_variable_list (common/scripted_triggers/ai_cities_triggers.txt)
# Any scope. Parameter name = the global variable list to test.
pa_imp_has_non_empty_global_variable_list = {
	has_global_variable_list = $name$
	global_variable_list_size = {
		name = $name$
		value > 0
	}
}
