	OR = {
		# is_monotheist_religion = no - in Invictus only buddhism and jainism carry is_monotheistic = yes
		NOR = { #P:A TODO Approximate: Imperator is_monotheist_religion = no; the mod splits buddhism into schools, so all of them are excluded along with jain
			religion = religion:buddhism
			religion = religion:mahayana
			religion = religion:theravada
			religion = religion:sammitiya
			religion = religion:tibetan_buddhism

		}
		religion = religion:zoroastrian # zoroaster
		pa_imp_is_spiritualist_religion = yes
		pa_imp_is_zoroastrian_religion = yes
	}
}
