is_hafsid_governor_trigger = {
	OR = {
		has_or_had_tag = TUN
		has_or_had_tag = BIJ
		is_subject_type = familial_governor
		AND = {
			has_variable = was_familial_governor
			ruler_or_heir_if_regent ?= {
				dynasty = dynasty:hafsid_dynasty
			}
		}
	}
}

can_be_targetted_by_tun_tax_raid = {
	location_rank ?= location_rank:rural_settlement
	NOT = { has_location_modifier = tun_tax_raid_modifier }
}

can_become_familial_governor_trigger = {
	is_ruler = no
	age_in_years > 15
	is_female = no
}

can_potentially_become_familial_governor_trigger = {
	is_alive = yes
	save_temporary_scope_as = evaluated_scope
	custom_tooltip = {
		text = tim_is_related_to_ruler_tt
		$country$.ruler ?= {
			dynasty ?= {
				scope:evaluated_scope.dynasty ?= this
			}
		}
	}
	is_loyal = yes
}

can_select_location_for_familial_governor = {
	location_rank != location_rank:town
	is_capital = no
}
