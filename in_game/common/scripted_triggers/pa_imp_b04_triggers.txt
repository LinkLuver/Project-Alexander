#P:A Imported from Imperator: rival_brothers_trigger (common/scripted_triggers/00_character_events.txt). Country scope.
# Two adult sons of the ruler (18-30, healthy) who are brothers.
pa_imp_rival_brothers_trigger = {
	government_type = government_type:monarchy
	exists = capital
	exists = ruler
	any_character = {
		is_eligible_heir = root #P:A TODO Approximate: Imperator any_successor (the line of succession); EU5 characters eligible as heir
		is_child_of = root.ruler
		# from_ruler_family = yes #P:A TODO Approximate: Imperator from_ruler_family (member of the ruling family); EU5 relies on is_child_of the ruler
		is_female = no
		age_in_years >= 18
		age_in_years <= 30
		pa_imp_has_any_illness_trigger = no
		employer = root
		save_temporary_scope_as = rival_brother_1_temp_t
		OR = {
			AND = {
				# can_get_rivals = yes
				root = {
					any_character = { # Imperator any_sibling
						is_sibling_of = scope:rival_brother_1_temp_t
						is_eligible_heir = root #P:A TODO Approximate: Imperator is_successor; EU5 is_eligible_heir
						is_child_of = root.ruler
						is_female = no
						age_in_years >= 18
						age_in_years <= 30
						# can_get_rivals = yes
						pa_imp_has_any_illness_trigger = no
						employer = root
						# NOT = { is_rival = scope:rival_brother_1_temp_t }
					}
				}
			}
			# AND = { num_of_rivals >= 1 any_rival = { is_successor = yes is_sibling_of = scope:rival_brother_1_temp_t ... } }
			#P:A TODO No EU5 equivalent: rival slots and existing rivalries between the brothers (EU5 characters have no rivals)
		}
	}
}
