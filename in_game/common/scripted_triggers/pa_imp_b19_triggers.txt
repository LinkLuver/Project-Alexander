#P:A Imported from Invictus: ai_should_counter_bribe_mercenary (common/scripted_triggers/ai_war_triggers.txt). Country scope; reads scope:mercenary_to_bribe.
pa_imp_ai_should_counter_bribe_mercenary = {
	# Counter bribe only armies, not navies
	scope:mercenary_to_bribe = {
		any_mercenary_sub_unit = { always = yes } #P:A TODO Approximate: Imperator "is_general = yes" (army mercenaries only); EU5 checks the company has sub-units
	}

	# Check if we have anywhere near enough money in the treasury first for performance optimization
	gold >= pa_imp_ai_treasury_for_counter_bribing_base

	# [HIGH PRIO] There's simply no comparable mercenary we can recruit instead of the one we are going to lose
	trigger_if = {
		limit = {
			save_temporary_scope_as = counter_bribing_country
			NOT = {
				any_mercenary = { # Imperator: any character of the mercenary holder country c:MER
					# Newly recruited mercenary must be at least close in power (= maintenance) to the bribed one.
					pa_imp_mercenary_strength_maintenance >= scope:mercenary_to_bribe.pa_imp_mercenary_adjusted_maintenance_75_percent
					pa_imp_ai_can_recruit_mercenary_without_bribed = yes
				}
			}
		}
		gold >= pa_imp_ai_treasury_for_counter_bribing_high_prio
	}

	# [MEDIUM PRIO] The war would be affected
	trigger_else_if = {
		limit = {
			at_war = yes #P:A TODO Approximate: Imperator [HIGH PRIO] when the mercenary army was besieging a fort or had warring armies in or next to its territory, and [MEDIUM PRIO] when it stood in or next to a war participant's territory, each with morale/strength thresholds; EU5 mercenary companies expose no unit position or morale, so being at war stands in for both
		}
		gold >= pa_imp_ai_treasury_for_counter_bribing_medium_prio
	}

	# [LOW PRIO] The immediate impact would be relatively low, counter bribe only if rich enough to just do it anyways
	trigger_else = {
		#P:A TODO No EU5 equivalent: Imperator also required the mercenary units at 95% morale and strength (not exiled), else never counter-bribed
		gold >= pa_imp_ai_treasury_for_counter_bribing_low_prio
	}
}

#P:A Imported from Invictus: ai_can_recruit_mercenary_without_bribed (common/scripted_triggers/ai_war_triggers.txt). Mercenary scope; reads scope:counter_bribing_country, scope:mercenary_to_bribe.
pa_imp_ai_can_recruit_mercenary_without_bribed = {
	pa_imp_mercenary_can_be_recruited_by_ai = yes

	# Check if mercenaries are in diplomatic range and are close enough so the AI can properly use them
	save_temporary_scope_as = mercenary_to_recruit
	scope:counter_bribing_country = {
		save_temporary_scope_as = potential_employer
	}
	mercenary_home ?= { #P:A TODO Approximate: Imperator checked the leader's current location; EU5 mercenary companies only expose their home location
		pa_imp_ai_can_recruit_mercenary_in_territory = yes
	}

	# Not exceeding the maintenance the country can cover
	pa_imp_mercenary_adjusted_maintenance <= scope:counter_bribing_country.pa_imp_ai_free_mercenary_maintenance_without_bribed # Imperator: no single unit of the mercenary above it
}

#P:A Imported from Invictus: mercenary_can_be_recruited_by_ai (common/scripted_triggers/ai_war_triggers.txt). Mercenary scope (Imperator: mercenary leader).
pa_imp_mercenary_can_be_recruited_by_ai = {
	# Basic checks for mercenary to be eligible
	has_customer = no # Imperator: is_mercenary = yes, employer = c:MER (the holder country of unhired mercenaries)
	# Only armies for now (Imperator: is_general = yes, in_command = yes, with a unit of size > 0)
	any_mercenary_sub_unit = { always = yes }
	#P:A TODO No EU5 equivalent: Imperator also skipped mercenaries whose units were in combat, moving or below full strength (EU5 mercenary companies expose no unit state)
}

#P:A Imported from Invictus: is_mercenary_unit (common/scripted_triggers/ai_war_triggers.txt). Unit scope.
pa_imp_is_mercenary_unit = {
	any_sub_unit = { #P:A TODO Approximate: Imperator checked that the unit's commander is a mercenary; EU5 checks for mercenary sub-units
		is_mercenary = yes
	}
}

#P:A Imported from Invictus: unit_pays_maintenance (common/scripted_triggers/ai_war_triggers.txt). Unit scope.
pa_imp_unit_pays_maintenance = {
	trigger_if = {
		limit = {
			is_army = yes
		}
		OR = {
			pa_imp_is_legion_unit = yes
			pa_imp_is_mercenary_unit = yes
		}
	}
	trigger_else_if = {
		limit = {
			is_navy = yes
		}
		always = yes
	}
	trigger_else = {
		always = no
	}
}

#P:A Imported from Invictus: ai_is_facing_existential_threat (common/scripted_triggers/ai_war_triggers.txt). Country scope.
pa_imp_ai_is_facing_existential_threat = {
	at_war = yes
	save_temporary_scope_as = country_at_war
	OR = {
		# Civil wars are no joke
		in_civil_war = yes
		# Is at war with player(s) with somewhat comparable combined population
		pa_imp_country_population_of_enemy_players >= pa_imp_country_population_40_percent
		# There's a war where...
		any_current_war = {
			OR = {
				# ...enemy side has much larger amount of territories than ally side
				pa_imp_num_of_territories_of_enemy_side >= pa_imp_num_of_territories_of_ally_side_plus_three_fifths
				# ...enemy side has much larger amount of population than ally side
				pa_imp_country_population_of_enemy_side >= pa_imp_country_population_of_ally_side_plus_three_fifths
			}
		}
		# A lot of our territories are occupied
		any_owned_location = {
			percent >= 0.50
			controller = {
				is_at_war_with = scope:country_at_war
			}
		}
	}
}

#P:A Imported from Invictus: ai_can_recruit_mercenary_in_territory (common/scripted_triggers/ai_war_triggers.txt). Location scope; reads scope:potential_employer.
pa_imp_ai_can_recruit_mercenary_in_territory = {
	# Prevent recruiting mercs in the enemy territory and spamming mercs on your own but occupied territories
	trigger_if = {
		limit = {
			exists = controller
		}
		NOT = {
			controller = {
				is_at_war_with = scope:potential_employer
			}
		}
	}
	trigger_else_if = {
		limit = {
			exists = owner
		}
		NOT = {
			owner = {
				is_at_war_with = scope:potential_employer
			}
		}
	}
	# Recruit mercs from unsettled territories only if they are adjacent to territories of the recruiting country
	trigger_else = {
		any_neighbor_location = {
			trigger_if = {
				limit = {
					exists = controller
				}
				controller = scope:potential_employer
			}
			trigger_else_if = {
				limit = {
					exists = owner
				}
				owner = scope:potential_employer
			}
			trigger_else = {
				always = no
			}
		}
	}
	#P:A TODO No EU5 equivalent: Imperator also required no enemy army in the territory, a land path out of the Rhodope enclave, and (for owned-by-others territories) a governorship capital within diplomatic range and presence in the same or a neighbouring region (EU5 has no governorships; mercenary range is engine-side)
	# Normally mercs are available by diplomatic range
	OR = {
		AND = {
			exists = owner
			owner = scope:potential_employer
		}
		NOT = {
			exists = owner
		}
		owner = {
			within_diplomatic_range = scope:potential_employer #P:A TODO Approximate: Imperator in_diplomatic_range plus a distance check from the employer's governorship capitals
		}
	}
	# If it's island territory, recruiting country must be owner
	trigger_if = {
		limit = {
			pa_imp_is_island_territory = yes
		}
		owner ?= scope:potential_employer
	}
	# If it's island area, recruiting country must own non-island territory here
	trigger_else_if = {
		limit = {
			area = {
				pa_imp_is_island_area = yes
			}
		}
		area = {
			any_location_in_area = {
				pa_imp_is_island_territory = no
				owner ?= scope:potential_employer
			}
		}
	}
	# It's on the land, now check if owner is present on non-island territories
	trigger_else = {
		scope:potential_employer = {
			any_owned_location = {
				pa_imp_is_island_territory = no
				area = {
					pa_imp_is_island_area = no
				}
			}
		}
	}
}

#P:A Imported from Invictus: is_island_area (common/scripted_triggers/ai_war_triggers.txt). Area scope.
pa_imp_is_island_area = {
	OR = {
		# Corsica
		this = location:sartene.area # Imperator p:3508
		# Sardinia
		this = location:terranova_pausania.area #P:A TODO Approximate location: Imperator Longones (p:3506) mapped to terranova_pausania
		this = location:glarus.area # Imperator p:3483 (Nura)
		# No neighbouring land areas
		NOT = {
			any_neighbor_area = {
				is_area_sea = no
			}
		}
	}
}
