	OR = {
		NOR = {
			government_type = government_type:republic
			government_type = government_type:theocracy
		}
		AND = {
			government_type = government_type:republic
			has_reform = government_reform:noble_elite
		}
		AND = {
			government_type = government_type:theocracy
			has_reform = government_reform:military_order_reform
		}
	}
}

country_rank_is_what = {
	trigger_if = {
		limit = {
			exists = scope:character
			or = {
				has_ruler = no
				not = { ruler = scope:character }
			}
			scope:character ?= {
				exists = "rule_end_date(root)"
			}
		}
		"country_rank_on_date(scope:character.rule_end_date(root))" ?= $target_rank$
	}
	trigger_else = {
		country_rank ?= $target_rank$
	}
}

can_upgrade_country_rank_empire = {
	is_great_power = yes
	trigger_if = {
		limit = { 
			country_exists = c:PAP
			NOT = { tag = PAP }
			religion = religion:catholic 
		}
		c:PAP = {
			opinion = { target = root value > 100 }
		}
	}
	OR = {
		total_population > 10000
		total_true_faith_population > 7500
		total_accepted_culture_population > 5000
	}
	prestige > 70
	is_subject = no
}
can_upgrade_country_rank_kingdom = {
	OR = {
		total_population > 2000
		total_true_faith_population > 1500
		total_accepted_culture_population > 1000
		AND = {
			has_game_rule = rank_restriction_by_numbers
			is_member_of_international_organization = international_organization:hre
			OR = {
				total_population > 2000
				total_true_faith_population > 1500
				total_accepted_culture_population > 1000
			}
			international_organization:hre = {
				any_international_organization_member = {
					country_rank_level = 3
					count < 6
				}
			}
		}
	}
	prestige > 50		
	is_subject = no
}

		total_accepted_culture_population > 100
	}
	prestige > 25
}

alliance_can_exist = {
	$first$ = {
		opinion = { target = $second$ value >= 0 }
		not = { has_blocked_treaties = $second$ }
		not = { is_subject_of = $second$ }
		not = { is_rival_of = $second$ }
		modifier:allowed_alliance = yes
		NOT = {
			is_member_of_international_organization_of_type = {
				type = coalition
				target = $second$
			}
		}
		NOT = {
			in_union_with = $second$
		}
		is_rebel_country = no
	}
	$second$ = {
		opinion = { target = $first$ value >= 0 }
		not = { has_blocked_treaties = $first$ }
		not = { is_subject_of = $first$ }
		not = { is_rival_of = $first$ }
		modifier:allowed_alliance = yes
		NOT = {
			is_member_of_international_organization_of_type = {
				type = coalition
				target = $first$
			}
		}
		is_rebel_country = no
	}
	can_create_subject_alliance = {
		actor = $first$
		recipient = $second$
	}
	NOT = {
		 AND = {
			$first$ = {
				is_hegemon = yes
			}
			$second$ = {
				is_hegemon = yes
			}
		}
	}
}


can_change_primary_culture = {
	$country$ = {
		trigger_if = {
			limit = {
				is_subject = yes
			}
			trigger_if = {
				limit = {
					subject_type ?= { only_allowed_overlord_primary_culture = yes }
				}
				custom_description = {
					text = can_change_primary_culture_must_match_overlord_culture
					overlord ?= {
						culture = $culture$
					}
				}
			}
			trigger_else_if = {
				limit = {
					subject_type ?= { only_allowed_overlord_primary_or_kindred_culture = yes }
				}
				custom_description = {
					text = can_change_primary_culture_must_match_overlord_or_kindred_culture
					overlord ?= {
						culture ?= {
							OR = {
								this = $culture$
								cultural_view = {
									target = $culture$
									value = kindred
								}
							}
						}
					}
				}
			}
			trigger_else = {
				hidden_trigger = { always = yes }
			}
		}
	}
}

is_eligible_for_the_great_ruler_event = {
	has_ruler = yes
	ruler ?= {
		total_abilities >= 250
	}
	is_subject = no
	ruler_reign >= 5
	has_or_had_great_ruler = no	#There has never been any "the Great" before (historical "the Great"s excluded)
	average_control_in_home_region >= 0.33
	OR = {
		country_rank_level >= 3
		is_great_power = yes
	}
	stability >= 75
	has_any_active_disaster = no
	prestige >= 75
	is_during_bankruptcy = no
	government_power >= 90
	war_exhaustion < 2
	num_works_of_art >= 4
	monthly_balance > 0
	OR = {
		at_war = no
		#Is Winning against a significant power
		any_great_power = {
			is_at_war_with = ROOT
			ROOT = {
				war_score_versus = {
					target = PREV
					value >= 0.4
				}
			}
		}
	}
}

	exists = union
	exists = scope:recipient
	union ?= {
		modifier:union_integration_level > 0
	}
	trigger_if = {
		limit = { scope:recipient = { has_variable = integrated_in_union } }
		save_temporary_scope_value_as = {
			name = union_cost_factor_multiplier
			value = {
				value = union.modifier:union_integration_level
				subtract = scope:recipient.var:integrated_in_union
			}
		}
	}
	trigger_if = {
		limit = {
			country_economical_base > 0
			scope:recipient.country_economical_base > 0
		}
		save_temporary_scope_value_as = {
			name = union_special_status_multiplier
			value = {
				value = country_economical_base
				divide = scope:recipient.country_economical_base
				add = 1
			}
		}
	}
	trigger_else = {
		save_temporary_scope_value_as = {
			name = union_cost_factor_multiplier
			value = union.modifier:union_integration_level
		}
	}
	trigger_if = {
		limit = { exists = scope:union_cost_factor_multiplier }
		custom_tooltip = {
			text = bribe_estate_enough_government_power_tt
			government_power >= {
				value = government_power_weak_penalty
				multiply = scope:union_cost_factor_multiplier
				if = {
					limit = { exists = scope:union_special_status_multiplier }
					multiply = scope:union_special_status_multiplier
				}
				max = 50
			}
		}

		gold >= union_parliament_gold_bribe_recipient
	}
}
