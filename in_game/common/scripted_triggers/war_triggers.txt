is_religious_war = {
	custom_description = {
		text = is_religious_war_text
		is_no_cb_war = no
		has_casus_belli = yes
		casus_belli ?= {
			has_tag = religious_casus_belli
		}
	}
}

scaligeri_peace_treaty_demand_allow_MLO = {
	trigger_if = {
		limit = {
			var:MLO_demands = flag:brescia_demand
		}
		province_definition:brescia_province = {
			any_location_in_province_definition = {
				owner ?= { 
					scope:war = {
						is_a_defender = PREV
					}
				}
			}
		}
	}
	trigger_else = {
		province_definition:parma_province = {
			any_location_in_province_definition = {
				owner ?= { 
					scope:war = {
						is_a_defender = PREV
					}
				}
			}
		}
	}
}

scaligeri_peace_treaty_tooltip_fallback_scopes = {
	trigger_if = {
		limit = {
			NOT = { exists = scope:war }
		}
		any_war = {
			casus_belli ?= { this = casus_belli:cb_scaligeri_war }
			save_temporary_scope_as = war
		}
		c:VER = {
			save_temporary_scope_as = loser
		}
	}
}
