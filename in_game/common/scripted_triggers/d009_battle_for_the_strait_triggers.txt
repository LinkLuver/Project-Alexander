is_battle_for_the_strait_principal = {
	OR = {
		AND = {
			country_exists = c:CAS
			this = c:CAS
		}
		AND = {
			country_exists = c:MOR
			this = c:MOR
		}
		AND = {
			country_exists = c:GRA
			this = c:GRA
		}
	}
}

has_ascendancy_over_granada = {
	trigger_if = {
		limit = { scope:actor = { has_or_had_tag = MOR } }
		situation:battle_for_the_strait = {
			has_variable = influence_over_granada
			var:influence_over_granada >= 75
		}
	}
	trigger_else = {
		situation:battle_for_the_strait = {
			has_variable = influence_over_granada
			var:influence_over_granada <= -75
		}
	}
}

can_demand_andalusi_submission = {
	has_dlc = "d009_across_the_pillars"
	is_situation_active = situation:battle_for_the_strait
	country_exists = c:GRA
	OR = {
		has_or_had_tag = CAS
		has_or_had_tag = MOR
	}
	c:GRA ?= {
		NOT = { is_subject_or_below_of = root }
		relative_strength = {
			target = root
			value <= 0.5
		}
	}
}
