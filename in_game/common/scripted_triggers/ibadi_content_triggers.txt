can_be_changed_to_via_generic_action = {
	generic_action:$action$ = {
		custom_tooltip = {
			text = can_be_changed_to_with_THIS_action
			always = no
		}
	}
}

is_forced_into_kitman_state = {
	religion_population_in_country = {
		religion = religion:ibadi
		value < 100
	}
}

country_allows_a_weak_imam = {
	OR = {
		at_war = yes
		war_exhaustion > 1
	}
}

can_become_weak_imam = {
	$country$ = {
		save_temporary_scope_as = country
	}
	"heir_score(scope:country)" < 275
}

can_become_strong_imam = {
	$country$ = {
		save_temporary_scope_as = country
	}
	"heir_score(scope:country)" >= 275
}

rustamid_legacy_trigger = {
	owns = location:tahert
	religion_population_in_country = {
		religion = religion:ibadi
		value >= 250 # 392k at game start in maghreb
	}
	region:maghreb_region = {
		any_ownable_location_in_region = {
			count >= 100
			owner ?= scope:actor
		}
	}
}
