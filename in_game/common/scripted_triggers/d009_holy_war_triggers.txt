# Shared Crusade / Jihad holy-war scripted triggers.

can_summon_iberian_military_orders = {
	has_dlc = "d009_across_the_pillars"
	religion = religion:catholic
	capital = { region = region:iberia_region }
	trigger_if = {
		limit = { exists = international_organization:crusade }
		has_special_status_in_international_organization = {
			type = special_status:crusade_temporal_leader
			international_organization = international_organization:crusade
		}
		international_organization:crusade.international_organization_target ?= {
			religion.group ?= religion_group:muslim
			capital = {
				OR = {
					region = region:iberia_region
					region = region:maghreb_region
				}
			}
		}
	}
	trigger_else = {
		always = no
	}
}
