mesoamerica_age_name_trigger = {
	capital ?= {
		OR = {
			region = region:mesoamerica_region
			region = region:central_america_region
		}
	}
}

greece_age_name_trigger = {
	culture = { has_culture_group = culture_group:greek_group }
}

#Below the greek loc so it doesnt have priority
near_east_age_name_trigger = {
	capital ?= {
		OR = {
			sub_continent = sub_continent:middle_east
			region = region:caucasus_region
			region = region:egypt_region
		}
	}
}

india_age_name_trigger = {
	capital ?= { sub_continent = sub_continent:south_asia }
}

china_age_name_trigger = {
	capital ?= {
		sub_continent = sub_continent:east_asia
		NOT = {
			region = region:japan_region
		}
	}
}

game_is_playing_on_easy_mode = { #P:A Vanilla Entry
	OR = {
		has_game_rule = player_very_easy_difficulty
		has_game_rule = player_easy_difficulty
	}
	OR = {
		has_game_rule = ai_very_easy_difficulty
		has_game_rule = ai_easy_difficulty
	}
}

game_is_playing_on_hard_mode = { #P:A Vanilla Entry
	NOR = { 
		has_game_rule = player_very_easy_difficulty
		has_game_rule = player_easy_difficulty
		has_game_rule = player_normal_difficulty
		has_game_rule = ai_very_easy_difficulty
		has_game_rule = ai_easy_difficulty
		has_game_rule = ai_normal_difficulty
	}
	OR = {
		AND = {
			has_game_rule = player_normal_difficulty
			NOT = { has_game_rule = ai_normal_difficulty }
		}
		AND = {
			has_game_rule = ai_normal_difficulty
			NOT = { has_game_rule = player_normal_difficulty }
		}
		AND = {
			NOT = { has_game_rule = player_normal_difficulty }
			NOT = { has_game_rule = ai_normal_difficulty }
		}
	}
}

	}
}
