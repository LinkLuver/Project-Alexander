#P:A Imported from Invictus: citizenship_restrictions_trigger (common/scripted_triggers/00_character_events_inv.txt). Character scope.
pa_imp_citizenship_restrictions_trigger = {
	exists = culture
	exists = employer
	exists = employer.culture
	trigger_if = {
		limit = {
			NOT = { culture = employer.culture }
		}
		employer = {
			has_primary_or_accepted_culture = prev.culture #P:A TODO Approximate: Imperator required one of the employer's country cultures to have the citizenship_by_adoption modifier (integrated cultures); EU5 requires the character's culture to be primary or accepted in the employer
		}
	}
}
