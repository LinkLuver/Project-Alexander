# Imperator scripted triggers for batch b14 of the Invictus/Indomita event import (2026-10-19).

# Batch-local helper (not an Imperator trigger): Imperator storms.txt "terrain = ocean", i.e. an open sea zone.
# Location scope.
#P:A TODO Approximate: Imperator had one open-sea terrain (ocean); EU5 splits open water into ocean, deep_ocean, inland_sea (the Mediterranean, Red Sea, Caspian) and narrows (straits, the Aegean)
pa_imp_b14_open_sea_trigger = {
	is_land = no
	OR = {
		topography = ocean
		topography = deep_ocean
		topography = inland_sea
		topography = narrows
	}
}

#P:A Imported from Invictus: marriage approval triggers of the seek a spouse scheme (common/scripted_triggers)
# Evaluated in scheme.21 at root (the country, which holds the variables suitor_age and love_target_age);
# scope:negotiator = head of the target family, scope:love_target = the proposed partner.
#P:A TODO Approximate: EU5 characters have no loyalty; every character counts as loyal (Imperator loyalty >= 50), so the *_disapproves_loyalty triggers are always false and the *_approves triggers only check the age gap

pa_imp_hof_disapproves_loyalty = {
	exists = scope:negotiator
	# scope:negotiator = { loyalty < 50 }
	always = no #P:A TODO No EU5 equivalent: the head of family has loyalty below 50 (EU5 characters have no loyalty)
}

pa_imp_hof_disapproves_age = {
	exists = scope:negotiator
	var:suitor_age >= pa_imp_get_love_target_age_svalue
}

pa_imp_self_disapproves_loyalty = {
	NOT = {
		exists = scope:negotiator
	}
	# scope:love_target = { loyalty < 50 }
	always = no #P:A TODO No EU5 equivalent: the proposed partner has loyalty below 50 (EU5 characters have no loyalty)
}

pa_imp_self_disapproves_age = {
	NOT = {
		exists = scope:negotiator
	}
	var:suitor_age >= pa_imp_get_love_target_age_svalue
}

pa_imp_hof_approves = {
	exists = scope:negotiator
	# scope:negotiator = { loyalty >= 50 } #P:A TODO Approximate: loyalty >= 50 of the head of family dropped (EU5 characters have no loyalty)
	var:suitor_age < pa_imp_get_love_target_age_svalue
}

pa_imp_self_approves = {
	NOT = {
		exists = scope:negotiator
	}
	# scope:love_target = { loyalty >= 50 } #P:A TODO Approximate: loyalty >= 50 of the proposed partner dropped (EU5 characters have no loyalty)
	var:suitor_age < pa_imp_get_love_target_age_svalue
}
