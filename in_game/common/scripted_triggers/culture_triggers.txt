	}

		text = is_religion_pagan_tt

	}
}

is_swiss_for_advances = {
	culture = { has_culture_group = culture_group:swiss_group }
	capital ?= {
		OR = {
			area = area:western_switzerland_area
			area = area:central_switzerland_area
			area = area:eastern_switzerland_area
		}
	}
}

roman_culture_movement_location_trigger = {
	has_owner = yes
	is_port = yes
	is_city = yes
	dominant_language = { language_family ?= language_family:romance_language_family }
	owner = {
		modifier:allow_roman_movement = yes
		OR = {
			has_or_had_tag = BYZ
			has_or_had_tag = ROM
		}
	}
	average_satisfaction > 0.5
}

hellenism_religion_movement_location_trigger = {
	has_owner = yes
	is_holy_site_for = religion:hellenism_religion
	owner = { religion.group = religion_group:christian }
}
