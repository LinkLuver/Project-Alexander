academia_plausible_location = {
	owner ?= {
		produced_in_country:paper > 5
	}
	has_institution = institution:philosophy

	num_pop_type:burghers > 1
	has_building_with_at_least_one_level = academy

	owner = {
		societal_value:traditionalist_vs_innovative > 30
	}
}

military_standardization_plausible_location = {
	#continent = continent:europe
	has_owner = yes

	custom_tooltip = {
		text = dominant_culture_is_owners_culture
		dominant_culture = owner.culture
	}
	modifier:local_manpower > 0

	owner = {
		societal_value:quality_vs_quantity < -20
	}
}
public_works_plausible_location = {
	has_institution = institution:metallurgy
	has_institution = institution:urbanization

	is_market_center = yes
	OR = {
		location_rank = location_rank:city
		location_rank = location_rank:megalopolis
	}
	market = {
		total_goods_value_traded > 500
	}
}
