# Imperator scripted triggers for batch b17 of the Invictus/Indomita event import (2026-10-19).
# The friend/rival triggers are character-scope. In Imperator their ROOT was the character of a character event; in the
# converted events that character is scope:pa_character (root is the country), so the comparisons use scope:pa_character.

#P:A Imported from Indomita (common/scripted_triggers): char_can_be_friend_trigger (character scope)
pa_imp_char_can_be_friend_trigger = {
	is_adult = yes
	is_ruler = no
	# can_get_friends = yes #P:A TODO No EU5 equivalent: EU5 characters have no friend slots
	# is_mercenary = no #P:A TODO No EU5 equivalent: EU5 has no mercenary characters
	exists = THIS
	# NOT = { is_rival = ROOT } #P:A TODO No EU5 equivalent: character rivals
	# NOT = { is_friend = ROOT } #P:A TODO No EU5 equivalent: character friends
	trigger_if = {
		limit = { exists = scope:pa_character }
		NOT = { this = scope:pa_character }
		trigger_if = { # Imperator NOT = { has_same_family = ROOT }
			limit = {
				exists = dynasty
				exists = scope:pa_character.dynasty
			}
			NOT = { dynasty = scope:pa_character.dynasty }
		}
	}
}
