country_flavor_heir = {
	type = country

	text = {
		localization_key = country_flavor_heir_successor
		trigger = {
			OR = {
				government_type = government_type:theocracy
				government_type = government_type:republic
			}
		}
	}


	text = {
		localization_key = rank_inca_heir
		trigger = {
			country_rank_level > 1
			government_type = government_type:monarchy
			exists = culture
			exists = culture.language
			culture.language = language:quechuan_language
		}
	}
	
	text = {
		localization_key = rank_kingdom_maya
		trigger = {
			country_rank_is_kingdom = yes
			court_language ?= language:maya_language
		}
	}
	text = {
		localization_key = rank_maya_heir
		trigger = {
			court_language ?= language:maya_language
		}
	}

	text = {
		localization_key = country_flavor_heir_crown_prince
		trigger = {
			country_rank_level >= 3 # kingdom
			government_type = government_type:monarchy
		}
	}

	text = {
		localization_key = country_flavor_heir
		fallback = yes
	}
}
