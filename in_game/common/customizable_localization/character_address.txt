# style, or form of address
character_address = {
	type = character
	
	# RULERS



	text = {
		localization_key = character_address_majesty

		trigger = {
			character_address_royal_ruler_trigger = yes
		}
	}

	text = {
		localization_key = character_address_grace_duke

		trigger = {
			any_ruling_countries = {
				government_type = government_type:monarchy
				country_rank = country_rank:rank_duchy
			}
		}
	}


	text = {
		localization_key = character_address_royal_highness

		trigger = {
			OR = {
				any_parent = {
					character_address_royal_ruler_trigger = yes
				}
				any_spouse = {
					character_address_royal_ruler_trigger = yes
				}
			}
		}
	}

	text = {
		localization_key = character_address_highness

		trigger = {
			any_parent = {
				any_parent = {
					OR = {
						character_address_imperial_ruler_trigger = yes
						character_address_royal_ruler_trigger = yes
					}
				}
			}
		}
	}

	text = {
		localization_key = character_address_excellency

		trigger = {
			is_ruler = yes
		}
	}


	# NON-RULERS

	text = {
		localization_key = character_address_lordship_low_noble

		trigger = {
			is_dynasty_head = yes
		}
	}
}
