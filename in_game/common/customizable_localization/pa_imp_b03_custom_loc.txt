pa_imp_get_republic_type = {
	type = country
	text = {
		localization_key = get_democratic_republic
		trigger = {
			government_type = government_type:republic
			has_reform = government_reform:sortition #P:A TODO Approximate: Imperator democratic_republic -> republic with the sortition reform
		}
	}
	text = {
		localization_key = get_theocratic_republic
		trigger = {
			government_type = government_type:theocracy #P:A TODO Approximate: Imperator theocratic_republic -> EU5 theocracy (EU5 has no theocratic republic)
		}
	}
	text = {
		localization_key = get_plutocratic_republic
		trigger = {
			government_type = government_type:republic
			has_reform = government_reform:merchant_republic #P:A TODO Approximate: Imperator plutocratic_republic -> republic with the merchant_republic reform
		}
	}
	# Imperator had no text for other republic forms (the sentence then showed nothing); this fallback is new.
	text = {
		localization_key = get_republic_type_fallback
		fallback = yes
	}
}
