# Imperator custom localization for batch b12 of the Invictus/Indomita event import (2026-10-13).

#P:A Imported from Invictus: get_fortune_goddess (common/customizable_localization/01_god_names_inv_1_0.txt)
# Imperator roman_pantheon is the Hellenic faith of Greeks and Romans; the mod splits it into hellenic_pantheon
# (Hellenic) and roman_pantheon (Italic), so every Imperator roman_pantheon check below accepts both.
pa_imp_get_fortune_goddess = {
	type = country

	text = { # Roman
		localization_key = get_fortuna
		trigger = {
			NOT = { culture = culture:etruscan }
			culture = { has_culture_group = culture_group:italic_group }
			OR = { # Imperator roman_pantheon
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	text = { # Etruscan
		localization_key = get_nortia
		trigger = {
			culture = culture:etruscan
			OR = { # Imperator roman_pantheon
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
				religion = religion:etruscan_pantheon #P:A TODO Approximate: added; Imperator Etruscans followed roman_pantheon, the mod gives them their own etruscan_pantheon
			}
		}
	}

	text = { # Hellenic
		localization_key = get_tyche
		trigger = {
			NOT = { culture = { has_culture_group = culture_group:italic_group } } # Imperator country_culture_group = latin
			OR = { # Imperator roman_pantheon
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	text = { # Fallback
		localization_key = get_fortune_goddess_default
		fallback = yes
	}
}

#P:A Imported from Invictus: get_wine_god (common/customizable_localization/01_god_names_inv_1_0.txt)
# Imperator roman_pantheon is the Hellenic faith of Greeks and Romans; the mod splits it into hellenic_pantheon
# (Hellenic) and roman_pantheon (Italic), so every Imperator roman_pantheon check below accepts both.
# get_sabazios (batch b02) and get_osiris (batch b07) are existing loc keys, reused here.
pa_imp_get_wine_god = {
	type = country

	text = { # Roman
		localization_key = get_bacchus
		trigger = {
			NOT = { culture = culture:etruscan }
			culture = { has_culture_group = culture_group:italic_group }
			OR = { # Imperator roman_pantheon
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	text = { # Etruscan
		localization_key = get_fufluns
		trigger = {
			culture = culture:etruscan
			OR = { # Imperator roman_pantheon
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
				religion = religion:etruscan_pantheon #P:A TODO Approximate: added; Imperator Etruscans followed roman_pantheon, the mod gives them their own etruscan_pantheon
			}
		}
	}

	text = { # Hellenic
		localization_key = get_dionysus
		trigger = {
			NOT = { culture = { has_culture_group = culture_group:italic_group } } # Imperator country_culture_group = latin
			OR = { # Imperator roman_pantheon
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	text = { # Anatolian
		localization_key = get_sabazios
		trigger = {
			OR = { # Imperator anatolian_religion
				religion = religion:luwian_pantheon
				religion = religion:cybele_pantheon
			}
		}
	}

	text = { # Kemetic
		localization_key = get_osiris
		trigger = {
			religion = religion:egyptian_pantheon
		}
	}

	text = { # Canaanite
		localization_key = get_marqod
		trigger = {
			OR = { # Imperator carthaginian_pantheon
				religion = religion:punic_pantheon
				religion = religion:canaanite_pantheon
			}
		}
	}

	text = { # Fallback
		localization_key = get_wine_god_default
		fallback = yes
	}
}
