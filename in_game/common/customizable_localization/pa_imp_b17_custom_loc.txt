#P:A Imported from Imperator (vanilla common/customizable_localization/00_event.txt): get_knowledge_of_alexander
pa_imp_get_knowledge_of_alexander = {
	type = country
	text = {
		trigger = {
			culture = { has_culture_group = culture_group:greek_group } # Imperator hellenic
		}
		localization_key = alexander_greek_cloc
	}
	text = {
		trigger = {
			culture = {
				OR = {
					has_culture_group = culture_group:indo_aryan_group # Imperator aryan
					has_culture_group = culture_group:pracyan_group
					# has_culture_group = south_levantine #P:A TODO No EU5 equivalent: Imperator south_levantine culture group has no mapped mod group
					has_culture_group = culture_group:dravidian_group # Imperator indian
					has_culture_group = culture_group:tibetan_group #P:A TODO Approximate: Imperator tibetan group -> the mod's tibetan_group (not in the mapping table)
				}
			}
		}
		localization_key = alexander_eastern_cloc
	}
	text = {
		trigger = {
			always = yes
		}
		localization_key = alexander_default_cloc
	}
}
