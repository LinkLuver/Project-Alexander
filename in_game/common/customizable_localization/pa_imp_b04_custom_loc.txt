		trigger = {
			is_heir = no #P:A TODO Approximate: Imperator is_pretender (a claimant); EU5 has no pretenders, any non-heir is a claimant here

		}

#P:A Imported from Invictus: get_surveillance_paragraph_ip3
pa_imp_get_surveillance_paragraph_ip3 = {
	type = character
	text = {
		localization_key = no_surveillance_ip3
		trigger = {
			employer = {
				OR = {
					NOT = { has_variable = surveillance }
					AND = {
						has_variable = surveillance
						var:surveillance < 1
					}
				}
			}
		}
	}
	text = {
		localization_key = surveillance_acquired_ip3
		trigger = {
			employer = {
				has_variable = surveillance
				var:surveillance >= 1
				NOT = {
					has_variable = surveillance_discovered
				}
			}
		}
	}
	text = {
		localization_key = surveillance_discovered_ip3
		trigger = {
			has_variable = successor_warned
		}
	}
	text = {
		localization_key = no_surveillance_ip3_default
		trigger = {
			always = yes
		}
	}
}

#P:A Imported from Invictus: get_love_goddess (common/customizable_localization/01_god_names_inv_1_0.txt)
# Invictus used roman_pantheon for both Romans and Greeks; the mod splits them into roman_pantheon (Italic),
# etruscan_pantheon and hellenic_pantheon, so each branch below accepts the matching mod religion too (as pa_imp_get_sky_god does).
pa_imp_get_love_goddess = {
	type = country

	text = { # Roman
		localization_key = get_venus
		trigger = {
			NOT = { culture = culture:etruscan }
			culture = { has_culture_group = culture_group:italic_group }
			religion = religion:roman_pantheon
		}
	}

	text = { # Etruscan
		localization_key = get_turan
		trigger = {
			culture = culture:etruscan
			OR = { #P:A TODO Approximate: Imperator checked roman_pantheon only; the mod gives Etruscans their own etruscan_pantheon
				religion = religion:roman_pantheon
				religion = religion:etruscan_pantheon
			}
		}
	}

	text = { # Hellenic
		localization_key = get_aphrodite
		trigger = {
			OR = { #P:A TODO Approximate: Imperator was "roman_pantheon and not a Latin culture group"; the mod's Greeks follow hellenic_pantheon
				religion = religion:hellenic_pantheon
				AND = {
					NOT = { culture = { has_culture_group = culture_group:italic_group } }
					religion = religion:roman_pantheon
				}
			}
		}
	}

	text = { # Canaanite
		localization_key = get_astarte
		trigger = {
			religion = religion:punic_pantheon
		}
	}

	text = { # Kemetic
		localization_key = get_isis
		trigger = {
			religion = religion:egyptian_pantheon
		}
	}

	text = { # Chaldean
		localization_key = get_ninlil
		trigger = {
			religion = religion:mesopotamian_pantheon
		}
	}

	text = { # Fallback
		localization_key = get_love_goddess_default
		fallback = yes
	}
}
