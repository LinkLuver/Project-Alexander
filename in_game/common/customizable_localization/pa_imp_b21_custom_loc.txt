# Imperator custom localization for b21 of the Invictus/Indomita event import (2026-10-20): the Olympic Games starters.

#P:A Imported from Imperator: get_olympic_highlight (common/customizable_localization/00_event.txt)
# Imperator read local_var:event_selector set by olympic.5; here the global variable olympic_event_selector_global,
# because olympic.6 also runs in the other competing countries.
pa_imp_get_olympic_highlight = {
	type = country

	text = {
		localization_key = pankration_event
		trigger = { global_var:olympic_event_selector_global ?= 1 }
	}
	text = {
		localization_key = diaulos_event
		trigger = { global_var:olympic_event_selector_global ?= 2 }
	}
	text = {
		localization_key = pygmachia_event
		trigger = { global_var:olympic_event_selector_global ?= 3 }
	}
	text = {
		localization_key = hoplitodromos_event
		trigger = { global_var:olympic_event_selector_global ?= 4 }
	}
	text = {
		localization_key = synoris_event
		trigger = { global_var:olympic_event_selector_global ?= 5 }
	}
	text = {
		localization_key = stadion_event
		fallback = yes
	}
}
