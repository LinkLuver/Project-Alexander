# Character scope: the option text of traits.1, by how good the new trait is (traits_option_variable 1 good, 2 neutral, 3 bad).
pa_imp_traits_option_selector = {
	type = character
	text = {
		localization_key = traits_option_selector.good
		trigger = {
			has_variable = traits_option_variable
			var:traits_option_variable = 1
		}
	}
	text = {
		localization_key = traits_option_selector.neutral
		trigger = {
			has_variable = traits_option_variable
			var:traits_option_variable = 2
		}
	}
	text = {
		localization_key = traits_option_selector.bad
		trigger = {
			has_variable = traits_option_variable
			var:traits_option_variable = 3
		}
	}
	text = {
		localization_key = traits_option_selector.default
		fallback = yes
	}
}
