#P:A Imported from Invictus: festival_type (common/customizable_localization/01_event_inv_1_0.txt)
# Imperator roman_pantheon -> OR of hellenic_pantheon and roman_pantheon (the mod splits the Imperator Hellenic faith)
pa_imp_festival_type = {
	type = country

	text = {
		localization_key = saturnalia
		trigger = {
			culture = { has_culture_group = culture_group:italic_group } # Imperator country_culture_group = latin
			OR = {
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	text = {
		localization_key = pyanepsia
		trigger = {
			culture = { has_culture_group = culture_group:greek_group } # Imperator country_culture_group = hellenic
			OR = {
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	text = {
		localization_key = marzeah
		trigger = {
			# has_deity_in_pantheon = deity:omen_melqart
			OR = { #P:A TODO Approximate: Imperator required Melqart in the country's pantheon (EU5 has no deities); the Punic and Canaanite faiths stand in
				religion = religion:punic_pantheon
				religion = religion:canaanite_pantheon
			}
			NOR = {
				religion = religion:hellenic_pantheon
				religion = religion:roman_pantheon
			}
		}
	}

	# text = { localization_key = adonia trigger = { has_deity_in_pantheon = deity:omen_astarte NOT = { has_deity_in_pantheon = deity:omen_melqart } NOT = { religion = roman_pantheon } } }
	# text = { localization_key = mayumas trigger = { has_deity_in_pantheon = deity:deity_atargatis NOT = { has_deity_in_pantheon = deity:omen_astarte } NOT = { has_deity_in_pantheon = deity:omen_melqart } NOT = { religion = roman_pantheon } } }
	#P:A TODO No EU5 equivalent: "Adonia" (Astarte in the pantheon) and "Mayumas" (Atargatis in the pantheon) need pantheon deities, which EU5 lacks; those countries get the fallback "Festival"

	text = {
		localization_key = festival_key
		trigger = {
			always = yes
		}
	}
}

#P:A Imported from Invictus: get_age_adjective (common/customizable_localization)
pa_imp_get_age_adjective = {
	type = character
	random_valid = yes

	text = {
		localization_key = get_maiden_nn
		trigger = {
			is_female = yes
			age_in_years <= 21
		}
	}
	text = {
		localization_key = get_master_nn
		trigger = {
			is_female = no
			age_in_years <= 21
		}
	}
	text = {
		localization_key = get_woman_nn
		trigger = {
			is_female = yes
			age_in_years > 21
			age_in_years <= 44
		}
	}
	text = {
		localization_key = get_man_nn
		trigger = {
			is_female = no
			age_in_years > 21
			age_in_years <= 44
		}
	}
	text = {
		localization_key = get_woman_of_age_nn
		trigger = {
			is_female = yes
			age_in_years > 44
		}
	}
	text = {
		localization_key = get_gentleman_nn
		trigger = {
			is_female = no
			age_in_years > 44
		}
	}
}
