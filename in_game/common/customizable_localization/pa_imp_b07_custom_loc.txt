#P:A Imported from Invictus: get_local_terrain (common/customizable_localization/01_event_inv_1_0.txt)
# Imperator: any owned province in the capital's area with the terrain. EU5: any owned location in the capital's area.
pa_imp_get_local_terrain = {
	type = country
	random_valid = yes

	text = {
		localization_key = get_mountain
		trigger = {
			any_owned_location = {
				area = root.capital.area
				topography = mountains
			}
		}
	}

	text = {
		localization_key = get_desert
		trigger = {
			any_owned_location = {
				area = root.capital.area
				vegetation = desert #P:A TODO Approximate: Imperator terrain desert; EU5 desert vegetation
			}
		}
	}

	text = {
		localization_key = get_jungle
		trigger = {
			any_owned_location = {
				area = root.capital.area
				vegetation = jungle
			}
		}
	}

	text = {
		localization_key = get_marsh
		trigger = {
			any_owned_location = {
				area = root.capital.area
				topography = wetlands
			}
		}
	}

	text = {
		localization_key = get_snow
		trigger = {
			any_owned_location = {
				area = root.capital.area
				winter_level = severe # Imperator has_province_modifier = severe_winter
			}
		}
	}

	text = {
		localization_key = get_forest
		trigger = {
			any_owned_location = {
				area = root.capital.area
				vegetation = forest
			}
		}
	}

	text = {
		localization_key = get_hillside
		trigger = {
			any_owned_location = {
				area = root.capital.area
				topography = hills
			}
		}
	}
	text = {
		localization_key = get_oasis
		trigger = {
			always = no #P:A TODO No EU5 equivalent: Imperator oasis terrain
		}
	}
	text = {
		localization_key = get_desert_hills
		trigger = {
			any_owned_location = {
				area = root.capital.area
				topography = hills #P:A TODO Approximate: Imperator terrain desert_hills; EU5 hills in an arid climate
				climate = arid
			}
		}
	}
	text = {
		localization_key = get_flood_plain
		trigger = {
			any_owned_location = {
				area = root.capital.area
				vegetation = farmland #P:A TODO Approximate: Imperator terrain flood_plain; EU5 farmland vegetation
			}
		}
	}
	text = {
		localization_key = get_steppes
		trigger = {
			any_owned_location = {
				area = root.capital.area
				vegetation = grasslands #P:A TODO Approximate: Imperator terrain steppes; EU5 grasslands in a cold arid climate
				climate = cold_arid
			}
		}
	}
	text = {
		localization_key = get_remote
		trigger = {
			always = yes
		}
	}
}

#P:A Imported from Invictus: get_god_of_dead (common/customizable_localization/01_god_names_inv_1_0.txt)
# Invictus used roman_pantheon for both Romans and Greeks; the mod splits them into roman_pantheon (Italic),
# etruscan_pantheon and hellenic_pantheon, so each branch accepts the matching mod religion too (as pa_imp_get_sky_god of batch b02 does).
pa_imp_get_god_of_dead = {
	type = country

	text = { # Roman
		localization_key = get_dis
		trigger = {
			NOT = { culture = culture:etruscan }
			culture = { has_culture_group = culture_group:italic_group }
			OR = { religion = religion:hellenic_pantheon religion = religion:roman_pantheon } #P:A Imperator roman_pantheon is Hellenic; the mod splits it into hellenic_pantheon and roman_pantheon (Italic)
		}
	}

	text = { # Etruscan
		localization_key = get_aita
		trigger = {
			culture = culture:etruscan
			OR = { #P:A TODO Approximate: Imperator checked roman_pantheon only; the mod gives Etruscans their own etruscan_pantheon
				OR = { religion = religion:hellenic_pantheon religion = religion:roman_pantheon } #P:A Imperator roman_pantheon is Hellenic; the mod splits it into hellenic_pantheon and roman_pantheon (Italic)
				religion = religion:etruscan_pantheon
			}
		}
	}

	text = { # Hellenic
		localization_key = get_hades
		trigger = {
			OR = { #P:A TODO Approximate: Imperator was "roman_pantheon and not a Latin culture group"; the mod's Greeks follow hellenic_pantheon
				religion = religion:hellenic_pantheon
				AND = {
					NOT = { culture = { has_culture_group = culture_group:italic_group } }
					OR = { religion = religion:hellenic_pantheon religion = religion:roman_pantheon } #P:A Imperator roman_pantheon is Hellenic; the mod splits it into hellenic_pantheon and roman_pantheon (Italic)
				}
			}
		}
	}

	text = { # Canaanite
		localization_key = get_mot
		trigger = {
			OR = { religion = religion:punic_pantheon religion = religion:canaanite_pantheon } #P:A Imperator carthaginian_pantheon covers both; the mod splits it into punic_pantheon and canaanite_pantheon
		}
	}

	text = { # Kemetic
		localization_key = get_osiris
		trigger = {
			religion = religion:egyptian_pantheon
		}
	}

	text = { # Druidic
		localization_key = get_cernunnos
		trigger = {
			religion = religion:gallic_pantheon #P:A TODO Approximate: Imperator druidism mapped to gallic_pantheon
		}
	}

	text = { # Chaldean
		localization_key = get_ninazu
		trigger = {
			religion = religion:mesopotamian_pantheon
		}
	}

	text = { # Fallback
		localization_key = get_god_of_dead_default
		fallback = yes
	}
}
