#P:A Imported from Invictus: get_sky_god (common/customizable_localization/01_god_names_inv_1_0.txt)
# Invictus used roman_pantheon for both Romans and Greeks; the mod splits them into roman_pantheon (Italic),
# etruscan_pantheon and hellenic_pantheon, so each branch below accepts the matching mod religion too.
pa_imp_get_sky_god = {
	type = country

	text = { # Roman
		localization_key = get_jupiter
		trigger = {
			NOT = { culture = culture:etruscan }
			culture = { has_culture_group = culture_group:italic_group }
			OR = { religion = religion:hellenic_pantheon religion = religion:roman_pantheon } #P:A Imperator roman_pantheon is Hellenic; the mod splits it into hellenic_pantheon and roman_pantheon (Italic)
		}
	}

	text = { # Etruscan
		localization_key = get_tinia
		trigger = {
			culture = culture:etruscan
			OR = { #P:A TODO Approximate: Imperator checked roman_pantheon only; the mod gives Etruscans their own etruscan_pantheon
				OR = { religion = religion:hellenic_pantheon religion = religion:roman_pantheon } #P:A Imperator roman_pantheon is Hellenic; the mod splits it into hellenic_pantheon and roman_pantheon (Italic)
				religion = religion:etruscan_pantheon
			}
		}
	}

	text = { # Hellenic
		localization_key = get_zeus
		trigger = {
			OR = { #P:A TODO Approximate: Imperator was "roman_pantheon and not a Latin culture group"; the mod's Greeks follow hellenic_pantheon
				religion = religion:hellenic_pantheon
				AND = {
					NOT = { culture = { has_culture_group = culture_group:italic_group } }
					OR = { religion = religion:hellenic_pantheon religion = religion:roman_pantheon } #P:A Imperator roman_pantheon is Hellenic; the mod splits it into hellenic_pantheon and roman_pantheon (Italic)
				}
			}
		}
	}

	text = { # Anatolian
		localization_key = get_sabazios
		trigger = {
			OR = { religion = religion:luwian_pantheon religion = religion:cybele_pantheon } #P:A Imperator anatolian_religion covers both; the mod splits it into luwian_pantheon and cybele_pantheon | P:A TODO Approximate: Imperator anatolian_religion mapped to luwian_pantheon (the mod also has cybele_pantheon and lydian_pantheon)
		}
	}

	text = { # Carthaginian
		localization_key = get_baal_hammon
		trigger = {
			culture = culture:carthaginian
			OR = { religion = religion:punic_pantheon religion = religion:canaanite_pantheon } #P:A Imperator carthaginian_pantheon covers both; the mod splits it into punic_pantheon and canaanite_pantheon
		}
	}

	text = { # Canaanite
		localization_key = get_baalshamin
		trigger = {
			NOT = { culture = culture:carthaginian }
			OR = { religion = religion:punic_pantheon religion = religion:canaanite_pantheon } #P:A Imperator carthaginian_pantheon covers both; the mod splits it into punic_pantheon and canaanite_pantheon
		}
	}

	text = { # Kemetic
		localization_key = get_horus
		trigger = {
			religion = religion:egyptian_pantheon
		}
	}

	text = { # Druidic
		localization_key = get_taranis
		trigger = {
			religion = religion:gallic_pantheon #P:A TODO Approximate: Imperator druidism mapped to gallic_pantheon
		}
	}

	text = { # Assyrian Chaldean
		localization_key = get_Ashshur
		trigger = {
			culture = culture:assyrian
			religion = religion:mesopotamian_pantheon
		}
	}

	text = { # Chaldean
		localization_key = get_enlil
		trigger = {
			NOT = { culture = culture:assyrian }
			religion = religion:mesopotamian_pantheon
		}
	}

	text = { # Fallback
		localization_key = get_sky_god_default
		fallback = yes
	}
}

#P:A Imported from Imperator: get_court_type (common/customizable_localization/00_generic.txt)
pa_imp_get_court_type = {
	type = country
	text = {
		localization_key = court_type_athens
		trigger = {
			government_type = government_type:republic
			tag = ATH
		}
	}
	text = {
		localization_key = court_type_greek_oligarchy #Gerousia
		trigger = {
			government_type = government_type:republic
			# government = oligarchic_republic
			always = no #P:A TODO No EU5 equivalent: Imperator oligarchic_republic government form (EU5 has no republic sub-forms)
			culture = { has_culture_group = culture_group:greek_group }
		}
	}
	text = {
		localization_key = court_type_greek # Boule
		trigger = {
			government_type = government_type:republic
			culture = { has_culture_group = culture_group:greek_group }
		}
	}
	text = {
		localization_key = court_type_punic #Adirim
		trigger = {
			government_type = government_type:republic
			OR = {
				culture = culture:carthaginian
				culture = culture:phoenician
			}
		}
	}
	text = {
		localization_key = court_type_rome # Rome
		trigger = {
			government_type = government_type:republic
			tag = ROM
		}
	}
	text = {
		localization_key = court_type_synedrion # Mac/Epi
		trigger = {
			government_type = government_type:republic
			OR = {
				tag = MAC
				tag = EPI
				tag = COT # Imperator KOR (Corinth)
				culture = culture:macedonian
				culture = culture:epirote
			}
		}
	}
	text = {
		localization_key = court_type_sangha # Sanskrit
		trigger = {
			government_type = government_type:republic
			culture = {
				OR = {
					has_culture_group = culture_group:indo_aryan_group
					has_culture_group = culture_group:pracyan_group
					has_culture_group = culture_group:dravidian_group
				}
			}
		}
	}
	text = {
		localization_key = court_type_monarchy
		trigger = {
			government_type = government_type:monarchy
		}
	}
	text = {
		localization_key = court_type_republic
		trigger = {
			government_type = government_type:republic
		}
	}
	text = {
		localization_key = court_type_tribal
		trigger = {
			government_type = government_type:tribe
		}
	}
	text = { #P:A EU5 theocracies had no Imperator line; they fall back to "court"
		localization_key = court_type_monarchy
		fallback = yes
	}
}

#P:A Imported from Imperator: get_age_description (common/customizable_localization/00_event.txt)
pa_imp_get_age_description = {
	type = character
	text = {
		localization_key = got_child
		trigger = {
			is_adult = no
		}
	}
	text = {
		localization_key = got_youth
		trigger = {
			is_adult = yes
			age_in_years < 22
		}
	}
	text = {
		localization_key = got_average_age
		trigger = {
			age_in_years >= 22
			age_in_years < 35
		}
	}
	text = {
		localization_key = got_older_person
		trigger = {
			age_in_years >= 35
			age_in_years < 50
		}
	}
	text = {
		localization_key = got_ancient_person
		trigger = {
			age_in_years >= 50
			age_in_years < 70
		}
	}
	text = {
		localization_key = got_methuselah
		trigger = {
			age_in_years >= 70
			age_in_years < 100
		}
	}
	text = {
		localization_key = got_centenarian
		trigger = {
			age_in_years >= 100
		}
	}
}
