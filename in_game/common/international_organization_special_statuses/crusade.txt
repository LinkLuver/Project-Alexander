crusade_spiritual_leader = {
	leader = yes
	max_countries = 1
	priority = 15

	can_bestow_trigger = {
		is_leader_of_international_organization = scope:recipient
	}
	auto_bestowal_trigger = {
		always = yes
	}
	auto_rescind_trigger = {
		NOT = { is_leader_of_international_organization = scope:recipient }
	}

	on_bestowed_effect = {
	}
	on_rescinded_effect = {
	}

	modifier = {
		monthly_religious_influence = 0.05
	}
}

crusade_temporal_leader = {
	max_countries = 1
	priority = 10

	can_bestow_trigger = {
		is_subject = no
		is_member_of_international_organization = scope:recipient
		NOT = { is_leader_of_international_organization = scope:recipient }
	}
	auto_bestowal_trigger = {
	}
	auto_rescind_trigger = {
		OR = {
			NOT = { is_member_of_international_organization = scope:recipient }
			is_leader_of_international_organization = scope:recipient
		}
	}

	on_bestowed_effect = {
	}
	on_rescinded_effect = {
	}

	modifier = {
		damage_done_versus_heathens_modifier = 0.05
	}
}
