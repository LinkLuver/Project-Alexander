scorch_earth = {
	toggle = no
	army_only = yes
	
	map = yes


	allow = {
		has_only_prisoners = no
		is_army = yes
		in_combat = no
		is_moving = no
		is_exiled = no
		in_siege = no
		unit_location.owner ?= owner
		unit_location.controller ?= owner
		unit_location.province.capital ?= unit_location

		unit_location.province ?= {
			province_food_percentage > 0.2
			NOT = { has_province_modifier = scorched_earth }
		}
	}
	
	start_effect = {
		unit_location.province = {
			add_province_modifier = {
				modifier = scorched_earth
				years = 1
				mode = add_and_extend
			}
		}
	}
	ai_will_do = {
		add = -10
	}
}
