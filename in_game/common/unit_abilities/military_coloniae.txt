military_coloniae = {
	toggle = no
	army_only = yes
	
	map = yes
	
	hidden = {
		owner ?= {
			or = {
			 	government_type = government:tribe
			 	government_type = government:amerindian_tribe
			 	government_type = government:amerindian_chiefdom
			 	government_type = government:amerindian_theocracy
			 	government_type = government:amerindian_republic
			 	government_type = government:steppe_horde
			}
		}
	}

	allow = { 
		owner = {
			manpower >= 5
		}
		unit_location = {
			dominant_culture != root.owner.culture
			rank_index > 2
			not = { has_location_modifier = pa_military_coloniae }
		}
	}
	
	start_effect = {
		owner = {
			add_manpower = -5
		}
		unit_location ={
			add_pop = {
				culture = root.owner.culture
				religion =  root.owner.religion
				type = pop_type:soldiers
				size = {
					value = 5
				}
			}
			add_location_modifier = {
				name = pa_military_coloniae
				years = 5
			}
			#add_migration = {
			#	from = 
			#	to_location = this

			#	religion =  root.owner.religion
			#	type = pop_type:soldiers
			#	months = 36

			#}

		}
	}
}
