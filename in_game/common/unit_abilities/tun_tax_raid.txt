tun_tax_raid = {
	army_only = yes
	cancel_on_combat_end = no

	hidden = {
		owner ?= {
			NOT = { has_estate_privilege = estate_privilege:tun_ungovernable_agriculturalists }
		}
	}

	allow = {
		is_army = yes
		is_exiled = no
		in_retreat = no
		has_only_prisoners = no
		unit_location = {
			can_be_targetted_by_tun_tax_raid = yes
			owner ?= root.owner
		}
	}

	highlight = {
		unit_location.province ?= {
			every_location_in_province = {
				limit = { can_be_targetted_by_tun_tax_raid = yes }
				add_to_list = target
			}
		}
	}
	
	start_effect = {
		owner ?= {
			add_gold = {
				value = 0
				root.unit_location.province ?= {
					every_location_in_province = {
						limit = { can_be_targetted_by_tun_tax_raid = yes }
						add = location_potential_tax_base
					}
				}
				multiply = 12
			}
		}
		unit_location = {
			custom_tooltip = {
				text = tun_tax_raid_everyhighlighted_province
				province ?= {
					every_location_in_province = {
						limit = { can_be_targetted_by_tun_tax_raid = yes }
						change_prosperity = prosperity_extreme_penalty
						add_location_modifier = {
							modifier = tun_tax_raid_modifier
							years = 5
							mode = add
						}
					}
				}
			}
			show_as_tooltip = {
				change_prosperity = -0.25
				add_location_modifier = {
					modifier = tun_tax_raid_modifier
					years = 5
					mode = add
				}
			}
		}
	}

	ai_will_do = {
		value = 1 #AI will not actively seek to do it, but if they will move their armies, why not?
	}
}
