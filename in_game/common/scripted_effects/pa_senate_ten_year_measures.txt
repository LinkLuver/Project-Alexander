pa_senate_start_ten_year_measure = {
	if = {
		limit = { tag = ROM government_type = government_type:republic is_active_parliament = no has_variable = pa_senate_initialized var:pa_senate_size > 0 NOT = { has_variable = pa_senate_motion_cooldown } exists = international_organization:pa_roman_senate international_organization:pa_roman_senate = { is_active_parliament = no } }
		remove_variable = pa_senate_agenda_stability
		remove_variable = pa_senate_agenda_holdings
		remove_variable = pa_senate_agenda_works
		remove_variable = pa_senate_agenda_law
		remove_variable = pa_senate_agenda_war
		remove_variable = pa_senate_agenda_alliance
		set_variable = { name = pa_senate_ten_year_proposal value = $measure$ }
		set_variable = { name = pa_senate_vote_opt value = { value = var:pa_senate_optimates_seats multiply = var:pa_senate_optimates_approval divide = 125 subtract = var:pa_senate_optimates_defectors } }
		set_variable = { name = pa_senate_vote_boni value = { value = var:pa_senate_boni_seats multiply = var:pa_senate_boni_approval divide = 125 subtract = var:pa_senate_boni_defectors } }
		set_variable = { name = pa_senate_vote_pop value = { value = var:pa_senate_populares_seats multiply = var:pa_senate_populares_approval divide = 125 subtract = var:pa_senate_populares_defectors } }
		pa_senate_open_motion_debate = { issue = parliament_issue:pa_senate_ten_year_debate }
		set_variable = { name = pa_senate_motion_cooldown value = yes months = 6 }
	}
}
