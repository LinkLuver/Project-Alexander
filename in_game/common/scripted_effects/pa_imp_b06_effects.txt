#P:A Imported from Imperator: move_country_with_entire_family_effect (common/scripted_effects/00_event_effects.txt). Character scope.
# Imperator: moves the character, the members of their family at the same court, their spouse and their children to $COUNTRY$.
pa_imp_move_country_with_entire_family_effect = {
	$COUNTRY$ = { save_scope_as = move_country_with_entire_family_employer }
	save_scope_as = move_country_with_entire_family_character
	hidden_effect = {
		if = {
			limit = {
				exists = dynasty #P:A TODO Approximate: Imperator is_minor_character = no (minor characters have no family); EU5 checks that the character has a dynasty
			}
			dynasty = {
				save_scope_as = move_country_with_entire_family_family
				# move_family = $COUNTRY$
				#P:A TODO No EU5 equivalent: moved the family itself to the new country (EU5 dynasties belong to no country)
			}
			scope:move_country_with_entire_family_family = {
				every_character_in_dynasty = {
					limit = {
						is_ruler = no
						#P:A Removed: prisoner = no is always true in EU5
						employer = scope:move_country_with_entire_family_character.employer
					}
					move_country = $COUNTRY$
				}
			}
		}
		pa_imp_remove_all_positions = yes
		move_country = $COUNTRY$
		remove_character_modifier = foreign_citizen
		# set_as_minor_character = THIS
		#P:A TODO No EU5 equivalent: made the character a minor character (EU5 has no minor characters)
		if = {
			limit = {
				exists = scope:move_country_with_entire_family_family
				NOT = { dynasty = scope:move_country_with_entire_family_family }
			}
			change_dynasty = scope:move_country_with_entire_family_family
		}
		every_spouse = { #P:A TODO Approximate: Imperator moved the single spouse link; EU5 characters can have several spouses, each is checked
			limit = {
				is_ruler = no
				#P:A Removed: prisoner = no is always true in EU5
				employer = scope:move_country_with_entire_family_character.employer # Imperator PREV.employer: the character, already moved
			}
			pa_imp_remove_all_positions = yes
			move_country = $COUNTRY$
			remove_character_modifier = foreign_citizen
			# set_as_minor_character = THIS
			#P:A TODO No EU5 equivalent: made the spouse a minor character (EU5 has no minor characters)
			if = {
				limit = {
					exists = scope:move_country_with_entire_family_family
					NOT = { dynasty = scope:move_country_with_entire_family_family }
				}
				change_dynasty = scope:move_country_with_entire_family_family
			}
		}
		if = {
			limit = { num_of_children >= 1 }
			every_child = {
				limit = {
					is_ruler = no
					#P:A Removed: prisoner = no is always true in EU5
					employer = scope:move_country_with_entire_family_character.employer # Imperator PREV.employer: the character, already moved
				}
				pa_imp_remove_all_positions = yes
				move_country = $COUNTRY$
				remove_character_modifier = foreign_citizen
				# set_as_minor_character = THIS
				#P:A TODO No EU5 equivalent: made the child a minor character (EU5 has no minor characters)
				if = {
					limit = {
						exists = scope:move_country_with_entire_family_family
						NOT = { dynasty = scope:move_country_with_entire_family_family }
					}
					change_dynasty = scope:move_country_with_entire_family_family
				}
			}
		}
	}
	custom_tooltip = move_country_with_entire_family_effect_tt
}
