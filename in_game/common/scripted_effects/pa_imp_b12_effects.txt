#P:A Imported from Invictus: trade_good_decision_spread_effect (common/scripted_effects/00_india_effects.txt)
# Location scope (Imperator: province); root = the country. Parameters: REGION = yes/no, TRADE_GOOD = an EU5 goods key
# (callers pass the mapped EU5 good, not the Imperator trade good).
# REGION = yes: a location of root in the same region, without the good or a valuable good, in an area without the good,
# starts producing it. REGION = no: the same in a neighboring area that has such a location of root.
#P:A TODO Approximate: EU5 areas and regions are not Imperator's (EU5 geography differs), so the spread reaches a different set of locations
pa_imp_trade_good_decision_spread_effect = {
	if = {
		limit = {
			always = $REGION$
		}
		region = {
			random_location_in_region = {
				limit = {
					owner ?= root
					NOR = {
						raw_material = goods:$TRADE_GOOD$
						pa_imp_valuable_trade_good_trigger = yes
					}
					area = {
						NOT = {
							any_location_in_area = {
								raw_material = goods:$TRADE_GOOD$
							}
						}
					}
				}
				change_raw_material = goods:$TRADE_GOOD$
			}
		}
	}
	else = {
		area = {
			random_neighbor_area = {
				limit = {
					any_location_in_area = {
						owner ?= root
						NOR = {
							raw_material = goods:$TRADE_GOOD$
							pa_imp_valuable_trade_good_trigger = yes
						}
					}
				}
				random_location_in_area = {
					limit = {
						owner ?= root
						NOT = {
							raw_material = goods:$TRADE_GOOD$
						}
					}
					change_raw_material = goods:$TRADE_GOOD$
				}
			}
		}
	}
}
