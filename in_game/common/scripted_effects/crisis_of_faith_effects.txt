seed_crisis_of_faith_heresiarch_effect = {
	ordered_character = {
		limit = {
			has_estate = estate_type:clergy_estate
			is_adult = yes
			is_ruler = no
			is_regent = no
		}
		order_by = total_abilities
		max = 1
		check_range_bounds = no
		save_scope_as = crisis_of_faith_heresiarch_scope
	}
	if = {
		limit = { NOT = { exists = scope:crisis_of_faith_heresiarch_scope } }
		create_character = {
			estate = estate_type:clergy_estate
			min_age = 30
			max_age = 60
			save_scope_as = crisis_of_faith_heresiarch_scope
		}
	}
	if = {
		limit = { exists = scope:crisis_of_faith_heresiarch_scope }
		set_variable = {
			name = crisis_of_faith_heresiarch_variable
			value = scope:crisis_of_faith_heresiarch_scope
		}
	}
}

clear_crisis_of_faith_heresiarch_effect = {
	if = {
		limit = { exists = var:crisis_of_faith_heresiarch_variable }
		remove_variable = crisis_of_faith_heresiarch_variable
	}
	if = {
		limit = { has_variable = crisis_of_faith_heresiarch_revealed_variable }
		remove_variable = crisis_of_faith_heresiarch_revealed_variable
	}
	set_variable = {
		name = crisis_of_faith_heresiarch_denounced_variable
		months = 12
	}
}
