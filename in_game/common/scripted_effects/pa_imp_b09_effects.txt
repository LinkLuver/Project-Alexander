#P:A Imported from Invictus: pass_down_bloodline_character_effect (common/scripted_effects/00_bloodlines_inheritance_inv.txt)
# Character scope. $BLOODLINE$ = a bloodline trait key. Only call it with traits that exist in the mod (none do yet).
pa_imp_pass_down_bloodline_character_effect = {
	if = {
		limit = {
			exists = father
			father = {
				has_trait = $BLOODLINE$
			}
			NOT = {
				has_trait = $BLOODLINE$
			}
		}
		add_trait = trait:$BLOODLINE$ # Imperator force_add_trait
	}
	if = {
		limit = {
			exists = mother
			mother = {
				has_trait = $BLOODLINE$
			}
			NOT = {
				has_trait = $BLOODLINE$
			}
		}
		add_trait = trait:$BLOODLINE$
	}
}

#P:A Imported from Imperator: appointment_office_skill_effect (common/scripted_effects/00_political_objective_effects.txt)
# Any scope; needs scope:appointment_target (a character). Stores the target's best skill in its variable appointment_office_skill:
# 1 = martial (EU5 mil), 2 = finesse (EU5 adm), 3 = charisma (EU5 dip), 4 = zeal (no EU5 equivalent, never set).
pa_imp_appointment_office_skill_effect = {
	if = {
		limit = {
			scope:appointment_target.mil >= scope:appointment_target.adm
			scope:appointment_target.mil >= scope:appointment_target.dip
			# scope:appointment_target.martial >= scope:appointment_target.zeal #P:A TODO No EU5 equivalent: zeal skill (EU5 characters have no zeal)
		}
		scope:appointment_target = {
			set_variable = {
				name = appointment_office_skill
				value = 1
			}
		}
	}
	else_if = {
		limit = {
			scope:appointment_target.adm >= scope:appointment_target.mil
			scope:appointment_target.adm >= scope:appointment_target.dip
			# scope:appointment_target.finesse >= scope:appointment_target.zeal #P:A TODO No EU5 equivalent: zeal skill
		}
		scope:appointment_target = {
			set_variable = {
				name = appointment_office_skill
				value = 2
			}
		}
	}
	else = {
		# Imperator: else_if charisma is the highest skill -> 3, else (zeal highest) -> 4
		#P:A TODO No EU5 equivalent: the zeal branch (value 4); without zeal, dip is the best skill whenever mil and adm are not
		scope:appointment_target = {
			set_variable = {
				name = appointment_office_skill
				value = 3
			}
		}
	}
}
