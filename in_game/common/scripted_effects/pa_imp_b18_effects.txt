#P:A Imported from Imperator: increment_state_investment_count (common/scripted_effects/00_event_effects.txt)
# Province scope (Imperator: state). Counts the investments made in the province.
pa_imp_increment_state_investment_count = {
	if = {
		limit = {
			NOT = {
				has_variable = num_state_investments
			}
		}
		set_variable = {
			name = num_state_investments
			value = 1
		}
	}
	else = {
		change_variable = {
			name = num_state_investments
			add = 1
		}
	}
}

#P:A Imported from Invictus: found_city_effect (common/scripted_effects/00_city_effects_inv.txt)
# Location scope (Imperator: province). Raises a settlement to a city (EU5: town).
pa_imp_found_city_effect = {
	change_location_rank = location_rank:town # Imperator set_city_status = city
	hidden_effect = {
		every_neighbor_location = {
			limit = {
				is_land = yes # Imperator NOR = { is_sea = yes terrain = riverine_terrain } (EU5 rivers and seas are water locations)
				location_rank = location_rank:rural_settlement # Imperator has_city_status = no
			}
			pa_imp_add_neighbor_city_modifier_effect = yes
		}
	}
	if = {
		limit = {
			pa_imp_food_trade_good_trigger = yes
			# NOT = { terrain = oasis } (EU5 has no oasis terrain)
		}
		# Remembers the food good (Imperator: "Must not be cleared!"; read when city status is revoked)
		if = {
			limit = { raw_material = goods:wheat } # Imperator grain
			set_variable = had_grain
		}
		else_if = {
			limit = { raw_material = goods:fish }
			set_variable = had_fish
		}
		else_if = {
			limit = { raw_material = goods:livestock } # Imperator cattle
			set_variable = had_cattle
		}
		else_if = {
			limit = { raw_material = goods:legumes } # Imperator vegetables
			set_variable = had_vegetables
		}
		else_if = {
			limit = { raw_material = goods:fruit } # Imperator fruits / dates (both map to EU5 fruit)
			set_variable = had_fruits
		}
		else_if = {
			limit = { raw_material = goods:olives } # Imperator olive
			set_variable = had_olive
		}
		hidden_effect = {
			pa_imp_replace_food_after_city_upgrade_effect = { VARIABLE_OR_SET_TRADE_GOODS = set_variable } #See comments in scripted effect before tampering with this
		}
		pa_imp_replace_food_after_city_upgrade_finish_effect = yes
		while = {
			count = 3
			random_pop = { #P:A TODO Approximate: Imperator turned 3 whole pops into citizens; EU5 turns size 1 of a pop into burghers each time (one Imperator pop converted 1:1 to EU5 size)
				limit = {
					NOT = { pop_type = pop_type:burghers } # Imperator citizen
				}
				if = {
					limit = { pop_size > 1 }
					split_pop = {
						size = 1
						type = pop_type:burghers
					}
				}
				else = {
					change_pop_type = pop_type:burghers
				}
			}
		}
	}
}

#P:A Imported from Invictus: replace_food_after_city_upgrade_effect (common/scripted_effects/00_event_effects_inv_1_0.txt)
# Location scope. VARIABLE_OR_SET_TRADE_GOODS = set_variable picks the replacement good as a location variable.
# Imperator also accepted set_trade_goods (to show the change in a tooltip); EU5 has no such effect and the swap itself is disabled (see pa_imp_replace_food_after_city_upgrade_finish_effect), so only set_variable is valid.
pa_imp_replace_food_after_city_upgrade_effect = {
	random_list = { #If set_variable is used it must be cleared, use the scripted effect after!
		5 = {
			modifier = {
				factor = 2
				OR = {
					pa_imp_in_region_upper_egypt_region = yes
					pa_imp_in_region_lower_egypt_region = yes
				}
			}
			$VARIABLE_OR_SET_TRADE_GOODS$ = papyrus
		}
		15 = { #Always pretty high chance
			$VARIABLE_OR_SET_TRADE_GOODS$ = cloth
			modifier = {
				factor = 1.5
				pa_imp_city_is_in_india_trigger = yes # Imperator: the nine Indian regions
			}
		}
		10 = { #Often high chance
			modifier = {
				factor = 2
				OR = {
					topography = wetlands # Imperator marsh
					climate = arid # Imperator desert / desert_hills (approximate)
				}
			}
			$VARIABLE_OR_SET_TRADE_GOODS$ = earthware
		}
		10 = {
			modifier = {
				factor = 2
				topography = wetlands # Imperator marsh
			}
			$VARIABLE_OR_SET_TRADE_GOODS$ = dye
		}
		5 = {
			modifier = {
				factor = 0
				NOR = {
					topography = mountains
					is_coastal = yes
				}
			}
			$VARIABLE_OR_SET_TRADE_GOODS$ = salt
		}
		5 = { #Unusual but not very dependent on where you are
			$VARIABLE_OR_SET_TRADE_GOODS$ = glass
		}
		10 = { #Only in certain regions
			modifier = {
				factor = 0
				NOR = {
					pa_imp_in_region_asia_region = yes
					pa_imp_in_region_greece_region = yes
					pa_imp_in_region_cilicia_region = yes
					pa_imp_in_region_assyria_region = yes
					pa_imp_in_region_mesopotamia_region = yes
					pa_imp_in_region_persis_region = yes
					pa_imp_in_region_media_region = yes
					pa_imp_in_region_bactriana_region = yes
					pa_imp_in_region_ariana_region = yes
					pa_imp_in_region_parthia_region = yes
					pa_imp_in_region_syria_region = yes
					pa_imp_in_region_palestine_region = yes
					pa_imp_city_is_in_india_trigger = yes #P:A TODO Approximate: Imperator listed the Indian regions except maru_region
					pa_imp_in_region_sogdiana_region = yes
					pa_imp_in_region_lower_egypt_region = yes
				}
			}
			modifier = {
				factor = 0.2
				OR = {
					pa_imp_in_region_asia_region = yes
					pa_imp_in_region_greece_region = yes
					pa_imp_in_region_cilicia_region = yes
					pa_imp_in_region_syria_region = yes
					pa_imp_in_region_palestine_region = yes
					pa_imp_in_region_lower_egypt_region = yes
				}
			}
			$VARIABLE_OR_SET_TRADE_GOODS$ = silk
		}
	}
}
