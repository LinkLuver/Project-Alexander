#P:A Imported from Imperator: rival_brothers_scope_effect (common/scripted_effects/00_event_effects.txt). Country scope.
# Saves the oldest qualifying son as scope:rival_brother_1 and one of his brothers as scope:rival_brother_2.
pa_imp_rival_brothers_scope_effect = {
	ordered_character = {
		limit = {
			is_eligible_heir = root #P:A TODO Approximate: Imperator ordered_successor (the line of succession); EU5 characters eligible as heir
			is_child_of = root.ruler
			# from_ruler_family = yes #P:A TODO Approximate: Imperator from_ruler_family; EU5 relies on is_child_of the ruler
			is_female = no
			age_in_years >= 18
			age_in_years <= 30
			pa_imp_has_any_illness_trigger = no
			employer = root
			save_temporary_scope_as = rival_brother_1_temp_e
			OR = {
				AND = {
					# can_get_rivals = yes
					root = {
						any_character = { # Imperator any_sibling
							is_sibling_of = scope:rival_brother_1_temp_e
							is_eligible_heir = root
							is_child_of = root.ruler
							is_female = no
							age_in_years >= 18
							age_in_years <= 30
							# can_get_rivals = yes # Is not rival and could be
							pa_imp_has_any_illness_trigger = no
							employer = root
							# NOT = { is_rival = scope:rival_brother_1_temp_e }
						}
					}
				}
				# AND = { num_of_rivals >= 1 any_rival = { ... } }
				#P:A TODO No EU5 equivalent: rival slots and existing rivalries between the brothers (EU5 characters have no rivals)
			}
		}
		order_by = age_in_days
		save_scope_as = rival_brother_1
		root = {
			random_character = { # Save second brother (Imperator random_sibling)
				limit = {
					is_sibling_of = scope:rival_brother_1
					is_child_of = root.ruler
					is_female = no
					age_in_years >= 18
					age_in_years <= 30
					# OR = { AND = { can_get_rivals = yes NOT = { is_rival = scope:rival_brother_1 } } AND = { num_of_rivals >= 1 is_rival = scope:rival_brother_1 } }
					#P:A TODO No EU5 equivalent: the brother can become (or already is) a rival (EU5 characters have no rivals)
					pa_imp_has_any_illness_trigger = no
				}
				# weight = { title_importance_modifier = yes } #P:A TODO No EU5 equivalent: Imperator weighting by office/title importance
				save_scope_as = rival_brother_2
			}
		}
	}
}

				trigger = {
					scope:rival_brothers_loser = {
						NOT = { has_trait = maimed }
					}
				}
				change_variable = {
					name = rival_brothers_wound
					add = 2

			}
			10 = {

					name = rival_brothers_wound
					add = 3
				}
			}
			20 = {
				trigger = {
					scope:rival_brothers_loser = {
						has_trait = maimed
						has_trait = scarred #P:A TODO Approximate: Imperator "wounded" trait; scarred used
					}
				}
				change_variable = {
					name = rival_brothers_wound
					add = 4
				}
			}
		}
		if = {
			limit = { var:rival_brothers_wound = 1 }

			limit = { var:rival_brothers_wound = 2 }

		}
		else_if = {
			limit = { var:rival_brothers_wound = 3 }
			scope:rival_brothers_loser = {

		}
		kill_character = { #P:A TODO Approximate: Imperator death_stabbing; EU5 death reason combat

			killer = scope:rival_brothers_victor
		}
		if = {
			limit = {
				scope:rival_brothers_victor = { is_alive = yes }
			}
			if = {
				limit = {
					scope:rival_brothers_victor = {
