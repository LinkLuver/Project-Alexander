bfts_seed_subject_year_count_effect = {
	if = {
		limit = {
			has_dlc = "d009_across_the_pillars"
			is_situation_active = situation:battle_for_the_strait
			country_exists = c:GRA
			this = c:GRA
		}
		if = {
			limit = {
				country_exists = c:MOR
				c:MOR ?= scope:overlord
			}
			situation:battle_for_the_strait = {
				if = {
					limit = { NOT = { has_variable = GRA_years_subject_under_MOR_variable } }
					set_variable = {
						name = GRA_years_subject_under_MOR_variable
						value = 1
					}
				}
			}
		}
		else_if = {
			limit = {
				country_exists = c:CAS
				c:CAS ?= scope:overlord
			}
			situation:battle_for_the_strait = {
				if = {
					limit = { NOT = { has_variable = GRA_years_subject_under_CAS_variable } }
					set_variable = {
						name = GRA_years_subject_under_CAS_variable
						value = 1
					}
				}
			}
		}
	}
}

cas_appoint_adelantado_mayor_effect = {
	set_variable = {
		name = cas_adelantado_mayor_character
		value = $target$
		years = modifier_duration_years_normal
	}
	if = { limit = { has_country_modifier = cas_adelantado_crown_estate_modifier } remove_country_modifier = cas_adelantado_crown_estate_modifier }
	else_if = { limit = { has_country_modifier = cas_adelantado_nobles_estate_modifier } remove_country_modifier = cas_adelantado_nobles_estate_modifier }
	if = {
		limit = { $target$ = { has_estate = estate_type:crown_estate } }
		add_country_modifier = { 
			modifier = cas_adelantado_crown_estate_modifier 
			years = modifier_duration_years_normal
			mode = add_and_extend 
		}
	}
	else_if = { 
		limit = { 
			$target$ = { has_estate = estate_type:nobles_estate } 
		}
		add_country_modifier = { 
			modifier = cas_adelantado_nobles_estate_modifier 
			years = modifier_duration_years_normal 
			mode = add_and_extend 
		}
	}
}

cas_repeal_adelantado_mayor_effect = {
	remove_variable = cas_adelantado_mayor_character
	if = { limit = { has_country_modifier = cas_adelantado_crown_estate_modifier } remove_country_modifier = cas_adelantado_crown_estate_modifier }
	if = { limit = { has_country_modifier = cas_adelantado_nobles_estate_modifier } remove_country_modifier = cas_adelantado_nobles_estate_modifier }
}
