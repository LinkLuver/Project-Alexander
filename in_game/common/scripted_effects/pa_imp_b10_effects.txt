#P:A Imported from Invictus (common/scripted_effects/00_event_effects_inv_1_0.txt)
# Country scope: rolls the outcome of a royal hunt into var:hunting_variable (< 0 death, 0 wounded, > 0 success).
# $ANIMAL_VALUE$: difficulty of the hunted animal (negative = more dangerous).
pa_imp_hunt_calc_effect = {
	hidden_effect = {
		set_variable = {
			name = hunting_variable
			value = {
				value = 0
				if = {
					limit = {
						exists = ruler
					}
					add = ruler.mil
				}
				divide = 5 # Imperator current_ruler.martial; EU5 skills are 5x Imperator's
				round = yes # Imperator skills are whole numbers, so the 'wounded' outcome (exactly 0) stays reachable
			}
			days = 30 # variable expires after 30 days, no need to remove
		}
		change_variable = {
			name = hunting_variable
			add = $ANIMAL_VALUE$
		}
		random_list = {
			1 = { # really bad hunt
				change_variable = {
					name = hunting_variable
					add = -5
				}
			}
			1 = { # bad hunt
				change_variable = {
					name = hunting_variable
					add = -3
				}
			}
			1 = { #unlucky hunt
				change_variable = {
					name = hunting_variable
					add = -1
				}
			}
			1 = { #lucky hunt
				change_variable = {
					name = hunting_variable
					add = 1
				}
			}
			1 = { #good hunt
				change_variable = {
					name = hunting_variable
					add = 3
				}
			}
			1 = { #really successful hunt
				change_variable = {
					name = hunting_variable
					add = 5
				}
			}
		}
	}
}
