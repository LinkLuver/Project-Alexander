#P:A Imported from Invictus (common/scripted_effects/00_event_effects_inv_1_0.txt)
# Country scope: rolls the outcome of a royal hunt into var:hunting_variable (< 0 death, 0 wounded, > 0 success).
# $ANIMAL_VALUE$: difficulty of the hunted animal (negative = more dangerous).
pa_imp_hunt_calc_effect = {
	hidden_effect = {
		set_variable = {
			name = hunting_variable
			value = {
				value = 0
				if = {
					limit = {
						exists = ruler
					}
					add = ruler.mil
				}
				divide = 5 # Imperator current_ruler.martial; EU5 skills are 5x Imperator's
				round = yes # Imperator skills are whole numbers, so the 'wounded' outcome (exactly 0) stays reachable
			}
			days = 30 # variable expires after 30 days, no need to remove
		}
		change_variable = {
			name = hunting_variable
			add = $ANIMAL_VALUE$
		}
		random_list = {
			1 = { # really bad hunt
				change_variable = {
					name = hunting_variable
					add = -5
				}
			}
			1 = { # bad hunt
				change_variable = {
					name = hunting_variable
					add = -3
				}
			}
			1 = { #unlucky hunt
				change_variable = {
					name = hunting_variable
					add = -1
				}
			}
			1 = { #lucky hunt
				change_variable = {
					name = hunting_variable
					add = 1
				}
			}
			1 = { #good hunt
				change_variable = {
					name = hunting_variable
					add = 3
				}
			}
			1 = { #really successful hunt
				change_variable = {
					name = hunting_variable
					add = 5
				}
			}
		}
	}
}

#P:A Imported from Indomita (common/scripted_effects/MPR_scripted_effects.txt)
# Location scope (Imperator province): adds up to $count$ special pops of $type$ (priest/merchant/architect/artist) as stacks
# of the <type>_in_province_mod location modifier, counted in var:<type>_in_territory and, on the owner, var:number_of_<type>_var (max 5 per country).
# The <type>_in_province_mod modifiers are owned by batch b20 (same names).
pa_imp_add_special_pop = {
	# SCOPE = location
	# text = tooltip to use - add_1_artist_tt, add_2_artist_tt, add_3_artist_tt, add_4_artist_tt, add_5_artist_tt, add_1_priest_tt, add_2_priest_tt, add_3_priest_tt, add_4_priest_tt, add_5_priest_tt, add_1_merchant_tt, add_2_merchant_tt, add_3_merchant_tt, add_4_merchant_tt, add_5_merchant_tt, add_1_architect_tt, add_2_architect_tt, add_3_architect_tt, add_4_architect_tt, add_5_architect_tt
	# type = priest/merchant/architect/artist
	# count = number of special pops to add (max is 5)
	custom_tooltip = $text$
	hidden_effect = {
		if = {
			limit = {
				NOT = { has_variable = $type$_in_territory }
			}
			set_variable = {
				name = $type$_in_territory
				value = 0
			}
		}

		if = {
			limit = {
				owner = {
					NOT = { has_variable = number_of_$type$_var }
				}
			}
			owner = {
				set_variable = {
					name = number_of_$type$_var
					value = 0
				}
			}
		}

		while = {
			count = $count$
			limit = {
				owner.var:number_of_$type$_var < 5
				var:$type$_in_territory >= 0
			}
			add_location_modifier = { # Imperator add_permanent_province_modifier
				modifier = $type$_in_province_mod
				days = -1
				mode = add
			}
			if = {
				limit = {
					NOT = { has_variable = $type$_in_territory }
				}
				set_variable = {
					name = $type$_in_territory
					value = 1
				}
			}
			else = {
				change_variable = {
					name = $type$_in_territory
					add = 1
				}
			}
			owner = {
				change_variable = {
					name = number_of_$type$_var
					add = 1
				}
			}
		}
	}
}
