#P:A Imported from Invictus (common/scripted_effects): annexation_character_cleanup
# Character scope: takes the character off every annexation list. Cleans lists, nothing else.
pa_imp_annexation_character_cleanup = {
	if = { limit = { is_in_list = first_annex_family_list } remove_from_list = first_annex_family_list }
	if = { limit = { is_in_list = second_annex_family_list } remove_from_list = second_annex_family_list }
	if = { limit = { is_in_list = third_annex_family_list } remove_from_list = third_annex_family_list }

#P:A Imported from Invictus (common/scripted_effects): country_kill_counter_effect
# Country scope: counts those who die but not those who vanish into obscurity.
pa_imp_country_kill_counter_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = country_kill_counter
			}
		}
		set_variable = {
			name = country_kill_counter
			value = 1
		}
	}
	else = {
		change_variable = {
			name = country_kill_counter
			add = 1
		}
	}
}

#P:A Imported from Imperator (common/scripted_effects/00_medical.txt): add_result_affliction_effect
# Character scope: gives the permanent affliction chosen by the calling event (variable medical_result_acquired),
# then clears the variable.
pa_imp_add_result_affliction_effect = {
	if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 1
			# NOT = { has_trait = barren }
		}
		# add_trait = barren #P:A TODO No EU5 equivalent: Imperator trait barren
	}
	else_if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 2
			NOT = {
				has_trait = castrated
			}
		}
		add_trait = trait:castrated
	}
	else_if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 3
			NOT = {
				has_trait = sickly # frail
			}
		}
		add_trait = trait:sickly # frail
	}
	else_if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 4
			NOT = {
				has_trait = blind
			}
		}
		if = {
			limit = { has_trait = one_eyed }
			remove_trait = trait:one_eyed
			add_trait = trait:blind
		}
		else = {
			add_trait = trait:one_eyed
		}
	}
	else_if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 5
			# NOT = { has_trait = paralysis }
		}
		# add_trait = paralysis #P:A TODO No EU5 equivalent: Imperator trait paralysis
	}
	else_if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 6
			NOT = {
				has_trait = blind
			}
		}
		if = {
			limit = { has_trait = one_eyed }
			remove_trait = trait:one_eyed
		}
		add_trait = trait:blind
	}
	else_if = {
		limit = {
			has_variable = medical_result_acquired
			var:medical_result_acquired = 7
			# NOT = { has_trait = brain_damaged }
		}
		# add_trait = brain_damaged #P:A TODO No EU5 equivalent: Imperator trait brain_damaged
	}
	if = {
		limit = {
			has_variable = medical_result_acquired
		}
		remove_variable = medical_result_acquired
	}
}
