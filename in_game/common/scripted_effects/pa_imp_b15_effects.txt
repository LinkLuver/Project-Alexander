#P:A Imported from Invictus (common/scripted_effects/00_galatian_invasion_effects.txt)
# Adds 1 to variable $VAR$, or sets it to 1 if it does not exist yet.
pa_imp_change_or_set_variable_effect = {
	if = {
		limit = { has_variable = $VAR$ }
		change_variable = {
			name = $VAR$
			add = 1
		}
	}
	else = {
		set_variable = {
			name = $VAR$
			value = 1
		}
	}
}

#P:A Imported from Imperator (common/scripted_effects/00_event_effects.txt)
# Location scope (Imperator: province). Rewards for sparing a conquered great city. Needs scope:special_province.
pa_imp_merciful_ruler_effect = {
	owner = {
		if = {
			limit = {
				government_type = government_type:monarchy
				exists = ruler
			}
			add_legitimacy = 10 # legitimacy_large
		}
		# current_ruler = { add_popularity = popularity_large }
		#P:A TODO No EU5 equivalent: +15 ruler popularity (EU5 characters have no popularity)
		scope:special_province = {
			add_location_modifier = {
				modifier = province_improvement_merciful_conqueror
				days = 3650
			}
		}
	}
}

#P:A Imported from Invictus (common/scripted_effects/00_event_effects_inv_1_0.txt)
# Location scope. Moderate looting of a conquered great city (scope:special_province).
pa_imp_neutral_conqueror_effect = {
	owner = {
		add_gold = {
			value = monthly_income_total # Imperator has_monthly_income
			multiply = 6
			min = 25
			max = 2000
			ceiling = yes
		}
		# current_ruler = { add_prominence = prominence_medium add_popularity = subtract_popularity_small }
		#P:A TODO No EU5 equivalent: +5 ruler prominence and -5 ruler popularity
		scope:special_province = {
			if = {
				limit = {
					has_location_modifier = province_diminishment_medium_looted
				}
				hidden_effect = {
					remove_location_modifier = province_diminishment_medium_looted
				}
			}
			add_location_modifier = {
				modifier = province_diminishment_medium_looted
				days = 3650
			}
			custom_tooltip = neutral_conqueror_effect_tooltip
			hidden_effect = {
				every_pop = { #P:A TODO Approximate: Imperator killed the least happy eighth of the pops; EU5 shrinks every pop by 12.5%
					add_pop_size = {
						value = pop_size
						multiply = -0.125
					}
				}
			}
			set_variable = {
				name = sacked_province
				days = 3650
			}
		}
	}
}

#P:A Imported from Invictus (common/scripted_effects/00_galatian_invasion_effects.txt)
# Character scope. Adds $POINTS$ to the character's tournament_points (Galatian tournament, galatian_invasion.100-.112, skipped).
pa_imp_award_tournament_points_effect = {
	if = {
		limit = { has_variable = tournament_points }
		change_variable = {
			name = tournament_points
			add = $POINTS$
		}
	}
	else = {
		set_variable = {
			name = tournament_points
			value = $POINTS$
		}
	}
}
