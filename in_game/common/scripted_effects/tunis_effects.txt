# country_first: country
# country_second: country
# For when two hafsid states get the same monarch. If there's a human involved, have the human country take precedence
# If TUN tag is involved, make sure the annexing country is TUN or that it becomes TUN
hafsid_succession_unite_two_countries = {
	if = {
		limit = {
			$country_first$ = {
				tag = TUN
			}
		}
		$country_first$ = {
			save_scope_as = tun_tag
		}
		$country_second$ = {
			save_scope_as = non_tun_tag
		}
	}
	else_if = {
		limit = {
			$country_second$ = {
				tag = TUN
			}
		}
		$country_second$ = {
			save_scope_as = tun_tag
		}
		$country_first$ = {
			save_scope_as = non_tun_tag
		}
	}


	if = {
		limit = {
			$country_first$ = {
				NOT = { this = $country_second$ }
			}
		}
		if = {
			limit = {
				$country_first$ = {
					is_ai = no
				}
			}
			$country_first$ = {
				annex_country = {
					country = $country_second$
					reason = CivilWar
					transfer = yes
				}
				$country_second$ = {
					save_scope_as = target
				}
				trigger_event_non_silently = flavor_tun.16
			}
		}
		else_if = {
			limit = {
				$country_second$ = {
					is_ai = no
				}
			}
			$country_second$ = {
				annex_country = {
					country = $country_first$
					reason = CivilWar
					transfer = yes
				}
				$country_first$ = {
					save_scope_as = target
				}
				trigger_event_non_silently = flavor_tun.16
			}
		}
		else_if = {
			limit = {
				exists = scope:tun_tag
			}
			scope:tun_tag = {
				annex_country = {
					country = scope:non_tun_tag
					reason = CivilWar
					transfer = yes
				}
			}
		}
		else = {
			$country_first$ = {
				annex_country = {
					country = $country_second$
					reason = CivilWar
					transfer = yes
				}
			}
		}
	}
}

hafsid_succession_effect = {
	# Special case for when Umar acceeds that needs to take precedence over normal hafsid succession mechanics
	if = {
		limit = {
			has_variable = tunisian_successor
			var:tunisian_successor = scope:new_ruler
		}
		if = {
			limit = {
				scope:old_ruler = character:tun_abu_yahya_abu_bakr_ii
				var:tunisian_successor = {
					is_alive = yes
					OR = {
						AND = {
							country_exists = c:GFS
							c:GFS.ruler ?= this
						}
						AND = { 
							country_exists = c:BIJ
							c:BIJ.ruler ?= this
						}
					}
				}
				character:tun_abu_hafs_umar = {
					is_alive = yes
					owner ?= ROOT
				}
			}
			trigger_event_non_silently = flavor_tun.7
		}
		remove_variable = tunisian_successor
	}
	# Otherwise, apply regular succession
	else_if = {
		limit = {
			is_hafsid_governor_trigger = yes
			NOT = { is_subject_type = fiefdom }
		}
		if = {
			limit = {
				scope:new_ruler = {
					any_ruling_countries = {
						NOT = { this = root }
						is_hafsid_governor_trigger = yes
						NOT = { is_subject_type = fiefdom }
					}
				}
			}
			save_scope_as = first_country
			scope:new_ruler = {
				random_ruling_countries = {
					limit = {
						NOT = { this = root }
						is_hafsid_governor_trigger = yes
						NOT = { is_subject_type = fiefdom }
					}
					save_scope_as = second_country
				}
			}
			trigger_event_silently = {
				on_action = on_hafsid_governor_death
			}
		}
	}
}

add_hafsid_governor_content = {
	set_variable = was_familial_governor
	if = {
		limit = {
			scope:actor ?= {
				has_reform = government_reform:tun_strong_familial_governors
			}
		}
		add_reform = government_reform:tun_strong_familial_governors
	}
	else_if = {
		limit = {
			scope:actor ?= {
				has_estate_privilege = estate_privilege:tun_arab_bedouins
			}
		}
		grant_estate_privilege = estate_privilege:tun_arab_bedouins
	}
	change_culture = $country$.culture
	change_religion = $country$.religion
}

# Create a local variable map "preferred_city" that will point the location to the closest eligible city.
cache_distance_map = {
	$country$ = {
		save_temporary_scope_as = country_temp_scope
	}
	scope:country_temp_scope = {
		every_owned_non_rural_location = {
			limit = {
				can_select_location_for_familial_governor = yes
				is_capital = no
			}
			add_to_list = cities
		}
		every_owned_location = {
			# First, establish default values pointing to capital
			add_to_local_variable_map = {
				name = preferred_city
				key = this
				value = scope:country_temp_scope.capital
			}
			add_to_local_variable_map = {
				name = preferred_city_score
				key = this
				value = "distance_to(scope:country_temp_scope.capital)"
			}

			save_scope_as = number_one
			# Now compare each city against the current lowest score
			every_in_list = {
				list = cities
				save_scope_as = number_two
				if = {
					limit = {
						"scope:number_one.distance_to(scope:number_two)" < "local_variable_map(preferred_city_score|scope:number_one)"
					}
					add_to_local_variable_map = {
						name = preferred_city
						key = scope:number_one
						value = scope:number_two
					}
					add_to_local_variable_map = {
						name = preferred_city_score
						key = scope:number_one
						value = "scope:number_one.distance_to(scope:number_two)"
					}
				}
			}
		}
	}
}

create_familial_governor_in_location_effect = {
	$location$ = {
		hidden_effect = { cache_distance_map = { country = $country$ } }# Build a map of locations closest to each city
		if = {
			limit = {
				$location$ = location:bejaia
				NOT = { country_exists = c:BIJ }
			}
		}
		create_country_from_location = {
			set_capital = $location$
			set_new_ruler = $character$
			make_subject_of = {
				target = $country$
				type = subject_type:familial_governor
			}
			add_hafsid_governor_content = { country = $country$ }
			save_scope_as = target_country
		}

		# Grant the new governor locations in the city's zone
		hidden_effect = {
			$location$ ={
				floodfill_locations = {
					limit = { 
						"local_variable_map(preferred_city|this)" ?= $location$
						owner ?= { NOT = { this = scope:target_country } }
					}
					change_location_owner = scope:target_country
				}
			}
		}
	}
}
