		limit = {
			NOT = { has_variable = crown_masterships_held }
		}

			name = crown_masterships_held
			value = 0
		}
	}
	change_variable = {
		name = crown_masterships_held
		add = 1
	}

		limit = {

			var:crown_masterships_held >= 3

			id = d009_iberian_order_events.63
			months = { 3 9 }
		}

		move_country = root
	}
	scope:target_order = {
		set_head_character = scope:crown_appointee
		change_religious_order_zeal = religious_order_moderate_penalty
	}
}
