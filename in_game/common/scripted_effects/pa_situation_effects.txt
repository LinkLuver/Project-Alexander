diadochi_wars_recalculate_top_3 = {
	if = {
		limit = {
			any_in_global_list = {
				variable = eligible_diadochi_list
				OR = {
					NOT = { country_exists = this }
					is_subject = yes
				}
			}
		}
		every_in_global_list = {
			variable = eligible_diadochi_list

			limit = { 
				OR = {
					NOT = { country_exists = this }
					is_subject = yes
				}
			}

			save_temporary_scope_as = target_country
			remove_list_global_variable = { 
				name = eligible_diadochi_list 
				target = scope:target_country
			}
		}
	}
	ordered_in_global_list = {  
		variable = eligible_diadochi_list
		limit = { is_subject = no }
		order_by = {
			
			add = {
				value = military_strength
			}
			add = {
				value = country_tax_base
			}
		} 
		max = 1
		check_range_bounds = no 
		situation:diadochi_wars = {
			set_variable = {
				name = strongest_diadochi_variable
				value = prev
			}
			set_variable = {
				name = strongest_strongest_diadochi_total_score_variable
				value = {
					add = var:strongest_diadochi_variable.military_strength
					add = var:strongest_diadochi_variable.country_tax_base 
				}
			}
		}
	}
	
	if = {
		limit = {
			global_variable_list_size = {
				name = eligible_diadochi_list
				value > 1
			}
		}
		ordered_in_global_list = {	#2nd strongest  
			variable = eligible_diadochi_list
			limit = { 
				this != situation:diadochi_wars.var:strongest_diadochi_variable 
				is_subject = no
			}
			order_by = {
				
				add = {
					value = military_strength
				}
				add = {
					value = country_tax_base
				}
			} 
			max = 1
			check_range_bounds = no
			situation:diadochi_wars = {
				set_variable = {
					name = second_strongest_strongest_diadochi_variable
					value = prev
				}
				set_variable = {
					name = second_strongest_strongest_diadochi_total_score_variable
					value = {
						add = var:second_strongest_strongest_diadochi_variable.military_strength
						add = var:second_strongest_strongest_diadochi_variable.country_tax_base 
					}
				}
			}
		}
	}
	else = {
		situation:diadochi_wars = {
			remove_variable = second_strongest_strongest_diadochi_variable
			remove_variable = second_strongest_strongest_diadochi_total_score_variable
		}
	}

	if = {
		limit = {
			global_variable_list_size = {
				name = eligible_diadochi_list
				value > 2
			}
		}
		ordered_in_global_list = { #3rd Strongest 
			variable = eligible_diadochi_list
			limit = { 
				this != situation:diadochi_wars.var:strongest_diadochi_variable 
				this != situation:diadochi_wars.var:second_strongest_strongest_diadochi_variable 
				is_subject = no
			}
			order_by = {
				
				add = {
					value = military_strength
				}
				add = {
					value = country_tax_base
				}
			} 
			max = 1
			check_range_bounds = no 
			situation:diadochi_wars = {
				set_variable = {
					name = third_strongest_strongest_diadochi_variable
					value = prev
				}
				set_variable = {
					name = third_strongest_strongest_diadochi_total_score_variable
					value = {
						add = var:third_strongest_strongest_diadochi_variable.military_strength
						add = var:third_strongest_strongest_diadochi_variable.country_tax_base
					}
				}
			}
		}
	}
	else = {
		situation:diadochi_wars = {
			remove_variable = third_strongest_strongest_diadochi_variable
			remove_variable = third_strongest_strongest_diadochi_total_score_variable
		}
	}
}


on_diadochi_occupy_location_effect = {

	if = {
		limit = {
			scope:recipient.ruler = {
				has_character_modifier = diadochi_wants_to_play
			} 
			scope:previous_owner.ruler = {
				has_character_modifier = diadochi_wants_to_play
				#is_subject = no
			}
		}
		#trigger_event_non_silently = {id = }
		scope:target_location = {	
			change_location_owner_forcefully = scope:recipient
		}
	}
	else_if = {
		limit = {
			scope:target_location = scope:previous_owner.capital
		}
		scope:previous_owner = {
			trigger_event_non_silently = {id = diadochi_wars.03 }
		}
	}
	else = {
		# offer to subjects to swap sides

	}
}
