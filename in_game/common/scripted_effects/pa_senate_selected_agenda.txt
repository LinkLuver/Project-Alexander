# Country scope. The calling interaction supplies a selected Parliament issue.
pa_senate_start_selected_agenda = {
	remove_variable = pa_senate_agenda_stability
	remove_variable = pa_senate_agenda_holdings
	remove_variable = pa_senate_agenda_works
	remove_variable = pa_senate_agenda_law
	remove_variable = pa_senate_agenda_war
	remove_variable = pa_senate_agenda_alliance
	set_variable = { name = pa_senate_vote_opt value = { value = var:pa_senate_optimates_seats multiply = var:pa_senate_optimates_approval divide = 125 subtract = var:pa_senate_optimates_defectors } }
	set_variable = { name = pa_senate_vote_boni value = { value = var:pa_senate_boni_seats multiply = var:pa_senate_boni_approval divide = 125 subtract = var:pa_senate_boni_defectors } }
	set_variable = { name = pa_senate_vote_pop value = { value = var:pa_senate_populares_seats multiply = var:pa_senate_populares_approval divide = 125 subtract = var:pa_senate_populares_defectors } }
	if = {
		limit = { $issue$ = parliament_issue:pa_senate_war_debate }
		set_variable = pa_senate_agenda_war
		change_variable = { name = pa_senate_vote_opt add = 8 }
		change_variable = { name = pa_senate_vote_boni subtract = 4 }
		change_variable = { name = pa_senate_vote_pop add = 5 }
		if = { limit = { var:pa_senate_optimates_approval < 35 } set_variable = { name = pa_senate_vote_opt value = 0 } }
		if = { limit = { var:pa_senate_boni_approval < 70 } set_variable = { name = pa_senate_vote_boni value = 0 } }
		if = { limit = { var:pa_senate_populares_approval < 40 } set_variable = { name = pa_senate_vote_pop value = 0 } }
	}
	else_if = {
		limit = { $issue$ = parliament_issue:pa_senate_alliance_debate }
		set_variable = pa_senate_agenda_alliance
		change_variable = { name = pa_senate_vote_opt add = 2 }
		change_variable = { name = pa_senate_vote_boni add = 8 }
		change_variable = { name = pa_senate_vote_pop add = 3 }
		if = { limit = { var:pa_senate_optimates_approval < 50 } set_variable = { name = pa_senate_vote_opt value = 0 } }
		if = { limit = { var:pa_senate_boni_approval < 35 } set_variable = { name = pa_senate_vote_boni value = 0 } }
		if = { limit = { var:pa_senate_populares_approval < 55 } set_variable = { name = pa_senate_vote_pop value = 0 } }
	}
	else_if = {
		limit = { $issue$ = parliament_issue:pa_senate_holdings_debate }
		set_variable = pa_senate_agenda_holdings
		change_variable = { name = pa_senate_vote_opt add = 9 }
		change_variable = { name = pa_senate_vote_boni add = 2 }
		change_variable = { name = pa_senate_vote_pop subtract = 5 }
		if = { limit = { var:pa_senate_optimates_approval < 30 } set_variable = { name = pa_senate_vote_opt value = 0 } }
		if = { limit = { var:pa_senate_boni_approval < 45 } set_variable = { name = pa_senate_vote_boni value = 0 } }
		if = { limit = { var:pa_senate_populares_approval < 70 } set_variable = { name = pa_senate_vote_pop value = 0 } }
	}
	else_if = {
		limit = { $issue$ = parliament_issue:pa_senate_works_debate }
		set_variable = pa_senate_agenda_works
		change_variable = { name = pa_senate_vote_opt subtract = 4 }
		change_variable = { name = pa_senate_vote_boni add = 9 }
		change_variable = { name = pa_senate_vote_pop add = 3 }
		if = { limit = { var:pa_senate_optimates_approval < 65 } set_variable = { name = pa_senate_vote_opt value = 0 } }
		if = { limit = { var:pa_senate_boni_approval < 35 } set_variable = { name = pa_senate_vote_boni value = 0 } }
		if = { limit = { var:pa_senate_populares_approval < 45 } set_variable = { name = pa_senate_vote_pop value = 0 } }
	}
	else = {
		if = { limit = { $issue$ = parliament_issue:pa_senate_stability_debate } set_variable = pa_senate_agenda_stability }
		change_variable = { name = pa_senate_vote_opt add = 2 }
		change_variable = { name = pa_senate_vote_boni add = 8 }
		change_variable = { name = pa_senate_vote_pop add = 2 }
		if = { limit = { var:pa_senate_optimates_approval < 45 } set_variable = { name = pa_senate_vote_opt value = 0 } }
		if = { limit = { var:pa_senate_boni_approval < 25 } set_variable = { name = pa_senate_vote_boni value = 0 } }
		if = { limit = { var:pa_senate_populares_approval < 45 } set_variable = { name = pa_senate_vote_pop value = 0 } }
	}
	pa_senate_open_motion_debate = { issue = $issue$ }
	set_variable = { name = pa_senate_motion_cooldown value = yes months = 6 }
}
