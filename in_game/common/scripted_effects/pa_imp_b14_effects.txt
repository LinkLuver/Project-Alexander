#P:A Imported from Imperator: demand_office_skill_tooltip_effect (common/scripted_effects/00_event_effects.txt)
# Any scope: tooltip naming the skill the office scope:demand_office_event_scope wants needs (value set by pa_imp_demand_office_skill_effect).
pa_imp_demand_office_skill_tooltip_effect = {
	if = {
		limit = { scope:demand_office_event_scope.var:demand_office_skill = 1 }
		custom_tooltip = scheme.8.martialtooltip
	}
	else_if = {
		limit = { scope:demand_office_event_scope.var:demand_office_skill = 2 }
		custom_tooltip = scheme.8.finessetooltip
	}
	else_if = {
		limit = { scope:demand_office_event_scope.var:demand_office_skill = 3 }
		custom_tooltip = scheme.8.charismatooltip
	}
	else_if = {
		limit = { scope:demand_office_event_scope.var:demand_office_skill = 4 }
		custom_tooltip = scheme.8.zealtooltip
	}
}

#P:A Imported from Imperator: reduce_province_rank_effect (common/scripted_effects/00_city_rank_effects.txt)
# Location scope: lowers the location one rank.
#P:A TODO Approximate: Imperator city -> settlement and metropolis -> city mapped to EU5 town -> rural_settlement and city -> town
pa_imp_reduce_province_rank_effect = {
	if = {
		limit = {
			location_rank = location_rank:town # Imperator has_province_rank = city
		}
		change_location_rank = location_rank:rural_settlement # Imperator set_city_status = settlement
	}
	else_if = {
		limit = {
			location_rank = location_rank:city # Imperator has_province_rank = city_metropolis
		}
		change_location_rank = location_rank:town # Imperator set_city_status = city
	}
}
