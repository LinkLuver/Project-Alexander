# Holy-war (Crusade / Jihad) shared scripted effects.

crusade_track_contribution = {
	every_international_organization_member = {
		if = {
			limit = { NOT = { has_variable = holy_war_contribution } }
			set_variable = { name = holy_war_contribution value = 0 }
		}
		change_variable = {
			name = holy_war_contribution
			add = "payment_contribution(root|crusade_treasury_contribution)"
		}
	}
}

holy_war_snapshot_participation = {
	if = {
		limit = { exists = var:active_war }
		save_scope_as = hw_io
		var:active_war = { save_scope_as = hw_current_war }
		remove_variable = min_holy_war_participation
		every_international_organization_member = {
			set_variable = {
				name = holy_war_participation
				value = "war_participation(scope:hw_current_war)"
			}
			if = {
				limit = {
					OR = {
						NOT = { exists = scope:hw_io.var:min_holy_war_participation }
						var:holy_war_participation < scope:hw_io.var:min_holy_war_participation
					}
				}
				scope:hw_io = {
					set_variable = {
						name = min_holy_war_participation
						value = prev.var:holy_war_participation
					}
				}
			}
		}
	}
}

holy_war_payout_equal_share = {
	if = {
		limit = {
			total_members > 0
			gold > 0
		}
		save_scope_as = hw_io
		set_variable = {
			name = hw_payout_each
			value = {
				value = gold
				multiply = 0.2
				divide = total_members
			}
		}
		every_international_organization_member = {
			add_gold = scope:hw_io.var:hw_payout_each
		}
		add_gold = {
			value = var:hw_payout_each
			multiply = total_members
			multiply = -1
		}
		remove_variable = hw_payout_each
	}
}

holy_war_payout_war_end = {
	if = {
		limit = {
			total_members > 0
			gold > 0
		}
		save_scope_as = hw_io
		set_variable = { name = hw_remaining value = gold }

		set_variable = { name = hw_total_weight value = 0 }
		every_international_organization_member = {
			limit = { has_variable = holy_war_participation }
			scope:hw_io = {
				change_variable = {
					name = hw_total_weight
					add = prev.var:holy_war_participation
				}
			}
		}
		if = {
			limit = { var:hw_total_weight > 0 }
			every_international_organization_member = {
				limit = { has_variable = holy_war_participation }
				add_gold = {
					value = scope:hw_io.var:hw_remaining
					multiply = var:holy_war_participation
					divide = scope:hw_io.var:hw_total_weight
				}
			}
		}
		else = {
			set_variable = { name = hw_total_weight value = 0 }
			every_international_organization_member = {
				limit = { has_variable = holy_war_contribution }
				scope:hw_io = {
					change_variable = {
						name = hw_total_weight
						add = prev.var:holy_war_contribution
					}
				}
			}
			if = {
				limit = { var:hw_total_weight > 0 }
				every_international_organization_member = {
					limit = { has_variable = holy_war_contribution }
					add_gold = {
						value = scope:hw_io.var:hw_remaining
						multiply = var:holy_war_contribution
						divide = scope:hw_io.var:hw_total_weight
					}
				}
			}
			else = {
				set_variable = {
					name = hw_remaining_each
					value = {
						value = var:hw_remaining
						divide = total_members
					}
				}
				every_international_organization_member = {
					add_gold = scope:hw_io.var:hw_remaining_each
				}
				remove_variable = hw_remaining_each
			}
		}

		add_gold = {
			value = var:hw_remaining
			multiply = -1
		}
		every_international_organization_member = {
			remove_variable = holy_war_contribution
			remove_variable = holy_war_participation
		}
		remove_variable = hw_remaining
		remove_variable = hw_total_weight
		remove_variable = min_holy_war_participation
	}
}

holy_war_discipline_penalty_effect = {
	if = {
		limit = { scope:discipline_io = { this = international_organization:crusade } }
		if = {
			limit = { excommunicate_target_trigger = yes }
			excommunicate_effect = yes
		}
		else = {
			add_religious_influence = {
				value = religious_influence_severe_bonus
				multiply = -1
			}
		}
	}
	else = {
		add_legitimacy = {
			value = legitimacy_severe_bonus
			multiply = -1
		}
	}
}

holy_war_desertion_penalty_effect = {
	every_international_organization_member = {
		limit = { this != root }
		add_opinion = { target = root modifier = opinion_abandoned_holy_war }
	}
	root = {
		add_prestige = prestige_severe_penalty
		if = {
			limit = { religion = { has_religious_influence = yes } }
			add_religious_influence = religious_influence_severe_penalty
		}
		else = {
			add_legitimacy = legitimacy_severe_penalty
		}
	}
}

holy_war_levy_payment_effect = {
	set_variable = { name = hw_levy_before value = gold }
	add_gold = {
		value = monthly_income_trade_and_tax
		multiply = $tithe$
		min = $min$
		max = $max$
		multiply = -1
	}
	if = {
		limit = {
			exists = scope:levy_io
			exists = var:hw_levy_before
		}
		scope:levy_io = {
			add_gold = {
				value = prev.var:hw_levy_before
				subtract = prev.gold
			}
		}
	}
	remove_variable = hw_levy_before
}

holy_war_plunder_to_treasury_effect = {
	if = {
		limit = { exists = scope:recipient.var:war_target }
		scope:recipient.var:war_target = { save_scope_as = plunder_target }
		custom_tooltip = plunder_the_conquered_tt
		hidden_effect = {
			scope:actor = { set_variable = { name = hw_plunder_before value = gold } }
		}
		scope:actor = {
			every_controlled_location = {
				limit = { owner = scope:plunder_target }
				scope:actor = { loot_location = prev }
			}
		}
		hidden_effect = {
			scope:actor = {
				if = {
					limit = { exists = var:hw_plunder_before }
					if = {
						limit = { gold > var:hw_plunder_before }
						set_variable = {
							name = hw_plunder_gained
							value = {
								value = gold
								subtract = var:hw_plunder_before
							}
						}
						add_gold = {
							value = var:hw_plunder_gained
							multiply = -1
						}
						scope:recipient = {
							add_gold = { value = prev.var:hw_plunder_gained }
						}
						remove_variable = hw_plunder_gained
					}
					remove_variable = hw_plunder_before
				}
			}
			scope:recipient = {
				change_variable = { name = religious_zeal add = 5 }
				clamp_variable = { name = religious_zeal max = 100 min = 0 }
			}
		}
	}
}

holy_war_scaled_gold_transfer_effect = {
	if = {
		limit = { exists = $payee$ }
		$payer$ = {
			add_gold = {
				save_temporary_value_as = {
					name = hw_transfer_sum
					value = {
						value = capital_wealth
						multiply = 0.2
						add = {
							value = country_economical_base
							multiply = 0.05
						}
						multiply = $scale$
						min = 0
						max = {
							value = 5000
							multiply = $scale$
						}
					}
				}
				value = scope:hw_transfer_sum
				multiply = -1
			}
		}
		$payee$ = {
			add_gold = { value = scope:hw_transfer_sum }
		}
	}
	else = {
		custom_tooltip = holy_war_scaled_gold_transfer_fallback_tt
		hidden_effect = {
			$payer$ = {
				add_gold = {
					save_temporary_value_as = {
						name = hw_transfer_sum
						value = {
							value = capital_wealth
							multiply = 0.2
							add = {
								value = country_economical_base
								multiply = 0.05
							}
							multiply = $scale$
							min = 0
							max = {
								value = 5000
								multiply = $scale$
							}
						}
					}
					value = scope:hw_transfer_sum
					multiply = -1
				}
			}
		}
	}
}
