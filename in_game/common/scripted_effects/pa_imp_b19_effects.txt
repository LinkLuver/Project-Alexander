#P:A Imported from Invictus: release_AIG_effect (common/scripted_effects/00_achaea_effects.txt). Country scope.
# Releases Aigion (Imperator p:444) as its own country if we own it.
pa_imp_release_AIG_effect = {
	if = {
		limit = {
			#NOT = { exists = c:AIG }
			owns = location:vostitsa # Imperator p:444 Aigion
		}
		custom_tooltip = dhe_achaean_league.7.AIG.TT
		hidden_effect = {
			location:vostitsa = {
				create_country_from_location = { #P:A TODO Approximate: Imperator created the country with tag AIG (change_country_tag, its flag, colour and heritage aigion_heritage); EU5 creates a new country named with the mod's AIG loc
					name = {
						name = AIG
						adjective = AIG_ADJ
					}
					save_scope_as = aigion_country
					change_religion = religion:hellenic_pantheon # Imperator roman_pantheon (Greek cities)
					change_government_type = government_type:republic #P:A TODO Approximate: Imperator oligarchic_republic
				}
			}
		}
	}
}

#P:A Imported from Indomita: set_imperial_power (common/scripted_effects/EE_scripted_effects.txt). Country scope.
# Applies the imperial modifier of the chosen focus (var:imperial_power = flag:political / flag:warfare / flag:economic).
pa_imp_set_imperial_power = {
	if = {
		limit = {
			has_variable = imperial_power
		}
		if = {
			limit = {
				var:imperial_power = flag:political
			}
			pa_imp_add_imperial_political_mods = yes
		}
		if = {
			limit = {
				var:imperial_power = flag:warfare
			}
			pa_imp_add_imperial_military_mods = yes
		}
		if = {
			limit = {
				var:imperial_power = flag:economic
			}
			pa_imp_add_imperial_economic_mods = yes
		}
	}
}

#P:A Imported from Indomita: add_imperial_economic_mods (common/scripted_effects/EE_scripted_effects.txt). Country scope.
pa_imp_add_imperial_economic_mods = {
	pa_imp_remove_all_imperial_modifier = yes
	if = {
		limit = {
			num_locations >= 3000 #P:A TODO Approximate: Imperator num_of_cities (territories); EU5 owned locations
		}
		add_country_modifier = {
			modifier = imperial_economic_mod_1
			years = -1
		}
		remove_variable = imperial_power_level
		set_variable = {
			name = imperial_power_level
			value = 5
		}
	}
	else_if = {
		limit = {
			num_locations >= 2000
			num_locations < 3000
		}
		add_country_modifier = {
			modifier = imperial_economic_mod_2
			years = -1
		}
		remove_variable = imperial_power_level
		set_variable = {
			name = imperial_power_level
			value = 4
		}
	}
	else_if = {
		limit = {
			num_locations >= 1500
			num_locations < 2000
		}
		add_country_modifier = {
			modifier = imperial_economic_mod_3
			years = -1
		}
		remove_variable = imperial_power_level
		set_variable = {
			name = imperial_power_level
			value = 3
		}
	}
	else_if = {
		limit = {
			num_locations >= 1000
			num_locations < 1500
		}
		add_country_modifier = {
			modifier = imperial_economic_mod_4
			years = -1
		}
		remove_variable = imperial_power_level
		set_variable = {
			name = imperial_power_level
			value = 2
		}
	}
	else = {
		add_country_modifier = {
			modifier = imperial_economic_mod_5
			years = -1
		}
		remove_variable = imperial_power_level
		set_variable = {
			name = imperial_power_level
			value = 1
		}
	}
}
