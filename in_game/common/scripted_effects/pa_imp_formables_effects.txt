#P:A Imported from Imperator: add_1_free_province_investments (common/scripted_effects/00_decision_effects.txt)
# Country scope.
pa_imp_add_1_free_province_investments = {
	#P:A TODO No EU5 equivalent: free province investments (Imperator province investments do not exist in EU5); the variable is still kept, but nothing in EU5 reads it
	if = {
		limit = { NOT = { has_variable = free_investments } }
		set_variable = { name = free_investments value = 1 }
	}
	else = {
		change_variable = { name = free_investments add = 1 }
	}
}

#P:A Imported from Imperator: capital_formable_small_effect (common/scripted_effects/00_event_effects.txt)
# Imperator province scope -> EU5 location scope. Imperator created 2 citizen and 2 freemen pops of the state's culture and religion.
pa_imp_capital_formable_small_effect = {
	custom_tooltip = capital_formable_small_effect_tooltip
	hidden_effect = {
		ordered_pop = {
			order_by = pop_size
			max = 1
			check_range_bounds = no
			save_temporary_scope_as = pa_majority_pop
		}
		if = { #P:A TODO Approximate: Imperator pop count converted 1:1 to EU5 size; the state culture/religion is taken from the location's largest pop
			limit = { exists = scope:pa_majority_pop }
			add_pop = { type = pop_type:burghers size = 2 culture = scope:pa_majority_pop.culture religion = scope:pa_majority_pop.religion } # Imperator citizen
			add_pop = { type = pop_type:peasants size = 2 culture = scope:pa_majority_pop.culture religion = scope:pa_majority_pop.religion } # Imperator freemen
		}
	}
}

#P:A Imported from Imperator: formable_capital_modifier_tribal_effect (common/scripted_effects/00_event_effects.txt)
# Imperator province scope -> EU5 location scope.
pa_imp_formable_capital_modifier_tribal_effect = {
	if = {
		limit = {
			NOT = { has_location_modifier = formable_capital_tribal_modifier }
		}
		add_location_modifier = {
			modifier = formable_capital_tribal_modifier
			days = -1
		}
	}
}

#P:A Imported from Imperator: formable_capital_modifier_large_effect (common/scripted_effects/00_event_effects.txt)
# Imperator province scope -> EU5 location scope.
pa_imp_formable_capital_modifier_large_effect = {
	if = {
		limit = { has_location_modifier = formable_capital_tribal_modifier }
		remove_location_modifier = formable_capital_tribal_modifier
	}
	if = {
		limit = { has_location_modifier = formable_capital_modifier }
		remove_location_modifier = formable_capital_modifier
	}
	if = {
		limit = {
			NOT = { has_location_modifier = formable_capital_huge_modifier }
		}
		add_location_modifier = {
			modifier = formable_capital_huge_modifier
			days = -1
		}
	}
}

#P:A Imported from Invictus: tribal_formable_government_change_effect (common/scripted_effects)
# Country scope. Tribes that are not yet a federation become one; everyone else gets 150 political influence.
pa_imp_tribal_formable_government_change_effect = {
	if = {
		limit = {
			government_type = government_type:tribe # Imperator is_tribal = yes
			NOT = { has_reform = government_reform:tribal_federation }
		}
		pa_imp_change_ir_government_effect = { TYPE = tribe REFORM = tribal_federation }
	}
	else = {
		add_political_influence = 150
	}
}

#P:A Imported from Invictus: formable_area_conquest_effect (common/scripted_effects)
# Country scope, parameter TRIGGER (a location-scope pa_imp_in_area/region trigger). Every location of the Imperator area held by an
# AI country goes to the new country (player-owned land stays).
#P:A TODO Approximate: Imperator also spared its province 2108; EU5 uses the geography trigger standing in for the Imperator area
pa_imp_formable_conquest_effect = {
	every_country = {
		limit = {
			is_ai = yes
			NOT = { this = root }
		}
		every_owned_location = {
			limit = { $TRIGGER$ = yes }
			change_location_owner = root
		}
	}
}

#P:A Imported from Invictus: formable_province_conquest_effect (common/scripted_effects)
# Imperator province scope -> EU5 location scope. An AI-held location goes to the new country (root).
pa_imp_formable_province_conquest_effect = {
	if = {
		limit = {
			NOT = { owner ?= root }
			owner ?= { is_ai = yes }
		}
		change_location_owner = root
	}
}
