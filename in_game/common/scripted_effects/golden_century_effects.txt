gc_change_colonial_dominance = {
	custom_description = {
		text = gc_change_colonial_dominance
		value = $value$
		situation:golden_century = {
			change_variable = { name = gc_colonial_dominance add = $value$ }
		}
	}
	gc_mirror_dominance_to_country = yes
}

gc_change_european_dominance = {
	custom_description = {
		text = gc_change_european_dominance
		value = $value$
		situation:golden_century = {
			change_variable = { name = gc_european_dominance add = $value$ }
		}
	}
	gc_mirror_dominance_to_country = yes
}

gc_mirror_dominance_to_country = {
	if = {
		limit = { exists = global_var:battle_for_the_strait_winner_country }
		global_var:battle_for_the_strait_winner_country = {
			set_variable = {
				name = gc_european_dominance
				value = situation:golden_century.var:gc_european_dominance
			}
			set_variable = {
				name = gc_colonial_dominance
				value = situation:golden_century.var:gc_colonial_dominance
			}
		}
	}
}
