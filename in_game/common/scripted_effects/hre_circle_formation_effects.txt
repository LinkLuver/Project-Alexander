# Scripted effects for Imperial Circle formation (HRE-specific)

# Kicks off the one-time dynamic formation process when the Imperial Circles law passes.
hre_start_circle_formation = {
	hidden_effect = {
		international_organization:hre = {
			hre_start_circle_formation_effect = { }
		}
	}
}

# Adds scope:actor to the specified formation circle.
# Usage: hre_join_imperial_circle_effect = { circle = <circle_scope> }
hre_join_imperial_circle_effect = {
	$circle$ = {
		add_country_to_circle = scope:actor
	}
}
