pacify_rebel_pops = {

	custom_tooltip = {
		text = every_supporter_pop_gains_20_satisfaction_tt
		every_pops_supporting_rebel = { add_pop_satisfaction = pop_satisfaction_extreme_bonus }
	}
}


grant_benefits_to_estate = {

	# a new privilege
	root.owner = {
		"estate(root.estate_type)" ?= {
			random_possible_privilege = {
				root.owner = {
					grant_estate_privilege = prev
				}
			}
		}
	}
	
	
	# a law change
	root.owner = {
		random_possible_policy = {
			limit = {
				any_estate_type_preferring = { this = root.estate_type }
				law = {
					NOT = { is_locked_for = root.owner }
				}
			}
			root.owner = {
				add_policy = prev
			}
		}
	}
	
	owner = {
		add_estate_satisfaction = {
			type = root.estate_type
			value = estate_satisfaction_extreme_bonus
		}
	}
}
