#P:A Imported from Imperator: befriend_event_counter_variable_effect (common/scripted_effects/00_event_effects.txt)
# Character scope (the befriending character): counts the befriend events seen so far.
pa_imp_befriend_event_counter_variable_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = friend_event_counter
			}
		}
		set_variable = {
			name = friend_event_counter
			value = 1
		}
	}
	else = {
		change_variable = {
			name = friend_event_counter
			add = 1
		}
	}
}

#P:A Imported from Imperator: fire_next_friend_event_effect (common/scripted_effects/00_event_effects.txt)
# Character scope (the befriending character): fires the next step of the befriend chain through the custom on_actions
# befriend_resolution (progress >= 6), befriend_fail (3+ events seen) or befriend_pulse. Imperator fired them on the character;
# the befriend events are country events here, so they are fired on the character's employer with the character saved as scope:pa_character.
pa_imp_fire_next_friend_event_effect = {
	save_scope_as = pa_character
	if = {
		limit = {
			has_variable = friend_progress
			var:friend_progress >= 6
		}
		employer = {
			trigger_event_non_silently = {
				on_action = befriend_resolution
				days = { 15 60 }
			}
		}
	}
	else_if = {
		limit = {
			has_variable = friend_event_counter
			var:friend_event_counter >= 3
		}
		employer = {
			trigger_event_non_silently = {
				on_action = befriend_fail
				days = { 15 60 }
			}
		}
	}
	else = {
		employer = {
			trigger_event_non_silently = {
				on_action = befriend_pulse
				days = { 15 60 }
			}
		}
	}
}

#P:A Imported from Imperator: demand_office_skill_effect (common/scripted_effects/00_event_effects.txt)
# Any scope: stores on scope:demand_office_event_scope which skill it is best at (1 martial, 2 finesse, 3 charisma, 4 zeal).
#P:A TODO Approximate: Imperator martial/finesse/charisma -> EU5 mil/adm/dip; zeal has no EU5 equivalent, so the zeal comparisons are dropped and value 4 (a zeal office) is never picked
pa_imp_demand_office_skill_effect = {
	if = {
		limit = {
			scope:demand_office_event_scope.mil >= scope:demand_office_event_scope.adm
			scope:demand_office_event_scope.mil >= scope:demand_office_event_scope.dip
			# scope:demand_office_event_scope.martial >= scope:demand_office_event_scope.zeal
		}
		scope:demand_office_event_scope = {
			set_variable = {
				name = demand_office_skill
				value = 1
			}
		}
	}
	else_if = {
		limit = {
			scope:demand_office_event_scope.adm >= scope:demand_office_event_scope.mil
			scope:demand_office_event_scope.adm >= scope:demand_office_event_scope.dip
			# scope:demand_office_event_scope.finesse >= scope:demand_office_event_scope.zeal
		}
		scope:demand_office_event_scope = {
			set_variable = {
				name = demand_office_skill
				value = 2
			}
		}
	}
	else_if = {
		limit = {
			scope:demand_office_event_scope.dip >= scope:demand_office_event_scope.mil
			scope:demand_office_event_scope.dip >= scope:demand_office_event_scope.adm
			# scope:demand_office_event_scope.charisma >= scope:demand_office_event_scope.zeal
		}
		scope:demand_office_event_scope = {
			set_variable = {
				name = demand_office_skill
				value = 3
			}
		}
	}
	else = {
		scope:demand_office_event_scope = {
			set_variable = {
				name = demand_office_skill
				value = 4
			}
		}
	}
}

#P:A Imported from Imperator: demand_office_military_scope_effect (common/scripted_effects/00_event_effects.txt)
# Country scope.
#P:A TODO Approximate: Imperator looked for the holder of a martial office (monarchy marshal / master of the guard, republic tribune of the treasury / of the soldiers, tribe warchief / bodyguard), preferring one with no better martial skill than scope:demand_office_event_scope, else any holder of those offices. #P:A Imported from Imperator: demand_office_transfer_effect (common/scripted_effects/00_event_effects.txt)
# Country scope: scope:demand_office_event_scope takes the office it wants, displacing scope:demand_office_old_officer_scope.
#P:A TODO Approximate: Imperator gave the character the old holder's office (or, with no holder, the office matching its best skill: monarchy marshal / physician / royal tutor / philosopher, republic tribune of the treasury / plebeian aedile / praetor / augur, tribe warchief / apothecary / arbitrator / wise person) and stripped both characters' positions. EU5 has no offices: both lose their positions (pa_imp_remove_all_positions: cabinet seat, command) and the character takes a free cabinet seat
pa_imp_demand_office_transfer_effect = {
	hidden_effect = {
		scope:demand_office_event_scope = { pa_imp_remove_all_positions = yes }
		if = {
			limit = {
				exists = scope:demand_office_old_officer_scope
			}
			scope:demand_office_old_officer_scope = { pa_imp_remove_all_positions = yes }
		}
	}
	if = {
		limit = {
			any_cabinet = {
				NOT = { exists = cabinet_member }
			}
		}
		random_cabinet = {
			limit = {
				NOT = { exists = cabinet_member }
			}
			add_to_cabinet = scope:demand_office_event_scope
		}
	}
}
