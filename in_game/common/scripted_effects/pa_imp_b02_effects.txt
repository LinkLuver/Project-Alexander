#P:A Imported from Invictus: set_primary_culture_cleanup_effect (common/scripted_effects/00_decision_effects_inv.txt)
# Country scope. Parameters:
#   NEW_PRIMARY_CULTURE = a bare mod culture key (e.g. macedonian), as the Imperator callers pass it
#   MAKE_OLD_CULTURE_INTEGRATED = yes/no: keep the old primary culture as an accepted culture
pa_imp_set_primary_culture_cleanup_effect = {
	if = {
		limit = {
			NOT = { culture = culture:$NEW_PRIMARY_CULTURE$ }
		}
		#P:A TODO No EU5 equivalent: Imperator removed the country-culture rights modifiers of the old and new primary
		#   culture (trial_right, citizenship_by_adoption, land_protection, patronizing_literature, bestowed_city_rights,
		#   promoted_administrators_in_subcultures, honor_guard, right_of_marriage, torture_protection,
		#   right_to_enter_contracts, inheritance_right, colonia_founded, officer_right and their country modifiers)
		#   and integrated the new culture; EU5 has no country-culture modifiers or integration
		culture = {
			save_scope_as = old_primary_culture
		}
		change_culture = culture:$NEW_PRIMARY_CULTURE$
		set_variable = {
			name = integrate_culture_yes_no
			value = $MAKE_OLD_CULTURE_INTEGRATED$
		}
		if = {
			limit = {
				always = $MAKE_OLD_CULTURE_INTEGRATED$
				NOT = { has_accepted_culture = scope:old_primary_culture }
			}
			add_accepted_culture = scope:old_primary_culture #P:A TODO Approximate: Imperator kept the old primary culture integrated; EU5 accepted culture stands in
		}
	}
}

#P:A Imported from Imperator: taken_bribe_effect (common/scripted_effects/00_event_effects.txt)
# Character scope: counts the bribes the character has taken.
pa_imp_taken_bribe_effect = {
	if = {
		limit = {
			NOT = {
				has_variable = taken_bribe
			}
		}
		set_variable = {
			name = taken_bribe
			value = 1
		}
	}
	else = {
		change_variable = {
			name = taken_bribe
			add = 1
		}
	}
}

#P:A Imported from Imperator: destroy_building_effect (common/scripted_effects/00_event_effects.txt)
# Location scope. Removes one level of a building in the location: a fixed order in rural locations, a weighted random pick in towns and cities.
# No b02 event uses it any more (the Yuezhi invasion was skipped as date-bound); kept so a later port can.
pa_imp_destroy_building_effect = {
	save_scope_as = destroy_random_building_scope
	if = {
		limit = { location_rank = location_rank:rural_settlement } # has_city_status = no
		# Imperator order: basic_settlement_infratructure_building, hill_fort, slave_mine_building, latifundia_building, local_forum_building, barracks_building, fortress_building
		#P:A TODO No EU5 equivalent: basic_settlement_infratructure_building, hill_fort, slave_mine_building and local_forum_building have no EU5 building; they are left out of the order
		if = {
			limit = { has_building = building_type:latifundia }
			change_building_level_in_location = {
				building = building_type:latifundia
				value = -1
			}
		}
		else_if = {
			limit = { has_building = building_type:training_fields } #P:A TODO Approximate: Imperator barracks_building -> EU5 training_fields
			change_building_level_in_location = {
				building = building_type:training_fields
				value = -1
			}
		}
		else_if = {
			limit = { has_building = building_type:castle } #P:A TODO Approximate: Imperator fortress_building -> EU5 castle
			change_building_level_in_location = {
				building = building_type:castle
				value = -1
			}
		}
	}
	else = { # has_city_status = yes
		if = {
			limit = { any_buildings_in_location = { always = yes } } # Imperator num_of_used_building_slots > 0
			custom_tooltip = destroy_building_effect.tt
		}
		hidden_effect = {
			random_list = {
				# Unlimited Buildings
				# population_building, military_building, town_hall_building (99 each), workshop_building (10)

						building = building_type:aqueduct_system
						value = -1
					}
				}
				99 = {
					trigger = { has_building = building_type:marketplace } # Imperator commerce_building
					change_building_level_in_location = {
						building = building_type:marketplace
						value = -1
					}
				}
				99 = {
					trigger = { has_building = building_type:library }
					change_building_level_in_location = {
						building = building_type:library
						value = -1
					}
				}
				10 = {
					trigger = { has_building = building_type:dock } #P:A TODO Approximate: Imperator port_building -> EU5 dock
					change_building_level_in_location = {
						building = building_type:dock

#P:A Imported from Invictus: process_states_that_just_revolted (common/scripted_effects/ai_war_effects.txt)
# Any scope. Invictus AI bookkeeping: the current wars that a newly seceded country fights as attacker are recorded as independence wars.
#P:A TODO No EU5 equivalent: Invictus fills the global list territories_that_just_revolted from its own AI secession scripts (on_action ai_on_state_secession), and reads war_of_independence_for_countries in its AI war logic; nothing in EU5 does either, so this effect has nothing to process. Its only caller, ai_war_events.1, was skipped as plumbing.
pa_imp_process_states_that_just_revolted = {
	if = {
		limit = {
			pa_imp_has_non_empty_global_variable_list = {
				name = territories_that_just_revolted
			}
		}
		every_in_global_list = {
			variable = territories_that_just_revolted
			if = {
				limit = {
					exists = owner
					owner = {
						at_war = yes
					}
				}
				owner = {
					save_temporary_scope_as = country_that_just_revolted
					every_current_war = {
						limit = {
							is_an_attacker = scope:country_that_just_revolted # Imperator any_war_attacker = { this = scope:country_that_just_revolted }
						}
						add_to_variable_list = {
							name = war_of_independence_for_countries
							target = scope:country_that_just_revolted
						}
					}
				}
			}
		}
		clear_global_variable_list = territories_that_just_revolted
	}
}
