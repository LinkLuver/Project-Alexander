#P:A Imported from Indomita (common/scripted_effects/MPR_scripted_effects.txt)
# Location scope (Imperator province): makes the number of stacked <type>_in_province_mod location modifiers equal to
# var:<type>_in_territory (1-5) for each special pop type (priest, merchant, architect, artist). Imperator wrote out the
# same block four times; it is one helper call per type here.
pa_imp_add_special_pop_province_modifier_effect = {
	pa_imp_b20_special_pop_stack_effect = { type = priest }
	pa_imp_b20_special_pop_stack_effect = { type = merchant }
	pa_imp_b20_special_pop_stack_effect = { type = architect }
	pa_imp_b20_special_pop_stack_effect = { type = artist }
}

# Helper of pa_imp_add_special_pop_province_modifier_effect (b20), location scope, one special pop type.
pa_imp_b20_special_pop_stack_effect = {
	# type = priest/merchant/architect/artist
	if = {
		limit = {
			has_variable = $type$_in_territory
		}
		if = {
			limit = {
				NOT = { has_location_modifier = $type$_in_province_mod }
			}
			add_location_modifier = { # Imperator add_permanent_province_modifier
				modifier = $type$_in_province_mod
				days = -1
				mode = add
			}
		}
		if = {
			limit = {
				has_location_modifier = $type$_in_province_mod
			}
			if = {
				limit = { var:$type$_in_territory = 1 }
				remove_location_modifier = $type$_in_province_mod
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
			}
			else_if = {
				limit = { var:$type$_in_territory = 2 }
				remove_location_modifier = $type$_in_province_mod
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
			}
			else_if = {
				limit = { var:$type$_in_territory = 3 }
				remove_location_modifier = $type$_in_province_mod
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
			}
			else_if = {
				limit = { var:$type$_in_territory = 4 }
				remove_location_modifier = $type$_in_province_mod
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
			}
			else_if = {
				limit = { var:$type$_in_territory = 5 }
				remove_location_modifier = $type$_in_province_mod
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
				add_location_modifier = { modifier = $type$_in_province_mod days = -1 mode = add }
			}
		}
	}
}

#P:A Imported from Invictus (common/scripted_effects/00_india_effects.txt)
# Location scope (Imperator province). Converts part of the location's people to $RELIGION$ (EU5 religion key, e.g. jain).
# Imperator converted a random 1/4 to 1/8 of all pops of the province that did not already follow $RELIGION$.
pa_imp_spread_religion_in_province = {
	save_scope_as = province_name
	religion:$RELIGION$ = {
		save_scope_as = pa_spread_religion
	}
	custom_tooltip = pa_b20_spread_religion_in_province_tt # Imperator spread_$RELIGION$_in_province_tt (one key per religion)
	hidden_effect = {
		every_pop = { #P:A TODO Approximate: Imperator turned 12.5-25% of the province's pops (whole pops) to the religion; EU5 splits 12.5-25% off every pop of another religion
			limit = {
				NOT = { religion = religion:$RELIGION$ }
			}
			split_pop = {
				fraction = { 0.125 0.25 }
				religion = religion:$RELIGION$
			}
		}
	}
}
