# Plotters are actual dissenting senators, not anonymous rebels.
pa_senate_attempt_tyrant_plot = {
	random_character = {
		limit = {
			owner = root is_alive = yes is_adult = yes this != root.ruler
			OR = { has_variable = pa_senate_optimates_dissenter has_variable = pa_senate_boni_dissenter has_variable = pa_senate_populares_dissenter }
		}
		save_scope_as = pa_senate_plotter
	}
	if = {
		limit = { exists = scope:pa_senate_plotter exists = root.ruler }
		set_variable = { name = pa_senate_plotter value = scope:pa_senate_plotter }
		set_variable = { name = pa_senate_plot_target value = root.ruler }
		set_variable = { name = pa_senate_plot_cooldown value = yes years = 5 }
		random_list = {
			30 = { trigger_event_non_silently = pa_senate.30 }
			70 = { trigger_event_non_silently = pa_senate.31 }
		}
	}
}
