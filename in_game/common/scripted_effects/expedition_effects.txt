# Establishes a Portuguese overseas post at a location.
# Root = expedition country; this = target location (must be a port, gated by option trigger).
#
# Uninhabited (has_owner = no): Portugal takes ownership, constructs a por_trading_post
#   (unique special building: +0.33 local max control, +0.25 merchant capacity, burgher pop),
#   and starts colonist migration from the capital.
# Foreign-owned: por_feitoria is placed in the existing polity's territory (no ownership
#   transfer, which avoids the diplomacy precompute assert in expedition context).
#
# Usage:  scope:location = { establish_cape_route_post = { gold_scale = -3 feitoria_count = 1 } }
establish_cape_route_post = {
	if = {
		limit = { has_owner = no }
		change_location_owner = root
		change_integration_level = colonized
		change_location_controller = root
		construct_building = {
			building_type = building_type:por_trading_post
			cost_multiplier = 0
			cost_multiplier_reason = "game_concept_event"
			instant = yes
		}
		add_migration = {
			owner = root
			from_location = root.capital
			to_location = this
			amount = 0.1
			months = 12
			keep_nationality = yes
		}
	}
	else_if = {
		limit = { top_owner ?= root }
		if = {
			limit = {
				owner = root
				NOT = { has_building = building_type:por_trading_post }
			}
			construct_building = {
				building_type = building_type:por_trading_post
				cost_multiplier = 0
				cost_multiplier_reason = "game_concept_event"
				instant = yes
			}
		}
	}
	else = {
		save_scope_as = por_feitoria_location
		root = {
			save_scope_as = por_feitoria_builder
		}
		construct_building = {
			building_type = building_type:por_feitoria
			owner = root
			cost_multiplier = 0
			cost_multiplier_reason = "game_concept_event"
			instant = yes
		}
		hidden_effect = {
			owner = {
				if = {
					limit = {
						exists = scope:por_feitoria_consenting_host
						scope:por_feitoria_consenting_host ?= this
					}
					trigger_event_non_silently = cape_route_to_india_events.31
				}
				else = {
					trigger_event_non_silently = cape_route_to_india_events.18
				}
			}
		}
	}
	root = {
		if = {
			limit = { $gold_scale$ != 0 }
			change_gold_effect = { scale = $gold_scale$ }
		}
		change_variable = { name = cape_feitorias_established add = $feitoria_count$ }
	}
}

relic_expedition_grant_relic_effect = {
	scope:expedition.expedition_current_location = {
		create_art = {
			type = work_of_art_type:holy_relic
			quality = $quality$
			key = $key$
			save_scope_as = recovered_relic
		}
	}
	scope:recovered_relic = {
		move_art_and_owner = { location = root.capital country = root }
	}
	scope:expedition = {
		set_variable = { name = relic_hunt_found_relic value = scope:recovered_relic }
	}
}
