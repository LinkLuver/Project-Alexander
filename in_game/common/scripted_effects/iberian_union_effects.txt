iu_establish_iberian_wedding_succession_effect = {
	custom_tooltip = {
		text = iu_establish_iberian_wedding_succession_tt
		save_temporary_scope_as = iu_pact_io
		random_international_organization_member = {
			save_temporary_scope_as = iu_crown_a
			scope:iu_pact_io = {
				scope:iu_crown_a = {
					change_heir_selection = heir_selection:iberian_union_succession
				}
				every_international_organization_member = {
					limit = { NOT = { this = scope:iu_crown_a } }
					create_relation = {
						first = this
						second = scope:iu_crown_a
						type = relation_type:iberian_wedding_pact
					}
					change_heir_selection = heir_selection:iberian_union_succession
				}
			}
		}
	}
}
