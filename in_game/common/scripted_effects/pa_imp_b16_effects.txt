#P:A Imported from Invictus: add_2_free_province_investments (common/scripted_effects)
# Country scope. Invictus spent var:free_investments on its province investment buttons.
pa_imp_add_2_free_province_investments = {
	custom_tooltip = gain_2_free_investment_tt
	#P:A TODO No EU5 equivalent: free province investments (Imperator province investments do not exist in EU5); the variable is still kept
	if = {
		limit = {
			NOT = {
				has_variable = free_investments
			}
		}
		set_variable = {
			name = free_investments
			value = 2
		}
	}
	else = {
		change_variable = {
			name = free_investments
			add = 2
		}
	}
}

#P:A Imported from Imperator: capital_formable_medium_effect (common/scripted_effects)
# Imperator state scope -> EU5 province scope. Imperator created 4 citizen and 4 freemen pops of the state's
# culture and religion (create_state_pop).
pa_imp_capital_formable_medium_effect = {
	custom_tooltip = capital_formable_medium_effect_tooltip
	hidden_effect = {
		province_capital ?= {
			ordered_pop = {
				order_by = pop_size
				max = 1
				check_range_bounds = no
				save_temporary_scope_as = pa_majority_pop
			}
			if = { #P:A TODO Approximate: Imperator pop count converted 1:1 to EU5 size, added to the province capital; the state culture/religion is taken from its largest pop
				limit = {
					exists = scope:pa_majority_pop
				}
				add_pop = {
					type = pop_type:burghers # Imperator citizen
					size = 4
					culture = scope:pa_majority_pop.culture
					religion = scope:pa_majority_pop.religion
				}
				add_pop = {
					type = pop_type:peasants # Imperator freemen
					size = 4
					culture = scope:pa_majority_pop.culture
					religion = scope:pa_majority_pop.religion
				}
			}
		}
	}
}
