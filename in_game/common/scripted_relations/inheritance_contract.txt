inheritance_contract = {
	type = diplomacy
	relation_type = oneway
	# scope:first = heirless country (grants succession right, receives gold)
	# scope:second = paying country (buys the right to inherit, pays gold)

	block_when_at_war = yes
	break_on_war = yes
	break_on_becoming_subject = yes
	can_get_without_buying = no
	always_shown_when_unaffordable = yes

	uses_diplo_capacity = receiving
	diplomatic_capacity_cost = inheritance_contract_upkeep_cost

	buy_price = inheritance_contract_cost
	buy_price_modifier = {
	    value = 1
		add = {
			value = scope:recipient.total_development
			multiply = 1.5
		}
		min = 1
		max = 10
	}

	category = {
		offer = CATEGORY_FRIENDLY_ACTIONS
		request = CATEGORY_FRIENDLY_ACTIONS
		cancellation = CATEGORY_HOSTILE_ACTIONS
		break = CATEGORY_HOSTILE_ACTIONS
	}

	relation_type_for_ai = friendly

	offer_visible = { always = no }

	request_enabled = {
		scope:actor = {
			government_type = government_type:monarchy
			is_subject = no
			has_ruler = yes
			ruler ?= {
				is_alive = yes
				is_regent = no
			}
			not = { in_union_with = scope:recipient }
		}
		scope:recipient = {
			government_type = government_type:monarchy
			has_heir = no
			is_subject = no
			has_ruler = yes
			not = { giving_scripted_relation_of_type = relation_type:inheritance_contract }
			ruler ?= {
				is_alive = yes
				is_regent = no
			}
		}
	}

	will_expire_trigger = {
		OR = {
			scope:first = { NOT = { government_type = government_type:monarchy } }
			scope:second = { NOT = { government_type = government_type:monarchy } }
			scope:first = { is_subject = yes }
			scope:second = { is_subject = yes }
			scope:first = { has_heir = yes }
			scope:second = {
				ruler_or_heir_if_regent ?= { is_alive = no }
			}
		}
	}

	offer_effect = {
		scope:recipient = {
			scope:actor = { save_scope_as = contract_partner }
			trigger_event_non_silently = { id = inheritance_contract.1 days = 1 }
		}
	}

	request_effect = {
		scope:actor = {
			scope:recipient = { save_scope_as = contract_partner }
			trigger_event_non_silently = { id = inheritance_contract.1 days = 1 }
		}
	}

	offer_declined_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
		}
	}

	request_declined_effect = {
		scope:actor = {
			add_prestige = prestige_mild_penalty
		}
	}

	cancel_effect = {
		scope:actor = {
			add_prestige = prestige_extreme_penalty
		}
		scope:recipient = {
			add_opinion = {
				target = scope:actor
				modifier = opinion_broke_inheritance_contract
			}
		}
	}

	wants_to_give_diplo_chance = {
		base = -50
		actor_is_rival = -1000
		trust_in_actor = 1
		positive_opinion = 0.3
		negative_opinion = -2
		same_religion = 5
		different_religion = -20
		royal_ties = 5
		in_debt = 2
		too_many_loans = 15
	}

	wants_to_give = {
		if = {
			limit = {
				scope:actor = { culture ?= { has_shared_culture_group = scope:recipient.culture } }
			}
			add = { desc = "inheritance_contract_same_culture_group" value = 20 }
		}
		if = {
			limit = {
				scope:actor = {
					is_in_same_international_organization = {
						international_organization = international_organization:hre
						target = scope:recipient
					}
				}
			}
			add = { desc = "inheritance_contract_hre_members" value = 20 }
		}
	}
}
