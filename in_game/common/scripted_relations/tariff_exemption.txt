tariff_exemption = {
	type = diplomacy
	relation_type = oneway
	break_on_war = yes
	is_exempt_from_import_tariff = yes
	category = {
		offer = CATEGORY_ECONOMY_ACTIONS
		request = CATEGORY_ECONOMY_ACTIONS
		cancellation = CATEGORY_ECONOMY_ACTIONS
		break = CATEGORY_ECONOMY_ACTIONS
	}

	relation_type_for_ai = friendly

	offer_enabled = {
		scope:actor = { has_markets = yes }
	}

	request_enabled = {
		scope:recipient = { has_markets = yes }
	}

	will_expire_trigger = {
		scope:first = { has_markets = no }
	}

	wants_to_give_diplo_chance = {
		base = -10
		trust_in_actor = 0.3
		positive_opinion = 0.1
	}

	wants_to_receive_diplo_chance = {
		base = 30
		actor_is_rival = -100
		trust_in_actor = 0.2
		positive_opinion = 0.1
	}

	wants_to_give = {
		desc = "game_concept_subject"
		if = {
			limit = {
				scope:recipient = {
					is_subject_of = scope:actor
				}
			}
			value = 1000
		}
		else = {
			add = {
				desc = "BASE"
				value = -75
			}

			add = {
				desc = "THEIR_TRUST_TOOLTIP"
				value = "scope:recipient.trust(scope:actor)"
			}

			add = {
				desc = "THEIR_OPINION_TOOLTIP"
				value = "scope:recipient.opinion(scope:actor)"
				if = {
					limit = { "scope:recipient.opinion(scope:actor)" > 0 }
					divide = 5
				}
			}

			if = {
				limit = {
					scope:recipient = {
						is_rival_of = scope:actor
					}
				}
				add = {
					desc = "game_concept_rival"
					value = -100
				}
			}
		}
	}

	wants_to_receive = {
		add = {
			desc = "DIPLOREASON_GOOD_FOR_US"
			value = 100
		}

		if = {
			limit = {
				scope:actor = {
					is_rival_of = scope:recipient
				}
			}
			add = {
				desc = "game_concept_rival"
				value = -200
			}
		}
	}

	wants_to_keep = {
		if = {
			limit = {
				scope:actor = {
					receiving_scripted_relation = {
						target = scope:recipient
						type = relation_type:tariff_exemption
					}
					NOT = {
						giving_scripted_relation = {
							target = scope:recipient
							type = relation_type:tariff_exemption
						}
					}
				}
			}
			add = {
				desc = "DIPLOREASON_GOOD_FOR_US"
				value = 100
			}
		}
		else = {
			add = {
				desc = "THEIR_OPINION_TOOLTIP"
				value = "scope:actor.opinion(scope:recipient)"
				if = {
					limit = { "scope:actor.opinion(scope:recipient)" > 0 }
					divide = 5
				}
			}

			add = {
				desc = "THEIR_TRUST_TOOLTIP"
				value = "scope:actor.trust(scope:recipient)"
				divide = 2
			}

			subtract = {
				desc = "BASE"
				value = 25
			}
		}
	}
}
