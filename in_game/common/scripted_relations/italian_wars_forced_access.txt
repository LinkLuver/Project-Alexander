italian_wars_forced_access = {
	type = diplomacy
	relation_type = oneway
	break_on_war = yes
	annulled_by_peace_treaty = yes
	military_access = yes
	food_access = yes
	receiving_color = define:NMapColors|DIPLOMACY_MIL_ACCESS_COLOR

	category = {
		offer = CATEGORY_ACCESS_ACTIONS
		request = CATEGORY_ACCESS_ACTIONS
		cancellation = CATEGORY_ACCESS_ACTIONS
		break = CATEGORY_ACCESS_ACTIONS
	}

	# Only created via the violate_italian_sovereignty interaction effect
	offer_visible = { always = no }
	request_visible = { always = no }

	break_visible = { always = yes }

	# Receiver cannot break for 1 year after creation
	break_enabled = {
		custom_tooltip = {
			text = cannot_break_unless_war_or_relation_expires_tt
			always = no
		}
	}

	wants_to_keep = {
		# Giver always wants to keep; receiver wants to break once unlocked
		if = {
			limit = {
				scope:actor = {
					giving_scripted_relation = {
						type = relation_type:italian_wars_forced_access
						target = scope:recipient
					}
				}
			}
			add = 100
		}
		else = {
			add = -100
		}
	}
}
