	itinerant_court_policy = {
		allow = {
			societal_value:centralization_vs_decentralization >= 10
		}
		country_modifier = {
			capital_movement_cost_modifier = -0.33
			land_cost_on_distance_from_capital_speed_propagation = 0.05
			monthly_towards_decentralization = societal_value_large_monthly_move
			can_call_rural_parliaments = yes
			pure_tooltip_entry = itinerant_court_capital_projection_tt
			pure_tooltip_entry = itinerant_court_policy_at_risk_tt
		}
		location_modifier = {
			potential_trigger = {
				is_capital = yes
				NOT = { has_building = building_type:itinerant_court_seat }
			}
			local_proximity_source = -20
		}
		on_activate = {
			capital ?= {
				if = {
					limit = { owner ?= { total_building_levels_including_construction:itinerant_court_seat < 1 } }
					change_building_level_in_location = {
						building = building_type:itinerant_court_seat
						value = 1
						owner = root
					}
				}
			}
			if = {
				limit = {
					total_building_levels_including_construction:local_governor > {
						value = effective_num_local_governors
						subtract = 1
						subtract = total_building_levels_including_construction:kings_manor
						if = {
							limit = { NOT = { has_dlc = "d009_across_the_pillars" } }
							subtract = total_building_levels_including_construction:lieutenancy
						}
					}
				}
				random_owned_location = {
					limit = { has_building = building_type:local_governor }
					change_building_level_in_location = {
						building = building_type:local_governor
						value = -1
					}
				}
			}
			custom_tooltip = { text = itinerant_court_reseat_tt }
		}
		on_deactivate = {
			every_owned_location = {
				limit = { has_building = building_type:itinerant_court_seat }
				change_building_level_in_location = {
					building = building_type:itinerant_court_seat
					value = -1
				}
			}
		}
		estate_preferences = {
			peasants_estate
		}
		years = 2
	}

	foreign_envoys_policy = {
		country_modifier = {
			diplomatic_reputation = diplomatic_reputation_weak_bonus
