	type = international_organization
	law_category = election

		leader_type = country
		on_activate = {
			hidden_effect = {
				international_organization_chooses_new_leader = this
			}
		}
		on_deactivate = {
			hidden_effect = {
				remove_law_from_international_organization = law:league_election_method
				remove_law_from_international_organization = law:league_ruling_period
				remove_law_from_international_organization = law:league_election_frequency
			}
		}
		wants_this_policy_bias = {
			if = {
				limit = { NOT = { scope:recipient = { international_organization_has_leader = yes } } }
				add = {
					desc = "VOTE_BASE_VALUE"
					value = 10
				}
				#our own dip rep in hope we get it
				add = {
					desc = "game_concept_diplomatic_reputation"
					value = modifier:diplomatic_reputation
				}
				add = io_members_and_potential_members_opinion_value
			}
			if = {
				limit = { modifier:bias_for_militarist_policies > 0 }
				add = {
					desc = "PROVIDES_POWER_OPPORTUNITY"
					value = {
						value = modifier:bias_for_militarist_policies
						multiply = {
							value = country_tax_base
							subtract = io_get_average_tax_base
						}
					}
				}
			}
		}
		wants_propose_policy = {
			add = league_default_leader_policy_desire
		}
