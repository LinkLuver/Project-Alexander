	league_koinon_foundation = {
		#A federal union of same cultured city states. Cohesion mainly depends on the culture opinion among the members' main cultures. Historical examples: epirote league,achaean league, aetolian league, boeotian league, lycian league, thessalian league, arcadian league, chalcidian league, acarnanian league, nesiotic league, euboean league, cretan league, phocian league. The leader of this league is the country leader of the strongest member country, for life.
	}
	league_isopoliteia_foundation = {
		#A league focused on trade and migration treaties, usually between colonies and home cities (generally playing tall). Cannot result into a member becoming hegemonic(more on that later), improves relations between countries and cultures, less selective about members (and their distance). Cohesion mainly depends from country opinions and trade volume between countries. This league cannot have a leader.
	}
	league_amphictyony_foundation = {
		#An alliance between custodians of important temples, could impose fines or wage sacred wars with religious pretexts. These types of leagues could also be government reforms, since some of them are too small to portray. Historical examples:ionian league, etruscan league*, Dorian Hexapolis, Onchestos Amphictyony, Calaurian Amphictyony, Delian Amphictyony, Delphic Amphictyony, early Latin League. Cohesion mainly depends on members' religious unity. The leader of this league is the country leader the protected temple is in.
	}
	league_symacheia_foundation = {
		# Military alliance against a specific enemy. It starts out as a less cohesive koinon, but there are no culture requirements. Historical examples: Delian league (w/ Athens), peloponnesian league, league of corinth, hellenic league, late latin league, samnite league. Cohesion mainly depends on hegemon's relative power OR main adversary's relative power. The leader of this league is the country leader of the strongest member country, for life
	}
	league_confederation_foundation = {
		#Temporary military alliance against a specific enemy. Once formed it has high base cohesion, but there are limits to who is invited similar to the koinon. Limited to tribes, they lose massive amounts of cohesion upon the leader's death. Historical examples: Vercingetorix’s Confederation, The Caledonian Confederacy, The Marcomannic Confederation. The leader of this league is the country leader of the strongest member country, for life.
	}
