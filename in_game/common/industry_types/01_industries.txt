# Industry types are granted to an area via the Promote Industry system (see the Urban Rights screen).
# Each industry covers a set of goods. The location/country modifiers are GENERATED in C++ at load
# from that goods list (see CIndustryType::BuildGeneratedModifiers), so you only list the goods + the
# two magnitudes here:
#   - per good, country-wide pop demand   -> global_<good>_pop_demand    = pop_demand
#   - per good, in-area production output  -> local_<good>_output_modifier = output
#
# color             = <named_color or hsv>   # map highlight for this industry
# kept_at_conquest  = no                      # (default no) revoke when the owner loses the area
# potential = {}                              # root = country; hides the industry in the picker when false
# allow = {}                                  # root = country, scope:target = area; gates granting
# goods = { ... }                             # the goods this industry promotes
# output    = <value>                         # location modifier magnitude per good (local_<good>_output_modifier)
# pop_demand = <value>                        # country modifier magnitude per good
#
# (location_modifier / country_modifier blocks are still honored if you need a bespoke entry, but
#  prefer the goods list so the system stays a single source of truth.)
