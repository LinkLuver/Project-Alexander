andalusi_refugee_town_rights = {
	kept_at_conquest = no

	location_modifier = {
		local_migration_attraction = 0.2
		local_production_efficiency = 0.025
	}

	potential = {
		has_dlc = "d009_across_the_pillars"
		modifier:allow_andalusi_refugee_town_rights = yes
		NOT = {
			global_var:battle_for_the_strait_winner_country ?= {
				religion.group = religion_group:muslim
			}
		}
	}

	allow = {
		scope:target = {
			"culture_percentage(culture:andalusi)" > 0
			NOR = {
				has_town_rights = town_rights_type:medina_charter_town_rights
				has_town_rights = town_rights_type:mzab_pentapolis_town_rights
				has_town_rights = town_rights_type:tribal_souk_town_rights
			}
		}
	}

	color = map_tunis
}
