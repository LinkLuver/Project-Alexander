### The aim to increase City specialization. They increase the efficiency of a specific Good, but decrease the efficiency of all others.

royal_tooling_rights = {
	location_modifier = {
		local_tools_output_modifier = 0.3

		local_production_efficiency = town_right_efficiency_penalty
	}
	
	color = goods_tools
}

royal_jewelry_rights = {
	location_modifier = {
		local_jewelry_output_modifier = 0.2
		local_jewelry_guild_building_levels = 10

		local_production_efficiency = town_right_efficiency_penalty
	}
	
	color = goods_jewelry
}

royal_naval_rights = {

	allow = {
		custom_tooltip = {
			text = not_forcefully_revoked_last_10_years_tt
			scope:target = { 
				NOT = { has_variable = recently_forced_to_revoke_town_rights_variable }
			}
		}
	}
	location_modifier = {
		local_naval_supplies_output_modifier = 0.3
		local_tar_output_modifier = 0.3

		local_production_efficiency = town_right_efficiency_penalty
	}
	
	color = goods_naval_supplies
}
