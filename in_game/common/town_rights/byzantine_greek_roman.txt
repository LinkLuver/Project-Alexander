imperial_breadbasket_rights = {
	kept_at_conquest = no
	potential = {
		culture = { has_culture_group = culture_group:latin_group }

	}
	
	allow = {
		scope:target = {
			NOT = { has_town_rights = town_rights_type:granary_town }
			custom_tooltip = {
				text = imperial_breadbasket_rights_rank_tt
				OR = {
					location_rank = location_rank:city
					location_rank = location_rank:megalopolis
				}
			}
			custom_tooltip = {
				text = imperial_breadbasket_rights_granary_tt
				any_buildings_in_location = {
					building_type = building_type:granary
					building_level >= 5
				}
			}
		}
	}
	
	location_modifier = {
		local_food_capacity = 1000
		local_monthly_prosperity = 0.002
		local_population_growth = 0.002

		local_pop_food_consumption = 0.2
	}

	color = goods_coffee
}
