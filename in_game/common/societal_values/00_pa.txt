	allow = {
		OR = {

			AND = {
				exists = capital
				capital = { 
					OR = {
						sub_continent ?= sub_continent:eastern_europe

						sub_continent ?= sub_continent:western_europe
						sub_continent ?= sub_continent:north_africa
					}
				}
			}
			AND = {
				exists = international_organization:achaemenid_argead
				culture ?= { has_culture_group = culture_group:greek_group }
				# is_member_of_international_organization = international_organization:middle_kingdom
			}
		}
	}

		legislative_efficiency = 0.25
		research_speed_modifier = 0.1

		stability_cost = -0.33
		
		peasants_estate_levy_size = 0.5

	}
}
