expedition_regency = {
	internally_assigned = yes
	hide_duration = yes
	allow_extend_regency = no

	allow = {
		ruler ?= { is_expedition_leader = yes }
	}

	modifier = {
		country_cabinet_efficiency = -0.10
	}

	start_effect = {
		if = {
			limit = {
				"estate_power(estate_type:nobles_estate)" > 0.33
			}
			if = {
				limit = {
					any_character = {
						can_become_a_regent = yes
						has_estate = estate_type:nobles_estate
					}
				}
				random_character = {
					limit = {
						can_become_a_regent = yes
						has_estate = estate_type:nobles_estate
					}
					save_scope_as = expedition_regent
				}
				set_regent = scope:expedition_regent
			}
			else = {
				hidden_effect = {
					create_character = {
						culture = root.culture
						religion = root.religion
						estate = estate_type:nobles_estate
						age = 35
						female = no
						save_scope_as = expedition_regent
					}
					set_regent = scope:expedition_regent
				}
			}
		}
		else_if = {
			limit = {
				"estate_power(estate_type:clergy_estate)" > 0.33
			}
			if = {
				limit = {
					any_character = {
						can_become_a_regent = yes
						has_estate = estate_type:clergy_estate
						religion = root.religion
					}
				}
				random_character = {
					limit = {
						can_become_a_regent = yes
						has_estate = estate_type:clergy_estate
						religion = root.religion
					}
					save_scope_as = expedition_regent
				}
				set_regent = scope:expedition_regent
			}
			else = {
				hidden_effect = {
					create_character = {
						culture = root.culture
						religion = root.religion
						estate = estate_type:clergy_estate
						age = 35
						female = no
						save_scope_as = expedition_regent
					}
					set_regent = scope:expedition_regent
				}
			}
		}
		else_if = {
			limit = {
				"estate_power(estate_type:burghers_estate)" > 0.33
			}
			if = {
				limit = {
					any_character = {
						can_become_a_regent = yes
						has_estate = estate_type:burghers_estate
					}
				}
				random_character = {
					limit = {
						can_become_a_regent = yes
						has_estate = estate_type:burghers_estate
					}
					save_scope_as = expedition_regent
				}
				set_regent = scope:expedition_regent
			}
			else = {
				hidden_effect = {
					create_character = {
						culture = root.culture
						religion = root.religion
						estate = estate_type:burghers_estate
						age = 35
						female = no
						save_scope_as = expedition_regent
					}
					set_regent = scope:expedition_regent
				}
			}
		}
		else_if = {
			limit = {
				"estate_power(estate_type:peasants_estate)" > 0.33
			}
			if = {
				limit = {
					any_character = {
						can_become_a_regent = yes
						has_estate = estate_type:peasants_estate
					}
				}
				random_character = {
					limit = {
						can_become_a_regent = yes
						has_estate = estate_type:peasants_estate
					}
					save_scope_as = expedition_regent
				}
				set_regent = scope:expedition_regent
			}
			else = {
				hidden_effect = {
					create_character = {
						culture = root.culture
						religion = root.religion
						estate = estate_type:peasants_estate
						age = 35
						female = no
						save_scope_as = expedition_regent
					}
					set_regent = scope:expedition_regent
				}
			}
		}
		else_if = {
			limit = {
				any_cabinet_character = {
					has_character_modifier = head_of_the_cabinet_modifier
					can_become_a_regent = yes
				}
			}
			random_cabinet_character = {
				limit = {
					has_character_modifier = head_of_the_cabinet_modifier
					can_become_a_regent = yes
				}
				save_scope_as = expedition_regent
			}
			set_regent = scope:expedition_regent
		}
		else_if = {
			limit = {
				has_consort = yes
				consort ?= {
					is_adult = yes
					modifier:blocked_from_cabinet = no
				}
				NOR = {
					#religion.group = religion_group:muslim #P:A Removed vanilla reference
					succession_law = heir_selection:tribal_matrilineal
				}
			}
			set_regent = root.consort
		}
		else = {
			hidden_effect = {
				create_character = {
					culture = root.culture
					religion = root.religion
					estate = estate_type:nobles_estate
					age = 35
					female = no
					save_scope_as = expedition_regent
				}
				set_regent = scope:expedition_regent
			}
		}
	}

	end_effect = {
	}
}
