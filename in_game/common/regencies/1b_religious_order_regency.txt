religious_order_regency = {

	allow = {
		has_any_religious_order_presence = yes
	}

	modifier = {
		global_clergy_estate_power = 0.1
	}

	start_effect = {
		random_religious_order_in_country = {
			save_scope_as = regency_order
		}
		if = {
			limit = {
				any_character = {
					trigger_if = {
						limit = { exists = scope:regency_order }
						can_become_a_regent = yes
						is_religious_order_member = yes
						religious_order = scope:regency_order
					}
					trigger_else = { always = no }
				}
			}
			random_character = {
				limit = {
					trigger_if = {
						limit = { exists = scope:regency_order }
						can_become_a_regent = yes
						is_religious_order_member = yes
						religious_order = scope:regency_order
					}
					trigger_else = { always = no }
				}
				save_scope_as = new_leader
			}
			set_regent = scope:new_leader
		}
		else = {
			hidden_effect = {
				create_character = {
					religion = root.religion
					estate = estate_type:clergy_estate
					age = 35
					female = no
					save_scope_as = new_leader
				}
				scope:new_leader = {
					join_religious_order = scope:regency_order
				}
				set_regent = scope:new_leader
			}
		}
	}
}
