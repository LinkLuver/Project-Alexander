cardinal_regency = {
	internally_assigned = yes

	allow = {
		any_cabinet_character = {
			religious_figure_type = cardinal
		}
	}

	modifier = {
		monthly_religious_influence = 0.2
		global_clergy_estate_power = 0.1
	}

	start_effect = {
		random_cabinet_character = {
			limit = {
				religious_figure_type = cardinal
			}
			save_scope_as = new_leader
		}
		set_regent = scope:new_leader
	}
}
