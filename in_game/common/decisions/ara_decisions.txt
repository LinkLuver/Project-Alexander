ara_unite_barcelona_dynasty = {
	decision_category = country_specific_decisions
	only_once = yes

	potential = {
		has_variable = ara_unite_de_barcelona_dynasty
		has_or_had_tag = ARA
		ruler_or_heir_if_regent ?= {
			has_dynasty = yes
		}
		is_subject = no
		country_exists = c:SIC

		c:SIC = {
			is_subject = no
			ruler_or_heir_if_regent.dynasty ?= root.ruler_or_heir_if_regent.dynasty
            is_junior_partner = no
		}
	}

	allow = {
		c:SIC = {
            at_war = no
        }

        great_power_score > c:SIC.great_power_score
	}

	ai_will_do = {
		value = 100
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }

		ruler_or_regent ?= {
			save_scope_as = target_character
		}

		c:SIC = {
            ruler_or_regent ?= {
                save_scope_as = target_character2
            }
        }

		save_scope_as = from_country
	}

	option = {
		name = ara_unite_barcelona_dynasty.a
		ai_chance = { value = 100 }
		effect = {
			remove_variable = ara_unite_de_barcelona_dynasty

            c:SIC = {
                trigger_event_silently = flavor_ara.42
            }

			show_as_tooltip = {
				drop_antagonism_bomb = {
					target = c:SIC.capital
					modifier = antagonism_ignore_sicilian_treaty
				}
			}

			custom_tooltip = ara_effects_only_take_place_if_sic_accepts_tt
		}
	}

    option = {
		name = ara_unite_barcelona_dynasty.b
		ai_chance = { value = 100 }
		effect = {
			remove_variable = ara_unite_de_barcelona_dynasty

            c:SIC = {
                trigger_event_silently = flavor_ara.43
            }

			show_as_tooltip = {
				drop_antagonism_bomb = {
					target = c:SIC.capital
					modifier = antagonism_ignore_sicilian_treaty_harsher
				}
			}

			custom_tooltip = ara_effects_only_take_place_if_sic_accepts_tt
		}
	}
}
