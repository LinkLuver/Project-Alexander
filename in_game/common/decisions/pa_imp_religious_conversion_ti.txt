convert_to_manichaean = {
	decision_category = religious_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		pa_imp_potential_change_religion_trigger = { RELIGION = manichaeism }
	}

	allow = {
		pa_imp_could_change_religion_trigger = { RELIGION = manichaeism }
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				capital ?= { dominant_religion = religion:manichaeism } # Imperator dominant_province_religion
			}
			add = 1
		}
		if = {
			limit = {
				OR = {
					tag = JUD
					tag = SMR # Imperator SMA (Samaria)
				}
			}
			multiply = 0
		}
	}

	option = {
		name = convert_to_manichaean.a
		ai_chance = { value = 100 }
		effect = {
			pa_imp_change_religion_effect = { RELIGION = manichaeism }
		}
	}
}
