#P:A Imported from Invictus: decisions/enshrine_equal_rights.txt
#   Imperator switched the country flag gender_equality. EU5 has no such switch; these decisions add or remove
#   pa_imp_social_equity_modifier (main_menu/common/static_modifiers/pa_imp_decisions_modifiers.txt), which grants the
#   EU5 gender_equality modifier.
#P:A TODO No EU5 equivalent: Imperator only showed these decisions under the game rule gender_rules (or the Invictus global variable game_rule_gender_rules); EU5 has no such rule, the check is dropped from both

enshrine_social_equity = {
	decision_category = government_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		modifier:gender_equality = no #P:A TODO Approximate: Imperator gender_equality = no
		# OR = {
		# 	gender_rules = yes
		# 	has_global_variable = game_rule_gender_rules
		# }
	}

	allow = {
		political_influence >= 25 # influence_large
		trigger_if = {
			limit = {
				has_variable = social_equity_var
			}
			custom_tooltip = {
				text = social_equity_tooltip
				NOT = {
					has_variable = social_equity_var
				}
			}
		}
	}

	ai_will_do = {
		value = 0
	}

	option = {
		name = enshrine_social_equity.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = -25 # subtract_influence_large
			# if = {
			# 	limit = {
			# 		is_republic = yes
			# 		has_party_type = traditionalist_party
			# 	}
			# 	add_party_approval = {
			# 		party_type = traditionalist_party
			# 		value = -10
			# 	}
			# }
			#P:A TODO No EU5 equivalent: republics lost 10 approval with the traditionalist party (EU5 has no parties)
			set_variable = {
				name = social_equity_var
				days = 7300
			}
			add_country_modifier = { #P:A TODO Approximate: Imperator set_gender_equality = yes
				modifier = pa_imp_social_equity_modifier
				days = -1
			}
			hidden_effect = {
				set_global_variable = enshrine_social_equity_used
			}
		}
	}
}
