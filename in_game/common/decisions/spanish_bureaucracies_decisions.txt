spa_seize_control_of_the_orders = {
	decision_category = government_decisions
	only_once = yes

	potential = {
		has_dlc = "d009_across_the_pillars"
		culture = {
			has_culture_group = culture_group:iberian_group
		}
		religion = religion:catholic
		has_reform = government_reform:system_of_councils_reform
	}

	allow = {
		at_war = no
		any_religious_order_in_country = {
			order_type ?= religious_order_type:iberian_military_order_type
		}
		region:iberia_region = {
			NOT = {
				any_location_in_region = {
					is_ownable = yes
					has_owner = yes
					owner ?= {
						NOT = {
							religion.group = religion_group:christian
						}
					}
				}
			}
		}
	}

	illustration_scope = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = spa_seize_control_of_the_orders.a
		ai_chance = { value = 100 }
		effect = {
			every_religious_order_in_country = {
				limit = {
					order_type ?= religious_order_type:iberian_military_order_type
				}
				change_religious_order_zeal = religious_order_severe_penalty
			}
			custom_tooltip = {
				text = establish_council_of_military_orders_tt
				set_variable = decision_taken_to_establish_council_of_military_orders
			}
		}
		
	}
}
