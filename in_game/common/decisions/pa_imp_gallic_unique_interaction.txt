gallic_smithy = { #Martial
	decision_category = other_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		always = no #P:A TODO No EU5 equivalent: needs the Invictus invention gallic_noric_steel_inv; EU5 has no inventions
		OR = {
			culture = { has_culture_group = culture_group:gallic_group } # Imperator country_culture_group = gallic
			# country_culture_group = celto_pannonian_group #P:A TODO No EU5 equivalent: the mod has no Celto-Pannonian group (its Boii are in gallic_group, already listed); add the Taurisci/Scordisci cultures here once they exist
			culture = { has_culture_group = culture_group:belgae_group }
			culture = culture:etruscan # Imperator primary_culture = etruscan
		}
		NAND = {
			has_variable = gallic_item_1_variable
			has_variable = gallic_item_2_variable
			has_variable = gallic_item_3_variable
			has_variable = gallic_item_4_variable
			has_variable = gallic_item_5_variable
			has_variable = gallic_item_6_variable
			has_variable = gallic_item_7_variable
			has_variable = gallic_item_8_variable
		}
	}

	allow = {
		custom_tooltip = {
			text = gallic_smithy.tt
			NOT = { has_variable = master_smith_is_working }
		}
		gold >= 250 # Imperator treasury
		political_influence >= 25
	}

	ai_will_do = {
		value = 100 # Imperator had no ai_will_do (the AI took it whenever it could)
	}

	option = {
		name = gallic_smithy.a
		ai_chance = { value = 100 }
		effect = {
			set_variable = {
				name = master_smith_is_working
				days = 1825 #5 yrs
			}
			add_gold = -250
			add_political_influence = -25
			trigger_event_non_silently = { #the item
				id = gallic_smithy_inv.1
				days = { 30 150 }
			}
			trigger_event_non_silently = { #sending for the smith
				id = gallic_smithy_inv.2
			}
		}
	}
}
