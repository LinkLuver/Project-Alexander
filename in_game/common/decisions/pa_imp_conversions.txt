#   in_game/common/government_reforms/pa_imperator_governments.txt (branch dev): "government = X" is written as
#   "has_reform = government_reform:X", "change_government = X" as pa_imp_change_ir_government_effect.

become_tribal_chiefdom = {
	decision_category = government_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		OR = {
			has_reform = government_reform:tribal_federation
			has_reform = government_reform:tribal_kingdom
		}
	}

	allow = {
		societal_value:centralization_vs_decentralization > 25 #P:A TODO Approximate: Imperator centralization < -25
		# capital_scope = {
		# 	custom_tooltip = {
		# 		text = pops_to_decivilize_pop
		# 		num_of_tribesmen > num_of_citizen
		# 		num_of_tribesmen > num_of_freemen
		# 	}
		# }
		#P:A TODO No EU5 equivalent: the capital needed more tribesmen than citizens and than freemen (no trigger compares two pop-type sizes; a script value could)
		# custom_tooltip = {
		# 	text = tribal_reform_decision_trigger_tt
		# 	NOT = {
		# 		any_character = {
		# 			is_clan_chief = yes
		# 			loyalty < 33
		# 		}
		# 	}
		# }
		#P:A TODO No EU5 equivalent: no clan chief could be below 33 loyalty (EU5 has no clan chiefs or loyalty)
	}

	ai_will_do = {
		value = 1
		if = {
			limit = {
				OR = {
					NOT = {
						has_reform = government_reform:tribal_kingdom
					}
					any_owned_location = {
						has_location_modifier = tribal_civilization_pmod
					}
					at_war = yes
				}
			}
			multiply = 0
		}
	}

	option = {
		name = become_tribal_chiefdom.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = 25 # influence_large
			if = {
				limit = {
					has_reform = government_reform:tribal_kingdom
				}
				add_stability = -10
			}
			else = {
				add_stability = -30
			}
			pa_imp_change_ir_government_effect = { TYPE = tribe REFORM = tribal_chiefdom }
			pa_imp_switch_government_type_event_clearup_effect = yes
			if = {
				limit = {
					any_owned_location = { has_location_modifier = tribal_civilization_pmod }
				}
				every_owned_location = {
					limit = { has_location_modifier = tribal_civilization_pmod }
					remove_location_modifier = tribal_civilization_pmod
				}
			}
		}
	}
}

tribal_reform_decision = {

		}
		is_ai = no
		NOT = {
			has_variable = can_tribal_reform
		}
	}

	allow = {
		stability >= 40
		societal_value:centralization_vs_decentralization <= -60 #P:A TODO Approximate: Imperator centralization >= 60

		# 		}
		# 	}
		# }
		#P:A TODO No EU5 equivalent: no clan chief could be below 33 loyalty (EU5 has no clan chiefs or loyalty)
	}

	ai_will_do = {
		value = 1
		if = {
			limit = {
				ruler ?= {
					pa_imp_char_is_bad_leader_trigger = yes
				}
			}
			multiply = 0
		}
	}

	option = {
		name = tribal_reform_decision.a
		ai_chance = { value = 100 }
		effect = {
			custom_tooltip = allows_tribal_reform_mission
			set_variable = {
				name = can_tribal_reform
			}
		}
	}
}

tribal_civilization = {
	decision_category = government_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		government_type = government_type:tribe # Imperator is_tribal = yes
		NOT = {
			capital ?= { has_location_modifier = tribal_civilization_pmod }
		}
	}

	allow = {
		pa_imp_tribal_civilization_allow = yes
		# custom_tooltip = {
		# 	text = tri_civ_cla_chi_tt
		# 	NOT = { # No clan chiefs are too disloyal
		# 		any_character = {
		# 			is_clan_chief = yes
		# 			loyalty < 50
		# 		}
		# 	}
		# }
		#P:A TODO No EU5 equivalent: no clan chief could be below 50 loyalty (EU5 has no clan chiefs or loyalty)
	}

	ai_will_do = {
		value = 1000
		if = {
			limit = {
				at_war = yes
				# OR = { war = yes AND = { is_advanced_ai_enabled = yes ... rank_by_num_territories >= regional_power } }
				#P:A TODO No EU5 equivalent: the Invictus advanced-AI game rule also turned this off for large tribes
			}
			multiply = 0
		}
	}

	option = {
		name = tribal_civilization.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = -100
			add_gold = -50
			capital ?= {
				add_location_modifier = {
					modifier = tribal_civilization_pmod
					days = -1
				}
				change_development = 5 #P:A TODO Approximate: Imperator add_civilization_value = 10 (0-100) read as EU5 development 5
			}
			if = {
				limit = {
					any_owned_location = {
						is_capital = no
						has_location_modifier = tribal_civilization_pmod
					}
				}
				every_owned_location = {
					limit = {
						is_capital = no
						has_location_modifier = tribal_civilization_pmod
					}
					remove_location_modifier = tribal_civilization_pmod
				}
			}
		}
	}
}
