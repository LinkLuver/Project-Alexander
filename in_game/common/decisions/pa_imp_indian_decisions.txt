choice_of_personal_deity_hindu = {
	decision_category = religious_decisions

	potential = {
		always = no #P:A TODO No EU5 equivalent: the decision only fires hindu_flavor.1, which the event import holds false (every option gives an Imperator personal_deity_* trait the mod lacks); release both together
		num_locations >= 1 # Imperator num_of_cities
		OR = {
			religion = religion:hindu
			religion = religion:agama
		}
		NOT = {
			ruler ?= {
				has_variable = personal_deity_chosen
			}
		}
	}

	allow = {
		political_influence >= 25
		NOT = {
			ruler ?= {
				has_variable = personal_deity_chosen
			}
		}
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = choice_of_personal_deity_hindu.a
		ai_chance = { value = 100 }
		effect = {
			add_political_influence = -25
			trigger_event_non_silently = { ##Deity Choice Event
				id = hindu_flavor.1
			}
		}
	}
}

# This decision is mainly so players can more easily conquer the Maldives and for AI as they otherwise get stuck never invading it
#P:A TODO Verify placement, was Male (8342) in IR
subjugate_maldives_decision = {
	decision_category = expansion_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		#P:A TODO Approximate: Imperator owns_or_subject_owns_area on one of 6 South Indian / Sri Lankan areas (keralaputa, north/south satiyaputa, pandiya, north/south tambapamni); no pa_imp_in_area trigger exists for them, so owning or having a subject own any location in the EU5 malabar_area, pandya_nadu or sailan_area stands in
		OR = {
			any_owned_location = {
				OR = {
					area = area:malabar_area
					province = province:pandya_nadu
					area = area:sailan_area
				}
			}
			any_subject = {
				any_owned_location = {
					OR = {
						area = area:malabar_area
						province = province:pandya_nadu
						area = area:sailan_area
					}
				}
			}
		}
		location:male_atoll.owner ?= {
			is_subject = no
			num_locations <= 20 #P:A TODO Approximate: Imperator num_of_cities <= 20 (territories); EU5 owned locations
		}
		NOT = {
			owns_or_non_sovereign_subject_owns = location:male_atoll
		}
	}

	allow = {
		#P:A TODO Approximate: Imperator owns_or_subject_owns_area = keralaputa_area, north_satiyaputa_area and south_satiyaputa_area (the Malabar coast) -> every location of the EU5 malabar_area owned by the country or its subjects
		area:malabar_area = {
			any_location_in_area = {
				count = all
				OR = {
					owner ?= root
					owner ?= { is_subject_of = root }
				}
			}
		}
		num_locations >= 50 #P:A TODO Approximate: Imperator num_of_cities >= 50 (territories); EU5 owned locations
		at_war = no
		political_influence >= 30
		location:male_atoll.owner ?= {
			is_subject = no
			num_locations <= 10 #P:A TODO Approximate: Imperator num_of_cities <= 10 (territories); EU5 owned locations
		}
	}

	ai_will_do = {
		value = 100 # Imperator had no ai_will_do (the AI took it whenever it could)
	}

	option = {
		name = subjugate_maldives_decision.a
		ai_chance = { value = 100 }
		effect = {
			location:male_atoll.owner ?= {
				make_subject_of = {
					target = root
					type = subject_type:client_state
				}
			}
			if = {
				limit = { is_ai = no }
				add_political_influence = -30
			}
		}
	}
}
