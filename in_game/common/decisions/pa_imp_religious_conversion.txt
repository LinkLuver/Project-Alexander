#P:A Imported from Invictus: decisions/religious_conversion.txt
#   Religion conversion decisions. Imperator religions are mapped with .claude/imperator_import/events/maps/tables.json.

convert_to_buddhist = {
	decision_category = religious_decisions

	potential = {
		num_locations >= 1 # Imperator num_of_cities
		pa_imp_potential_change_religion_trigger = { RELIGION = buddhism }
	}

	allow = {
		pa_imp_could_change_religion_trigger = { RELIGION = buddhism }
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				capital ?= { dominant_religion = religion:buddhism } # Imperator dominant_province_religion
			}
			add = 1
		}
		if = {
			limit = {
				NOT = { tag = MRY }
				NOT = { tag = BAC }
			}
			multiply = 0
		}
	}

	option = {
		name = convert_to_buddhist.a
		ai_chance = { value = 100 }
		effect = {
			pa_imp_change_religion_effect = { RELIGION = buddhism }
		}
	}
}
