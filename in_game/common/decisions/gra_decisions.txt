expand_the_alhambra = {
	decision_category = country_specific_decisions

	tags = { GRA ADU }

	potential = {
		has_dlc = "d009_across_the_pillars"
		OR = {
			has_variable = may_upgrade_alhambra_variable
			global_var:battle_for_the_strait_winner_country ?= root
		}
		OR = {
			NOT = { has_variable = alhambra_extension_level }
			var:alhambra_extension_level < 3
		}
	}

	allow = {
		owns = location:granada
		location:granada = { is_capital = yes }
        at_war = no
        is_during_bankruptcy = no
        gold > 250
	}

	ai_will_do = {
		value = 100
	}

	option = {
		name = expand_the_alhambra.a
		ai_chance = { value = 100 }
		effect = {
			custom_tooltip = {
				text = gra_construct_alhambra_increment_tt
				if = {
					limit = {
						NOT = { has_variable = alhambra_extension_level }
					}
					set_variable = {
						name = alhambra_extension_level
						value = 1
					}
				}
				else = {
					change_variable = { name = alhambra_extension_level add = 1 }
				}
			}
			custom_tooltip = {
				text = expand_the_alhambra_construct_tt
				capital ?= {
					construct_building = {
						building_type = building_type:alhambra
					}
				}
			}
			remove_variable = may_upgrade_alhambra_variable
		}
	}
}
