#P:A TODO In very special cases, (maybe pirates?) make it so the main fleet is the main source of control instead of the capital, with higher control growth/less control loss, meaning the ruler can tour his realm and keep relative control across vast distances
admiralty_regime_reform = { #Rule of the Navarch
	age = age_1_princes #P:A Vanilla age: age_3_discovery
	major = yes
	potential = {
		num_of_ports > 0
		NOT = { government_type = government_type:steppe_horde }
		NOT = { government_type = government_type:tribe }
	}

	allow = {
		NOT = { has_reform = government_reform:military_regime_reform }
	}

	country_modifier = {
		monthly_towards_naval = societal_value_large_monthly_move
		navy_maintenance_efficiency = 0.15
		monthly_navy_tradition = 0.1
	}

	on_activate = {
		hidden_effect = {
			if = {
				limit = { exists = succession_law }
				set_variable = { name = previous_heir_selection value = succession_law }
			}
		}
		change_heir_selection = heir_selection:admiralty_regime_heir_selection
	}
	on_deactivate = {
		if = {
			limit = { has_variable = previous_heir_selection }
			change_heir_selection = var:previous_heir_selection
			remove_variable = previous_heir_selection
		}
		else = {
			set_to_default_heir_selection = yes
		}
	}
	years = 2
}
