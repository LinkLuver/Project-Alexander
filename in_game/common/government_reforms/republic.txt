peasant_republic_reform = { #Agrarian Democracy
	major = yes
	government = republic

	potential = { country_has_estate = estate_type:peasants_estate }

	allow = {
		OR = {
			AND = {
				"estate_power(estate_type:peasants_estate)" > 0.1
				societal_value:serfdom_vs_free_subjects > 80
			}

			#This is added for countries that should be given access to this through an event etc
			has_reform = government_reform:peasant_republic_reform
		}
	}

	country_modifier = {
		global_peasants_estate_power = 1
		global_nobles_estate_power = -0.25
		base_peasants_estate_power_modifier = 9.0
		base_nobles_estate_power_modifier = -0.9
		global_raw_material_output = 0.33
		peasants_estate_target_satisfaction = medium_permanent_target_satisfaction
		nobles_estate_target_satisfaction = small_permanent_target_satisfaction_penalty
		monthly_towards_decentralization = societal_value_monthly_move
		monthly_towards_free_subjects = societal_value_monthly_move
		peasants_estate_can_participate_in_parliament = yes
		government_reform_slots = 1
		court_language_is_common_language_importance_modifier = 15
	}

	years = 3
}
