iberian_military_order_type = {
	color = { 200 50 50 }
	graphical_cultures = { military_order_gfx }
	location_overlay_loc_key = ORDER_HOLDING
	can_field_levies = yes
	starting_regiment = a_order_knights
	default_calling = crusaders_calling
	valid_callings = { 
		crusaders_calling 
		wardens_calling 
		confidants_calling 
		mariners_calling
	}
	founding_name_pattern = ORDER_OF_LOCATION_NAME_PATTERN
	founding_cost_gold = 100
	holding_goods_demand = religious_order_holding_demand
	default_founding_country_allow = {
		has_dlc = "d009_across_the_pillars"
		religion = religion:catholic
		culture = { has_culture_group = culture_group:iberian_group }
		any_owned_location = {
			OR = {
				region = region:iberia_region
				region = region:maghreb_region
			}
		}
	}
	default_founding_location_allow = {
		owner ?= scope:actor
		OR = {
			region = region:iberia_region
			region = region:maghreb_region
		}
	}
	
	on_leader_death = {
		scope:religious_order = {
			random_pop_in_religious_order_holding_location = {
				age = { 30 50 }
				female = no
				valid_pop_trigger = {
					religion = scope:religious_order.religion
				}
			}
		}
	}
}
