#scope for allow is the character.

orthodox_education =
{
	potential = {
		has_dlc = "d008_fate_of_the_phoenix"
		owner = { religion = religion:orthodox }
	}
	allow = {
		owner = {
			is_leader_of_international_organization_of_type = {
				type = autocephalous_patriarchate
			}
		}
	}

	price_to_select = select_orthodox_education
	price_to_deselect = deselect_orthodox_education

	modifier = {
		character_child_education = 0.8
	}
	country_modifier = {
		monthly_religious_influence = -0.1
	}

	on_education_start_effect = {
		if = {
			limit = { religion != owner.religion }
			change_character_religion = owner.religion
		}
	}
}
