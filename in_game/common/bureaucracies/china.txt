# YUAN

central_secretariat_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = CHI
	}

	estates_that_like = {
		crown_estate
		burghers_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	#neutral
	neutral_modifier = {
		government_size = 1
	}

	#positive - zhongshu_sheng advance raises scale by 50%
	positive_modifier = {
		scale = {
			value = scope:maintenance
			if = {
				limit = { root ?= { has_advance = zhongshu_sheng } }
				multiply = 1.5
			}
		}
		monthly_legitimacy = 0.2
		monthly_celestial_authority = 0.02
	}

	#negative - administrative chaos from an underfunded secretariat raises the cost of political decisions
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		stability_cost_efficiency = -0.1
		monthly_celestial_authority = -0.04
	}
}

# MING

six_boards_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_or_had_tag = CHI
	}

	estates_that_like = {
		crown_estate
	}
	estates_that_dislike = {
		nobles_estate
	}

	on_deactivate = {
		if = {
			limit = {
				has_law = six_ministries_overseer_law
			}
			custom_tooltip = six_boards_deactivate_disables_oversight_law_tt
		}
	}

	#neutral
	neutral_modifier = {
		government_size = 2
	}

	#positive - three_departments_system reform raises scale by 50%; chi_grand_secretariat adds 20%; chi_direct_report adds 10%
	positive_modifier = {
		scale = {
			value = scope:maintenance
			if = {
				limit = { root ?= { has_reform = government_reform:three_departments_system } }
				multiply = 1.5
			}
			if = {
				limit = { root ?= { has_policy = chi_grand_secretariat } }
				multiply = 1.2
			}
			if = {
				limit = { root ?= { has_policy = chi_direct_report } }
				multiply = 1.1
			}
		}
		global_manpower_modifier = 0.1
		monthly_celestial_authority = 0.02
	}

	#negative - six entrenched ministries without funding become fiefdoms of powerful officials
	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		global_estate_power = 0.25
		monthly_celestial_authority = -0.03
	}
}
