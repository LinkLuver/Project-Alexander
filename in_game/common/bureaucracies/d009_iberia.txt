# Spanish Bureaucracies
# Two-tier council system for Castile and Spain.
# Statewide Councils: become available when system_of_councils_reform is adopted.
# Regional Councils: become available when geographic ownership thresholds are met.
# Each Regional Council gates one geographic zone for lieutenancy construction.


# Statewide Councils

spa_council_of_state_bureaucracy = {
	implementation_price = price:implement_bureaucracy_price
	maintenance_price = price:maintain_bureaucracy_price
	removal_price = price:remove_bureaucracy_price
	maintenance_price_modifier = {
		value = country_economical_base
		multiply = 0.004
	}

	potential = {
		has_dlc = "d009_across_the_pillars"
		OR = {
			has_or_had_tag = CAS
			has_or_had_tag = SPA
		}
	}
	
	allow = {
		has_reform = government_reform:system_of_councils_reform
	}

	estates_that_like = {
		nobles_estate
		clergy_estate
	}
	estates_that_dislike = {
		burghers_estate
		peasants_estate
	}

	neutral_modifier = {
		monthly_diplomats = 0.1
	}

	positive_modifier = {
		scale = { value = scope:maintenance }
		global_crown_estate_power = 0.15
	}

	negative_modifier = {
		scale = {
			value = 1
			subtract = scope:maintenance
		}
		parliament_base_support = -0.075
	}
}
