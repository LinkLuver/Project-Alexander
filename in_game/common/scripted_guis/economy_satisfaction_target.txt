# Toggle auto minting (zero inflation). Root = country. 1 = on, 0 = off.
toggle_auto_coin_minting = {
	effect = {
		root = {
			if = { limit = { exists = var:auto_coin_minting }
				if = { limit = { var:auto_coin_minting > 0 }
					set_variable = { name = auto_coin_minting value = 0 }
				} else = {
					set_variable = { name = auto_coin_minting value = 1 }
				}
			} else = {
				set_variable = { name = auto_coin_minting value = 1 }
			}
		}
	}
}
