pa_senate_accept_estate_agenda = {
	scope = country
	saved_scopes = { target agenda }
	effect = {
		root = {
			if = {
				limit = {
					tag = ROM
					is_active_parliament = yes
					has_parliament_agendas = yes
					exists = scope:target
					exists = scope:agenda
					scope:agenda = {
						agenda_for_estate_type = scope:target
						is_available_for = root
					}
				}
				grant_parliament_agenda_for_estate = scope:target
			}
		}
	}
}
