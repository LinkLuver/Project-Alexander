# Country-scoped senate prototype. Values are initialized only once per country.
pa_senate_initialize = {
    effect = {
        root = {
            if = {
                limit = { tag = ROM government_type = government_type:republic NOT = { has_variable = pa_senate_initialized } }
                set_variable = pa_senate_initialized
                set_variable = { name = pa_senate_optimates_approval value = 50 }
                set_variable = { name = pa_senate_boni_approval value = 50 }
                set_variable = { name = pa_senate_populares_approval value = 50 }
                set_variable = { name = pa_senate_optimates_seats value = 120 }
                set_variable = { name = pa_senate_boni_seats value = 105 }
                set_variable = { name = pa_senate_populares_seats value = 75 }
                set_variable = { name = pa_senate_size value = 300 }
                set_variable = { name = pa_senate_pass_threshold value = 153 }
                set_variable = { name = pa_senate_compromise_threshold value = 90 }
                set_variable = { name = pa_senate_authority value = 50 }
                set_variable = { name = pa_senate_last_vote value = 0 }
                set_variable = { name = pa_senate_vote_opt value = 0 }
                set_variable = { name = pa_senate_vote_boni value = 0 }
                set_variable = { name = pa_senate_vote_pop value = 0 }
                set_variable = pa_senate_last_motion_waiting
                set_variable = { name = pa_senate_optimates_defectors value = 0 }
                set_variable = { name = pa_senate_boni_defectors value = 0 }
                set_variable = { name = pa_senate_populares_defectors value = 0 }
                set_variable = { name = pa_senate_optimates_campaign value = 0 }
                set_variable = { name = pa_senate_boni_campaign value = 0 }
                set_variable = { name = pa_senate_populares_campaign value = 0 }
                set_variable = { name = pa_senate_optimates_social_base value = 0 }
                set_variable = { name = pa_senate_boni_social_base value = 0 }
                set_variable = { name = pa_senate_populares_social_base value = 0 }
                set_variable = { name = pa_senate_optimates_noble_support value = 0 }
                set_variable = { name = pa_senate_optimates_burgher_support value = 0 }
                set_variable = { name = pa_senate_optimates_peasant_support value = 0 }
                set_variable = { name = pa_senate_boni_noble_support value = 0 }
                set_variable = { name = pa_senate_boni_burgher_support value = 0 }
                set_variable = { name = pa_senate_boni_peasant_support value = 0 }
                set_variable = { name = pa_senate_populares_noble_support value = 0 }
                set_variable = { name = pa_senate_populares_burgher_support value = 0 }
                set_variable = { name = pa_senate_populares_peasant_support value = 0 }
                set_variable = pa_senate_agenda_stability
                # The founding Senate serves a full term. Elections thereafter
                # are resolved by the yearly pulse when this five-year mandate ends.
                set_variable = { name = pa_senate_election_cooldown value = yes days = 1825 }
            }
        }
    }
}

    effect = { root = {

# The consul calls a country-Parliament debate; seats are awarded only on passage.
pa_senate_expand_membership = {
    effect = { root = {
        if = {
            limit = { tag = ROM government_type = government_type:republic has_ruler = yes has_variable = pa_senate_initialized gold >= 100 var:pa_senate_size > 0 var:pa_senate_size <= 790 is_active_parliament = no NOT = { has_variable = pa_senate_expansion_cooldown } NOT = { has_variable = pa_senate_motion_cooldown } exists = international_organization:pa_roman_senate international_organization:pa_roman_senate = { is_active_parliament = no } }
            parliament_issue:pa_senate_country_expansion = { save_scope_as = parliament_issue }
            set_parliament_location = capital
            set_parliament_active = yes
            set_parliament_issue = parliament_issue:pa_senate_country_expansion
            pa_senate_country_initial_backing = yes
            set_variable = { name = pa_senate_expansion_cooldown value = yes years = 30 }
            set_variable = { name = pa_senate_motion_cooldown value = yes months = 6 }
            set_variable = { name = pa_senate_country_session_guard value = yes days = 101 }
            trigger_event_silently = { id = pa_senate.41 days = 100 }
        }
    } }
}
# A war mandate is an actual permission consumed by the next Roman declaration
# of war. Its vote is faction-weighted in the same way as every other motion.
pa_senate_authorize_war = {
    effect = { root = {
        if = { limit = { is_active_parliament = no has_variable = pa_senate_initialized var:pa_senate_size > 0 NOT = { has_variable = pa_senate_motion_cooldown } exists = international_organization:pa_roman_senate international_organization:pa_roman_senate = { is_active_parliament = no } }
            remove_variable = pa_senate_agenda_stability
            remove_variable = pa_senate_agenda_holdings
            remove_variable = pa_senate_agenda_works
            remove_variable = pa_senate_agenda_law
            set_variable = pa_senate_agenda_war
            set_variable = { name = pa_senate_vote_opt value = { value = var:pa_senate_optimates_seats multiply = var:pa_senate_optimates_approval divide = 125 } }
            set_variable = { name = pa_senate_vote_boni value = { value = var:pa_senate_boni_seats multiply = var:pa_senate_boni_approval divide = 125 } }
            set_variable = { name = pa_senate_vote_pop value = { value = var:pa_senate_populares_seats multiply = var:pa_senate_populares_approval divide = 125 } }
            change_variable = { name = pa_senate_vote_opt subtract = var:pa_senate_optimates_defectors }
            change_variable = { name = pa_senate_vote_boni subtract = var:pa_senate_boni_defectors }
            change_variable = { name = pa_senate_vote_pop subtract = var:pa_senate_populares_defectors }
            # The Optimates and Populares lean toward expansion; the Boni ask
            # for a stronger mandate before accepting it.
            change_variable = { name = pa_senate_vote_opt add = 8 }
            change_variable = { name = pa_senate_vote_boni subtract = 4 }
            change_variable = { name = pa_senate_vote_pop add = 5 }
			# A bloc that dislikes this issue supplies no votes at all. Approval,
			# defections and the issue's ideology therefore all matter.
			if = { limit = { var:pa_senate_optimates_approval < 35 } set_variable = { name = pa_senate_vote_opt value = 0 } }
			if = { limit = { var:pa_senate_boni_approval < 70 } set_variable = { name = pa_senate_vote_boni value = 0 } }
			if = { limit = { var:pa_senate_populares_approval < 40 } set_variable = { name = pa_senate_vote_pop value = 0 } }
			change_variable = { name = pa_senate_vote_opt add = 0 min = 0 max = var:pa_senate_optimates_seats }
			change_variable = { name = pa_senate_vote_boni add = 0 min = 0 max = var:pa_senate_boni_seats }
			change_variable = { name = pa_senate_vote_pop add = 0 min = 0 max = var:pa_senate_populares_seats }
            set_variable = { name = pa_senate_last_vote value = { value = var:pa_senate_vote_opt add = var:pa_senate_vote_boni add = var:pa_senate_vote_pop } }
            set_variable = { name = pa_senate_pass_threshold value = { value = var:pa_senate_size multiply = 51 divide = 100 } }
            set_variable = { name = pa_senate_compromise_threshold value = { value = var:pa_senate_size multiply = 30 divide = 100 } }
            remove_variable = pa_senate_last_motion_waiting
            remove_variable = pa_senate_last_motion_passed
            remove_variable = pa_senate_last_motion_compromise
            remove_variable = pa_senate_last_motion_rejected
			remove_variable = pa_senate_vote_defied
			remove_variable = pa_senate_war_vote_defied
			set_variable = pa_senate_last_motion_waiting
			set_variable = { name = pa_senate_half_seats value = { value = var:pa_senate_size divide = 2 } }
			save_scope_as = pa_senate_proposing_rome
			international_organization:pa_roman_senate = {
				set_parliament_location = scope:pa_senate_proposing_rome.capital
				set_parliament_active = yes
				set_parliament_issue = parliament_issue:pa_senate_war_debate
				scope:pa_senate_proposing_rome = {
					if = {
						limit = { var:pa_senate_last_vote > var:pa_senate_half_seats }
						set_participated_in_parliament = { international_organization = international_organization:pa_roman_senate vote = yes }
					}
					else = { set_participated_in_parliament = { international_organization = international_organization:pa_roman_senate vote = no } }
				}
				set_parliament_issue_support = { value = scope:pa_senate_proposing_rome.var:pa_senate_last_vote divide = scope:pa_senate_proposing_rome.var:pa_senate_size }
			}
            set_variable = { name = pa_senate_motion_cooldown value = yes months = 6 }
        }
    } }
}
# The consul may overrule a failing debate, paying tyranny and factional anger.
# With no active debate this remains the emergency war-mandate action.
pa_senate_defy_senate_for_war = {
    effect = { root = {
        if = { limit = {
			has_variable = pa_senate_initialized
			exists = international_organization:pa_roman_senate
			NOT = { has_variable = pa_senate_vote_defied }
			OR = {
				AND = { is_active_parliament = yes has_ongoing_parliament_debate = yes parliament_issue_support <= 0.5 }
				international_organization:pa_roman_senate = { has_ongoing_parliament_debate = yes parliament_issue_support <= 0.5 }
				AND = { is_active_parliament = no international_organization:pa_roman_senate = { is_active_parliament = no } NOT = { has_variable = pa_senate_war_mandate } }
			}
		}
            change_variable = { name = pa_senate_tyranny add = 15 max = 100 }
            change_variable = { name = pa_senate_authority subtract = 12 min = 0 }
            change_variable = { name = pa_senate_boni_approval subtract = 8 min = 0 }
            change_variable = { name = pa_senate_boni_crisis add = 2 max = 12 }
			if = {
				limit = { is_active_parliament = yes has_ongoing_parliament_debate = yes }
				set_variable = pa_senate_vote_defied
				set_parliament_issue_support = 1
			}
			else_if = {
				limit = {
					international_organization:pa_roman_senate = { has_ongoing_parliament_debate = yes }
				}
				set_variable = pa_senate_vote_defied
				set_variable = pa_senate_war_vote_defied
				unset_participated_in_parliament = international_organization:pa_roman_senate
				set_participated_in_parliament = { international_organization = international_organization:pa_roman_senate vote = yes }
				international_organization:pa_roman_senate = { set_parliament_issue_support = 1 }
			}
			else = { set_variable = { name = pa_senate_war_mandate value = yes days = 365 } }
        }
    } }
}
