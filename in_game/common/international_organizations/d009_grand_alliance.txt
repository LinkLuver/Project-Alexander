grand_alliance = {
	unique = yes
	has_target = yes
	embargo = yes
	show_on_diplomatic_map = yes
	show_strength_comparison_with_target = yes

	has_leader_country = yes
	leader_type = country
	leader_change_method = score
	leader_change_trigger_type = timed
	months_between_leader_changes = 12
	disband_if_no_leader = no
	promote_strongest_member_to_war_leader = yes
	leader_score = {
		value = great_power_score
	}

	gives_food_access_to_members = yes
	has_military_access = {
		scope:recipient ?= { is_member_of_international_organization = root }
		scope:actor ?= { is_member_of_international_organization = root }
	}

	expel_members_who_are_targets_of_other_members = yes

	leader_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_LEADER_COLOR
	target_color = define:NMapColors|INTERNATIONAL_ORGANIZATION_TARGET_COLOR

	diplomatic_capacity_cost = {
		scope:recipient = {
			add = {
				desc  = "UPKEEP_NUM_MEMBERS"
				value = total_members
				multiply = 0.1
			}
			add = {
				desc  = "UPKEEP_TOTAL_ECO_BASE"
				value = international_organization_economical_base
				subtract = root.country_economical_base
				multiply = 0.00025
			}
		}
	}

	join_defensive_wars_always = {
		scope:target ?= { this = root.international_organization_target }
		NOT = { has_cooldown = ga_declare_war_on_hegemon }
	}
	join_offensive_wars_always = {
		scope:target ?= { this = root.international_organization_target }
		trigger_if = {
			limit = { exists = scope:casus_belli }
			scope:casus_belli = casus_belli:cb_grand_alliance
		}
		trigger_else = { always = no }
	}
	can_declare_war = {
		trigger_if = {
			limit = {
				scope:attacker ?= { is_member_of_international_organization = scope:recipient }
				scope:defender ?= { this = scope:recipient.international_organization_target }
			}
			custom_tooltip = {
				text = ga_dominance_threshold_tt
				trigger_if = {
					limit = {
						situation:golden_century = {
							has_variable = gc_european_dominance
							has_variable = gc_colonial_dominance
						}
					}
					OR = {
						situation:golden_century.var:gc_european_dominance >= 75
						situation:golden_century.var:gc_colonial_dominance >= 75
					}
				}
				trigger_else = { always = no }
			}
		}
		trigger_else = { always = yes }
	}

	modifier = {
		potential_trigger = { situation:golden_century.var:gc_european_dominance >= 25 }
		global_defensive = 0.1
		diplomatic_reputation = 1
	}
	modifier = {
		potential_trigger = { situation:golden_century.var:gc_european_dominance >= 50 }
		monthly_war_exhaustion = -0.05
		monthly_diplomats = 0.1
	}
	modifier = {
		potential_trigger = { situation:golden_century.var:gc_european_dominance >= 75 }
		global_war_score_efficiency = 0.1
		global_garrison_size_modifier = 0.2
	}
	modifier = {
		potential_trigger = { situation:golden_century.var:gc_european_dominance >= 100 }
		global_manpower_modifier = 0.1
		discipline = 0.05
	}

	modifier = {
		potential_trigger = { situation:golden_century.var:gc_colonial_dominance >= 25 }
		spy_network_construction = 0.33
		ship_build_speed = 0.2
	}
	modifier = {
		potential_trigger = { situation:golden_century.var:gc_colonial_dominance >= 50 }
		blockade_efficiency   = 0.5
		navy_light_ship_power = 0.1
	}
	modifier = {
		potential_trigger = { situation:golden_century.var:gc_colonial_dominance >= 75 }
		naval_morale = 0.1
		naval_range_modifier = 0.1
	}
	modifier = {
		potential_trigger = { situation:golden_century.var:gc_colonial_dominance >= 100 }
		naval_damage_taken = -0.05
		naval_damage_done  = 0.05
	}

	create_visible_trigger = { always = no }
	invite_visible_trigger = { always = yes }

	potential_target_trigger = {
		trigger_if = {
			limit = { has_global_variable = battle_for_the_strait_winner_country }
			this = global_var:battle_for_the_strait_winner_country
		}
		
	}
	can_target_trigger = {
		trigger_if = {
			limit = { has_global_variable = battle_for_the_strait_winner_country }
			this = global_var:battle_for_the_strait_winner_country
		}
	}

	can_join_trigger = {
		trigger_if = {
			limit = { at_war = yes }
			is_at_war_with = scope:target
		}
		exists = scope:target
		NOT = { this = scope:target }
		is_rebel_country = no
		custom_tooltip = {
			text = grand_alliance_no_recent_leave_tt
			NOT = { has_variable = recently_left_grand_alliance }
		}
		OR = {
			is_subject = no
			is_subject_type = tributary
		}
		knows_country          = scope:target
		within_diplomatic_range = scope:target
		OR = {
			AND = {
				OR = {
					is_rival_of = scope:target
					is_enemy_of = scope:target
				}
				border_distance_to = {
					country = scope:target
					value < 500
				}
			}
			border_distance_to = {
				country = scope:target
				value < 250
			}
		}
		NOT = { is_subject_or_below_of = scope:target }
		NOT = { is_allied_with = { target = scope:target } }
		NOT = { in_union_with = scope:target }
		OR = {
			AND = {
				antagonism = { target = scope:target value > 30 }
				opinion    = { target = scope:target value < 0 }
			}
			is_rival_of = scope:target
			is_enemy_of = scope:target
		}
	}

	can_leave_trigger = {
		trigger_if = {
			limit = { root = scope:recipient.leader_country }
			custom_tooltip = {
				text = grand_alliance_leader_cannot_leave_tt
				NOT = { scope:recipient.leader_country ?= root }
			}
		}
		trigger_if = {
			limit = { exists = scope:target }
			NOT = { is_at_war_with = scope:target }
		}
	}

	auto_leave_trigger = {
		NOT = { scope:recipient.leader_country ?= root }
		trigger_if = {
			limit = { exists = scope:target }
			OR = {
				AND = {
					is_subject = yes
					NOT = { is_subject_type = tributary }
				}
				is_subject_or_below_of = scope:target
				in_union_with          = scope:target
				is_allied_with = { target = scope:target }
				AND = {
					not = { is_at_war_with = scope:target }
					not = { is_rival_of   = scope:target }
					not = { is_enemy_of   = scope:target }
					antagonism = { target = scope:target value < 5 }
					opinion    = { target = scope:target value > 25 }
				}
				AND = {
					not = { is_at_war_with = scope:target }
					not = { this = scope:recipient.leader_country }
					trigger_if = {
						limit = {
							situation:golden_century = {
								has_variable = gc_european_dominance
								has_variable = gc_colonial_dominance
							}
						}
						situation:golden_century.var:gc_european_dominance < 25
						situation:golden_century.var:gc_colonial_dominance < 25
					}
					trigger_else = { always = no }
				}
			}
		}
		trigger_else = { always = yes }
	}

	auto_disband_trigger = {
		OR = {
			total_members < 1
			not = { country_exists = scope:target }
			NOT = { is_situation_active = situation:golden_century }
		}
	}
	
	on_left = {
		remove_casus_belli = {
			target = scope:target
			type   = casus_belli:cb_grand_alliance
		}
		set_variable = {
			name = recently_left_grand_alliance
			days = 1825
		}
	}

	ai_desire_to_join = {
		add = {
			desc = "grand_alliance_hostility"
			if = {
				limit = {
					OR = {
						is_enemy_of    = scope:target
						is_rival_of    = scope:target
					}
				}
				value = 100
			}
		}
		add = {
			desc = "game_concept_antagonism"
			scope:target = {
				value = prev_antagonism_towards_this
			}
		}
		add = {
			desc     = "OPINION"
			value    = "opinion(scope:target)"
			multiply = -1
		}
		add = {
			desc = "DISTANCE_TO_TARGET"
			scope:target = {
				value    = "border_distance_to(root)"
				multiply = -0.1
			}
			min = -100
			max = 0
		}
		add = {
			desc = "ga_colossus_on_our_border"
			value = 0
			if = {
				limit = { is_neighbor_of = scope:target }
				value = 20
			}
		}
		add = {
			desc = "grand_alliance_reputable_leader"
			if = {
				limit = {
					exists = scope:recipient
					exists = scope:recipient.leader_country
				}
				value    = scope:recipient.leader_country.modifier:diplomatic_reputation
				subtract = scope:target.modifier:diplomatic_reputation
				multiply = 4
			}
		}
		add = {
			desc = "ga_fear_of_dominance"
			value = 0
			if = {
				limit = {
					situation:golden_century = {
						has_variable = gc_european_dominance
						has_variable = gc_colonial_dominance
					}
				}
				add = situation:golden_century.var:gc_european_dominance
				add = situation:golden_century.var:gc_colonial_dominance
				multiply = 0.25
			}
			min = 0
			max = 50
		}
		add = {
			desc = "ga_dominance_tipping_point"
			value = 0
			if = {
				limit = {
					situation:golden_century = {
						has_variable = gc_european_dominance
						has_variable = gc_colonial_dominance
						OR = {
							var:gc_european_dominance >= 50
							var:gc_colonial_dominance >= 50
						}
					}
				}
				value = 15
			}
		}
	}

	ai_desire_to_allow_new_member = {
		add = {
			desc  = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}
}
