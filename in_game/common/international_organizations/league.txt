league = {
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no
	show_leave_message = no

	has_leader_country = no
	land_ownership_rule = league_land_ownership
	gives_military_access_to_all_when_at_war = yes
	can_declare_war = {
		always = yes
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}
	can_declare_war = {
		trigger_if = {
			limit = { 
				scope:attacker ?= { is_member_of_international_organization = scope:recipient } 
				or = {
					international_organization_has_law = law:league_military_assistance
					and = {
						international_organization_has_law = law:league_legal_jurisdiction
						international_organization_has_policy = policy:league_dikai
					}
				}

			}
			scope:defender ?= { NOT = { is_member_of_international_organization = scope:recipient } }
		}
		trigger_else = {
			always = yes
		}
	}
	can_join_trigger = {
		scope:recipient = {
			international_organization_has_law = law:sc_membership_regulation
		}
		is_rebel_country = no
	}
	can_leave_trigger = {
		always = no
	}
	auto_leave_trigger = {
		always = no

	}
	variables = {
		

	}
}
