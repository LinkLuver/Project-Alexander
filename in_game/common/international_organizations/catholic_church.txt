catholic_church = {
	has_target = no
	unique = yes
	expel_members_who_are_targets_of_other_members = no
	disband_if_no_leader = no
	show_leave_message = no

	max_active_resolutions = 1

	has_leader_country = yes
	leader_title_key = "CATHOLIC_CHURCH_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			ruler ?= {
				add_to_list = leaders
			}
		}
	}

	can_initiate_policy_votes = {
		is_member_of_international_organization = scope:recipient
		is_situation_active = situation:council_of_trent
	}
	
	can_declare_war = {
		always = yes
	}

	gold = yes
	international_organization_modifier = {
		unlock_withdraw_from_organization_treasury = yes
	}
	
	variables = {
		papal_authority = {
			format = "PAPAL_AUTHORITY_DISPLAY"
			change_format = "VARIABLE_CHANGE_FORMAT"
			min = 0
			max = 100
			start = 60
			monthly_change = {
				# Reform desire drain - main feedback loop
				subtract = {
					desc = "PA_REFORM_DESIRE"
					value = religion:catholic.reform_desire
					multiply = 0.2
				}

				# Internal peace bonus
				add = {
					desc = "INTERNAL_PEACE"
					if = {
						limit = { international_organization_has_internal_peace = no }
						value = 0
					}
					else = {
						value = 0.03
					}
				}

				# Doctrine: Ultramontanism
				if = {
					limit = { international_organization_has_policy = policy:ultramontanism }
					add = {
						desc = "PA_ULTRAMONTANISM"
						value = 0.02
					}
				}

				# Doctrine: Gallicanism
				if = {
					limit = { international_organization_has_policy = policy:gallicanism }
					subtract = {
						desc = "PA_GALLICANISM"
						value = 0.01
					}
				}

				# Doctrine: Conciliarism
				if = {
					limit = { international_organization_has_policy = policy:conciliarism }
					subtract = {
						desc = "PA_CONCILIARISM"
						value = 0.025
					}
				}

				# Doctrine: Invisible Church
				if = {
					limit = { international_organization_has_policy = policy:invisible_church }
					subtract = {
						desc = "PA_INVISIBLE_CHURCH"
						value = 0.05
					}
				}

				# PAP controls Rome
				if = {
					limit = {
						exists = leader_country
						leader_country = { owns = location:rome }
					}
					add = {
						desc = "PA_CONTROLS_ROME"
						value = 0.02
					}
				}

				# Cardinals across Catholic nations
				add = {
					desc = "PA_CARDINALS"
					value = 0
					every_international_organization_member = {
						add = num_cardinals
					}
					multiply = 0.002
				}

				# Excommunicated rulers still on throne
				subtract = {
					desc = "PA_EXCOMMUNICATED_RULERS"
					value = 0
					every_international_organization_member = {
						limit = { is_excommunicated = yes }
						add = 0.02
					}
				}

				# Members who can ignore papal bulls
				subtract = {
					desc = "PA_CAN_IGNORE_BULLS"
					value = 0
					every_international_organization_member = {
						limit = { modifier:can_ignore_papal_bulls = yes }
						add = 0.01
					}
				}

				# Interregnum - no Pope
				if = {
					limit = { international_organization_has_leader = no }
					subtract = {
						desc = "PA_INTERREGNUM"
						value = 0.5
					}
				}

				# Pope's monthly_papal_authority modifier (scaled by papal_authority_modifier)
				if = {
					limit = {
						international_organization_has_leader = yes
						NOT = { leader_country = { modifier:monthly_papal_authority = 0 } }
					}
					add = {
						desc = "FROM_MODIFIERS"
						value = leader_country.modifier:monthly_papal_authority
						multiply = leader_country.modifier:papal_authority_modifier
						add = leader_country.modifier:monthly_papal_authority
					}
				}
			}
		}

		#keep this as a var so it gets auto-added to the religion
		religion = {
			format = "RELIGION_FORMAT"
			start = religion:catholic
		}
	}

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_lead_trigger = {
		this = c:PAP
	}

	can_join_trigger = {
		religion = religion:catholic
		is_rebel_country = no
	}
	
	can_leave_trigger = {
		always = no
	}
	
	auto_leave_trigger = {
		not = {
			religion = religion:catholic
		}
	}

	auto_disband_trigger = {
		always = no
	}

	on_joined = {
		if = {
			limit = {
				scope:recipient = {
					or = {
						not = { has_variable = religion }
						not = { exists = var:religion }
					}
				}
			}
			#default the religion to that of the leader
			scope:recipient = {
				set_variable = {
					name = religion
					value = root.religion
				}
				var:religion = {
					add_international_organization = scope:recipient
				}
			}
		}
		if = {
			limit = { is_situation_active = situation:council_of_trent }
			add_to_global_variable_list = { name = council_of_trent_voters target = scope:actor }
		}
	}

	on_left = {
		if = {
			limit = { is_situation_active = situation:council_of_trent }
			remove_list_global_variable = { name = council_of_trent_voters target = scope:actor }
			remove_list_global_variable = { name = council_of_trent_unvoters target = scope:actor }
			remove_list_global_variable = { name = council_of_trent_voting_for_debate target = scope:actor }
			remove_list_global_variable = { name = council_of_trent_voting_against_debate target = scope:actor }
		}
	}
	
	ai_desire_to_join = {
	}

	payments_implemented = {
		tithe
	}

	special_statuses_implemented = {
		curia
		military_order
		bishopric
	}

	secondary_map_color_override = {
		#show cardinal locations
		if = {
			limit = {
				exists = cardinal
			}
			value = rgb { 250 250 70 }
		}
	}
	tooltip = {
		if = {
			limit = {
				exists = cardinal
			}
			custom_tooltip = { text = LOCATION_HAS_CARDINAL }
		}
	}
	
	ai_desire_to_allow_new_member = {
		add = {
			desc = "WE_WANT_THE_ORGANIZATION_TO_GROW"
			value = 20
		}
	}

	ai_issue_voting_bias = {
		add = {
			desc = "INERTIA"
			value = -1
		}
		
		#The Pope wants reforms only if desire is meaningfully present
		add = {
			desc = "POLICY_BIAS_REFORM_DESIRE"
			if = {
				limit = {
					scope:recipient = {
						leader_country = scope:actor
					}
				}
				#this should end up with 10 for high reform desire and -10 for low reform desire
				if = {
					limit = {
						religion:catholic = {
							reform_desire >= 0.5
						}
					}
					value = 1
				}
				else = {
					value = -1
				}
				multiply = 10
			}
		}

		#Member Vote
		add = {
			desc = "POLICY_BIAS_REFORM_DESIRE"
			if = {
				limit = { not = { scope:actor = scope:recipient.leader_country } }
				scope:recipient = {
					leader_country = {
						if = {
							limit = {
								is_set = scope:vote
							}
							value = 1
						}
						else = {
							value = -1
						}
						multiply = scope:actor.religion.reform_desire
						multiply = 20
					}
				}
			}
		}
		#Papal Authority - high PA means members align with the Pope, low PA means they resist
		add = {
			desc = "PAPAL_AUTHORITY"
			if = {
				limit = { not = { scope:actor = scope:recipient.leader_country } }
				value = scope:recipient.var:papal_authority
				subtract = 50
				multiply = 0.3
			}
		}
	}
}
