goods_camels.1 = {
	type = country_event
	title = goods_camels.1.title
	desc = goods_camels.1.desc
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_camels.dds"
	outcome = neutral

	trigger = {
		has_dlc = "d009_across_the_pillars"
		any_owned_location = {
			raw_material = goods:camels
		}
	}

	immediate = {
		random_owned_location = {
			limit = { raw_material = goods:camels }
			save_scope_as = target_location
		}
	}

	option = {
		name = goods_camels.1.a
		change_gold_effect = { scale = 2.0 }
		add_prestige = prestige_weak_bonus
		scope:target_location = {
			change_max_raw_material_workers = 1
		}
	}
	option = {
		name = goods_camels.1.b
		add_army_tradition = army_tradition_severe_bonus
	}
}

goods_camels.9 = {
	type = country_event
	title = goods_camels.9.title
	desc = goods_camels.9.desc
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_camels.dds"
	outcome = neutral

	trigger = {
		has_dlc = "d009_across_the_pillars"
		any_owned_location = { raw_material = goods:camels }
		has_ruler = yes
		any_country = {
			is_great_power = yes
			this != root
			within_diplomatic_range = root
			NOT = { is_at_war_with = root }
			any_cabinet_character = { always = yes }
		}
	}

	immediate = {
		random_country = {
			limit = {
				is_great_power = yes
				this != root
				within_diplomatic_range = root
				NOT = { is_at_war_with = root }
				any_cabinet_character = { always = yes }
			}
			save_scope_as = offended_country
			random_cabinet_character = {
				save_scope_as = target_character
			}
		}
	}

	option = {
		name = goods_camels.9.a
		change_gold_effect = { scale = -2.0 }
		scope:offended_country = {
			add_opinion = { target = root modifier = opinion_pleased }
		}
	}
	option = {
		name = goods_camels.9.b
		add_prestige = prestige_weak_bonus
		scope:offended_country = {
			add_opinion = { target = root modifier = opinion_insulted }
		}
	}
}
