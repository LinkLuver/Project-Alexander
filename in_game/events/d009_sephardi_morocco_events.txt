d009_sephardi_morocco.1 = { #The Megorashim Arrive
	type = country_event
	title = d009_sephardi_morocco.1.title
	desc = d009_sephardi_morocco.1.desc
	outcome = neutral
	historical_info = d009_sephardi_morocco.1.historical_info

	fire_only_once = yes

	dynamic_historical_event = {
		tag = MOR
		from = 1492.1.1
		to = 1520.1.1
		monthly_chance = 25
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		religion = religion:sunni
		owns = location:fez
		owns = location:tetouan

		any_in_global_list = {
			variable = crypto_jewish_community_locations
			owner = {
				culture = { has_culture_group = culture_group:iberian_group }
				NOT = { this = root }
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }

		random_in_global_list = {
			variable = crypto_jewish_community_locations
			limit = {
				owner = {
					culture = { has_culture_group = culture_group:iberian_group }
					NOT = { this = root }
				}
			}

			owner = { save_scope_as = origin_country }
		}
	}

	option = {
		name = d009_sephardi_morocco.1.a
		historical_option = yes

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_right
		}

		if = {
			limit = {
				is_situation_active = situation:golden_century
				has_global_variable = battle_for_the_strait_winner_country
				this = global_var:battle_for_the_strait_winner_country
			}
			gc_change_european_dominance = { value = gc_dominance_weak_bonus }
		}

		scope:origin_country = {
			d009_crypto_jewish_exodus_effect = {
				FRACTION = 0.2
				AMOUNT = 0.3
				DESTINATION = location:fez
			}
		}

		if = {
			limit = {
				location:fez = {
					"religion_population(religion:judaism)" > 0
				}
			}

			location:fez = {
				add_location_modifier = {
					modifier = mellah_split
					mode = add
				}
			}
		}
	}

	option = {
		name = d009_sephardi_morocco.1.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_left
		}

		scope:origin_country = {
			d009_crypto_jewish_exodus_effect = {
				FRACTION = 0.1
				AMOUNT = 0.15
				DESTINATION = location:tetouan
			}
		}
	}
}
