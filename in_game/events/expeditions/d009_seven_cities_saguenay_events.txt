	outcome = positive

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
		}
		location:stadacona = {
			random_present_culture_in_location = { save_scope_as = target_culture }
			random_present_religion_in_location = { save_scope_as = target_religion }
		}
		create_character = {
			age = { 20 35 }
			culture = scope:target_culture
			religion = scope:target_religion
			birth_location = location:stadacona
			estate = estate_type:peasants_estate
			save_scope_as = target_character
			create_in_limbo = yes
		}
	}

	option = {
		name = seven_cities_saguenay.10.a
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			random_list = {
				7 = {
					scope:expedition = { change_variable = { name = leads_found add = 1 } }
				}
				3 = { }
			}
		}
		custom_tooltip = {
			text = saguenay_chiefs_son_sails_tt
			scope:expedition = {
				set_variable = { name = saguenay_chief_aboard value = 1 }
				set_variable = {
					name = saguenay_chief_character_var
					value = scope:target_character
				}
			}
		}
	}

	option = {
		name = seven_cities_saguenay.10.b
		add_prestige = prestige_weak_bonus
		hidden_effect = { kill_character_silently = scope:target_character }
	}
}

seven_cities_saguenay.11 = {
#The White Sickness
	type = country_event
	title = seven_cities_saguenay.11.title
	desc = seven_cities_saguenay.11.desc
	historical_info = seven_cities_saguenay.11.historical_info
	outcome = negative

	location_backdrop = { backdrop_preset = backdrop_at_sea }

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
		}
		location:stadacona = { save_scope_as = target_location }
	}

	option = {
		name = seven_cities_saguenay.11.a
		change_gold_effect = { scale = -1 }
		if = {
			limit = { scope:expedition = { has_variable = saguenay_chief_aboard } }
			custom_tooltip = saguenay_annedda_cure
			custom_tooltip = {
				text = expedition_healer_joins_court_tt
				scope:expedition = {
					set_variable = {
						name = healer_location
						value = scope:target_location
					}
				}
			}
			if = {
				limit = {
					scope:target_location = {
						exists = dominant_culture
						dominant_culture = { NOR = { this = ROOT.culture is_accepted_in = ROOT } }
					}
				}
				custom_tooltip = {
					text = improve_root_culture_view_of_location_dominant_culture_tt
					ROOT.culture = {
						change_cultural_view = {
							target = scope:target_location.dominant_culture
							change = 1
						}
					}
				}
			}
		}
		else = {
			custom_tooltip = saguenay_strangers_mercy
			scope:expedition = { pause_expedition = { 10 20 } }
		}
	}

	option = {
		name = seven_cities_saguenay.11.b
		scope:expedition = { pause_expedition = { 15 25 } }
	}
}
