namespace = mining_survey_events

# mining_survey_events.1 — Departure (on_start)
mining_survey_events.1 = {
	type = country_event
	title = mining_survey_events.1.title
	desc = mining_survey_events.1.desc
	historical_info = mining_survey_events.1.historical_info
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_field_survey }

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= {
				save_scope_as = current_location
			}
		}
		stall_expedition = scope:expedition
	}

	# Outfit the team properly: assayers, smiths, surveyors
	option = {
		name = mining_survey_events.1.a
		change_gold_effect = { scale = -2 }
		add_prestige = prestige_weak_bonus
		scope:expedition = {
			pause_expedition = 10
		}
	}

	# The local headmen will provide what we need
	option = {
		name = mining_survey_events.1.b
		change_gold_effect = { scale = -1 }
	}

	after = {
		resume_expedition = scope:expedition
	}
}
