pacific_crossing_events.1 = {
	type = country_event
	title = pacific_crossing_events.1.title
	desc = pacific_crossing_events.1.desc
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = {
		any_character = { is_expedition_leader = yes }
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= { save_scope_as = current_location }
		}
		stall_expedition = scope:expedition
	}

	option = {
		name = pacific_crossing_events.1.a
		change_gold_effect = { scale = -2 }
		custom_tooltip = pacific_crossing_events.1.a.tt
		scope:expedition = {
			change_variable = { name = num_ships add = 1 }
			clamp_variable = { name = num_ships min = 0 max = 6 }
			pause_expedition = 30
		}
	}

	option = {
		name = pacific_crossing_events.1.b
		change_gold_effect = { scale = -1 }
	}

	option = {
		name = pacific_crossing_events.1.c
		trigger = {
			scope:target_leader = { has_trait = navigator }
		}
		add_prestige = prestige_weak_bonus
	}

	after = {
		resume_expedition = scope:expedition
	}

	historical_info = pacific_crossing_events.1.historical_info
}
