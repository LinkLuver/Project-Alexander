seven_cities_norumbega.12 = {
#Smoke on the River
	type = country_event
	title = seven_cities_norumbega.12.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { scope:expedition = { has_variable = norumbega_walker_aboard } }
				desc = seven_cities_norumbega.12.desc_walker
			}
			triggered_desc = { desc = seven_cities_norumbega.12.desc }
		}
	}
	historical_info = seven_cities_norumbega.12.historical_info
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_wilderness_landing }

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
			pause_expedition = 10
		}
		scope:location = { save_scope_as = target_location }
	}

	option = {
		name = seven_cities_norumbega.12.a
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			random_list = {
				1 = {
					scope:expedition = { change_variable = { name = leads_found add = 1 } }
				}
				1 = { }
			}
		}
		if = {
			limit = { scope:expedition = { NOT = { has_variable = norumbega_walker_aboard } } }
			scope:expedition = { pause_expedition = { 10 20 } }
		}
	}

	option = {
		name = seven_cities_norumbega.12.b
		scope:target_leader = {
			if = {
				limit = { 
					num_of_traits_of_category = { type = explorer value < 1 }
				}
				add_trait = trait:cartographer
			}
			else = { 
				add_adm = { 5 10 }
				add_dip = { 5 10 }
				add_mil = { 5 10 }
			}
		}
		capital = {
			create_art = {
				type = work_of_art_type:chronicle
				artist = scope:target_leader
				quality = 60
			}
		}
	}

	option = {
		name = seven_cities_norumbega.12.c
		trigger = { scope:target_leader = { has_trait = treasure_hunter } }
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 2 } }
		}
	}
}
