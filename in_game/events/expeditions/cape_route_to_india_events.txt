		}

		name = cape_route_to_india_events.4.a

		add_prestige = prestige_mild_bonus
		add_country_modifier = {
			modifier = kongo_diplomatic_contact
			days = 3650
		}
	}

		name = cape_route_to_india_events.4.b
		change_gold_effect = { scale = -1 }
		custom_tooltip = cape_route_to_india_events.supplies_up_10_tt

				name = cargo

					value = market_goods_weak_gain
