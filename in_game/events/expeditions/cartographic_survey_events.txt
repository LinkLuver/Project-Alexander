namespace = cartographic_survey_events

# cartographic_survey_events.1 — Departure (on_start)
cartographic_survey_events.1 = {
	type = country_event
	title = cartographic_survey_events.1.title
	desc = cartographic_survey_events.1.desc
	historical_info = cartographic_survey_events.1.historical_info
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= {
				save_scope_as = current_location
			}
		}
		stall_expedition = scope:expedition
	}

	# Commission the finest instruments and chart-makers
	option = {
		name = cartographic_survey_events.1.a
		change_gold_effect = { scale = -2 }
		add_prestige = prestige_weak_bonus
		scope:expedition = {
			pause_expedition = 10
		}
	}

	# The standard chart-makers will do
	option = {
		name = cartographic_survey_events.1.b
		change_gold_effect = { scale = -1 }
	}

	after = {
		resume_expedition = scope:expedition
	}
}
