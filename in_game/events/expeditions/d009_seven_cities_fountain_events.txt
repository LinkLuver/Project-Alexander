	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
			pause_expedition = 14
		}
		scope:location = { save_scope_as = target_location }
	}

	option = {
		name = seven_cities_fountain.1.a
		add_prestige = -5
		fail_expedition = scope:expedition
	}

	historical_info = seven_cities_fountain.1.historical_info
}

# .2 — medium outcome
seven_cities_fountain.2 = {
	type = country_event
	title = seven_cities_fountain.2.title
	desc = seven_cities_fountain.2.desc
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_wilderness_landing }

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
			pause_expedition = 14
		}
		scope:location = { save_scope_as = target_location }
	}

	option = {
		name = seven_cities_fountain.2.a
		add_prestige = prestige_mild_bonus
		change_gold_effect = { scale = 1 }
		scope:location = {
			add_location_modifier = {
				modifier = fountain_of_youth_discovered
				days = -1
			}
		}
		grant_seven_cities_expedition_loot_effect = yes
		if = {
			limit = { scope:expedition = { has_variable = rejuvenating_water_variable } }
			if = {
				limit = { has_ruler = yes }
				ruler = {
					add_character_modifier = {
						modifier = rejuvenating_waters_modifier
						years = { 5 10 }
						mode = extend
						size = { value = 1  add = scope:expedition.var:rejuvenating_water_variable }
					}
				}
			}
			if = {
				limit = { has_heir = yes }
				heir = {
					add_character_modifier = {
						modifier = rejuvenating_waters_modifier
						years = { 5 10 }
						mode = extend
						size = { value = 1  add = scope:expedition.var:rejuvenating_water_variable }
					}
				}
			}
		}
		end_expedition = scope:expedition
	}

	historical_info = seven_cities_fountain.2.historical_info
}
