hajj_caravan_events.1 = {
	type = country_event
	title = hajj_caravan_events.1.title
	desc = hajj_caravan_events.1.desc
	historical_info = hajj_caravan_events.1.historical_info
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			save_scope_as = hajj_expedition
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= { save_scope_as = current_location }
		}
		stall_expedition = scope:expedition
	}

	option = {
		name = hajj_caravan_events.1.a
		ai_chance = { base = 50 }
		add_prestige = prestige_weak_bonus
	}

	option = {
		name = hajj_caravan_events.1.b
		ai_chance = {
			factor = 30
			modifier = {
				factor = 0
				gold <= root.monthly_income_trade_and_tax
			}
		}
		change_gold_effect = { scale = -2 }
		if = {
			limit = { root.monthly_manpower > 0 }
			add_manpower = { value = root.monthly_manpower multiply = -3 }
		}
		else_if = {
			limit = { manpower > 0 }
			add_manpower = { value = manpower multiply = -0.3 }
		}
		add_prestige = prestige_weak_bonus
		custom_tooltip = hajj_caravan_events.guards_up_tt
		custom_tooltip = hajj_caravan_events.splendor_up_15_tt
		scope:expedition = {
			change_variable = { name = guards add = 25 }
			clamp_variable = { name = guards min = 0 max = 90 }
			change_variable = { name = splendor add = 15 }
			clamp_variable = { name = splendor min = 0 max = 100 }
		}
	}

	option = {
		name = hajj_caravan_events.1.c
		ai_chance = {
			factor = 20
			modifier = {
				factor = 0
				gold <= root.monthly_income_trade_and_tax
			}
		}
		change_gold_effect = { scale = -5 }
		if = {
			limit = { root.monthly_manpower > 0 }
			add_manpower = { value = root.monthly_manpower multiply = -3 }
		}
		else_if = {
			limit = { manpower > 0 }
			add_manpower = { value = manpower multiply = -0.3 }
		}
		custom_tooltip = hajj_caravan_events.guards_up_tt
		custom_tooltip = hajj_caravan_events.splendor_up_30_tt
		custom_tooltip = hajj_caravan_events.mahmal_gained_tt
		scope:expedition = {
			set_variable = mahmal
			change_variable = { name = guards add = 25 }
			clamp_variable = { name = guards min = 0 max = 90 }
			change_variable = { name = splendor add = 30 }
			clamp_variable = { name = splendor min = 0 max = 100 }
		}
		add_prestige = prestige_mild_bonus
		add_legitimacy = legitimacy_mild_bonus
	}

	after = {
		resume_expedition = scope:expedition
	}

}

			set_variable = raid_fired
		}
	}

	option = {
		name = hajj_caravan_events.9.a
		ai_chance = {
			factor = 60
			modifier = {
				factor = 2
				scope:expedition = { var:guards < 30 }
			}
		}
		change_gold_effect = { scale = -2 }
		custom_tooltip = hajj_caravan_events.splendor_down_10_tt

				factor = 2
				scope:expedition = { var:guards >= 50 }
			}
		}
		custom_tooltip = hajj_caravan_events.9.b.chance_tt
		set_variable = { name = raid_won value = 0 }
		hidden_effect = {
			random = {
				chance = { value = 5 add = scope:expedition.var:guards }
				set_variable = { name = raid_won value = 1 }
			}
		}
		if = {
			limit = { var:raid_won = 1 }
			add_prestige = prestige_weak_bonus
		}
		else = {
			if = {
				limit = { scope:expedition = { has_variable = mahmal } }
				scope:expedition = { remove_variable = mahmal }
			}
			custom_tooltip = hajj_caravan_events.guards_down_20_tt
			custom_tooltip = hajj_caravan_events.splendor_down_15_tt
			scope:expedition = {
				pause_expedition = 15
				change_variable = { name = guards add = -20 }
				clamp_variable = { name = guards min = 0 max = 90 }
				change_variable = { name = splendor add = -15 }
				clamp_variable = { name = splendor min = 0 max = 100 }
			}
		}
		remove_variable = raid_won
	}

	option = {
		name = hajj_caravan_events.9.c
		ai_chance = {
			factor = 30
			modifier = {
				factor = 2
				scope:target_leader = { has_trait = zealot }
			}
		}
		custom_tooltip = hajj_caravan_events.9.c.chance_tt
		set_variable = { name = raid_appeal_won value = 0 }
		hidden_effect = {
			random = {
				chance = {
					value = 45
					if = {
						limit = { scope:target_leader = { has_trait = zealot } }
						add = 20
					}
				}
				set_variable = { name = raid_appeal_won value = 1 }
			}
