namespace = expedition_succession_events

expedition_succession_events.1 = {
	type = country_event
	title = expedition_succession_events.1.title
	desc = expedition_succession_events.1.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		scope:expedition = {
			expedition_current_location ?= { save_scope_as = current_location }
			create_character = {
				estate = estate_type:burghers_estate
				age = { 30 55 }
				female = no
				adm = { 0 50 }
				dip = { 0 50 }
				mil = { 0 50 }
				trait_category = explorer
				save_scope_as = target_leader
			}
			set_expedition_leader = scope:target_leader
			change_variable = { name = morale add = -20 }
			clamp_variable = { name = morale min = 0 max = 100 }
		}
	}

	option = {
		name = expedition_succession_events.1.a
		custom_tooltip = expedition_succession_events.new_leader_tt
		custom_tooltip = expedition_succession_events.morale_down_tt
	}
}
