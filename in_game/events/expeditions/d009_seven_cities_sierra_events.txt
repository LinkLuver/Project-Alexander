seven_cities_sierra.10 = {
#Silver Out of the West
	type = country_event
	title = seven_cities_sierra.10.title
	desc = seven_cities_sierra.10.desc
	historical_info = seven_cities_sierra.10.historical_info
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
		}
		create_character = {
			age = { 30 45 }
			culture = root.culture
			religion = root.religion
			birth_location = root.capital
			estate = estate_type:peasants_estate
			save_scope_as = target_character
			create_in_limbo = yes
		}
	}

	option = {
		name = seven_cities_sierra.10.a
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}
		custom_tooltip = {
			text = plata_castaway_road_tt
			scope:expedition = {
				set_variable = { name = plata_castaway_road value = 1 }
				set_variable = {
					name = plata_castaway_character_var
					value = scope:target_character
				}
			}
		}
	}

	option = {
		name = seven_cities_sierra.10.b
		change_gold_effect = { scale = -1 }
		custom_tooltip = {
			text = plata_silver_in_hand_tt
			scope:expedition = {
				if = {
					limit = { NOT = { has_variable = uncovered_riches_variable } }
					set_variable = {
						name = uncovered_riches_variable
						value = 0
					}
				}
				change_variable = {
					name = uncovered_riches_variable
					add = 0.1
				}
			}
		}
		hidden_effect = { kill_character_silently = scope:target_character }
	}
}
