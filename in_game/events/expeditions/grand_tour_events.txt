# grand_tour_events.1 — Departure (on_start)
grand_tour_events.1 = {
	type = country_event
	title = grand_tour_events.1.title
	desc = grand_tour_events.1.desc
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= { save_scope_as = current_location }
		}
		stall_expedition = scope:expedition
	}

	# The great courts of Italy will teach them what books cannot
	option = {
		name = grand_tour_events.1.a
		add_gold = -250
	}

	option = {
		name = grand_tour_events.1.party
		trigger = {
			scope:expedition = { var:gt_diligence <= 75 }
		}
		add_gold = {
			value = root.monthly_income_trade_and_tax
			multiply = -1.5
			min = -500
		}
		custom_tooltip = grand_tour_events.diligence_up_30_tt
		scope:expedition = {
			change_variable = { name = gt_diligence add = 30 }
			clamp_variable = { name = gt_diligence min = 0 max = 100 }
		}
		gt_party_trait_roll_effect = yes
	}

	after = {
		resume_expedition = scope:expedition
	}
}
