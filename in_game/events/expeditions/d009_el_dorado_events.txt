d009_el_dorado_events.1 = { #A Map of Dubious Provenance
	type = country_event
	title = d009_el_dorado_events.1.title
	desc = d009_el_dorado_events.1.desc
	historical_info = d009_el_dorado_events.el_dorado_desc
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_wilderness_landing }

	trigger = {
		scope:expedition = { exists = expedition_leader }
		scope:location = {
			is_land = yes
			province_definition = {
				any_neighbor_province_definition = {
					any_location_in_province_definition = { is_land = yes }
				}
			}
		}

		scope:expedition = {
			NOT = { has_variable = el_dorado_1 }
		}
	}

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }

			set_variable = el_dorado_1
		}
		scope:location = { 
			save_scope_as = target_location 
			province_definition = {
				random_neighbor_province_definition = {
					limit = {
						any_location_in_province_definition = { is_land = yes }
					} 
					random_location_in_province_definition = {
						limit = { is_land = yes }
						save_scope_as = target_new_waypoint
					}
				}
			}    
		}
	}

	option = {
		name = d009_el_dorado_events.1.a

		change_gold_effect = { scale = -1.5 }

		scope:expedition = { add_new_waypoint = scope:target_new_waypoint }

		random_list = {
			2 = {
				show_chance = no
				desc = new_location_grants_clue_for_el_dorado

				custom_tooltip = {
					text = may_bring_us_closer_to_uncovering_goal_tt

					scope:expedition = { change_variable = { name = leads_found add = 1 } }
				}
				
			}
			7 = {
				show_chance = no
				desc = cartographer_cannot_read

				seven_cities_expedition_stall_effect = { MIN = 5 MAX = 15 }
			}
			1 = {
				show_chance = no
				desc = gasp_its_a_trap

				seven_cities_expedition_calamity_effect = yes
			}
		}
	}

	option = {
		name = d009_el_dorado_events.1.b
		custom_tooltip = d009_el_dorado_events.1.b.tt
	}
}
