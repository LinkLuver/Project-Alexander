d009_fountain_of_youth_events.1 = { #The Curative Spring
	type = country_event
	title = d009_fountain_of_youth_events.1.title
	desc = d009_fountain_of_youth_events.1.desc
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_wilderness_landing }

	trigger = {
		scope:expedition = { exists = expedition_leader }
		scope:location = { is_land = yes }

		scope:expedition = { NOT = { has_variable = fountain_1 } }
	}

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
			set_variable = fountain_1
		}
		scope:location = { save_scope_as = target_location }
	}

	option = {
		name = d009_fountain_of_youth_events.1.a

		change_gold_effect = { scale = -1 }

		random_list = {
			6 = {
				show_chance = no
				desc = the_spring_runs_potent

				custom_tooltip = {
					text = may_bring_us_closer_to_uncovering_goal_tt
					scope:expedition = { change_variable = { name = leads_found add = 1 } }
				}
				custom_tooltip = {
					text = the_waters_may_restore_vigor_tt
					scope:expedition = {
						if = {
							limit = { NOT = { has_variable = rejuvenating_water_variable } }
							set_variable = { name = rejuvenating_water_variable  value = 0 }
						}
						change_variable = { name = rejuvenating_water_variable  add = { 0.2 0.4 } }
					}
				}
			}
			4 = {
				show_chance = no
				desc = the_spring_is_ordinary

				custom_tooltip = {
					text = may_bring_us_closer_to_uncovering_goal_tt
					scope:expedition = { change_variable = { name = leads_found add = 1 } }
				}
			}
		}
	}

	option = {
		name = d009_fountain_of_youth_events.1.b
		custom_tooltip = d009_fountain_of_youth_events.1.b.tt
	}

	historical_info = d009_fountain_of_youth_events.fountain_desc
}

d009_fountain_of_youth_events.2 = { #The Healer's Price
	type = country_event
	title = d009_fountain_of_youth_events.2.title
	desc = d009_fountain_of_youth_events.2.desc
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_foreign_court_audience }

	trigger = {
		scope:expedition = { exists = expedition_leader }
		scope:location = {
			is_land = yes
			exists = dominant_culture
		}

		scope:expedition = { NOT = { has_variable = fountain_2 } }
	}

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
			set_variable = fountain_2
		}
		scope:location = { save_scope_as = target_location }
	}

	option = {
		name = d009_fountain_of_youth_events.2.a

		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}

		if = {
			limit = {
				scope:target_location = {
					exists = dominant_culture
					dominant_culture = { NOR = { this = ROOT.culture  is_accepted_in = ROOT } }
				}
			}
			custom_tooltip = {
				text = improve_root_culture_view_of_location_dominant_culture_tt
				ROOT.culture = { change_cultural_view = { target = scope:target_location.dominant_culture  change = 1 } }
			}
		}

		custom_tooltip = {
			text = the_healer_will_accompany_us_tt
			scope:expedition = { set_variable = { name = healer_location  value = scope:target_location } }
		}
	}

	option = {
		name = d009_fountain_of_youth_events.2.b

		change_gold_effect = { scale = -2 }

		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}
		custom_tooltip = {
			text = the_healer_will_accompany_us_tt
			scope:expedition = { set_variable = { name = healer_location  value = scope:target_location } }
		}
	}

	option = {
		name = d009_fountain_of_youth_events.2.c

		seven_cities_expedition_stall_effect = { MIN = 2 MAX = 5 }

		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}

		if = {
			limit = {

					exists = dominant_culture
