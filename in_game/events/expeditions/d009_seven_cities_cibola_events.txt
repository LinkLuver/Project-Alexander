seven_cities_cibola.10 = {
#The Friar's Sworn Report
	type = country_event
	title = seven_cities_cibola.10.title
	desc = seven_cities_cibola.10.desc
	historical_info = seven_cities_cibola.10.historical_info
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = { 
		scope:expedition = { exists = expedition_leader }
	}

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
		}

		create_character = {
			age = { 30 50 }
			culture = root.culture
			religion = root.religion
			birth_location = root.capital
			estate = estate_type:clergy_estate
			save_scope_as = target_character
			create_in_limbo = yes
		}
	}

	option = {
		name = seven_cities_cibola.10.a
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}
		custom_tooltip = {
			text = cibola_friar_sails_tt
			scope:expedition = { 
				set_variable = { name = cibola_friar_aboard value = 1 } 
				set_variable = {
					name = friar_character_var
					value = scope:target_character
				}
			}
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	option = {
		name = seven_cities_cibola.10.b
		custom_tooltip = {
			text = expedition_less_likely_to_stall_tt
			scope:expedition = {
				if = {
					limit = { NOT = { has_variable = expedition_receives_guides_and_porters_variable } }
					set_variable = expedition_receives_guides_and_porters_variable
				}
			}
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_penalty
		}
		hidden_effect = { kill_character_silently = scope:target_character }
	}
}
