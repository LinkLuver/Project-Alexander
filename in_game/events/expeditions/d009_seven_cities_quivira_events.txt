seven_cities_quivira.10 = {
#Two Guides
	type = country_event
	title = seven_cities_quivira.10.title
	desc = seven_cities_quivira.10.desc
	historical_info = seven_cities_quivira.10.historical_info
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = { scope:expedition = { exists = expedition_leader } }

	immediate = {
		scope:expedition = {
			expedition_leader = { save_scope_as = target_leader }
		}
		location:quivira = {
			random_present_culture_in_location = { save_scope_as = target_culture }
			random_present_religion_in_location = { save_scope_as = target_religion }
		}
		create_character = {
			age = { 40 55 }
			culture = scope:target_culture
			religion = scope:target_religion
			birth_location = location:quivira
			estate = estate_type:peasants_estate
			save_scope_as = target_character
			create_in_limbo = yes
		}
		create_character = {
			age = { 20 30 }
			culture = scope:target_culture
			religion = scope:target_religion
			birth_location = location:quivira
			estate = estate_type:peasants_estate
			save_scope_as = second_guide
			create_in_limbo = yes
		}
		scope:expedition = {
			random_list = {
				45 = { set_variable = quivira_elder_honest }
				55 = { }
			}
		}
	}

	option = {
		name = seven_cities_quivira.10.a
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}
		custom_tooltip = {
			text = quivira_gold_guide_tt
			scope:expedition = {
				set_variable = {
					name = quivira_guide_character_var
					value = scope:target_character
				}
				if = {
					limit = { has_variable = quivira_elder_honest }
					set_variable = { name = quivira_true_guide value = 1 }
					if = {
						limit = { NOT = { has_variable = expedition_receives_guides_and_porters_variable } }
						set_variable = expedition_receives_guides_and_porters_variable
					}
				}
				else = {
					set_variable = { name = quivira_false_guide value = 1 }
				}
			}
		}
		hidden_effect = { kill_character_silently = scope:second_guide }
	}

	option = {
		name = seven_cities_quivira.10.b
		custom_tooltip = {
			text = may_bring_us_closer_to_uncovering_goal_tt
			scope:expedition = { change_variable = { name = leads_found add = 1 } }
		}
		custom_tooltip = {
			text = quivira_grass_guide_tt
			scope:expedition = {
				set_variable = {
					name = quivira_guide_character_var
					value = scope:second_guide
				}
				if = {
					limit = { NOT = { has_variable = quivira_elder_honest } }
					set_variable = { name = quivira_true_guide value = 1 }
					if = {
						limit = { NOT = { has_variable = expedition_receives_guides_and_porters_variable } }
						set_variable = expedition_receives_guides_and_porters_variable
					}
				}
				else = {
					set_variable = { name = quivira_false_guide value = 1 }
				}
