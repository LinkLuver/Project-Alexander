circumnavigation_expedition_events.1 = {
	type = country_event
	title = circumnavigation_expedition_events.1.title
	desc = circumnavigation_expedition_events.1.desc
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = {
		any_character = { is_expedition_leader = yes }
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= { save_scope_as = current_location }
		}
	}

	option = {
		name = circumnavigation_expedition_events.1.magellan
		trigger = {
			character:por_magellan ?= { is_expedition_leader = yes }
		}
		add_prestige = prestige_weak_bonus
	}

	option = {
		name = circumnavigation_expedition_events.1.a
		trigger = {
			NOT = { character:por_magellan ?= { is_expedition_leader = yes } }
		}
		add_prestige = prestige_weak_bonus
	}

	historical_info = circumnavigation_expedition_events.1.historical_info
}

circumnavigation_expedition_events.2 = {
	type = country_event
	title = circumnavigation_expedition_events.2.title
	desc = circumnavigation_expedition_events.2.desc
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_foreign_port_arrival }

	trigger = {
		any_character = { is_expedition_leader = yes }
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			expedition_current_location ?= { save_scope_as = current_location }
			if = {
				limit = { has_variable = departure_port }
				var:departure_port = { save_scope_as = departure_port }
			}
		}
		stall_expedition = scope:expedition
	}

	option = {
		name = circumnavigation_expedition_events.2.a
		change_gold_effect = { scale = -2 }
		custom_tooltip = circumnavigation_expedition_events.provisioned_tt
		custom_tooltip = circumnavigation_expedition_events.morale_up_10_tt
		scope:expedition = {
			change_variable = { name = provisions add = 30 }
			clamp_variable = { name = provisions min = 0 max = 100 }
			change_variable = { name = morale add = 10 }
			clamp_variable = { name = morale min = 0 max = 100 }
		}
	}

	option = {
		name = circumnavigation_expedition_events.2.b
		add_prestige = prestige_weak_bonus
		custom_tooltip = circumnavigation_expedition_events.morale_up_5_tt
		scope:expedition = {
			change_variable = { name = morale add = 5 }
			clamp_variable = { name = morale min = 0 max = 100 }
		}
	}

	option = {
		name = circumnavigation_expedition_events.2.c
		ai_chance = { base = 0 }
		scope:expedition = { set_variable = circ_voluntary_retreat }
		fail_expedition = scope:expedition
	}

	after = {
		resume_expedition = scope:expedition
	}

	historical_info = circumnavigation_expedition_events.2.historical_info
}
