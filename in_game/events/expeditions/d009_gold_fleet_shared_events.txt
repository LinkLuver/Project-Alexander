gold_fleet_shared_events.2 = {
	type = country_event
	title = gold_fleet_shared_events.2.title
	desc = gold_fleet_shared_events.2.desc
	outcome = negative

	location_backdrop = { backdrop_preset = backdrop_at_sea }

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
			set_variable = gf_enemy_done
			pause_expedition = 14
		}
		scope:location = {
			random_unit_in_location = {
				limit = { owner = { is_enemy_of = root } }
				owner = { save_scope_as = intercepting_country }
			}
		}
	}

	option = {
		name = gold_fleet_shared_events.2.a
		random = {
			chance = {
				value = 90
				divide = scope:expedition.var:gold_fleet_danger
			}
			set_variable = gf_ambush_success
			add_prestige = prestige_weak_bonus
		}
		if = {
			limit = { NOT = { has_variable = gf_ambush_success } }
			custom_tooltip = gold_fleet_shared_events.warship_fight_fail_tt
			scope:expedition = {
				set_variable = gf_cargo_damaged
				set_variable = {
					name = gold_fleet_cargo
					value = {
						value = var:gold_fleet_cargo
						multiply = 0.25
					}
				}
			}
		}
		remove_variable = gf_ambush_success
		scope:intercepting_country ?= {
			trigger_event_non_silently = gold_fleet_shared_events.3
		}
	}

	option = {
		name = gold_fleet_shared_events.2.b
		custom_tooltip = gold_fleet_shared_events.cargo_loss_tt
		custom_tooltip = gold_fleet_shared_events.danger_up_1_tt
		scope:expedition = {
			set_variable = {
				name = gold_fleet_cargo
				value = {
					value = var:gold_fleet_cargo
					multiply = 0.85
				}
			}
			change_variable = { name = gold_fleet_danger add = 1 }
		}
		scope:intercepting_country ?= {
			trigger_event_non_silently = gold_fleet_shared_events.3
		}
	}

	option = {
		name = gold_fleet_shared_events.2.c
		random = {
			chance = {
				value = 50
				divide = scope:expedition.var:gold_fleet_danger
			}
			set_variable = gf_ambush_success
		}
		if = {
			limit = { NOT = { has_variable = gf_ambush_success } }
			custom_tooltip = gold_fleet_shared_events.warship_flee_fail_tt
			scope:expedition = {
				set_variable = gf_cargo_damaged
				set_variable = {
					name = gold_fleet_cargo
					value = {
						value = var:gold_fleet_cargo
						multiply = 0.5
					}
				}
			}
		}
		remove_variable = gf_ambush_success
		scope:intercepting_country ?= {
			trigger_event_non_silently = gold_fleet_shared_events.3
		}
	}

	historical_info = gold_fleet_shared_events.2.historical_info
}
