# western_ocean_voyage_events.1 — Departure (on_start)
western_ocean_voyage_events.1 = {
	type = country_event
	title = western_ocean_voyage_events.1.title
	desc = western_ocean_voyage_events.1.desc
	historical_info = western_ocean_voyage_events.1.historical_info
	outcome = neutral

	location_backdrop = { backdrop_preset = backdrop_home_departure }

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
		}
		capital ?= { save_scope_as = target_location }
		stall_expedition = scope:expedition
	}

	option = {
		name = western_ocean_voyage_events.1.a
		change_gold_effect = { scale = -2 }
		custom_tooltip = western_ocean_voyage_events.ship_gained_tt
		scope:expedition = {
			change_variable = { name = num_ships add = 1 }
			clamp_variable = { name = num_ships min = 0 max = 4 }
			pause_expedition = 30
		}
	}

	option = {
		name = western_ocean_voyage_events.1.b
		change_gold_effect = { scale = -1 }
	}

	option = {
		name = western_ocean_voyage_events.1.c
		trigger = {
			scope:target_leader = { has_trait = navigator }
			NOT = { scope:target_leader = character:christopher_columbus }
		}
		add_prestige = prestige_weak_bonus
	}

	option = {
		name = western_ocean_voyage_events.1.columbus
		trigger = {
			exists = character:christopher_columbus
			scope:target_leader = character:christopher_columbus
		}
		add_prestige = prestige_weak_bonus
	}

	after = {
		resume_expedition = scope:expedition
	}
}

# western_ocean_voyage_events.2 — The Canary Islands (on_arrive_to_waypoint)
western_ocean_voyage_events.2 = {
	type = country_event

	historical_info = western_ocean_voyage_events.2.historical_info

		exists = scope:expedition.expedition_leader

	option = {
		name = western_ocean_voyage_events.2.b
		trigger = {
			area:canary_islands_area = {
				any_location_in_area = { owner ?= root }
			}
		}

			pause_expedition = 7
			change_variable = { name = provisions add = 20 }
			clamp_variable = { name = provisions min = 0 max = 100 }
			change_variable = { name = morale add = 10 }
			clamp_variable = { name = morale min = 0 max = 100 }
		}
	}

	option = {
		name = western_ocean_voyage_events.2.c
		trigger = {
			area:canary_islands_area = {
				any_location_in_area = {
					is_land = yes
					has_owner = no
				}
			}
		}
		custom_tooltip = western_ocean_voyage_events.provisions_up_10_tt
		scope:expedition = {
			pause_expedition = 10
			change_variable = { name = provisions add = 10 }
			clamp_variable = { name = provisions min = 0 max = 100 }
		}
		random_list = {
			70 = {
				custom_tooltip = western_ocean_voyage_events.2.c.calm_tt
				custom_tooltip = western_ocean_voyage_events.morale_up_5_tt
				scope:expedition = {
					change_variable = { name = morale add = 5 }
					clamp_variable = { name = morale min = 0 max = 100 }
				}
			}
			30 = {
				custom_tooltip = western_ocean_voyage_events.2.c.clash_tt
				custom_tooltip = western_ocean_voyage_events.morale_down_10_tt
				scope:expedition = {
					change_variable = { name = morale add = -10 }
					clamp_variable = { name = morale min = 0 max = 100 }
				}
			}
		}
	}

		name = western_ocean_voyage_events.2.d

			scope:expedition = { var:num_ships < 3 }
			area:canary_islands_area = {

	}

		ai_chance = { base = 0 }

		fail_expedition = scope:expedition

	}

	# Generic — press on
	option = {
		name = western_ocean_voyage_events.3.a
		trigger = {
			OR = {
				NOT = { exists = character:christopher_columbus }
				NOT = { scope:target_leader = character:christopher_columbus }
			}
		}
		custom_tooltip = western_ocean_voyage_events.morale_down_15_tt
		scope:expedition = {
			change_variable = { name = morale add = -15 }
			clamp_variable = { name = morale min = 0 max = 100 }

		name = western_ocean_voyage_events.3.b

		fail_expedition = scope:expedition

	after = {
		resume_expedition = scope:expedition
	}
}

# western_ocean_voyage_events.4 — Land! (on_arrive_to_waypoint, Caribbean landfall)
western_ocean_voyage_events.4 = {
	type = country_event
	title = western_ocean_voyage_events.4.title
	desc = western_ocean_voyage_events.4.desc
	historical_info = western_ocean_voyage_events.4.historical_info
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_wilderness_landing }

	trigger = {
		exists = scope:expedition.expedition_leader
	}

	immediate = {
		set_variable = { name = landfall_set value = 2 }
		scope:expedition = {
			expedition_leader = {
				save_scope_as = target_leader
			}
		}
		stall_expedition = scope:expedition
		scope:location = {
			save_scope_as = target_location
			random_neighbor_location = {
				limit = { is_land = yes }
				save_scope_as = adjacent_land_loc
			}
		}
	}

	# Columbus plants the flag — we have found the Indies
	option = {
		name = western_ocean_voyage_events.4.columbus
		trigger = {
			exists = character:christopher_columbus
			scope:target_leader = character:christopher_columbus
		}
		add_prestige = prestige_severe_bonus
		if = {
			limit = { exists = scope:adjacent_land_loc }
			discover_area = scope:adjacent_land_loc.area
		}
		trigger_event_non_silently = western_ocean_voyage_events.6
	}

	# Generic — claim the land for the crown
	option = {
		name = western_ocean_voyage_events.4.a
		trigger = {
			OR = {
				NOT = { exists = character:christopher_columbus }
				NOT = { scope:target_leader = character:christopher_columbus }
			}
		}
		add_prestige = prestige_mild_bonus
		if = {
			limit = { exists = scope:adjacent_land_loc }
			discover_area = scope:adjacent_land_loc.area
		}
		trigger_event_non_silently = western_ocean_voyage_events.6
	}

	# Something is different — not the Indies
	option = {
		name = western_ocean_voyage_events.4.b
		add_legitimacy = legitimacy_mild_bonus
		if = {
			limit = { exists = scope:adjacent_land_loc }
			discover_area = scope:adjacent_land_loc.area
		}
		trigger_event_non_silently = western_ocean_voyage_events.6
	}

	after = {
		resume_expedition = scope:expedition
	}
}
