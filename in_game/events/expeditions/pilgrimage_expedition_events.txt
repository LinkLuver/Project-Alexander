# pilgrimage_expedition_events.1 — Jerusalem, the Holy City (on_arrive_to_waypoint)
pilgrimage_expedition_events.1 = {
	type = country_event
	title = pilgrimage_expedition_events.1.title
	desc = pilgrimage_expedition_events.1.desc
	historical_info = pilgrimage_expedition_events.1.historical_info
	outcome = positive

	location_backdrop = { backdrop_preset = backdrop_pilgrimage_approach }

	trigger = {
		has_ruler = yes
		ruler = { is_expedition_leader = yes }
	}

	immediate = {
		religion:catholic = { random_holy_site_in_religion = { limit = { location = location:jerusalem } save_scope_as = target_holy_site } }
		ruler ?= {
			save_scope_as = target_ruler
		}
		custom_tooltip = pilgrimage_expedition_events.retinue_up_15_tt
		custom_tooltip = pilgrimage_expedition_events.provisions_up_40_tt
		scope:expedition = {
			expedition_current_location ?= { save_scope_as = current_location }
			change_variable = { name = retinue add = 15 }
			change_variable = { name = provisions add = 40 }
			clamp_variable = { name = provisions min = 0 max = 100 }
		}
		stall_expedition = scope:expedition
	}

	option = {
		name = pilgrimage_expedition_events.1.a
		custom_tooltip = pilgrimage_expedition_events.piety_up_10_tt
		scope:expedition = {
			change_variable = { name = piety add = 10 }
			clamp_variable = { name = piety min = 0 max = 100 }
		}
	}

	option = {
		name = pilgrimage_expedition_events.1.b
		add_prestige = prestige_mild_bonus
		custom_tooltip = pilgrimage_expedition_events.piety_down_10_tt
		scope:expedition = {
			change_variable = { name = piety add = -10 }
			clamp_variable = { name = piety min = 0 max = 100 }
		}
	}

	after = {
		resume_expedition = scope:expedition
	}
}
