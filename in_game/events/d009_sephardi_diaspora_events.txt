d009_sephardi_diaspora.1 = { #The Exiles Arrive
	type = country_event
	title = d009_sephardi_diaspora.1.title
	desc = d009_sephardi_diaspora.1.desc
	outcome = neutral
	historical_info = d009_sephardi_diaspora.1.historical_info

	fire_only_once = yes

	dynamic_historical_event = {
		tag = TUR
		from = 1492.1.1
		to = 1520.1.1
		monthly_chance = 25
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		religion = religion:sunni
		owns = location:thessaloniki

		any_in_global_list = {
			variable = crypto_jewish_community_locations
			owner = {
				culture = { has_culture_group = culture_group:iberian_group }
				NOT = { this = root }
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }

		random_in_global_list = {
			variable = crypto_jewish_community_locations
			limit = {
				owner = {
					culture = { has_culture_group = culture_group:iberian_group }
					NOT = { this = root }
				}
			}

			owner = { save_scope_as = origin_country }
		}
	}

	option = {
		name = d009_sephardi_diaspora.1.a
		historical_option = yes

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_right
		}

		scope:origin_country = {
			d009_crypto_jewish_exodus_effect = {
				FRACTION = 0.2
				AMOUNT = 0.3
				DESTINATION = location:thessaloniki
			}
		}
	}

	option = {
		name = d009_sephardi_diaspora.1.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_left
		}

		scope:origin_country = {
			d009_crypto_jewish_exodus_effect = {
				FRACTION = 0.1
				AMOUNT = 0.15
				DESTINATION = location:thessaloniki
			}
		}
	}
}

d009_sephardi_diaspora.4 = { #Doña Gracia Mendes and the Ferrara Bible
	type = country_event
	title = d009_sephardi_diaspora.4.title
	desc = d009_sephardi_diaspora.4.desc
	outcome = neutral
	historical_info = d009_sephardi_diaspora.4.historical_info

	trigger = {
		has_dlc = "d009_across_the_pillars"
		owns = location:ferrara
		country_exists = c:POR

		NOT = {
			this = c:POR
		}

		NOT = {
			has_variable = d009_sephardi_diaspora_4_fired
		}

		c:POR = {
			religion.group = religion_group:christian

			any_owned_location = {
				has_location_modifier = crypto_jewish_community
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }

		set_variable = d009_sephardi_diaspora_4_fired

		create_character = {
			first_name = name_grace
			last_name = Mendes
			adm = { 60 80 }
			dip = { 60 80 }
			mil = { 10 30 }
			culture = culture:sephardi
			religion = religion:judaism
			female = yes
			min_age = 40
			estate = estate_type:burghers_estate
			create_in_limbo = yes
			save_scope_as = target_character
		}
	}

	option = {
		name = d009_sephardi_diaspora.4.a
		historical_option = yes

		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_move_to_left
		}

		scope:target_character = {
			move_country = root
		}

		c:POR = {
			d009_crypto_jewish_exodus_effect = {
				FRACTION = 0.1
				AMOUNT = 0.15
				DESTINATION = location:ferrara
			}
		}
	}

	option = {
		name = d009_sephardi_diaspora.4.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_move_to_right
		}

		hidden_effect = {
			kill_character_silently = scope:target_character
		}
	}
}
