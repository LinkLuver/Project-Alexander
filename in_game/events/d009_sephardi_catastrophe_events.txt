d009_sephardi_catastrophe.1 = { #The Preaching of Ferrand Martinez
	type = country_event
	title = d009_sephardi_catastrophe.1.title
	desc = d009_sephardi_catastrophe.1.desc
	outcome = neutral
	historical_info = d009_sephardi_catastrophe.1.historical_info

	fire_only_once = yes

	dynamic_historical_event = {
		tag = CAS
		tag = ARA
		from = 1391.1.1
		to = 1393.1.1
		monthly_chance = 50
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"

		OR = {
			AND = {
				owns = location:sevilla
				location:sevilla = {
					"religion_population(religion:judaism)" > 0
				}
			}

			AND = {
				owns = location:cordoba
				location:cordoba = {
					"religion_population(religion:judaism)" > 0
				}
			}
		}
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:peasants_estate }

		create_character = {
			first_name = name_ferdinand
			last_name = Martinez
			adm = { 40 60 }
			dip = { 60 80 }
			mil = { 10 30 }
			culture = culture:castilian
			religion = religion:catholic
			female = no
			min_age = 45
			estate = estate_type:clergy_estate
			create_in_limbo = yes
			save_scope_as = target_character
		}

		if = {
			limit = {
				owns = location:sevilla
				location:sevilla = {
					"religion_population(religion:judaism)" > 0
				}
			}
			location:sevilla = {
				save_scope_as = target_location
			}
		}
		else = {
			location:cordoba = {
				save_scope_as = target_location
			}
		}
	}

	option = {
		name = d009_sephardi_catastrophe.1.a

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
		add_stability = stability_mild_bonus

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_right
		}

		if = {
			limit = {
				is_situation_active = situation:battle_for_the_strait
				country_exists = c:GRA

				OR = {
					has_or_had_tag = CAS
					has_or_had_tag = MOR
				}
			}
			change_influence_over_granada_effect = { value = bfts_pull_toward_cas_small }
		}

		scope:target_character = {
			banish_character = yes
		}
	}

	option = {
		name = d009_sephardi_catastrophe.1.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}

		if = {
			limit = {
				is_situation_active = situation:battle_for_the_strait
				country_exists = c:GRA

				OR = {
					has_or_had_tag = CAS
					has_or_had_tag = MOR
				}
			}
			change_influence_over_granada_effect = { value = bfts_pull_toward_mor_small }
		}

		hidden_effect = {
			kill_character_silently = scope:target_character
		}

		scope:target_location = {
			add_location_modifier = {
				modifier = crypto_jewish_community
				years = crypto_community_duration
				mode = add
			}
			every_pop = {
				limit = {
					religion = religion:judaism
				}

				split_pop = {
					fraction = 0.5
					religion = religion:catholic
				}
			}
		}

		add_to_global_variable_list = { name = crypto_jewish_community_locations target = scope:target_location }
		set_global_variable = d009_pogroms_spreading
	}

	option = {
		name = d009_sephardi_catastrophe.1.c

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_severe_bonus }

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}

		scope:target_character = {
			move_country = root
		}

		scope:target_location = {
			add_location_modifier = {
				modifier = crypto_jewish_community
				years = crypto_community_duration
				mode = add
			}

			every_pop = {
				limit = {
					religion = religion:judaism
				}

				split_pop = {
					fraction = 0.8
					religion = religion:catholic
				}
			}
		}

		add_to_global_variable_list = { name = crypto_jewish_community_locations target = scope:target_location }
		set_global_variable = d009_pogroms_spreading
	}
}

	}

		location:lisbon = {

			any_location_in_scripted_geography = {

				is_coastal = yes
			}
		}
	}

	illustration_tags = {
		10 = angry

	}

	immediate = {

		scripted_geography:d009_maghreb_coastal_geography = {
			random_location_in_scripted_geography = {
				limit = {
					has_owner = yes
					owner != root
					dominant_religion ?= religion:sunni
					is_coastal = yes
				}

				save_scope_as = escape_target
			}
		}

		religion = {
			random_religious_order_in_religion = {
				limit = {
					any_location_in_religious_order = { this = location:bologna }
				}
				save_scope_as = massacre_dominican_order
			}
		}
	}

	option = {
		name = d009_sephardi_catastrophe.6.a

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }

		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_right
		}

		if = {
			limit = { exists = scope:massacre_dominican_order }
			scope:massacre_dominican_order = {
				change_religious_order_zeal = religious_order_mild_penalty
			}
		}
	}

	option = {
		name = d009_sephardi_catastrophe.6.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_left
		}

		add_migration = {
			owner = root
			from_location = location:lisbon
			to_location = scope:escape_target
			culture = culture:sephardi
			amount = {
				value = "location:lisbon.culture_population(culture:sephardi)"
				multiply = 0.3
			}
			months = 24
		}

		if = {
			limit = { exists = scope:massacre_dominican_order }
			scope:massacre_dominican_order = {
				change_religious_order_zeal = religious_order_mild_bonus
			}
		}
	}
}
