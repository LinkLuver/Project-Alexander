namespace = crisis_of_faith

crisis_of_faith.1 = {
	hide_portraits = yes
	type = country_event
	category = disaster_event
	title = crisis_of_faith.1.title
	desc = crisis_of_faith.1.desc
	outcome = neutral

	image = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	immediate = {
		create_rebel = {
			category = estate
			estate = clergy_estate
			name = schismatic_clergy_name
			rebel_demand = rebel_demand_crisis_of_faith_1
			save_scope_as = schismatic_clergy_scope
		}
		set_variable = {
			name = schismatic_clergy_rebel
			value = scope:schismatic_clergy_scope
		}

		seed_crisis_of_faith_heresiarch_effect = yes

		set_variable = {
			name = total_num_prov_variable
			value = 0
		}
		every_province = {
			limit = {
				OR = {
					province_average_control < 0.5
					province_satisfaction < 0.5
				}
				NOT = { any_location_in_province = { is_capital = yes } }
			}
			add_to_list = schism_provinces_list
			owner = {
				change_variable = {
					name = total_num_prov_variable
					add = 1
				}
			}
		}
		if = {
			limit = { list_size = { name = schism_provinces_list value = 0 } }
			ordered_province = {
				limit = { NOT = { any_location_in_province = { is_capital = yes } } }
				order_by = population
				max = 1
				check_range_bounds = no
				add_to_list = schism_provinces_list
			}
			set_variable = {
				name = total_num_prov_variable
				value = 1
			}
		}
	}

	option = {
		name = crisis_of_faith.1.a #Accommodate the reformers

		add_devotion = devotion_mild_penalty
		add_stability = stability_mild_bonus
		custom_tooltip = crisis_of_faith.1.a.tt
		hidden_effect = {
			if = {
				limit = { list_size = { name = schism_provinces_list value > 0 } }
				ordered_in_list = {
					list = schism_provinces_list
					order_by = population
					max = {
						value = var:total_num_prov_variable
						divide = 8
						ceiling = yes
					}
					every_location_in_province = {
						every_pop = {
							limit = {
								owner = root
								pop_type = pop_type:clergy
							}
							add_pop_satisfaction = pop_satisfaction_ultimate_penalty
							change_pop_allegiance = scope:schismatic_clergy_scope
						}
					}
				}
			}
		}
	}

	option = {
		name = crisis_of_faith.1.b #Suppress the schismatics

		add_stability = stability_extreme_penalty
		custom_tooltip = crisis_of_faith.1.tt
		hidden_effect = {
			if = {
				limit = { list_size = { name = schism_provinces_list value > 0 } }
				ordered_in_list = {
					list = schism_provinces_list
					order_by = population
					max = {
						value = var:total_num_prov_variable
						divide = 4
						ceiling = yes
					}
					every_location_in_province = {
						every_pop = {
							limit = { owner = root }
							add_pop_satisfaction = pop_satisfaction_ultimate_penalty
							change_pop_allegiance = scope:schismatic_clergy_scope
						}
					}
				}
			}
		}
	}

	after = {
		remove_variable = total_num_prov_variable
	}
}

crisis_of_faith.101 = { #Schismatic Doctrine Entrenched
	type = country_event
	category = disaster_event
	title = crisis_of_faith.101.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = var:crisis_of_faith_heresiarch_variable }
				desc = crisis_of_faith.101.desc.heresiarch
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = crisis_of_faith.101.desc
			}
		}
	}
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	immediate = {
		add_country_modifier = {
			modifier = entrenched_schism
			years = 10
		}
		if = {
			limit = { exists = var:crisis_of_faith_heresiarch_variable }
			var:crisis_of_faith_heresiarch_variable ?= { save_scope_as = crisis_of_faith_heresiarch_scope }
		}
		random_religious_aspect = {
			save_scope_as = crisis_of_faith_lost_aspect_scope
		}
	}

	option = {
		name = crisis_of_faith.101.a #The schism prevails
		trigger = { NOT = { exists = var:crisis_of_faith_heresiarch_variable } }

		if = {
			limit = { exists = scope:crisis_of_faith_lost_aspect_scope }
			remove_religious_aspect = scope:crisis_of_faith_lost_aspect_scope
		}
	}

	option = {
		name = crisis_of_faith.101.b
		trigger = { exists = var:crisis_of_faith_heresiarch_variable }

		set_new_ruler = scope:crisis_of_faith_heresiarch_scope
		if = {
			limit = { exists = scope:crisis_of_faith_lost_aspect_scope }
			remove_religious_aspect = scope:crisis_of_faith_lost_aspect_scope
		}
	}

	after = {
		clear_crisis_of_faith_heresiarch_effect = yes
	}
}

crisis_of_faith.7 = {
	type = country_event
	category = disaster_event
	title = crisis_of_faith.7.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { "estate_power(estate_type:crown_estate)" >= 0.3 }
				desc = crisis_of_faith.7.desc.ready
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = crisis_of_faith.7.desc.not_ready
			}
		}
	}
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	trigger = { has_variable = crisis_of_faith_heresiarch_revealed_variable }

	immediate = {
		var:crisis_of_faith_heresiarch_variable ?= { save_scope_as = crisis_of_faith_heresiarch_scope }
	}

	option = {
		name = crisis_of_faith.7.a

		if = {
			limit = { "estate_power(estate_type:crown_estate)" >= 0.3 }
			add_devotion = devotion_severe_bonus
			hidden_effect = {
				var:schismatic_clergy_rebel ?= {
					add_rebel_progress = -0.15
					every_pops_supporting_rebel = { add_pop_satisfaction = pop_satisfaction_mild_bonus }
				}
				clear_crisis_of_faith_heresiarch_effect = yes
			}
		}
		else = {
			add_stability = stability_severe_penalty
			custom_tooltip = crisis_of_faith.7.a.fail_tt
		}
	}
}
