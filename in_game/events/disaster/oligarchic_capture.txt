namespace = oligarchic_capture

oligarchic_capture.1 = {
	hide_portraits = yes
	type = country_event
	category = disaster_event
	title = oligarchic_capture.1.title
	desc = oligarchic_capture.1.desc
	outcome = neutral

	image = "gfx/interface/illustrations/disaster/coup_attempt.dds"

	immediate = {
		if = {
			limit = {
				any_estate = {
					estate_type = estate_type:burghers_estate
					power > "root.estate_power(estate_type:nobles_estate)"
				}
			}
			create_rebel = {
				category = estate
				estate = burghers_estate
				name = oligarchic_faction_name
				save_scope_as = oligarchic_faction_scope
			}
			ordered_character = {
				limit = {
					has_estate = estate_type:burghers_estate
					is_adult = yes
					is_ruler = no
					is_regent = no
				}
				order_by = total_abilities
				max = 1
				check_range_bounds = no
				save_scope_as = oligarchic_ringleader_scope
			}
		}
		else = {
			create_rebel = {
				category = estate
				estate = nobles_estate
				name = oligarchic_faction_name
				save_scope_as = oligarchic_faction_scope
			}
			ordered_character = {
				limit = {
					has_estate = estate_type:nobles_estate
					is_adult = yes
					is_ruler = no
					is_regent = no
				}
				order_by = total_abilities
				max = 1
				check_range_bounds = no
				save_scope_as = oligarchic_ringleader_scope
			}
		}
		set_variable = {
			name = oligarchic_faction_rebel
			value = scope:oligarchic_faction_scope
		}
		if = {
			limit = { exists = scope:oligarchic_ringleader_scope }
			set_variable = {
				name = oligarchic_ringleader_variable
				value = scope:oligarchic_ringleader_scope
			}
		}

		set_variable = {
			name = total_num_prov_variable
			value = 0
		}
		every_province = {
			limit = {
				OR = {
					province_average_control < 0.5
					province_food_percentage < 0.5
					province_satisfaction < 0.5
				}
				any_location_in_province = { is_capital = no }
			}
			add_to_list = oligarchic_provinces_list
			owner = {
				change_variable = {
					name = total_num_prov_variable
					add = 1
				}
			}
		}
	}

	option = {
		name = oligarchic_capture.1.a #Capitulate to the faction

		add_republican_tradition = republican_tradition_severe_penalty
		add_stability = stability_mild_bonus
	}

	option = {
		name = oligarchic_capture.1.b #Resist the faction

		add_stability = stability_extreme_penalty
		custom_tooltip = oligarchic_capture.1.tt
		hidden_effect = {
			if = {
				limit = { list_size = { name = oligarchic_provinces_list value > 0 } }
				ordered_in_list = {
					list = oligarchic_provinces_list
					order_by = population
					max = {
						value = var:total_num_prov_variable
						divide = 4
						ceiling = yes
					}
					every_location_in_province = {
						every_pop = {
							limit = { owner = root }
							add_pop_satisfaction = pop_satisfaction_ultimate_penalty
							change_pop_allegiance = scope:oligarchic_faction_scope
						}
					}
				}
			}
		}
	}

	after = {
		remove_variable = total_num_prov_variable
	}
}

oligarchic_capture.101 = { #Oligarchic Constitution Entrenched
	type = country_event
	category = disaster_event
	title = oligarchic_capture.101.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { exists = var:oligarchic_ringleader_variable }
				desc = oligarchic_capture.101.desc.ringleader
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = oligarchic_capture.101.desc
			}
		}
	}
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/coup_attempt.dds"

	immediate = {
		add_country_modifier = {
			modifier = oligarchic_entrenchment
			years = 10
		}
		if = {
			limit = { exists = var:oligarchic_ringleader_variable }
			var:oligarchic_ringleader_variable ?= { save_scope_as = oligarchic_ringleader_scope }
		}
	}

	option = {
		name = oligarchic_capture.101.a #The faction has won
		trigger = { NOT = { exists = var:oligarchic_ringleader_variable } }
	}

	option = {
		name = oligarchic_capture.101.b
		trigger = { exists = var:oligarchic_ringleader_variable }

		change_government_type = government_type:monarchy
		set_new_ruler = scope:oligarchic_ringleader_scope
	}

	after = {
		if = {
			limit = { exists = var:oligarchic_ringleader_variable }
			remove_variable = oligarchic_ringleader_variable
		}
		if = {
			limit = { has_variable = oligarchic_ringleader_revealed_variable }
			remove_variable = oligarchic_ringleader_revealed_variable
		}
	}
}

oligarchic_capture.7 = {
	type = country_event
	category = disaster_event
	title = oligarchic_capture.7.title
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { "estate_power(estate_type:crown_estate)" >= 0.3 }
				desc = oligarchic_capture.7.desc.ready
			}
			triggered_desc = {
				trigger = { always = yes }
				desc = oligarchic_capture.7.desc.not_ready
			}
		}
	}
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/coup_attempt.dds"

	trigger = { has_variable = oligarchic_ringleader_revealed_variable }

	immediate = {
		var:oligarchic_ringleader_variable ?= { save_scope_as = oligarchic_ringleader_scope }
	}

	option = {
		name = oligarchic_capture.7.a
		if = {
			limit = { "estate_power(estate_type:crown_estate)" >= 0.3 }
			custom_tooltip = oligarchic_capture.7.a.success_tt
			add_political_influence = political_influence_weak_bonus
			hidden_effect = {
				kill_character = {
					target = scope:oligarchic_ringleader_scope
					reason = execution
				}
				var:oligarchic_faction_rebel ?= { save_scope_as = oligarchic_faction_scope }
				scope:oligarchic_faction_scope ?= { add_rebel_progress = -0.15 }
				remove_variable = oligarchic_ringleader_variable
				remove_variable = oligarchic_ringleader_revealed_variable
			}
		}
		else = {
			add_stability = stability_severe_penalty
			custom_tooltip = oligarchic_capture.7.a.fail_tt
		}
	}
}
