        is_subject = yes
        country_exists = c:ARA
        overlord = c:ARA
        c:ARA ?= {
            has_ruler = yes
        }
    }

        event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
        ruler ?= {
            save_scope_as = our_ruler
        }	
        c:ARA ?= {
            save_scope_as = target_country
            ruler ?= {
                save_scope_as = ara_ruler
            }
        }
    }	

    option = { # The tense relationship between <ara_ruler> and <our_ruler> is on the verge of creating a rift between our two countries
        name = flavor_mll.1.a	
        custom_tooltip = repair_or_damage_relations_with_aragon_tt

        add_opinion = { target = c:ARA modifier = mll_ara_strained_relationship }
        reverse_add_opinion = { target = c:ARA modifier = mll_ara_strained_relationship } 	
        set_variable = {
            name = mll_authority_of_james_iii_events
