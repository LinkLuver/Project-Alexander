#Reclamation of Cordoba
flavor_gra.7 = {
	type = country_event
	title = flavor_gra.7.title
	desc = flavor_gra.7.desc
	outcome = positive

	dynamic_historical_event = {
		tag = ADU
		tag = GRA
		from = 1337.1.1
		to = 1821.1.1
		monthly_chance = 50
	}

	fire_only_once = yes

	trigger = {
		has_dlc = "d009_across_the_pillars"
		own_entire_province = province_definition:cordoba_province

		religion.group = religion_group:muslim
	}

	immediate = {
		location:cordoba = {
			save_scope_as = target_location
		}

		ruler_or_regent ?= {
			save_scope_as = target_character
		}
	}

	option = { #Convert the city
		name = flavor_gra.7.a

		scope:target_location = {
			custom_tooltip = {
				text = gra_convert_cordoba_to_root_religion
				convert_location_with_randomness = {
					religion = root.religion
					min = 0.1
					max = 0.3
				}
			}
		}

		scope:target_location.province = {
			custom_tooltip = {
				text = gra_cordoba_province_gains_core
				every_location_in_province = {
					add_core = root
					change_integration_level = core
				}
			}
		}

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_minor_move_to_left
		}

		check_situation_and_change_granadan_sovereignty = {
			value = 10
		}
	}

	option = { #Keep the rights of the citizens to follow their faith
		name = flavor_gra.7.b


		scope:target_location.province = {
			custom_tooltip = {
				text = gra_cordoba_province_gains_core
				every_location_in_province = {
					add_core = root
					change_integration_level = core
				}
			}
		}

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_minor_move_to_right
		}

		check_situation_and_change_granadan_sovereignty = {
			value = 10
		}
	}
}

#Capital of the Old Caliphate
flavor_gra.17 = {
	type = country_event
	title = flavor_gra.17.title
	desc = flavor_gra.17.desc
	outcome = positive

	dynamic_historical_event = {
		tag = GRA
		from = 1400.1.1
		to = 1600.1.1
		monthly_chance = 2
	}

	fire_only_once = yes

	trigger = {
		has_dlc = "d009_across_the_pillars"
		owns = location:cordoba
	}

	immediate = {
		ruler_or_regent ?= {
			save_scope_as = target_character
		}	
	}

	option = {
		name = flavor_gra.17.a
		
		location:cordoba = {
			add_location_modifier = {
				mode = add
				modifier = gra_old_capital_of_the_caliphate
				years = -1
			}

			change_prosperity = prosperity_mild_bonus
			change_development = development_mild_bonus
		}

		if = {
			limit = {
				capital != location:cordoba
			}
			set_capital = location:cordoba
		}
	}

	option = {
		name = flavor_gra.17.b

		trigger = {
			capital != location:cordoba
		}

		location:cordoba = {
			add_location_modifier = {
				mode = add
				modifier = gra_old_capital_of_the_caliphate
				years = -1
			}
		}
		
		capital ?= {
			change_prosperity = prosperity_extreme_bonus
			change_development = development_extreme_bonus
		}
	}
}

#Rebuilding Madinat Al-Zahra in Cordoba
flavor_gra.21 = {
	type = country_event
	title = flavor_gra.21.title
	desc = flavor_gra.21.desc
	outcome = neutral

	dynamic_historical_event = {
		tag = GRA
		tag = ADU
		from = 1440.1.1
		to = 1821.1.1
		monthly_chance = 2
	}

	fire_only_once = yes

	trigger = {
		has_dlc = "d009_across_the_pillars"
		owns = location:cordoba

		location:cordoba = {
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
	}

	immediate = {
		ruler_or_regent ?= {
			save_scope_as = target_character
		}

		unlock_building_effect = {
			type = madinat_al_zahra
		}
	}

	option = {
		name = flavor_gra.21.a
		
		location:cordoba = {
			construct_building = {
				building_type = building_type:madinat_al_zahra
			}
		}
	}

	option = {
		name = flavor_gra.21.b
		
		add_prestige = prestige_mild_penalty

		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_minor_move_to_right
		}

		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty
		}

		show_as_tooltip = {
			unlock_building_effect = {
				type = madinat_al_zahra
			}
		}
	}
}

#Restoring the Library of Cordoba
flavor_gra.22 = {
	type = country_event
	title = flavor_gra.22.title
	desc = flavor_gra.22.desc
	outcome = positive

	dynamic_historical_event = {
		tag = GRA
		tag = ADU
		from = 1440.1.1
		to = 1821.1.1
		monthly_chance = 2
	}

	fire_only_once = yes

	trigger = {
		has_dlc = "d009_across_the_pillars"
		owns = location:cordoba

		location:cordoba = {
			has_building_with_at_least_one_level = library_cordoba
		}
	}

	immediate = {
		ruler_or_regent ?= {
			save_scope_as = target_character
		}
	}

	option = {
		name = flavor_gra.22.a
		
		location:cordoba = {
			change_development = development_mild_bonus

			custom_tooltip = {
				text = add_pop_satisfaction_based_on_literacy
				every_pop = {
					limit = {
						owner ?= root
					}
					add_pop_satisfaction = {
						value = pop_satisfaction_extreme_bonus
						multiply = pop_literacy
						divide = 100
					}
				}
			}
		}

		add_research_progress = research_progress_ultimate_bonus

		change_societal_value = {
			type = traditionalist_vs_innovative
			value = societal_value_minor_move_to_right
		}
	}
}
