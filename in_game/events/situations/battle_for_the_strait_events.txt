battle_for_the_strait_events.3 = {
	type = country_event
	category = situation_event
	image = "gfx/interface/illustrations/situation/battle_for_the_strait.dds"

	title = battle_for_the_strait_events.3.title
	desc = battle_for_the_strait_events.3.desc
	outcome = positive


	major = yes
	major_trigger = {
		religion = religion:sunni
		capital.sub_continent = sub_continent:north_africa
	}

	immediate = {
		ruler_or_regent = { save_scope_as = target_ruler }
		c:CAS ?= {
			save_scope_as = target_cas_country
		}
		set_global_variable = {
			name = battle_for_the_strait_winner_country
			value = root
		}
		
	}

	option = {
		name = battle_for_the_strait_events.3.a

		add_country_modifier = {
			modifier = protector_of_andalus_modifier
			years = bfts_victory_reward_years
		}
		if = {
			limit = { NOT = { country_rank = country_rank:rank_empire } }
			set_country_rank = country_rank:rank_empire
		}

		custom_tooltip = bfts_andalus_formation_eased_tt

		custom_tooltip = {
			text = bfts_iberia_fully_integrated_tt
			area:granada_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = root
						NOT = { integration_level = core }
					}
					add_core = root
					change_integration_level = core
				}
			}
			area:andalusia_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = root
						NOT = { integration_level = core }
					}
					add_core = root
					change_integration_level = core
				}
			}
			area:murcia_area = {
				every_ownable_location_in_area = {
					limit = {
						owner = root
						NOT = { integration_level = core }
					}
					add_core = root
					change_integration_level = core
				}
			}
		}

		custom_tooltip = {
			text = bfts_taifa_annexation_boosted_tt
			every_subject = {
				limit = {
					is_subject_type = taifa
					is_being_annexed = yes
				}
				save_temporary_scope_as = taifa_subject
				root = {
					change_annexation_progress = {
						target = scope:taifa_subject
						value = {
							value = "annexation_cost(scope:taifa_subject)"
							multiply = 0.25
						}
					}
				}
			}
		}

		custom_tooltip = battle_for_the_strait_concludes_tt

		if = {
			limit = { has_or_had_tag = MOR }
			custom_tooltip = bfts_dominate_iberia_golden_century_tt
		}
	}
}

	outcome = neutral
	historical_info = battle_for_the_strait_events.55.historical_info

	major = yes
	major_trigger = {
		OR = {
			has_or_had_tag = GRA
			has_or_had_tag = CAS
			has_or_had_tag = MOR
		}
	}

	trigger = {
		country_exists = c:GRA
	}

	immediate = {
		ruler_or_regent = { save_scope_as = target_mam_ruler }
		c:GRA ?= {
			save_scope_as = target_gra_country
			ruler_or_regent = { save_scope_as = target_gra_ruler }
		}
	}

	option = {	# Forge an alliance, or deepen our trust
		name = battle_for_the_strait_events.56.a

		ai_will_select = {
			value = -100
			if = {
				limit = {
					exists = scope:target_gra_country
					opinion = { target = scope:target_gra_country value >= 50 }
				}
				value = 150
			}
		}

		if = {
			limit = {
				is_subject = no
				c:GRA ?= { is_subject = no }
				NOT = { is_allied_with = { target = c:GRA } }
			}
			create_alliance = { target = c:GRA }
		}

				value = 1
				target = c:GRA
			}
		}

			add_manpower = {
				value = c:MAM.monthly_manpower

					limit = { c:MAM = { has_advance = reform_the_mamluk_corps } }
					create_sub_unit_with_owner = {
						type = a_mamluk_horsemen_renaissance

					}
					create_sub_unit_with_owner = {
						type = a_mamluk_unit_renaissance
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
				}
				else = {
					create_sub_unit_with_owner = {
						type = a_mamluk_horsemen_traditions
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
					create_sub_unit_with_owner = {
						type = a_mamluk_unit_traditions
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
				}
			}
		}
	}

	option = {	# Send only manpower and a Mamluk contingent
		name = battle_for_the_strait_events.56.b

		ai_will_select = {
			value = -25
			if = {
				limit = {
					exists = scope:target_gra_country
					opinion = { target = scope:target_gra_country value >= 0 }
				}
				value = 60
			}
		}

		if = {
			limit = { exists = c:GRA }
			transfer_yearly_manpower = {
				value = 1
				target = c:GRA
			}
			transfer_yearly_gold = {
				value = 1
				target = c:GRA
			}
		}

		c:GRA ?= {
			capital = {
				if = {
					limit = { c:MAM = { has_advance = reform_the_mamluk_corps } }
					create_sub_unit_with_owner = {
						type = a_mamluk_horsemen_renaissance
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
					create_sub_unit_with_owner = {
						type = a_mamluk_unit_renaissance
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
				}
				else = {
					create_sub_unit_with_owner = {
						type = a_mamluk_horsemen_traditions
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
					create_sub_unit_with_owner = {
						type = a_mamluk_unit_traditions
						owner = scope:target_gra_country
						origin = c:MAM.capital
						experience = 100
					}
				}
			}
		}
	}

	option = {	# Thoughts and prayers and/or high-fives!
		name = battle_for_the_strait_events.56.c

		ai_will_select = {
			value = 25
		}

		custom_tooltip = gra_gets_nothing_tt

	}
}
