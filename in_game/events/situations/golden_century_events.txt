golden_century_events.5 = {	#Proclaim a Fatwa - choose the aspect; a random version of that aspect is rolled
	type = country_event
	category = situation_event
	image_list = "golden_century_illustration"

	title = golden_century_events.5.title
	desc = golden_century_events.5.desc
	historical_info = golden_century_events.5.desc.historical_info
	outcome = positive

	immediate = {
		ruler_or_regent = { save_scope_as = target_ruler }
		random_list = {
			1 = { set_variable = fatwa_version_1 }
			1 = { set_variable = fatwa_version_2 }
			1 = { set_variable = fatwa_version_3 }
		}
	}

	option = {		# Administration
		name = golden_century_events.5.a
		switch = {
			trigger = has_variable
			fatwa_version_1 = {
				ruler_or_regent = { add_adm = { 1 10 } }
			}
			fatwa_version_2 = {
				ordered_owned_non_rural_location = {
					order_by = population
					max = 4
					check_range_bounds = no
					change_development = development_mild_bonus
				}
			}
			fatwa_version_3 = {
				add_all_estate_satisfaction = { value = estate_satisfaction_severe_bonus }
			}
		}
	}

	option = {		# Diplomacy
		name = golden_century_events.5.b
		switch = {
			trigger = has_variable
			fatwa_version_1 = {
				add_country_modifier = { modifier = fatwa_concord_modifier years = 5 mode = add_and_extend }
			}
			fatwa_version_2 = {
				ruler_or_regent = { add_dip = { 1 10 } }
			}
			fatwa_version_3 = {
				add_country_modifier = { modifier = fatwa_trade_modifier years = 5 mode = add_and_extend }
			}
		}
	}

	option = {		# Military
		name = golden_century_events.5.c
		switch = {
			trigger = has_variable
			fatwa_version_1 = {
				add_yearly_manpower = 2
			}
			fatwa_version_2 = {
				ruler_or_regent = { add_mil = { 1 10 } }
			}
			fatwa_version_3 = {
				add_army_tradition = army_tradition_mild_bonus
				add_navy_tradition = navy_tradition_mild_bonus
			}
		}
	}

	after = {
		hidden_effect = {
			remove_variable = fatwa_version_1
			remove_variable = fatwa_version_2
			remove_variable = fatwa_version_3
		}
	}
}

golden_century_events.7 = {	#Muster the Ghuzat
	type = country_event
	category = situation_event
	image_list = "golden_century_illustration"

	title = golden_century_events.7.title
	desc = golden_century_events.7.desc
	outcome = positive

	immediate = {
		ruler_or_regent = { save_scope_as = target_ruler }

		ordered_country_in_diplomatic_range = {
			limit = { religion.group = religion_group:muslim }
			order_by = {
				value = num_provinces
				multiply = religious_unity
				add = "opinion(scope:actor)"
			}
			max = 10
			check_range_bounds = no
			add_to_list = guzat_origin_country_list
		}

		random_in_list = {
			list = guzat_origin_country_list
			save_scope_as = target_origin_country
		}

		scope:target_origin_country = {
			ordered_province = {
				limit = {
					owner ?= {
						NOT = { this = root }
						religion.group = religion_group:muslim
						within_diplomatic_range = root
					}
				}
				order_by = {
					value = population
					subtract = {
						value = "province_capital.distance_to_squared(root.capital)"
						multiply = 0.0001
					}
					multiply = "province_pop_type_population(pop_type:soldiers)"
					multiply = "province_pop_type_population(pop_type:laborers)"
				}
				max = {
					value = root.country_rank_level
					multiply = 2
				}
				check_range_bounds = no
				add_to_list = origin_provinces_list
			}
		}

		create_character = {
			culture = root.culture
			religion = root.religion
			adm = { 70 100 }
			dip = { 70 100 }
			mil = { 70 100 }
			estate = estate_type:peasants_estate
			add_random_trait_from_category = general
			save_scope_as = target_ghazi_general
			create_in_limbo = yes
		}
	}

	option = {
		name = golden_century_events.7.a
		if = {
			limit = { has_bureaucracy_of_type = bureaucracy_type:ghuzat_mujahidin_bureaucracy }
			scope:target_ghazi_general = { move_country = root }
		}
		else = {
			if = {
				limit = { exists = scope:target_ghazi_general }
				custom_tooltip = {
					text = missed_reward_ghuzat_mujahidin_tt
					kill_character_silently = scope:target_ghazi_general
				}
			}
		}

		every_in_list = {
			list = origin_provinces_list

			add_migration = {
				owner = scope:target_origin_country
				from = this.province_definition
				to = root.capital.province_definition
				type = pop_type:soldiers
				religion = root.religion
				amount = {
					value = "this.province_pop_type_population(pop_type:soldiers)"
					multiply = 0.25
				}
				months = 6
			}

			add_migration = {
				owner = scope:target_origin_country
				from = this.province_definition
				to = root.capital.province_definition
				type = pop_type:laborers
				religion = root.religion
				amount = {
					value = "this.province_pop_type_population(pop_type:laborers)"
					multiply = 0.15
				}
				months = 6
			}
		}
	}
}

			trigger = { exists = scope:seizing_country }

		}

			trigger = { NOT = { exists = scope:seizing_country } }
			text = golden_century_events.10.a.alt
		}

		ai_will_select = {
			add = 100
			add = "opinion(scope:seizing_country)"
			subtract = "opinion(scope:target_overlord)"

		}

		make_subject_of = {
			target = scope:seizing_country
			type = subject_type:colonial_nation
		}

		scope:target_overlord = {
			add_casus_belli = {
				target = scope:target_colony

			}

			add_opinion = {

			}

			}

		name = golden_century_events.10.b

			add = 100

			add = "opinion(scope:target_overlord)"

		}

	}
