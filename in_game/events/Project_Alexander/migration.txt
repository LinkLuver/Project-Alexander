namespace = migration

migration.100 = {
    type = country_event
    title = "migration.1.t"
    desc = "migration.1.desc"

    is_triggered_only = yes

    immediate = {
    	clear_variable = temp_total_pop
        set_variable = {
            name = temp_total_pop
            value = total_population 
        }
    }

    option = {
        name = "migration.1.a"
        add_gold = var:temp_total_pop
		capital = {
			while = {
				count = ROOT.var:temp_total_pop
				create_sub_unit = unit_type:a_migratory_band
			}
        }
        hidden_effect = {
	        every_owned_location = {
	            every_pop = {
	                destroy_pop = this
	            }
	        	owner = {
	    	        abandon_location = prev
	        	}
	        }	
        }
        custom_tooltip = "abandon_country"
		change_country_type = army

        remove_variable = temp_total_pop
    }

}
