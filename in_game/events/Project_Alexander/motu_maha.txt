namespace = motu_maha

# Removes the placeholder pops that main_menu/setup/start/06_pops.txt puts in motu_maha
# (one pop of every culture, so no culture starts without pops).
# Keeps only the local maori_culture / tapu_religion pop.
# Fired 7 days after game start by on_game_start_remove_placeholder_pops (in_game/common/on_action/pa_motu_maha.txt).
# A country event, not a location event: events need a country to fire for and motu_maha has no owner.
motu_maha.1 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	immediate = {
		location:motu_maha = {
			every_pop = {
				limit = {
					NOT = {
						AND = {
							culture = culture:maori_culture
							religion = religion:tapu_religion
						}
					}
				}
				destroy_pop = this
			}
		}
	}
}
