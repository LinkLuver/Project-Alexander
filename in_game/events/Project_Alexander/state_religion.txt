namespace = pa_state_religion

pa_state_religion.1 = {
	type = country_event
	hidden = yes

	trigger = {
		OR = {
			AND = {
				current_age = age_2_imperialism
				NOT = { has_variable = allow_imperial_religion }
				modifier:tolerance_own > 5
				religion_percentage_in_country = {
					religion = root.religion
					value > 0.8
				}
				has_ruler = yes
				ruler = { dip > 80 }
			}
			AND = {
				modifier:tolerance_own > 5
				religion_percentage_in_country = {
					religion = root.religion
					value > 0.8
				}
				has_ruler = yes
				ruler = { dip > 80 }
				knows_about_institution = institution:state_religion
				exists = capital
			}
		}
	}
	immediate = {
		set_variable = {
			name = allow_imperial_religion
			value = yes
		}
		if = {
			limit = {
				knows_about_institution = institution:state_religion
				exists = capital
			}
			capital = {
				change_institution_progress = {
					type = institution:state_religion
					value = 100
				}
			}
		}
	}
}
