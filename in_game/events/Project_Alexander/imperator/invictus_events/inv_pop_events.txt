# Riots among the freemen of a province; a local man brings the news to the capital.
inv_pop_events.1 = {
	type = country_event
	title = inv_pop_events.1.t
	desc = inv_pop_events.1.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		any_owned_location = {
			population_with_traits = { #P:A TODO Approximate: Imperator num_of_freemen >= 10 (freemen pops); EU5 checks 10 (thousand) peasants, pop count converted 1:1 to size
				limit = { pop_type = pop_type:peasants }
				value >= 10
			}
		}
		NOT = { has_variable = freemen_event_cooldown }
	}

	immediate = {
		ruler ?= { save_scope_as = pa_ruler }
		random_owned_location = {
			limit = {
				population_with_traits = {
					limit = { pop_type = pop_type:peasants }
					value >= 10
				}
			}
			save_scope_as = pop_province
		}
		set_variable = {
			name = freemen_event_cooldown
			days = 1825 # medium_cooldown
		}
		# Imperator: the province's holding owner if it had one (Invictus saved the province itself there), else a newly created man.
		#P:A TODO No EU5 equivalent: holdings and holding owners; the new character is always created
		hidden_effect = {
			create_character = {
				female = no
				birth_location = scope:pop_province
				save_scope_as = free_man
			}
			scope:free_man = {
				pa_imp_good_finesse_trait_effect = yes
			}
		}
	}

	option = {
		name = inv_pop_events.1.a
		ruler ?= {
			# add_popularity = -10
			#P:A TODO No EU5 equivalent: -10 popularity for the ruler (EU5 characters have no popularity)
			add_trait = trait:arrogant
		}
	}
	option = {
		name = inv_pop_events.1.b
		scope:pop_province = {
			add_location_modifier = {
				modifier = local_unrest_generic_unrest
				days = 3650
			}
		}
	}
}
