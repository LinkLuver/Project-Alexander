# A minor earthquake damages a fortress in a poor province.
inv_building_events.1 = {
	type = country_event
	title = inv_building_events.1.t
	desc = inv_building_events.1.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		NOT = { has_variable = military_building_cooldown }
		any_owned_location = {
			has_fort = yes #P:A TODO Approximate: Imperator num_of_fortress_building > 1 (fortress level 2+); EU5 forts have no stacked levels
			location_tax_base < 3 #P:A TODO Approximate: Imperator province_income < 3 (monthly province income); EU5 location tax base has a different scale
		}
	}

	immediate = {
		ruler ?= { save_scope_as = pa_ruler }
		set_variable = {
			name = military_building_cooldown
			days = 9125 # very_long_cooldown
		}
		random_owned_location = {
			limit = {
				has_fort = yes #P:A TODO Approximate: Imperator num_of_fortress_building > 1
				location_tax_base < 3 #P:A TODO Approximate: Imperator province_income < 3
			}
			save_scope_as = building_province
		}
	}

	option = {
		name = inv_building_events.1.a
		scope:building_province = {
			add_location_modifier = {
				modifier = local_unrest_generic_unrest
				days = 730
			}
		}
	}
	option = {
		name = inv_building_events.1.b
		add_gold = {
			value = 0
			subtract = root.monthly_income_total
			multiply = 3
			min = -6000
		}
	}
}

	}

	trigger = {

		any_owned_location = {
			has_fort = yes #P:A TODO Approximate: Imperator num_of_fortress_building > 1 (fortress level 2+); EU5 forts have no stacked levels
			# governor_policy = borderlands
			any_neighbor_location = { #P:A TODO Approximate: Imperator governor_policy = borderlands (EU5 has no governors or governor policies); a location bordering another country is used
				exists = owner

		ruler ?= { save_scope_as = pa_ruler }
		set_variable = {
			name = military_building_cooldown

		}
		random_owned_location = {
			limit = {
				has_fort = yes #P:A TODO Approximate: Imperator num_of_fortress_building > 1
				any_neighbor_location = { #P:A TODO Approximate: Imperator governor_policy = borderlands
					exists = owner
					NOT = { owner = root }
				}
			}
			save_scope_as = building_province
		}
		# Imperator: the fort guy was the province's holding owner, else its governor (if not the ruler), else a newly created man.
		#P:A TODO No EU5 equivalent: holding owners and governors; the new character is always created
		hidden_effect = {
			create_character = {
				female = no
				birth_location = scope:building_province
				save_scope_as = fort_guy
			}
			scope:fort_guy = {
				pa_imp_good_finesse_trait_effect = yes
			}
		}
	}

	option = {
		name = inv_building_events.3.a
		add_stability = 3
		# current_ruler = { add_popularity = 5 }
		#P:A TODO No EU5 equivalent: +5 popularity for the ruler (EU5 characters have no popularity)
	}
	option = {
		name = inv_building_events.3.b
		add_gold = {
			value = 0
			subtract = root.monthly_income_total
			multiply = 6
			floor = yes
			max = -20
			min = -2000
		}
		capital = {
			add_location_modifier = {
				modifier = fort_guy_assisting_local_fort
				days = 3650
			}
		}
	}
}
