# A minor earthquake damages a fortress in a poor province.
inv_building_events.1 = {
	type = country_event
	title = inv_building_events.1.t
	desc = inv_building_events.1.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		NOT = { has_variable = military_building_cooldown }
		any_owned_location = {
			has_fort = yes #P:A TODO Approximate: Imperator num_of_fortress_building > 1 (fortress level 2+); EU5 forts have no stacked levels
			location_tax_base < 3 #P:A TODO Approximate: Imperator province_income < 3 (monthly province income); EU5 location tax base has a different scale
		}
	}

	immediate = {
		ruler ?= { save_scope_as = pa_ruler }
		set_variable = {
			name = military_building_cooldown
			days = 9125 # very_long_cooldown
		}
		random_owned_location = {
			limit = {
				has_fort = yes #P:A TODO Approximate: Imperator num_of_fortress_building > 1
				location_tax_base < 3 #P:A TODO Approximate: Imperator province_income < 3
			}
			save_scope_as = building_province
		}
	}

	option = {
		name = inv_building_events.1.a
		scope:building_province = {
			add_location_modifier = {
				modifier = local_unrest_generic_unrest
				days = 730
			}
		}
	}
	option = {
		name = inv_building_events.1.b
		add_gold = {
			value = 0
			subtract = root.monthly_income_total
			multiply = 3
			min = -6000
		}
	}
}

				}
			}
			save_scope_as = building_province

				female = no
				birth_location = scope:building_province

			floor = yes
			max = -20
			min = -2000
		}
		capital = {
			add_location_modifier = {
				modifier = fort_guy_assisting_local_fort
				days = 3650
			}
		}
	}
}

		any_owned_location = {
			has_building = building_type:counting_house #P:A TODO Approximate: Imperator num_of_town_hall_building > 1 (Tax Office level 2+); EU5 counting_house (single level)
			population_with_traits = { #P:A TODO Approximate: Imperator num_of_slaves > 4 (slave pops); EU5 checks more than 4 (thousand) slaves, pop count converted 1:1 to size
				limit = { pop_type = pop_type:slaves }
				value > 4
			}
			population_with_traits = { #P:A TODO Approximate: Imperator num_of_nobles > 3 (noble pops); EU5 checks more than 3 (thousand) nobles, pop count converted 1:1 to size
				limit = { pop_type = pop_type:nobles }
				value > 3
			}
		}
	}

	immediate = {
		set_variable = {
			name = population_building_cooldown
			days = 3650 # long_cooldown
		}
		random_owned_location = {
			limit = {
				has_building = building_type:counting_house #P:A TODO Approximate: Imperator num_of_town_hall_building > 1
				population_with_traits = {
					limit = { pop_type = pop_type:slaves }
					value > 4
				}
				population_with_traits = {
					limit = { pop_type = pop_type:nobles }
					value > 3
				}
			}
			save_scope_as = building_province
		}
		scope:building_province = {
			ordered_pop = {
				limit = { pop_type = pop_type:nobles }
				order_by = pop_satisfaction # Imperator: pop_hapiness
				add_pop_size = { #P:A TODO Approximate: Imperator kill_pop removed 1-4 whole noble pops (the happiest); EU5 removes 1-4 (thousand) people from the most satisfied noble pop
					integer_range = {
						min = 1
						max = 4
					}
					max = pop_size
					multiply = -1
				}
			}
		}
	}
