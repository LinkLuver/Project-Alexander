# Hidden: a Macedonian successor has occupied every key place of a feudatory/satrapy of another Macedonian successor;
# the subject is offered a change of overlord (diadochi_liberation.2006).
# Imperator province event (root = the occupied province) fired by on_province_occupied. Written for EU5 on_siege_won and
# on_location_occupied (root = country, scope:target = the occupied location); the occupier is read from scope:target.controller.
diadochi_liberation.2001 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = scope:target
		scope:target = {
			exists = owner
			controller = {
				culture = culture:macedonian # Imperator primary_culture = macedonian
				has_variable = war_of_the_diadochs
			}
			owner = {
				is_subject = yes
				OR = {
					is_subject_type = feudatory
					is_subject_type = satrapy
				}
				NOT = { # Imperator any_owned_province = { count = all controller = root.controller filter = { ... } }: every owned state capital, fortress and the capital is held by the occupier
					any_owned_location = {
						OR = {
							is_province_capital = yes # Imperator is_state_capital
							has_fort = yes # Imperator num_of_fortress_building >= 1
							is_capital = yes
						}
						NOT = {
							controller = scope:target.controller
						}
					}
				}
			}
			owner.overlord = {
				culture = culture:macedonian # Imperator primary_culture = macedonian
				has_variable = war_of_the_diadochs
			}
		}
	}

	immediate = {
		scope:target = {
			controller = {
				save_scope_as = subject_flipper
			}
			owner.overlord = {
				save_scope_as = current_overlord
				#trigger_event = {
					#id = diadochi_liberation.2002#old
				#}
			}
			owner = {
				save_scope_as = flipping_subject
				trigger_event_non_silently = {
					id = diadochi_liberation.2006 #NEW
				}
			}
		}
	}
}

# The overlord learns that its subject is under occupation (old version of the chain; its call in diadochi_liberation.2001 is commented out in the Invictus source).
diadochi_liberation.2002 = {
	type = country_event
	title = diadochi_liberation.2002.t
	desc = diadochi_liberation.2002.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		scope:current_overlord.ruler ?= {
			save_scope_as = pa_current_overlord_ruler
		}
		scope:subject_flipper.ruler ?= {
			save_scope_as = pa_subject_flipper_ruler
		}
		scope:flipping_subject.ruler ?= {
			save_scope_as = pa_flipping_subject_ruler
		}
	}

	option = { # 10 - 20 days to relive it
		name = diadochi_liberation.2002.a
		scope:flipping_subject = {
			trigger_event_non_silently = {
				id = diadochi_liberation.2003
				days = { 10 20 }
			}
		}
	}
}

# The occupied subject decides whether to defect to the occupier (old version of the chain, fired by diadochi_liberation.2002).
diadochi_liberation.2003 = {
	type = country_event
	title = diadochi_liberation.2003.t
	desc = diadochi_liberation.2003.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		NOT = {
			any_owned_location = {
				controller = root
				OR = {
					is_province_capital = yes # Imperator is_state_capital
					has_fort = yes # Imperator num_of_fortress_building >= 1
				}
			}
		}
		capital = {
			controller = scope:subject_flipper
		}
		is_subject = yes
		overlord = {
			this = scope:current_overlord
		}
	}

	immediate = {
		scope:current_overlord.ruler ?= {
			save_scope_as = pa_current_overlord_ruler
		}
		scope:subject_flipper.ruler ?= {
			save_scope_as = pa_subject_flipper_ruler
		}
		scope:flipping_subject.ruler ?= {
			save_scope_as = pa_flipping_subject_ruler
		}
	}

	option = { # new overlord
		name = diadochi_liberation.2003.a
		custom_tooltip = diadochi_liberation.2003.at #joing war
		scope:subject_flipper = {
			if = {
				limit = {
					scope:flipping_subject = { is_subject_type = feudatory }
				}
				scope:current_overlord = {
					cancel_subject = scope:flipping_subject # Imperator release_subject
					remove_truce_with = scope:flipping_subject # Imperator end_truce
				}
				hidden_effect = {
					every_current_war = {
						limit = {
							any_war_participant = { this = root }
						}
						root = {
							leave_war = { # Imperator remove_from_war = root
								war = prev
								actor = scope:subject_flipper
							}
						}
					}
				}
				scope:flipping_subject = {
					remove_truce_with = scope:current_overlord # Imperator end_truce
					remove_truce_with = scope:subject_flipper # Imperator end_truce
					make_subject_of = {
						target = scope:subject_flipper
						type = subject_type:feudatory
					}
				}
			}
			else = {
				scope:current_overlord = {
					cancel_subject = scope:flipping_subject # Imperator release_subject
					remove_truce_with = scope:flipping_subject # Imperator end_truce
				}
				hidden_effect = {
					every_current_war = {
						limit = {
							any_war_participant = { this = root }
						}
						root = {
							leave_war = { # Imperator remove_from_war = root
								war = prev
								actor = scope:subject_flipper
							}
						}
					}
				}
				scope:flipping_subject = {
					remove_truce_with = scope:current_overlord # Imperator end_truce
					remove_truce_with = scope:subject_flipper # Imperator end_truce
					make_subject_of = {
						target = scope:subject_flipper
						type = subject_type:satrapy
					}
				}
			}
			hidden_effect = {
				every_current_war = {
					limit = {
						is_an_attacker = scope:subject_flipper # Imperator any_war_attacker
					}
					scope:flipping_subject = {
						join_war_as_attacker = { # Imperator add_to_war (attacker, not leader)
							war = prev
						}
					}
				}
				every_current_war = {
					limit = {
						is_a_defender = scope:subject_flipper # Imperator any_war_defender
					}
					scope:flipping_subject = {
						join_war_as_defender = { # Imperator add_to_war (defender, not leader)
							war = prev
						}
					}
				}
				# add_aggressive_expansion = {
				# 	value = scope:flipping_subject.num_of_cities
				# 	multiply = 0.375
				# 	multiply = scope:subject_flipper.modifier:agressive_expansion_impact
				# 	add = {
				# 		value = scope:flipping_subject.num_of_cities
				# 		multiply = 0.375
				# 	}
				# 	min = {
				# 		value = scope:flipping_subject.num_of_cities
				# 		multiply = 0.0375
				# 	}
				# }
				#P:A TODO No EU5 equivalent: the new overlord gains aggressive expansion scaled by the subject's city count (EU5 antagonism is per country, not a global value)
			}
		}
		ai_chance = { factor = 200 }
	}

	option = { # hold on a little longer
		name = diadochi_liberation.2003.b
		#custom_tooltip = diadochi_liberation.2003.bt
		scope:flipping_subject = {
			trigger_event_non_silently = {
				id = diadochi_liberation.2003
				days = { 60 180 }
			}
		}
		add_manpower = -1
		ai_chance = { factor = 0 }
	}
}
