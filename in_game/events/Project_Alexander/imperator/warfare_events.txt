# warfare.3/4/5: mercenary bribing. In Imperator fired by the character interaction merc_buyoff, the bribe-mercenary GUI
# button and the AI effect ai_recruit_mercenaries on the mercenary's current employer, with scope:actor = the bribing
# country and scope:target = the mercenary leader. Here scope:target is the EU5 mercenary company.
# warfare.6-9, 13, 17: Imperator character events, converted to country events (character = scope:pa_character).
namespace = warfare

# Another country tries to buy away our mercenaries: match the offer?
warfare.3 = {
	type = country_event
	title = warfare.3.t
	desc = warfare.3.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		exists = scope:target
		exists = scope:actor
	}

	immediate = {
		ruler ?= { save_scope_as = merc_employer_ruler }
		scope:actor = {
			ruler ?= { save_scope_as = merc_briber_ruler }
		}
		save_scope_as = counter_target
		scope:target = {
			save_scope_as = mercenary_to_bribe
		}
	}

	option = {
		name = warfare.3.a
		add_gold = {
			value = 0
			subtract = pa_imp_army_size_merc_estimation
		}
		scope:actor = {
			trigger_event_non_silently = {
				id = warfare.4
			}
		}
		ai_chance = {
			modifier = {
				factor = 0
				pa_imp_ai_should_counter_bribe_mercenary = no
			}
		}
	}

	option = {
		name = warfare.3.b
		scope:target = {
			become_hired_by = scope:actor #P:A TODO Approximate: Imperator change_mercenary_employer moved the mercenary leader and his units to the briber; EU5 has the company hired by the briber
		}
		scope:actor = {
			trigger_event_non_silently = {
				id = warfare.5
			}
		}
		ai_chance = {
			modifier = {
				factor = 0
				pa_imp_ai_should_counter_bribe_mercenary = yes
			}
		}
	}
}

# Battle Wound. Imperator: character event from the random_events of in_land_battle (root = a commander in the battle,
# weight 4 against 1098 for nothing). EU5 in_battle has root = the supreme commander (character); the caller must save
# that character as scope:pa_character and fire this on its employer.
warfare.6 = {
	type = country_event
	title = warfare.6.t
	desc = warfare.6.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				is_adult = yes
				NOT = { has_trait = scarred } #P:A TODO Approximate: Imperator NOT has_trait = wounded; the mod has no wounded trait, scarred is used
				NOT = { has_variable = wounded_in_battle }
			}
		}
		trigger_else = {
			any_character = {
				is_general = yes # not in Imperator: root was always a commander in battle
				is_adult = yes
				NOT = { has_trait = scarred }
				NOT = { has_variable = wounded_in_battle }
			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			exists = scope:pa_character
			add = {
				value = scope:pa_character.mil
				divide = 20 # Imperator martial / 4; EU5 skills are x5
				multiply = -1
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					is_general = yes
					is_adult = yes
					NOT = { has_trait = scarred }
					NOT = { has_variable = wounded_in_battle }
				}
				save_scope_as = pa_character
			}
		}
		scope:pa_character = {
			add_trait = trait:scarred #P:A TODO Approximate: Imperator added the temporary "wounded" trait; EU5 has no wounded trait, scarred used
			set_variable = {
				name = wounded_in_battle
				days = 60
			}
			if = {
				limit = {
					NOT = { has_trait = one_eyed }
					NOT = { has_trait = blind }
				}
				random_list = {
					5 = { add_trait = trait:one_eyed }
					95 = { }
				}
			}
		}
	}

	option = {
		name = warfare.6.a
		# scope:pa_character = { add_popularity = 5 }
		#P:A TODO No EU5 equivalent: +5 popularity for the wounded commander (EU5 characters have no popularity)
	}
}

# Hidden: a supported pretender's civil war breaks out; the pretender takes the rebel throne and the supporter is told.
# Imperator: character event from on_move_country (root = the pretender who just moved to the rebel country,
# scope:old_country = the country left). The pretender carries the variable successor_support_ci (= the supporting
# country), set by monarchy_events.102 (batch b13). Written for EU5 on_civil_war_start fired on the rebel country
# (scope:target there) or with scope:pa_character saved; without it the event looks for the marked pretender itself.
warfare.13 = {
	type = country_event
	hidden = yes

	trigger = {
		in_civil_war = yes # Imperator: employer = { civil_war_with = scope:old_country }
		trigger_if = {
			limit = { exists = scope:old_country }
			civil_war_opponent ?= scope:old_country
		}
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				has_variable = successor_support_ci
				exists = var:successor_support_ci
				var:successor_support_ci = {
					exists = capital
				}
			}
		}
		trigger_else = {
			any_character = {
				has_variable = successor_support_ci
				exists = var:successor_support_ci
				var:successor_support_ci = {
					exists = capital
				}
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					has_variable = successor_support_ci
					exists = var:successor_support_ci
				}
				save_scope_as = pa_character
			}
		}
		scope:pa_character = {
			var:successor_support_ci = {
				save_scope_as = origin_supporter
			}
			save_scope_as = supported_target
			remove_variable = successor_support_ci
		}
		set_new_ruler = scope:pa_character
		scope:origin_supporter = {
			trigger_event_non_silently = {
				id = warfare.12
			}
		}
	}
}

# Successor support: the pretender we backed won the civil war; make the kingdom our client?
# Imperator: on_civil_war_won/on_civil_war_lost (Invictus on_civil_war_won_inv.txt): when the winner's ruler carries
# complete_pretender_support (= the supporter, set by warfare.12.d) and the supporter is allied to the winner, the winner
# is saved as scope:puppet_kingdom and the supporter gets warfare.14 if it outranks the winner, else warfare.15.
warfare.14 = {
	type = country_event
	title = warfare.14.t
	desc = warfare.14.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		exists = scope:puppet_kingdom # not in Imperator: its only caller always saved it
	}

	immediate = {
		ruler ?= { save_scope_as = puppeteer_ruler }
		scope:puppet_kingdom = {
			ruler ?= { save_scope_as = puppet_ruler }
		}
	}

	option = {
		name = warfare.14.a
		# add_tyranny = 20
		#P:A TODO No EU5 equivalent: +20 tyranny (EU5 has no tyranny)
		add_stability = -10
		scope:puppet_kingdom = { # Imperator make_subject = { target = scope:puppet_kingdom type = client_state }
			make_subject_of = {
				target = root
				type = subject_type:client_state
			}
		}
		save_scope_as = puppeteer_country
		every_country = {
			limit = {
				is_ai = no
			}
			trigger_event_non_silently = {
				id = warfare.16
			}
		}
		ai_chance = {
			modifier = { factor = 100 }
		}
	}

	option = {
		name = warfare.14.b
		add_gold = 500
	}
}
