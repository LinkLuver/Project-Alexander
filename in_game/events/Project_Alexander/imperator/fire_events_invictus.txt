# A fire in a crowded city (scope:en_fuego). Invictus has no caller for fire_events.2 (the chain start), so it is dormant until hooked up.
namespace = fire_events

# A fire breaks out in one of our crowded cities.
fire_events.2 = {
	type = country_event # Imperator: country_event (a province_event in vanilla)
	title = fire_events.2.t
	desc = fire_events.2.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		any_owned_location = {
			NOT = { location_rank = location_rank:rural_settlement } # Imperator has_city_status = yes
			# OR = {
			# 	AND = { any_pops_in_province = { count > 20 } any_pops_in_province = { count >= population_cap } }
			#P:A TODO No EU5 equivalent: "more than 20 pops and at the population cap" (EU5 locations have no pop cap)
			population > 30 #P:A TODO Approximate: Imperator more than 30 pops, read as 30 (thousand) EU5 population
			# }
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		random_owned_location = {
			limit = {
				NOT = { location_rank = location_rank:rural_settlement }
				population > 30 #P:A TODO Approximate: Imperator more than 30 pops (or more than 20 at the pop cap), read as 30 (thousand) EU5 population
			}
			save_scope_as = en_fuego
		}
	}

	option = {
		name = fire_events.2.a
		trigger = {
			# invention = global_freemen_output_inv_4 # fire_brigade
			always = no #P:A TODO No EU5 equivalent: needs the Invictus invention "fire brigade" (global_freemen_output_inv_4); EU5 has no inventions
		}
		add_gold = -50
	}

	option = {
		name = fire_events.2.b
		scope:en_fuego = {
			random_pop = {
				random_list = {
					50 = {
						add_pop_size = { #P:A TODO Approximate: Imperator kill_pop removed 1 whole pop; EU5 removes up to 1 (thousand) people from a random pop
							value = pop_size
							multiply = -1
							min = -1
						}
					}
					50 = {}
				}
			}
		}
		hidden_effect = {
			random_list = {
				50 = {
					trigger_event_non_silently = {
						id = fire_events.3
						days = 1
					}
				}
				50 = {
					trigger_event_non_silently = {
						id = fire_events.5
						days = 1
					}
				}
			}
		}
	}
}
