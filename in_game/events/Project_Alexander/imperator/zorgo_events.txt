# A family member of the ruler maims a rival over a property dispute.
zorgo_event.1 = {
	type = country_event
	title = zorgo_event.1.t
	desc = zorgo_event.1.desc
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		is_subject = no
		exists = ruler
		ruler = { has_dynasty = yes }
		any_character = {
			is_dynasty_head = no
			is_ruler = no
			is_adult = yes
			is_female = no
			has_dynasty = yes
			dynasty = root.ruler.dynasty # from_ruler_family = yes
		}
	}

	immediate = {
		# get_event_picture = { cg = "hellenic" } #P:A Removed: only picked the Imperator event picture; EU5 uses illustration_tags
		ruler = {
			save_scope_as = pa_ruler
		}
		random_character = {
			limit = {
				is_dynasty_head = no
				is_ruler = no
				is_adult = yes
				is_female = no
				has_dynasty = yes
				dynasty = root.ruler.dynasty # from_ruler_family = yes
			}
			save_scope_as = random_dude
		}
		random_character = {
			limit = {
				is_dynasty_head = no
				is_ruler = no
				is_adult = yes
				is_female = no
				OR = { # from_ruler_family = no
					has_dynasty = no
					NOT = { dynasty = root.ruler.dynasty }
				}
				NOT = { this = scope:random_dude } # Imperator workaround "age != scope:random_dude.age" to avoid picking the same character
			}
			save_scope_as = random_dude_2
			hidden_effect = {
				add_trait = trait:one_eyed
			}
		}
	}

	fire_only_once = yes

	option = {
		name = zorgo_event.1.a
		add_gold = {
			value = root.monthly_income_total
			multiply = -4
			min = -6000
		}
		# scope:random_dude = { add_loyalty = spurned_family } #P:A TODO No EU5 equivalent: -loyalty ("spurned family") for the ruler's relative (EU5 characters have no loyalty)
		# scope:random_dude_2 = { add_loyalty = loyalty_love_encouraged } #P:A TODO No EU5 equivalent: +loyalty for the victim (EU5 characters have no loyalty)
	}

	option = {
		name = zorgo_event.1.b
		# add_tyranny = 3 #P:A TODO No EU5 equivalent: +3 tyranny (EU5 has no tyranny)
		# scope:random_dude = { add_loyalty = loyalty_love_encouraged } #P:A TODO No EU5 equivalent: +loyalty for the ruler's relative (EU5 characters have no loyalty)
		# scope:random_dude_2 = { add_loyalty = spurned_family } #P:A TODO No EU5 equivalent: -loyalty for the victim (EU5 characters have no loyalty)
		# current_ruler = { add_popularity = subtract_popularity_small } #P:A TODO No EU5 equivalent: -5 popularity for the ruler (EU5 characters have no popularity)
	}

# The ruler wants to lead a raid into a neighbor's lands.
# Imperator fired this from the "nomadic_raid" character interaction (tribal chiefdom / steppe horde rulers, not at war,
# costing 25 political influence), with the raided country as scope:recipient. EU5 has no such interaction: if a caller
# sets scope:recipient it is used, otherwise the event picks a neighboring target itself so it can run from a pulse.
zorgo_event.125 = {
	type = country_event
	title = me_zorgoomni.31_title
	desc = me_zorgoomni.31_desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		exists = ruler
		ruler = {
			is_adult = yes
			NOT = { has_variable = nomadic_raid_cooldown } # from the Imperator interaction
		}
		trigger_if = {
			limit = { exists = scope:recipient }
			scope:recipient = {
				exists = capital
				NOT = { has_variable = stop_raiding_me_every_day_please_var }
			}
		}
		trigger_else = { #P:A TODO Approximate: stands in for the Imperator nomadic_raid interaction conditions (government tribal_chiefdom or steppe_horde, or the ruler_pirate_raiding_allowed variable; not at war; target in diplomatic range); the target is limited to neighbors here
			government_type = government_type:tribe
			at_war = no
			any_neighbor_country = {
				exists = capital
				within_diplomatic_range = root # in_diplomatic_range = scope:actor
				NOT = { has_variable = stop_raiding_me_every_day_please_var }
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		root = {
			save_scope_as = nomadic_raid_country
		}
		if = { #P:A Added: pick the raided country when no caller supplied one
			limit = {
				NOT = { exists = scope:recipient }
			}
			random_neighbor_country = {
				limit = {
					exists = capital
					within_diplomatic_range = root
					NOT = { has_variable = stop_raiding_me_every_day_please_var }
				}
				save_scope_as = recipient
			}
		}
		scope:recipient = {
			ruler ?= {
				save_scope_as = recipient_ruler
			}
		}
		hidden_effect = {
			ruler = { #P:A Added: cooldown the Imperator interaction set on the raiding ruler
				set_variable = {
					name = nomadic_raid_cooldown
					days = 1825
				}
			}
		}
	}

	option = {
		name = me_zorgoomni.31.a
		custom_tooltip = me_zorgoomni.31_tooltip1
		trigger_event_non_silently = {
			id = zorgo_event.126
			days = 30
		}
		# hidden_effect = { capital_scope = { show_animated_text = raiding_party_leaves_animated_text } } #P:A TODO No EU5 equivalent: floating "Raiding party leaves!" text over the capital
	}
	option = {
		name = me_zorgoomni.31.b
		# current_ruler = { add_popularity = -5 } #P:A TODO No EU5 equivalent: -5 popularity for the ruler (EU5 characters have no popularity)
		custom_tooltip = me_zorgoomni.31_tooltip2
	}
}

		10 = angry
	}
