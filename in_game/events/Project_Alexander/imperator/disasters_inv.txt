# A river floods one of our flood plains.
disasters_inv.1 = { #flood in flood_plain
	type = country_event
	title = disasters_inv.1.t
	desc = disasters_inv.1.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		any_owned_location = {
			pa_imp_b14_flood_plain_trigger = yes # Imperator terrain = flood_plain
		}
		NOT = {
			has_variable = floods_cooldown
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		random_owned_location = {
			limit = {
				pa_imp_b14_flood_plain_trigger = yes # Imperator terrain = flood_plain
				population >= 4 #P:A TODO Approximate: Imperator total_population >= 4 pops, read as 4 (thousand) EU5 population
			}
			save_scope_as = the_day_after
			add_location_modifier = {
				modifier = flood_initial_damage
				days = 730
			}
			# ordered_pops_in_province = { order_by = pop_hapiness max = { value = total_population multiply = 0.15 } kill_pop = yes }
			every_pop = { #P:A TODO Approximate: Imperator killed the 15% unhappiest pops; EU5 shrinks every pop by 15%
				add_pop_size = {
					value = pop_size
					multiply = -0.15
				}
			}
			custom_tooltip = disasters_inv_destroyed_building
			hidden_effect = {
				pa_imp_b14_disaster_building_damage_effect = { NOTHING = 8 }
			}
		}
	}

	option = { #secure the area
		name = disasters_inv.1.a
		# ruler = { add_popularity = 5 } #P:A TODO No EU5 equivalent: +5 popularity for the ruler (EU5 characters have no popularity)
		add_gold = {
			value = monthly_income_total # Imperator subtract = has_monthly_income
			multiply = -6
			floor = yes
			max = -50
			min = -200
		}
		trigger_event_non_silently = {
			id = disasters_inv.3
			days = { 10 20 }
		}
	}

	option = { #greedy bastard
		name = disasters_inv.1.b
		hidden_effect = {
			if = {
				limit = {
					scope:the_day_after = {
						any_neighbor_location = {
							exists = owner
							owner = root
							OR = {
								pa_imp_b14_flood_plain_trigger = yes # Imperator terrain = flood_plain
								vegetation = farmland # Imperator terrain = farmland
								topography = flatland # Imperator terrain = plains
							}
						}
					}
				}
				random = {
					chance = 50
					trigger_event_non_silently = {
						id = disasters_inv.2
						days = { 1 7 }
					}
				}
			}
		}
	}

	after = {
		set_variable = {
			name = floods_cooldown
			days = 9125 #25yrs
		}
	}
}

###Avalanche
# An avalanche hits a mountain location; it may set off another one nearby.
disasters_inv.4 = { #Avalanche
	type = country_event
	title = disasters_inv.4.t
	desc = disasters_inv.4.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		any_owned_location = {
			topography = mountains # Imperator terrain = mountain
		}
		NOT = {
			has_variable = avalanche_cooldown
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		random_owned_location = {
			limit = {
				topography = mountains # Imperator terrain = mountain
				population >= 4 #P:A TODO Approximate: Imperator total_population >= 4 pops, read as 4 (thousand) EU5 population
			}
			if = {
				limit = {
					NOT = {
						has_variable = starting_point
					}
				}
				set_variable = {
					name = starting_point
					days = 2
				}
			}
			# Imperator switch = { trigger = has_variable starting_point = { save_scope_as = snowball } recurring_disaster = {} }
			if = {
				limit = {
					has_variable = starting_point
				}
				save_scope_as = snowball
			}
			add_location_modifier = {
				modifier = hit_by_avalanche
				days = 730
			}
			# ordered_pops_in_province = { order_by = pop_hapiness max = { value = total_population multiply = 0.35 } kill_pop = yes }
			every_pop = { #P:A TODO Approximate: Imperator killed the 35% unhappiest pops; EU5 shrinks every pop by 35%
				add_pop_size = {
					value = pop_size
					multiply = -0.35
				}
			}
			every_unit_in_location = { # Imperator wrapped this in if = { limit = { any_unit_in_province = { exists = yes } } }, a no-op check
				damage_unit_percent = 0.06
				damage_unit_morale_percent = 0.12
			}
			custom_tooltip = disasters_inv_destroyed_building
			hidden_effect = {
				pa_imp_b14_disaster_building_damage_effect = { NOTHING = 8 }
			}
		}
	}

	option = {
		name = disasters_inv.4.a
	}

	after = {
		if = {
			limit = {
				exists = scope:snowball
				scope:snowball = {
					any_neighbor_location = {
						topography = mountains # Imperator terrain = mountain
						NOT = {
							has_location_modifier = hit_by_avalanche
						}
					}
				}
			}
			scope:snowball = {
				clear_saved_scope = snowball
				random_neighbor_location = {
					limit = {
						topography = mountains # Imperator terrain = mountain
						NOT = {
							has_location_modifier = hit_by_avalanche
						}
					}
					save_scope_as = snowball
					set_variable = {
						name = recurring_disaster
						days = 7
					}
					random = {
						chance = 50
						owner ?= {
							trigger_event_non_silently = {
								id = disasters_inv.4
								days = { 3 5 }
							}
						}
					}
				}
			}
		}
		else = {
			set_variable = {
				name = avalanche_cooldown
				days = 9125 #25 yrs
			}
		}
	}
}

###Drought - May turn into a famine
# A province with no food reserves suffers a drought; we can share the water of the neighbouring areas or let it spread (disasters_inv.8).
# Imperator states are EU5 provinces: scope:i_am_thirsty is a province.
#P:A TODO Approximate: Imperator state food amounts (add_state_food) are applied 1:1 to the EU5 province food stockpile (change_province_food); the scales are not known to match
disasters_inv.7 = { #drought
	type = country_event
	title = disasters_inv.7.t
	desc = disasters_inv.7.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		any_province = { # Imperator any_country_state
			OR = {
				# state_monthly_food_income < 0 #P:A TODO No EU5 equivalent: the province's monthly food balance is negative (EU5 has no province food balance trigger)
				province_food <= 0 # Imperator has_state_food <= 0
			}
		}
		any_owned_location = {
			pa_imp_b14_desert_trigger = yes # Imperator terrain = desert / desert_hills
		}
		NOT = {
			has_variable = drought_cooldown
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		random_province = { # Imperator random_country_state
			limit = {
				OR = {
					# state_monthly_food_income < 0 #P:A TODO No EU5 equivalent: negative monthly food balance of the province
					province_food <= 0 # Imperator has_state_food <= 0
				}
			}
			save_scope_as = i_am_thirsty
			add_province_modifier = {
				modifier = grow_a_pear
				days = 1825 #5 yrs
			}
		}
	}

	option = { #try to save and feed your people
		name = disasters_inv.7.a
		scope:i_am_thirsty = {
			area = {
				every_neighbor_area = {
					every_province_in_area = { # Imperator every_area_state
						change_province_food = -100 # Imperator add_state_food = -100
						add_province_modifier = {
							modifier = i_bet_you_wish_you_had_my_water
							days = 365
						}
					}
				}
			}
			change_province_food = 200 # Imperator add_state_food = { value = 100 multiply = 2 } #count neighboring states
		}
		add_gold = {
			value = monthly_income_total # Imperator subtract = has_monthly_income
			multiply = -16
			min = -300
			max = -50
		}
		ruler ?= {
			# add_trait = benefactor #P:A TODO No EU5 equivalent: the ruler gains the benefactor trait (not in EU5)
		}
		hidden_effect = {
			if = {
				limit = {
					scope:i_am_thirsty.area = {
						any_neighbor_area = {
							any_province_in_area = { # Imperator any_area_state
								province_food <= 0 # Imperator has_state_food <= 0
							}
						}
					}
				}
				scope:i_am_thirsty.area = {
					every_neighbor_area = {
						every_province_in_area = { # Imperator every_area_state
							limit = {
								province_food <= 0 # Imperator has_state_food <= 0
							}
							add_province_modifier = {
								modifier = grow_a_pear
								days = 1825 # 5yrs
							}
						}
					}
				}
			}
		}
	}

	option = { #ignore the area
		name = disasters_inv.7.b
		ruler ?= {
			# add_trait = harsh #P:A TODO No EU5 equivalent: the ruler gains the harsh trait (not in EU5)
		}
		hidden_effect = {
			trigger_event_non_silently = {
				id = disasters_inv.8 #drought spreads
				days = { 30 100 }
			}
			scope:i_am_thirsty.area = {
				every_neighbor_area = {
					every_province_in_area = { # Imperator every_area_state
						limit = {
							owner = root
						}
						change_province_food = -150 # Imperator add_state_food = -150
					}
				}
			}
		}
	}
}

# Famine: the ignored drought spreads to the neighbouring areas and may keep spreading (fired by disasters_inv.7 option b and by itself;
# scope:i_am_thirsty (a province) carries over).
disasters_inv.8 = {
	type = country_event
	title = disasters_inv.8.t
	desc = disasters_inv.8.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		exists = scope:i_am_thirsty
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = drought_spread
			days = 500
			value = 5
		}
		scope:i_am_thirsty.area = {
			every_neighbor_area = {
				every_province_in_area = { # Imperator every_area_state
					add_province_modifier = {
						modifier = grow_a_pear
						days = 1825 # 5yrs
					}
				}
			}
		}
	}

	option = {
		name = disasters_inv.8.a
		random_list = {
			25 = {
				modifier = {
					factor = var:drought_spread
				}
				custom_tooltip = disasters_inv_8_a_1
				trigger_event_non_silently = {
					id = disasters_inv.8
					days = { 30 100 }
				}
			}
			75 = {
				custom_tooltip = disasters_inv_8_a_2
				set_variable = {
					name = drought_cooldown
					days = 9125 #25 yrs
				}
			}
		}
	}

	option = {
		name = disasters_inv.8.b
		custom_tooltip = disasters_inv_8_b
		trigger = {
			ruler ?= {
				OR = {
					has_trait = kind_hearted # Imperator merciful
					# has_trait = lunatic #P:A TODO No EU5 equivalent: lunatic trait
					# has_trait = lapsed #P:A TODO No EU5 equivalent: lapsed trait
					# has_trait = arbitrary #P:A TODO No EU5 equivalent: arbitrary trait
					# has_trait = easygoing #P:A TODO No EU5 equivalent: easygoing trait
				}
			}
			var:drought_spread >= 1
		}
		add_gold = {
			value = monthly_income_total # Imperator subtract = has_monthly_income
			multiply = -32
			min = -500
			max = -50
		}
		# highlight = yes #P:A Removed: Imperator option highlighting has no EU5 equivalent
	}

	after = {
		if = {
			limit = {
				var:drought_spread >= 2 # Imperator OR = { var:drought_spread = 5 / 4 / 3 / 2 }
				var:drought_spread <= 5
			}
			change_variable = {
				name = drought_spread
				add = -1
			}
		}
		scope:i_am_thirsty = {
			remove_province_modifier = grow_a_pear
			add_province_modifier = {
				modifier = grow_a_pear
				days = 3650 #10 yrs
			}
			clear_saved_scope = i_am_thirsty
			set_variable = {
				name = thirsty_bois
				days = 500
			}
			area = {
				random_neighbor_area = {
					random_province_in_area = { # Imperator random_area_state
						limit = {
							has_province_modifier = grow_a_pear
							NOT = {
								has_variable = thirsty_bois
							}
						}
						save_scope_as = i_am_thirsty
					}
				}
			}
		}
	}
}
