	}

	option = {
		name = ritual_sacrifice_inv.1.a
		if = {
			limit = { exists = scope:officiating_priest }

				target = scope:ritual_sacrifice
				killer = scope:officiating_priest

			kill_character = {

				reason = sacrifice
			}

		#P:A TODO No EU5 equivalent: the priest gained +1 zeal and a 15% chance of the "lunatic" trait (EU5 characters have no zeal and no lunatic trait)
		ruler = {
			if = {
				limit = {
					scope:ritual_sacrifice = {
						NOT = { religion = root.ruler.religion }
					}
				}
				random_list = {
					1 = {
						add_trait = trait:zealot # Imperator zealous
					}
					1 = {
						add_trait = trait:pious
					}
					1 = {
						add_trait = trait:pious # Imperator devout
					}
				}
			}
		}
		hidden_effect = {
			random_list = {
				5 = { # Tribe likes the sacrifice (as in the event)
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { mil > 35 } # Imperator martial > 7
					}
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { adm > 35 } # Imperator finesse > 7
					}
					# modifier = { factor = 2 trigger = { scope:ritual_sacrifice.zeal > 7 } }
					#P:A TODO No EU5 equivalent: x2 if the sacrifice had zeal above 7 (EU5 characters have no zeal)
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { dip > 35 } # Imperator charisma > 7
					}
					modifier = { # Heretic
						factor = 4
						scope:ritual_sacrifice = {
							NOT = { religion = root.ruler.religion }
						}
					}
					modifier = { # Cultural Heretic
						factor = 3
						scope:ritual_sacrifice = {
							NOT = { culture = root.ruler.culture }
						}
					}
					# modifier = { factor = 10 trigger = { current_ruler = { popularity > 50 } scope:ritual_sacrifice = { is_rival = employer.current_ruler } } }
					#P:A TODO No EU5 equivalent: x10 if the ruler had popularity above 50 and the sacrifice was the ruler's rival (EU5 has no popularity or character rivals)
					trigger_event_non_silently = {
						id = ritual_sacrifice_inv.2
						days = { 10 30 }
					}
				}
				5 = { # Tribe dislikes the sacrifice (as in the event)
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { mil < 20 } # Imperator martial < 4
					}
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { adm < 20 } # Imperator finesse < 4
					}
					# modifier = { factor = 2 trigger = { scope:ritual_sacrifice.zeal < 4 } }
					#P:A TODO No EU5 equivalent: x2 if the sacrifice had zeal below 4 (EU5 characters have no zeal)
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { dip < 20 } # Imperator charisma < 4
					}
					# modifier = { factor = 10 trigger = { current_ruler = { popularity > 50 } scope:ritual_sacrifice = { is_friend = employer.current_ruler } } }
					#P:A TODO No EU5 equivalent: x10 if the ruler had popularity above 50 and the sacrifice was the ruler's friend (EU5 has no popularity or character friends)
					modifier = {
						factor = 10
						scope:ritual_sacrifice = { age_in_years < 6 }
					}
					modifier = {
						factor = 2
						scope:ritual_sacrifice = { is_female = yes }
					}
					trigger_event_non_silently = {
						id = ritual_sacrifice_inv.3
						days = { 10 30 }
					}
				}
			}
		}
	}
}
