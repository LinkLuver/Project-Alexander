# Pretender questions the legitimacy of the monarch
monarchy_events.1 = {
	type = country_event
	title = monarchy_events.1.t
	desc = monarchy_events.1.desc
	outcome = negative
	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		government_type = government_type:monarchy
		exists = ruler
		any_character = {
			is_eligible_heir = root #P:A TODO Approximate: Imperator is_pretender (a claimant to the throne); EU5 is_eligible_heir who is not the ruler
			is_ruler = no
			is_adult = yes
			exists = father
			NOT = {
				is_child_of = root.ruler
			}
			father = {
				exists = father
				NOT = {
					is_child_of = root.ruler
				}
			}
			# loyalty < 50 #P:A TODO No EU5 equivalent: only claimants with loyalty below 50 deny the ruler's legitimacy (EU5 characters have no loyalty)
		}
		legitimacy > 10
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		random_character = {
			limit = {
				is_eligible_heir = root #P:A TODO Approximate: Imperator is_pretender; EU5 is_eligible_heir who is not the ruler
				is_ruler = no
				is_adult = yes
				exists = father
				# loyalty < 50
				NOT = {
					is_child_of = root.ruler
				}
				father = {
					exists = father
					NOT = {
						is_child_of = root.ruler
					}
				}
			}
			save_scope_as = legitimacy_denier
		}
	}

	option = { # ignore it
		name = monarchy_events.1.a
		add_legitimacy = -20 # subtract_legitimacy_large
	}

	option = { # publicly repudiate the claims
		name = monarchy_events.1.b
		pa_imp_remove_stability_effect = yes
		# scope:legitimacy_denier = { add_loyalty = loyalty_lies_revealed }
		#P:A TODO No EU5 equivalent: the pretender gets the loyalty_lies_revealed loyalty modifier (EU5 characters have no loyalty)
	}
}

# A philosopher offers to tutor the monarch (philosopher king chain)
monarchy_events.2 = {
	type = country_event
	title = monarchy_events.2.t
	desc = monarchy_events.2.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		culture = {
			has_culture_group = culture_group:greek_group # Imperator country_culture_group = hellenic
		}
		government_type = government_type:monarchy
		exists = ruler
		ruler = {
			NOT = {
				has_variable = philosopher_king_student
			}
		}
	}

	immediate = {
		hidden_effect = {
			create_character = {
				culture = root.culture # Imperator default: the country's culture and religion
				religion = root.religion
				# add_trait = wise #P:A TODO No EU5 equivalent: Imperator trait "wise"
				# add_trait = polymath #P:A TODO No EU5 equivalent: Imperator trait "polymath"
				add_trait = trait:scholar
				age = 40
				save_scope_as = philosopher_king_educator
			}
		}
		random_country = {
			limit = {
				culture = {
					has_culture_group = culture_group:greek_group # Imperator country_culture_group = hellenic
				}
				NOT = { this = root }
			}
			save_scope_as = philosopher_move_country
		}
		ruler = {
			save_scope_as = philosopher_king_student
			set_variable = philosopher_king_student
		}
	}

	option = { # Hire as your teacher
		name = monarchy_events.2.a
		add_legitimacy = 2 # legitimacy_small
		# scope:philosopher_king_educator = { give_office = office_philosopher }
		#P:A TODO No EU5 equivalent: the philosopher is given the court office "philosopher" (EU5 has no offices); monarchy_events.3/.4/.5 check that the philosopher still serves the country instead
		hidden_effect = {
			if = {
				limit = {
					ruler = {
						OR = {
							# has_trait = wise #P:A TODO No EU5 equivalent: Imperator trait "wise"
							# has_trait = polymath #P:A TODO No EU5 equivalent: Imperator trait "polymath"
							has_trait = scholar
							adm > 35 # Imperator finesse > 7
						}
					}
				}
				trigger_event_non_silently = {
					id = monarchy_events.3
					days = { 90 180 }
				}
			}
			else_if = {
				limit = {
					ruler = {
						OR = {
							# has_trait = dumb #P:A TODO No EU5 equivalent: Imperator trait "dumb"
							has_trait = arrogant
							# has_trait = narrowminded #P:A TODO No EU5 equivalent: Imperator trait "narrowminded"
							adm < 20 # Imperator finesse < 4
						}
					}
				}
				trigger_event_non_silently = {
					id = monarchy_events.5
					days = { 90 180 }
				}
			}
			else = {
				trigger_event_non_silently = {
					id = monarchy_events.4
					days = { 90 180 }
				}
			}
		}
	}

	option = { # Send the philosopher away
		name = monarchy_events.2.b
		# ruler = { add_popularity = 5 } # popularity_small
		#P:A TODO No EU5 equivalent: +5 popularity for the ruler (EU5 characters have no popularity)
		if = {
			limit = {
				exists = scope:philosopher_move_country
			}
			scope:philosopher_king_educator = {
				move_country = scope:philosopher_move_country
			}
		}
	}
}

# The philosopher king: the monarch learns a virtue
monarchy_events.3 = {
	type = country_event
	title = monarchy_events.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					var:philosopher_king_virtue = 1
				}
				desc = monarchy_events.3.desc.a
			}
			triggered_desc = {
				trigger = {
					var:philosopher_king_virtue = 2
				}
				desc = monarchy_events.3.desc.b
			}
			triggered_desc = {
				trigger = {
					var:philosopher_king_virtue = 3
				}
				desc = monarchy_events.3.desc.c
			}
		}
	}
	outcome = positive
	illustration_tags = {
		10 = interior
		10 = happy
	}

	trigger = {
		exists = ruler
		exists = scope:philosopher_king_student
		scope:philosopher_king_student = {
			pa_imp_char_valid_for_country_events_trigger = yes
			NOT = {
				has_variable = had_philosopher_king_outcome
			}
		}
		exists = scope:philosopher_king_educator
		scope:philosopher_king_educator = {
			# has_office = office_philosopher
			is_alive = yes #P:A TODO Approximate: Imperator required the philosopher to hold the court office "philosopher" (EU5 has no offices); here the philosopher must still serve the country
			employer = root
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		hidden_effect = {
			random_list = {
				33 = {
					set_variable = {
						name = philosopher_king_virtue
						value = 1
					}
				}
				33 = {
					set_variable = {
						name = philosopher_king_virtue
						value = 2
					}
				}
				33 = {
					set_variable = {
						name = philosopher_king_virtue
						value = 3
					}
				}
			}
			set_variable = { # set on the country, as in Invictus (the trigger checks the student)
				name = had_philosopher_king_outcome
				value = 1
			}
		}
	}

	option = {
		name = monarchy_events.3.a
		ruler = {
			add_adm = 5 # Imperator add_finesse = 1
		}
		# Monarch cultivates justice
		if = {
			limit = {
				var:philosopher_king_virtue = 1
				NOT = {
					ruler = {
						OR = {
							has_trait = just
							# has_trait = arbitrary #P:A TODO No EU5 equivalent: Imperator trait "arbitrary"
						}
					}
				}
			}
			ruler = {
				add_trait = trait:just
			}
		}
		else_if = {
			limit = {
				var:philosopher_king_virtue = 1
				ruler = {
					has_trait = just
				}
			}
			# add_political_influence = 25 # influence_large
			#P:A TODO No EU5 equivalent: +25 political influence (EU5 has no political influence)
		}
		# Monarch cultivates bravery
		else_if = {
			limit = {
				var:philosopher_king_virtue = 2
				NOT = {
					ruler = {
						OR = {
							# has_trait = brave #P:A TODO No EU5 equivalent: Imperator trait "brave"
							has_trait = craven # Imperator coward
						}
					}
				}
			}
			# ruler = { add_trait = brave }
			#P:A TODO No EU5 equivalent: the ruler gains the Imperator trait "brave"
		}
		else_if = {
			limit = {
				var:philosopher_king_virtue = 2
				# ruler = { has_trait = brave }
				always = no #P:A TODO No EU5 equivalent: Imperator trait "brave" (a ruler who already was brave gained stability instead)
			}
			pa_imp_add_stability_effect = yes
		}
		# Monarch becomes skilled at political discourse
		else_if = {
			limit = {
				var:philosopher_king_virtue = 3
				NOT = {
					ruler = {
						OR = {
							has_trait = silver_tongue
							# has_trait = blunt #P:A TODO No EU5 equivalent: Imperator trait "blunt"
						}
					}
				}
			}
			ruler = {
				add_trait = trait:silver_tongue
			}
		}
		else_if = {
			limit = {
				var:philosopher_king_virtue = 3
				ruler = {
					has_trait = silver_tongue
				}
			}
			# add_political_influence = 25 # influence_large
			#P:A TODO No EU5 equivalent: +25 political influence (EU5 has no political influence)
		}
		# scope:philosopher_king_educator = { add_loyalty = loyalty_teaching_praised_1 }
		#P:A TODO No EU5 equivalent: the philosopher gets the loyalty_teaching_praised_1 loyalty modifier (EU5 characters have no loyalty)
	}
}

# Citizens want to move to a neighbouring allied country
monarchy_events.41 = {
	type = country_event
	title = monarchy_events.41.t
	desc = monarchy_events.41.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		government_type = government_type:monarchy
		NOT = { has_variable = monarchy_events_41_cooldown }
		any_owned_location = {
			any_neighbor_location = {
				exists = owner
				owner = {
					is_allied_with = { target = root } # Imperator alliance_with
				}
			}
			any_pop = {
				pop_type = pop_type:burghers # Imperator citizen
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		random_owned_location = { # Find owned province for citizens to travel from
			limit = {
				any_neighbor_location = {
					exists = owner
					owner = {
						is_allied_with = { target = root }
					}
				}
				any_pop = {
					pop_type = pop_type:burghers
				}
			}
			save_scope_as = randomized_country_province
			random_neighbor_location = { # Find neighbor province for citizens to travel to
				limit = {
					exists = owner
					owner = {
						is_allied_with = { target = root }
					}
				}
				save_scope_as = randomized_neighbor_province
				owner = {
					save_scope_as = allied_neighbor
				}
			}
		}
		root = {
			save_scope_as = original_country
		}
		hidden_effect = {
			set_variable = {
				name = monarchy_events_41_cooldown
				days = 3650
			}
		}
	}

	option = { # Let citizens go to allied country
		name = monarchy_events.41.a
		scope:randomized_country_province = {
			random_pop = { #P:A TODO Approximate: Imperator move_pop moved one whole citizen pop to the allied location; EU5 splits off size 1 (one Imperator pop converted 1:1 to EU5 size)
				limit = {
					pop_type = pop_type:burghers # Imperator citizen
				}
				weight = {
					base = 1
					modifier = {
						factor = 3
						religion = root.religion
					}
					modifier = {
						factor = 3
						culture = root.culture
					}
				}
				split_pop = {
					size = 1
					location = scope:randomized_neighbor_province
				}
			}
		}
		scope:allied_neighbor = {
			add_opinion = {
				modifier = opinion_close_allies
				target = root
			}
		}
		hidden_effect = {
			scope:allied_neighbor = {
				trigger_event_non_silently = {
					id = monarchy_events.42
					days = { 5 10 }
				}
			}
		}
	}

	option = { # Make sure the citizens stick around in ROOT country by military force
		name = monarchy_events.41.b
		# add_tyranny = 1 # tyranny_small
		#P:A TODO No EU5 equivalent: +1 tyranny (EU5 has no tyranny)
	}

	option = { # Build up province they travel from, to make them refrain from traveling
		name = monarchy_events.41.c
		add_gold = {
			value = 0
			subtract = root.monthly_income_total
			multiply = 4
			min = -6000
		}
		scope:randomized_country_province = {
			add_location_modifier = {
				modifier = province_improvements_monarchy
				days = 3650
			}
		}
	}
}
