#P:A Imported from Invictus: events/storms.txt
#   (storm.32, storm.41-44 and storm.51 are Indomita-only additions, marked per event; Indomita storm.40 and storm.50 were not imported:
#   they are its renumbered copies of the Caspian/Aral storm.100 and the Eastern Europe storm.300, which Invictus already has)
# Every storm is a random sea/desert/snow location that gets a 60-day modifier and VFX, spreading to its neighbours. These are not
# EU5 weather systems (start_weather_system), so EU5 on_storm_reached_location never sees them.
# Imperator firing (vanilla common/on_action/00_yearly_country.txt, yearly_country_pulse effect): once a year, unless the global
# variable stormbringer is set, one random country with 25+ cities and not at war gets storm.1 (10-60 days), storm.11 (10-60 days),
# storm.21 (now) and storm.25 (2 days, clears the flag); then stormbringer is set. The storm events never use their root.
namespace = storm

# Hidden: a sea storm in the Atlantic.
storm.2 = {
	type = country_event
	hidden = yes

	immediate = {
		#Atlantic
		random_location_in_the_world = { # Imperator random_sea_and_river_zone
			limit = {
				pa_imp_b14_open_sea_trigger = yes # Imperator terrain = ocean
				pa_imp_in_region_atlantic_region = yes
				NOT = { has_location_modifier = stormy_seas }
			}
			add_vfx = {
				name = "seastorm"
				duration = 60
			}
			add_location_modifier = {
				modifier = stormy_seas
				days = 60
			}
			every_neighbor_location = {
				limit = {
					pa_imp_b14_open_sea_trigger = yes
					pa_imp_in_region_atlantic_region = yes
					NOT = { has_location_modifier = stormy_seas }
				}
				add_vfx = {
					name = "seastorm"
					duration = 60
				}
				add_location_modifier = {
					modifier = stormy_seas
					days = 60
				}
			}
		}
	}
}
