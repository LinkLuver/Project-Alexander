# Hidden: an AI tribe reforms into a despotic monarchy.
nation_forming.3 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = capital # Imperator has_land = yes
		is_ai = yes
	}

	immediate = {
		change_government_type = government_type:monarchy #P:A TODO Approximate: Imperator despotic_monarchy (EU5 has no government sub-forms)
		pa_imp_switch_government_type_event_clearup_effect = yes
		add_country_modifier = {
			modifier = civilized_tribe
			days = 3650
		}
		capital = {
			if = {
				limit = {
					NOT = {
						has_location_modifier = civilized_tribe_ppmod
					}
				}
				add_location_modifier = {
					modifier = civilized_tribe_ppmod
					days = -1
				}
			}
			else = {
				root = {
					add_stability = 5
				}
			}
		}
		pa_imp_add_2_free_province_investments = yes
	}
}
