trade_goods_flavour.4 = { #find cypriot dwarf elephants and dwarf hippo bones found
	type = country_event #I originally thought, that the elephants only died out in the BA. turns out it was around 12000 BCE; they'd get ivory
	title = trade_goods_flavour.4.t
	desc = trade_goods_flavour.4.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		# NOT = {
		# 	exists = treasure:cypriot_dwarf_elephant_bones #not that they'd recognise it...
		# }
		#P:A TODO No EU5 equivalent: "the Cypriot dwarf elephant bones treasure does not exist yet" (EU5 has no treasures); the global variable check below still blocks repeats
		has_global_variable = collecting_dwarf_bones #P:A TODO Invictus source bug: this variable is only set by option a of this same event, so the event can never fire; NOT = { has_global_variable = collecting_dwarf_bones } was probably meant (kept as in the source)
		any_owned_location = {
			pa_imp_in_area_cyprus_area = yes
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
	}

	option = { # collect the bones
		name = trade_goods_flavour.4.a
		set_global_variable = collecting_dwarf_bones
		trigger_event_non_silently = {
			id = trade_goods_flavour.5
			days = { 120 160 }
		}
		add_gold = -200
	}

	option = { #sell them
		name = trade_goods_flavour.4.b
		add_gold = monthly_income_total # Imperator has_monthly_income
	}

	option = { #ignore them
		name = trade_goods_flavour.4.c
	}
}

trade_goods_flavour.13 = { # Silphium - Silphium Farming
	type = country_event
	title = trade_goods_flavour.13.t
	desc = trade_goods_flavour.13.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		any_owned_location = {
			raw_material = goods:medicaments #P:A TODO Approximate: Imperator trade good silphium -> EU5 medicaments (EU5 has no silphium)
		}
		NOT = {
			has_variable = silphium_flavour_cooldown
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = silphium_flavour_cooldown
			days = 7300 # 20 years
		}
		random_owned_location = {
			limit = {
				raw_material = goods:medicaments #P:A TODO Approximate: Imperator trade good silphium -> EU5 medicaments
				NOT = {
					has_location_modifier = silphium_farm_pmod
				}
			}
			save_scope_as = silphium_farm
		}
	}

	option = { #Farm Silphium More
		name = trade_goods_flavour.13.a
		scope:silphium_farm = {
			add_location_modifier = {
				modifier = silphium_farm_pmod
				days = -1
			}
		}
	}

	option = { #Try to have it spread
		name = trade_goods_flavour.13.b
		trigger = {
			OR = {
				scope:silphium_farm = {
					any_neighbor_location = {
						owner = root
						NOT = {
							raw_material = goods:medicaments #P:A TODO Approximate: Imperator trade good silphium -> EU5 medicaments
						}
						OR = {
							topography = flatland # Imperator terrain plains
							topography = hills
						}
					}
				}
				any_owned_location = { #original silphium province disappeared
					pa_imp_is_original_silphium_province = yes
					NOT = {
						raw_material = goods:medicaments #P:A TODO Approximate: Imperator trade good silphium -> EU5 medicaments
					}
				}
			}
			scope:silphium_farm = {
				region = {
					any_location_in_region = {
						count <= 10 #at most 10 at any give time
						raw_material = goods:medicaments #P:A TODO Approximate: Imperator trade good silphium -> EU5 medicaments; EU5 regions are not Imperator regions
					}
				}
			}
		}
		if = {
			limit = {
				any_owned_location = { #original silphium province disappeared
					pa_imp_is_original_silphium_province = yes
					NOT = {
						raw_material = goods:medicaments
					}
				}
			}
			random = {
				chance = 75 #higher chance of Silphium reappearing
				random_owned_location = {
					limit = {
						pa_imp_is_original_silphium_province = yes
						NOT = {
							raw_material = goods:medicaments
						}
					}
					change_raw_material = goods:medicaments #P:A TODO Approximate: Imperator set_trade_goods = silphium -> EU5 medicaments
				}
			}
		}
		else = {
			random = {
				chance = 30
				scope:silphium_farm = {
					random_neighbor_location = {
						limit = {
							NOT = {
								raw_material = goods:medicaments
							}
							OR = {
								topography = flatland # Imperator terrain plains
								topography = hills
							}
						}
						change_raw_material = goods:medicaments #P:A TODO Approximate: Imperator set_trade_goods = silphium -> EU5 medicaments
					}
				}
			}
		}
	}

	option = { #We should not overexploit it
		name = trade_goods_flavour.13.c
		scope:silphium_farm = {
			add_location_modifier = {
				modifier = silphium_farm_pmod
				days = 3650 # 10 years
			}
		}
	}
}

	}

	immediate = {
		ruler ?= {

		}
		set_variable = {
			name = cedar_flavour_cooldown
			days = 7300 # 20 years
		}
	}

	option = {
		name = trade_goods_flavour.17.a
		trigger = {
			any_owned_location = {
				raw_material = goods:lumber #P:A TODO Approximate: Imperator trade good cedar -> EU5 lumber
				NOR = {
					# too lazy to create a script that would remove the provinces without the first modifier or that would reapply it in case of regrowth

					has_location_modifier = bred_cedars_future
				}
				NOR = {
					is_capital = yes
					location_rank = location_rank:city #P:A TODO Approximate: Imperator province rank city_metropolis -> EU5 city
				}
			}
		}
		add_country_modifier = {
			modifier = cedar_used_up
			days = 5475 # 15 years
		}
		random_owned_location = {
			limit = {
				raw_material = goods:lumber #P:A TODO Approximate: Imperator trade good cedar -> EU5 lumber
				NOR = {
					has_location_modifier = cedars_of_lebanon
					has_location_modifier = bred_cedars_future
				}
			}
			# set_trade_goods = wood
			#P:A TODO No EU5 equivalent: turned the cedar territory into ordinary wood; cedar and wood both map to EU5 lumber, so there is nothing to switch (the cost of this option is lost)
			set_variable = cedar_gone_variable #next event can restore forests; no growth of total amount of cedars
		}

		ai_chance = {
			factor = 0
		}
	}

	option = { #exploit just a little bit
		name = trade_goods_flavour.17.b
		add_country_modifier = {
			modifier = cedar_used_up
			days = 1825 # 5 years
		}

			limit = {

				days = 9125 # 25 years
			}
		}
	}

	option = { #do not touch more than we'd usually do
		name = trade_goods_flavour.17.c
	}
}
