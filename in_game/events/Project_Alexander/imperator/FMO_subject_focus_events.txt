# Commerce focus for all subjects.
subject_focus.1 = {
	type = country_event
	title = subject_focus_1_t
	desc = subject_focus_1_desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
	}

	option = {
		name = subject_focus_1.a #Economic Focus
		# if = { limit = { any_country = { count <= 6 is_subject_of = ROOT } } add_political_influence = -75 }
		# else_if = { limit = { any_country = { count > 6 is_subject_of = ROOT } } add_political_influence = -115 }
		#P:A TODO No EU5 equivalent: costs 75 political influence (115 with more than 6 subjects)
		custom_tooltip = subject_focus_6_ttt
		show_as_tooltip = {
			random_subject = {
				add_country_modifier = {
					modifier = commerce_focused_subject
					days = -1
				}
			}
		}
		every_subject = {
			limit = {
				NOT = {
					has_country_modifier = commerce_focused_subject
				}
			}
			hidden_effect = {
				if = {
					limit = {
						has_country_modifier = commerce_focused_subject
					}
					remove_country_modifier = commerce_focused_subject
				}
				else_if = {
					limit = {
						has_country_modifier = military_focused_subject
					}
					remove_country_modifier = military_focused_subject
				}
				else_if = {
					limit = {
						has_country_modifier = development_focused_subject
					}
					remove_country_modifier = development_focused_subject
				}
				add_country_modifier = {
					modifier = commerce_focused_subject
					days = -1
				}
			}
		}
	}

	option = {
		name = subject_focus_1.b
		if = {
			limit = { has_variable = recently_changed_focus }
			remove_variable = recently_changed_focus
		}
	}
}

# The events below are fired for one subject, saved as scope:recipient by the Indomita subject interaction buttons.

subject_focus_individual.1 = {  #Send Subject some gifts
	type = country_event
	title = subject_focus_individual.1.t
	desc = subject_focus_individual.1.desc
	illustration_tags = {
		10 = interior
		10 = happy
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		scope:recipient.ruler ?= {
			save_scope_as = recipient_ruler
		}
	}

	option = {
		name = subject_focus_individual.1.a
		#Shower them with gifts
		add_gold = -250
		# add_political_influence = -25 #P:A TODO No EU5 equivalent: costs 25 political influence
		scope:recipient = {
			add_gold = 250
			add_country_modifier = {
				modifier = loyalty_to_overlord_1
				days = 3650
			}
		}
	}

	option = {
		name = subject_focus_individual.1.b
		#A small sum will do
		add_gold = -100
		# add_political_influence = -25 #P:A TODO No EU5 equivalent: costs 25 political influence
		scope:recipient = {
			add_gold = 100
			add_country_modifier = {
				modifier = loyalty_to_overlord_2
				days = 3650
			}
		}
	}

	option = {
		name = subject_focus_individual.1.c
		remove_variable = recently_sent_subject_gifts
	}

	after = {
		#Change value so it doesn't refire on every click
		if = {
			limit = {
				has_variable = recently_sent_subject_gifts
			}
			change_variable = {
				name = recently_sent_subject_gifts
				add = 1
			}
		}
	}
}

								value = scope:pa_moved_pop.pop_size
								max = 1
							}
						}
					}
					add_pop_size = {
