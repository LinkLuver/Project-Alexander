# EU5 has no focus trees (same conventions as CFT_FMO_lifestyle_event_chain.txt, batch b08):
# - the ruler's focus (has_triggered_character_modifier = <x>_focus_triggered_mod) is approximated by the ruler's best
#   EU5 skill (oratory -> DIP, martial -> MIL, civic -> ADM; zeal -> a pious or zealous ruler);
# - focus experience (character_experience, add_focus_experience) has no EU5 equivalent; the calls go to the empty
#   stub pa_imp_add_focus_experience (b08) so a later port can fill it in.
# - events whose only effect has no EU5 equivalent are held false with always = no.

# Priestly support focus: the ruler's priests convert a province
lifestyle_events.16 = {
	type = country_event
	title = lifestyle_events.16.t
	desc = lifestyle_events.16.desc
	outcome = positive
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		exists = capital # Imperator has_land = yes
		ruler ?= {
			# has_triggered_character_modifier = priestly_support_focus_triggered_mod
			pa_imp_char_is_religious_trigger = yes #P:A TODO Approximate: Indomita priestly support focus (zeal tree); EU5 has no focus trees or zeal skill, a pious or zealous ruler stands in
			# character_experience > 80 #P:A TODO No EU5 equivalent: more than 80 focus-tree experience (Indomita character focus system)
		}
		any_owned_location = {
			NOT = { dominant_religion = root.religion }
		}
	}

	immediate = {
		ruler = { save_scope_as = pa_ruler }
		random_owned_location = {
			limit = {
				NOT = { dominant_religion = root.religion }
			}
			save_scope_as = conversion_province
		}
	}

	option = {
		name = lifestyle_events.16.a
		trigger = {
			scope:conversion_province = {
				population > 20 #P:A TODO Approximate: Imperator more than 20 pops; EU5 population converted 1:1 (one pop = 1)
			}
		}
		ruler = {
			pa_imp_add_focus_experience = { text = add_10_focus_xp amount = 10 }
		}
		scope:conversion_province = {
			while = {
				count = 7
				random_pop = { #P:A TODO Approximate: Imperator converted 7 pops; EU5 pops are larger groups, so up to 7 whole pops convert
					limit = {
						NOT = { religion = root.religion }
					}
					change_pop_religion = root.religion
				}
			}
		}
	}
	option = {
		name = lifestyle_events.16.b
		trigger = {
			scope:conversion_province = {
				population <= 20
			}
		}
		ruler = {
			pa_imp_add_focus_experience = { text = add_10_focus_xp amount = 10 }
		}
		scope:conversion_province = {
			while = {
				count = 3
				random_pop = { #P:A TODO Approximate: Imperator converted 3 pops; EU5 pops are larger groups, so up to 3 whole pops convert
					limit = {
						NOT = { religion = root.religion }
					}
					change_pop_religion = root.religion
				}
			}
		}
	}
}
