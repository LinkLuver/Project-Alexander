#P:A Imported from Indomita: events/mandate_events.txt
# Mandate of Heaven events for the Chinese Empire. In Indomita all ten were in the biyearly_country_pulse.
# Indomita gated them with government = chinese_empire; EU5 has no such government form, so they require a monarchy with
# the mandate variable that chinese_flavor.2 (batch b10) sets when the empire is formed.
namespace = mandate_events

# Yellow River Dikes.
mandate_events.1 = {
	type = country_event
	title = mandate_events.1.t
	desc = mandate_events.1.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		exists = capital
		government_type = government_type:monarchy #P:A TODO Approximate: Imperator government = chinese_empire; EU5 monarchy that holds the Mandate of Heaven variable
		has_variable = mandate_variable
		any_owned_location = {
			OR = {
				pa_imp_in_region_ji_region = yes
				pa_imp_in_region_yu_region = yes
			}
		}
	}

	immediate = {
		ruler ?= { save_scope_as = mandate_ruler }
		random_owned_location = {
			limit = {
				OR = {
					pa_imp_in_region_ji_region = yes
					pa_imp_in_region_yu_region = yes
				}
			}
			save_scope_as = target_province
		}
	}

	option = {
		name = mandate_events.1.a
		pa_imp_change_mandate_effect = { amount = 5 }
		custom_tooltip = add_5_mandate
		add_gold = {
			value = 0
			subtract = monthly_income_total
			multiply = 6
		}
		scope:target_province = {
			add_pop = { #P:A TODO Approximate: Imperator create_state_pop = freemen (one pop); EU5 adds a peasant pop of size 1 with the location's dominant culture and religion
				type = pop_type:peasants
				size = 1
				culture = dominant_culture
				religion = dominant_religion
			}
		}
	}

	option = {
		name = mandate_events.1.b
		pa_imp_change_mandate_effect = { amount = -5 }
		custom_tooltip = subtract_5_mandate
	}
}

# Trade Regulations: local officials stimulate commerce against official policy.
mandate_events.4 = {
	type = country_event
	title = mandate_events.4.t
	desc = mandate_events.4.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		exists = capital
		government_type = government_type:monarchy #P:A TODO Approximate: Imperator government = chinese_empire; EU5 monarchy that holds the Mandate of Heaven variable
		has_variable = mandate_variable
		any_owned_location = {
			has_building = building_type:marketplace # Imperator commerce_building
		}
	}

	immediate = {
		ruler ?= { save_scope_as = mandate_ruler }
		random_owned_location = {
			limit = {
				has_building = building_type:marketplace
			}
			save_scope_as = target_province
		}
	}

	option = {
		name = mandate_events.4.a
		pa_imp_change_mandate_effect = { amount = -5 }
		custom_tooltip = subtract_5_mandate
		scope:target_province = {
			add_location_modifier = {
				modifier = mandate_trade_regulations_mod
				days = 3650
			}
		}
	}

	option = {
		name = mandate_events.4.b
		pa_imp_change_mandate_effect = { amount = 5 }
		custom_tooltip = add_5_mandate
		scope:target_province = {
			add_location_modifier = {
				modifier = mandate_trade_regulations_neg_mod
				days = 3650
			}
		}
	}
}

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		exists = capital
		government_type = government_type:monarchy #P:A TODO Approximate: Imperator government = chinese_empire; EU5 monarchy that holds the Mandate of Heaven variable
		has_variable = mandate_variable

			any_buildings_in_location = { #P:A TODO Approximate: Imperator has_building = military_building; EU5 any building of the military category
				building_category = military_category
			}
		}
	}

	immediate = {
		ruler ?= { save_scope_as = mandate_ruler }
		random_owned_location = {
			limit = {
				any_buildings_in_location = {

				}
			}
			save_scope_as = target_province
		}
	}

	option = {
		name = mandate_events.6.a
		add_manpower = 4
	}

	option = {
		name = mandate_events.6.b
		scope:target_province = {
			add_pop = { #P:A TODO Approximate: Imperator create_state_pop = freemen twice (two pops); EU5 adds a peasant pop of size 2 with the location's dominant culture and religion
				type = pop_type:peasants
				size = 2
				culture = dominant_culture
				religion = dominant_religion
			}
		}
	}
}
