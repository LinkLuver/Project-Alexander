#P:A Imported from Invictus: events/after_battle.txt
# Imperator character events (root = the army or fleet commander), converted to country events.
# The hidden events after_battle.1-.10, .21-.30 and .41 were fired by Imperator on_actions with the commander as root.
# In EU5 they are written for battle on_actions (root = the country of the main unit, scope:actor = that unit,
# scope:target = the main enemy unit); the commander scope:actor.leader is saved as scope:pa_character:
#   winners: Invictus custom_on_great_battle_commander (fired from on_great_battle_won_country) -> EU5 on_great_battle_won
#   losers:  Imperator on_battle_lost (root = the losing commander)                            -> EU5 on_battle_lost
# The visible events after_battle.11-.20, .31-.40 and .51 are fired by the hidden ones.
namespace = after_battle

#Tactical Supremacy
# Hidden: after a great victory, the commander may become "victorious". For EU5 on_great_battle_won.
after_battle.1 = {
	type = country_event
	hidden = yes

	trigger = {
		always = no #P:A TODO No EU5 equivalent: only relays to after_battle.11, which exists to give the commander the Imperator trait victorious (EU5 has no such trait)
		exists = scope:actor.leader
		scope:actor.leader = {
			NOR = {
				# has_trait = victorious #P:A TODO No EU5 equivalent: Imperator trait victorious
				has_trait = conqueror
				# has_trait = comradely #P:A TODO No EU5 equivalent: Imperator trait comradely
			}
			is_adult = yes
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			exists = scope:actor.leader
			add = {
				value = scope:actor.leader.mil
				divide = 20 # Imperator martial / 4 (EU5 skills are x5)
			}
		}
	}

	immediate = {
		scope:actor.leader = {
			save_scope_as = pa_character
			save_scope_as = after_battle_1_target
		}
		trigger_event_non_silently = after_battle.11 # Imperator: employer = { trigger_event = ... } (root is the employer)
	}
}

#Leading from the Front
# Hidden: a brave or cowardly commander (not the ruler) in a lost battle. For EU5 on_battle_lost.
after_battle.4 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = scope:actor.leader
		scope:actor.leader = {
			is_ruler = no
			OR = {
				has_trait = craven # Imperator coward
				# has_trait = brave #P:A TODO No EU5 equivalent: Imperator trait brave
			}
		}
	}

	immediate = {
		scope:actor.leader = {
			save_scope_as = pa_character
			save_scope_as = after_battle_4_target
		}
		trigger_event_non_silently = after_battle.14 # Imperator: employer = { trigger_event = ... } (root is the employer)
	}
}

after_battle.14 = {
	type = country_event
	title = after_battle.4.t
	desc = after_battle.4.d
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		exists = scope:after_battle_4_target
		scope:after_battle_4_target = {
			is_alive = yes
			# NOT = { has_loyalty = loyalty_fled_battle } #P:A TODO No EU5 equivalent: Imperator character loyalty modifier "fled battle" (EU5 characters have no loyalty)
			is_adult = yes
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
	}

	option = {
		name = after_battle.4.a
		scope:after_battle_4_target = {
			random_list = { # Invictus weights: only the third outcome can happen
				0 = {
					# if = { limit = { has_trait = brave } add_popularity = 15 }
					# else = { if = { limit = { has_trait = coward } remove_trait = coward } add_trait = brave }
					#P:A TODO No EU5 equivalent: +15 popularity if brave, else coward -> brave (EU5 has no popularity and no brave trait); weight 0 in Invictus
					if = {
						limit = {
							has_trait = craven
						}
						remove_trait = trait:craven
					}
				}
				0 = {
					# add_popularity = 10 #P:A TODO No EU5 equivalent: +10 character popularity; weight 0 in Invictus
				}
				2 = {
					# if = { limit = { has_legion_trigger = yes } hidden_effect = { legion = { add_legion_history = { key = commander_death_history commander = scope:after_battle_4_target province = scope:after_battle_4_target.location } } } }
					#P:A TODO No EU5 equivalent: records the commander's death in the legion history (EU5 has no legions)
					kill_character = {
						target = scope:after_battle_4_target
						reason = battle
					}
				}
			}
		}
	}

	option = {
		name = after_battle.4.b
		scope:after_battle_4_target = {
			if = {
				limit = {
					has_trait = craven # Imperator coward
				}
				# add_loyalty = loyalty_fled_battle
				# add_popularity = -10
				#P:A TODO No EU5 equivalent: the coward gets the "fled battle" loyalty modifier and -10 popularity (EU5 characters have neither)
			}
			else = {
				# if = { limit = { has_trait = brave } remove_trait = brave }
				#P:A TODO No EU5 equivalent: removes the Imperator trait brave first
				add_trait = trait:craven # Imperator coward
			}
		}
	}
}

			}
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			exists = scope:actor.leader
			add = {
				value = scope:actor.leader.mil
				divide = 20 # Imperator martial / 4 (EU5 skills are x5)
			}
		}
	}

	immediate = {
		scope:actor.leader = {
			save_scope_as = pa_character

		}
		trigger_event_non_silently = after_battle.15 # Imperator: employer = { trigger_event = ... } (root is the employer)
	}
}
