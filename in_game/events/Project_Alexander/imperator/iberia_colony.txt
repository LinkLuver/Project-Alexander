# Hidden: an Iberian tribe owning Olisipo (Lisbon) is offered a Punic purchase of the harbour.
iberia_colony.4 = { #hidden event
	type = country_event
	hidden = yes

	trigger = {
		NOT = {
			has_global_variable = alis_ubbo_created
		}
		location:lisbon = {
			has_owner = yes
			owner = {
				culture = {
					OR = {

		}

iberia_colony.2 = { #Punic Purchase
	type = country_event
	title = iberia_colony.2.t
	desc = iberia_colony.2.desc
	outcome = positive

		10 = regular

		culture = {
			OR = {

				has_culture_group = culture_group:iberia_group

			save_scope_as = pa_ruler
		}
		scope:alis_ubbo_punic.ruler ?= {
			save_scope_as = pa_punic_ruler
		}
	}

	option = {
		name = iberia_colony.2.a
		custom_tooltip = iberia_colony.2.at

		hidden_effect = {
			location:lisbon = {
				add_location_modifier = {
					modifier = founding_city_in_progress
					days = 730
				}
				# trigger_event = { id = state_improvement.6 days = 730 }
				#P:A TODO No EU5 equivalent: state_improvement.6 (gives the location city status after 2 years) is not part of the import; the construction modifier simply expires
				add_pop = { #P:A TODO Approximate: Imperator pop count converted 1:1 to EU5 size (2 noble pops); carthaginian_pantheon -> punic_pantheon
					type = pop_type:nobles
					size = 2
					culture = culture:carthaginian
					religion = religion:punic_pantheon
				}
				add_pop = { #P:A TODO Approximate: Imperator pop count converted 1:1 to EU5 size (2 citizen pops -> burghers)
					type = pop_type:burghers
					size = 2
					culture = culture:carthaginian
					religion = religion:punic_pantheon

			}
		}

		name = iberia_colony.2.b
		custom_tooltip = iberia_colony.2.bt
		ai_chance = {
			factor = 10
		}
	}

# Hidden: Saguntum (AI-owned) is refounded as the Greek-cultured trading republic of Sagointium.
iberia_colony.1029 = {
	type = country_event
	hidden = yes

	trigger = {
		location:valencia = { #P:A TODO Approximate location: Imperator Saguntum (p:1029) mapped to valencia
			has_owner = yes
			owner = {
				is_ai = yes
			}
		}
	}

	immediate = {
		location:valencia.owner = {
			if = {
				limit = {
					culture = culture:edetanian
				}
				random_list = {
					20 = {
						save_scope_as = sagunt_country
					}
					40 = {
						random_list = {
							20 = {
							}
							20 = {
								change_country_name = EDT
								change_country_adjective = EDT_ADJ
								change_country_color = map_EDT
								# change_country_flag = EDT #P:A TODO No EU5 equivalent: the mod has no EDT coat of arms key to switch to
								# set_country_heritage = edetanian_heritage #P:A TODO No EU5 equivalent: Imperator country heritage
							}
						}
						location:valencia = {
							create_country_from_location = {
								change_country_name = SAGUNT_NAME
								change_country_adjective = SAGUNT_ADJ
								save_scope_as = sagunt_country
							}
						}
					}
				}
			}
			else = {
				save_scope_as = sagunt_old
				location:valencia = {
					create_country_from_location = {
						change_country_name = SAGUNT_NAME
						change_country_adjective = SAGUNT_ADJ
						save_scope_as = sagunt_country
					}
				}
			}
		}
		scope:sagunt_old ?= {
			save_temporary_scope_as = pa_sagunt_overlord
			scope:sagunt_country = {
				make_subject_of = {
					target = scope:pa_sagunt_overlord
					type = subject_type:tributary
				}
			}
		}
		scope:sagunt_country = {
			if = {
				limit = { exists = scope:sagunt_old }
				add_truce_with = {
					target = scope:sagunt_old
					years = 10
				}
			}
			change_country_name = SAGUNT_NAME
			change_government_type = government_type:republic #P:A TODO Approximate: Imperator plutocratic_republic; EU5 has no government sub-forms
			# change_country_color = sagunt_color
			# change_country_flag = SAGUNT_FLAG
			# set_country_heritage = arseken_heritage
			#P:A TODO No EU5 equivalent: Imperator gave Sagointium its own map colour, flag and heritage (none exist in the mod)
			change_country_adjective = SAGUNT_ADJ
			# p:1235 = { set_owned_by = scope:sagunt_country } #Sebelaci
			#P:A TODO Approximate location: Imperator Sebelaci (p:1235) also maps to valencia, the new country's capital, so there is no second location to cede
			location:valencia = {
				rename_location = sagointium
				remove_location_modifier = aegis_of_arse_modifier
				add_location_modifier = {
					modifier = shield_of_sagointium_modifier
					days = -1
				}
				add_pop = { #P:A TODO Approximate: Imperator pop count converted 1:1 to EU5 size (2 noble pops); Imperator roman_pantheon -> hellenic_pantheon (Massalian Greeks)
					type = pop_type:nobles
					size = 2
					culture = culture:massalian
					religion = religion:hellenic_pantheon
				}
			}
			if = {
				limit = {
					NOT = {
						culture = culture:massalian
					}
				}
				pa_imp_set_primary_culture_cleanup_effect = {
					NEW_PRIMARY_CULTURE = massalian
					MAKE_OLD_CULTURE_INTEGRATED = yes
				}
			}
			change_religion = religion:hellenic_pantheon # Imperator roman_pantheon (Greek culture -> hellenic_pantheon)
			every_character = {
				change_character_culture = culture:massalian
				change_character_religion = religion:hellenic_pantheon
			}
			every_owned_location = {
				limit = {
					any_neighbor_location = {
						is_coastal = yes
					}
				}
				ordered_pop = {
					order_by = pop_satisfaction # Imperator pop_hapiness
					max = 3
					check_range_bounds = no
					change_pop_culture = culture:massalian
					change_pop_religion = religion:hellenic_pantheon
				}
			}
			every_owned_location = {
				ordered_pop = {
					limit = {
						OR = {
							religion = religion:hellenic_pantheon
							religion = religion:roman_pantheon
						}
					}
					order_by = pop_satisfaction # Imperator pop_hapiness
					max = 10
					check_range_bounds = no
					change_pop_culture = culture:massalian
				}
			}
		}
	}
}
