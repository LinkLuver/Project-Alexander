# Melqart's temple at Tyre is sacked (desecration.2/.3) and the god is reborn at Gadir.
# Imperator Tyrus (p:743) -> location:sidon (approximate: no Tyre location in the mod map); Gadir (p:1344) -> location:cadiz.
# The pilot's desecration.2 fires cr_religious_flavour_events.2 on location:cadiz directly, so .1 is an unused relay for now.
namespace = cr_religious_flavour_events

# Hidden relay: if Tyre and Gadir have different owners, Melqart will be reborn at Gadir.
cr_religious_flavour_events.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				NOT = { location:sidon.owner ?= location:cadiz.owner } #P:A TODO Approximate location: Imperator Tyrus (p:743) mapped to sidon
			}
			location:cadiz = {
				trigger_event_non_silently = {
					id = cr_religious_flavour_events.2
					days = { 120 360 }
				}
			}
		}
		save_scope_as = cr_sacker_of_temples
	}
}
