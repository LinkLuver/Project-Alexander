#P:A Imported from Indomita: events/health.txt
# Imperator character events, converted to country events: the character is scope:pa_character.
# health.1/3/4/5 were fired by Imperator on_giving_birth (root = mother, scope:newborn). In EU5 they are meant for
# on_character_birth (root = country, scope:character = the newborn); they read the mother from the newborn.
namespace = health

# The mother dies in childbirth.
health.1 = {
	type = country_event
	title = health.1.t
	desc = health.1.desc
	outcome = negative
	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		exists = scope:character
		scope:character = {
			exists = mother
			mother = {
				is_alive = yes
				employer = root
			}
		}
	}

	immediate = {
		scope:character = {
			save_scope_as = newborn
			mother = {
				save_scope_as = pa_character
			}
		}
		# Imperator: death_reason = death_infection_during_childbirth if the mother had the "infection" trait, else death_childbirth
		kill_character = { #P:A TODO Approximate: EU5 has no childbirth death reason (or infection trait); sickly_death used
			target = scope:pa_character
			reason = sickly_death
		}
		if = {
			limit = {
				any_character = {
					is_spouse_of = scope:pa_character
					is_alive = yes
				}
			}
			random_character = {
				limit = {
					is_spouse_of = scope:pa_character
				}
				save_scope_as = grieving_husband
			}
		}
	}

	option = {
		name = health.1.a
		# if = {
		# 	limit = { exists = scope:grieving_husband scope:grieving_husband = { NOT = { has_trait = mourning } } }
		# 	random = { chance = 50 scope:grieving_husband = { add_trait = mourning trigger_event = { id = health.2 days = { 360 3600 } } } }
		# }
		#P:A TODO No EU5 equivalent: 50% chance the widower gains the "mourning" trait (EU5 has no mourning trait); health.2 removes it later
	}
}
