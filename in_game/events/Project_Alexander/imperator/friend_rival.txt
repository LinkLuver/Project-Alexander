# A romance: the character grows close to the one who courted them (friend_request).
friend_rival.1 = {
	type = country_event
	title = friend_rival.1.t
	desc = friend_rival.1.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = happy
	}

	trigger = {
		exists = scope:pa_character # Imperator root character, from the befriend interaction
		exists = scope:targeter
		scope:targeter = { is_alive = yes }
		always = no #P:A TODO No EU5 equivalent: the event exists to make the character and scope:targeter friends, and the befriend interaction / friend_request on_action that fired it do not exist in EU5
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 0.1
			exists = scope:targeter
			scope:targeter = { # Imperator is_same_gender = root
				OR = {
					AND = {
						is_female = yes
						scope:pa_character = { is_female = yes }
					}
					AND = {
						is_female = no
						scope:pa_character = { is_female = no }
					}
				}
			}
		}
	}

	option = {
		name = friend_rival.1.a
		# scope:pa_character = { add_friend = scope:targeter } #P:A TODO No EU5 equivalent: character friends
		ai_chance = {
			factor = 100
		}
	}
	option = {
		name = friend_rival.1.b
	}
}

# Mediation: the ruler settles a dispute over wealth between two courtiers, befriending one (monthly_ruler_pulse).
friend_rival.5 = {
	type = country_event
	title = friend_rival.5.t
	desc = friend_rival.5.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				is_ruler = yes
				pa_imp_char_can_have_friends_trigger = yes
				NOT = {
					has_variable = friend_rival_5_toggle
				}
			}
		}
		trigger_else = {
			ruler ?= {
				pa_imp_char_can_have_friends_trigger = yes
				NOT = {
					has_variable = friend_rival_5_toggle
				}
			}
		}
		any_character = {
			count >= 2
			pa_imp_char_can_be_friend_trigger = yes
			NOT = { this = root.ruler }
			# is_party_leader = no #P:A TODO No EU5 equivalent: Imperator party leaders
			# prisoner = no #P:A Removed: prisoner = no is always true in EU5
		}
		always = no #P:A TODO No EU5 equivalent: every option pays a courtier from personal wealth and/or makes them the ruler's friend; EU5 characters have neither wealth nor friends
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			ruler = { save_scope_as = pa_character }
		}
		scope:pa_character = {
			set_variable = {
				name = friend_rival_5_toggle
				days = 1825
			}
		}
		random_character = {
			limit = {
				pa_imp_char_can_be_friend_trigger = yes
				NOT = { this = scope:pa_character }
				# is_party_leader = no #P:A TODO No EU5 equivalent: Imperator party leaders
				# prisoner = no #P:A Removed: prisoner = no is always true in EU5
			}
			# weight = { title_importance_modifier = yes } #P:A TODO No EU5 equivalent: Imperator weighted the pick by office/title importance
			set_variable = {
				name = fr_5_first
				days = 450 #P:A TODO Approximate: Imperator used a random 180-720 days; set_variable takes a fixed duration, the midpoint is used
			}
			save_scope_as = friend_rival_5_target_one
		}
		random_character = {
			limit = {
				pa_imp_char_can_be_friend_trigger = yes
				NOT = { this = scope:pa_character }
				# is_party_leader = no #P:A TODO No EU5 equivalent: Imperator party leaders
				# prisoner = no #P:A Removed: prisoner = no is always true in EU5
				NOT = { has_variable = fr_5_first }
			}
			# weight = { title_importance_modifier = yes } #P:A TODO No EU5 equivalent: Imperator weighted the pick by office/title importance
			save_scope_as = friend_rival_5_target_two
		}
	}

	option = {
		name = friend_rival.5.a
		# scope:friend_rival_5_target_one = { add_gold = { value = root.employer.has_monthly_income multiply = 6 } add_friend = ROOT }
		#P:A TODO No EU5 equivalent: the first claimant gets 6 months of the country's income as personal wealth and becomes the ruler's friend (EU5 characters have no wealth or friends)
		hidden_effect = {
			scope:friend_rival_5_target_two = {
				save_scope_as = friend_rival_5_spurned
			}
			random_list = {
				50 = {
					trigger_event_non_silently = {
						id = friend_rival.6
						days = { 30 90 }
					}
				}
				50 = {}
			}
		}
	}

	option = {
		name = friend_rival.5.b
		# scope:friend_rival_5_target_two = { add_gold = { value = root.employer.has_monthly_income multiply = 6 } add_friend = ROOT }
		#P:A TODO No EU5 equivalent: the second claimant gets 6 months of the country's income as personal wealth and becomes the ruler's friend
		hidden_effect = {
			scope:friend_rival_5_target_one = {
				save_scope_as = friend_rival_5_spurned
			}
			random_list = {
				50 = {
					trigger_event_non_silently = {
						id = friend_rival.6
						days = 1
					}
				}
				50 = {}
			}
		}
	}

	option = {
		name = friend_rival.5.c
		# scope:friend_rival_5_target_one = { add_gold = { value = root.employer.has_monthly_income multiply = 3 } }
		# scope:friend_rival_5_target_two = { add_gold = { value = root.employer.has_monthly_income multiply = 3 } }
		#P:A TODO No EU5 equivalent: both claimants get 3 months of the country's income as personal wealth
		# add_popularity = popularity_small #P:A TODO No EU5 equivalent: +5 popularity for the ruler
	}
}
