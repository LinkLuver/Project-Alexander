food_events.1 = {

	title = food_events.1.t

	outcome = negative

		10 = exterior

	# right_portrait = scope:fe_city.governor

	trigger = {
		NOT = { has_variable = food_event_cooldown }
		any_province = {
			NOT = {
				any_location_in_province = { has_variable = food_event_cooldown_state }

				count >= 3

	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		ruler ?= {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = food_event_cooldown
			days = 3650 # 10 years
		}
		random_province = {
			limit = {
				NOT = {
					any_location_in_province = { has_variable = food_event_cooldown_state }
				}
				any_location_in_province = {
					count >= 3
					owner = root
					NOR = {
						climate = arid #P:A TODO Approximate: Imperator desert terrain -> EU5 arid climate
						NOT = { location_max_winter_level = none } #P:A Invictus checked has_winter = no here, the opposite of the trigger (so no province could match); aligned with the trigger
					}
				}
			}
			# Imperator weights were all factor = 1 (no effect): not the capital's state, wholly owned, storm-damaged
			ordered_location_in_province = {
				limit = {
					owner = root
					NOR = {
						climate = arid #P:A TODO Approximate: Imperator desert terrain -> EU5 arid climate
						NOT = { location_max_winter_level = none }
					}
				}
				order_by = population
				save_scope_as = fe_city
			}
		}
		pa_imp_famine_relocation_immediate_effect = yes
	}
