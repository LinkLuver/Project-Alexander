# Hidden: the Olympic Games at Olympia (Imperator p:439, EU5 location:pontikokastro) fall due every four years and are
# offered to its owner. Started from on_game_start by pa_imp_olympics_start
# (in_game/common/on_action/pa_imperator_events.txt), as Invictus did from on_game_initialized; then re-fires itself.
olympic.12 = {
	type = location_event
	hidden = yes

	immediate = {
		save_scope_as = olympic_games_location
		owner ?= { save_scope_as = olympia_owner_country }
		if = {
			limit = {
				exists = scope:olympia_owner_country
				controller ?= scope:olympia_owner_country # Imperator owner = controller: Olympia is not occupied
			}
			scope:olympia_owner_country = {
				trigger_event_silently = olympic.2
			}
		}
		if = { # counts olympiads
			limit = { has_global_variable = olympic_counter_global_var }
			change_global_variable = {
				name = olympic_counter_global_var
				add = 1
			}
		}
		else = {
			set_global_variable = {
				name = olympic_counter_global_var
				value = 1
			}
		}
		# Imperator also set a debug variable olympics_countdown here (not carried over)
		#P:A TODO Verify: EU5 fires a location event only for an owned location; if Olympia is ever left unowned, the four-year cycle stops (Imperator kept it going)
		trigger_event_silently = {
			id = olympic.12
			days = 1459 # 4 years - 11 April
		}
	}
}

		else = {
			scope:olympic_holder_country = {
				trigger_event_silently = olympic.8
			}
			every_country = {
