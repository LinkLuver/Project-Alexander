#P:A Imported from Indomita: events/FMO_MPE_events.txt
# Expiry of Indomita "special pops" (priest, merchant, architect, artist). Indomita's send_<type>_button scripted GUIs
# (common/scripted_guis/MPR_send_special_pop_buttons.txt, 40 political influence) added one to var:<type>_in_territory of the
# location and var:number_of_<type>_var of the owner, then fired the odd event on the location and the even event on the owner
# 9125 days (25 years) later to take it away again. Two separate events so that a conquest does not break the counts.
#P:A TODO Nothing fires these yet: the Indomita GUI buttons are not imported, and b10's pa_imp_add_special_pop does not schedule the expiry. A caller should fire the location event (silently) on the location and the country event on its owner, 9125 days after adding a special pop.
namespace = mpe_pop_events

# Hidden: one priest leaves the location.
mpe_pop_events.1 = {
	type = location_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_variable = priest_in_territory
			}
			change_variable = {
				name = priest_in_territory
				subtract = 1
			}
			if = {
				limit = { var:priest_in_territory > 0 }
				pa_imp_add_special_pop_province_modifier_effect = yes
			}
			else = {
				remove_variable = priest_in_territory
				if = {
					limit = {
						has_location_modifier = priest_in_province_mod
					}
					remove_location_modifier = priest_in_province_mod
				}
			}
		}
	}
}

# Hidden: one priest leaves the country's count (max 5 per country).
