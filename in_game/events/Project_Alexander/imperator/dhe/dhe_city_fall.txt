# Imperator province events fired by on_ownership_change (root = the province, owner = its new owner).
# EU5: location events for on_location_changed_owner (root = location, scope:winner = new owner, scope:loser = old owner).
# Each event checks its own city; dhe_city_fall.100 (hidden) marks the new owner so each city falls only once per country.

# Carthage
dhe_city_fall.1 = {
	type = location_event
	title = dhe_city_fall.1.t
	desc = {
		first_valid = {
			triggered_desc = { #special text for proto_europeans
				trigger = { owner = { culture = { has_culture_group = culture_group:nuragic_group } } } #P:A TODO Approximate: Imperator proto_european culture group mapped to nuragic_group
				desc = dhe_city_fall.1.desc_2
			}
			triggered_desc = { # vanilla text
				trigger = { always = yes }
				desc = dhe_city_fall.1.desc
			}
		}
	}
	outcome = neutral
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		this = location:tunis
		owner = {
			in_civil_war = no
			NOT = {
				has_variable = dhe_city_fall_1
			}
		}
		pa_imp_city_fall_trigger = yes
	}

	immediate = {
		save_scope_as = special_province
		owner = {
			ruler ?= { save_scope_as = pa_ruler }
		}
		hidden_effect = {
			if = {
				limit = {
					has_variable = cr_walls_of_carthage_exist
				}
				add_location_modifier = {
					modifier = cr_carthage_walls_of_carthage
					days = -1
				}
			}
		}
	}

	option = { # Treat them nicely
		name = dhe_city_fall.1.a
		pa_imp_merciful_ruler_effect = yes
		owner = {
			if = {
				limit = {
					culture = { has_culture_group = culture_group:nuragic_group }
				}
				custom_tooltip = nuragic_dhe_city_fall.1
				# random_list = { 66 = { current_ruler = { add_trait = submissive } } 33 = { current_ruler = { add_trait = weak_willed } } }
				#P:A TODO No EU5 equivalent: a Nuragic ruler showing mercy became submissive (66%) or weak_willed (33%); EU5 has neither trait
			}
		}
	}

	option = { # Bring a few things back home
		name = dhe_city_fall.1.b
		pa_imp_neutral_conqueror_effect = yes
		owner = {
			if = {
				limit = {
					culture = { has_culture_group = culture_group:nuragic_group }
				}
				custom_tooltip = nuragic_dhe_city_fall.1
				# random_list = { 66 = { current_ruler = { add_trait = submissive } } 16 = { current_ruler = { add_trait = weak_willed } } 18 = { } }
				#P:A TODO No EU5 equivalent: a Nuragic ruler became submissive (66%) or weak_willed (16%); EU5 has neither trait
			}
		}
	}

	option = { # Sack the city
		name = dhe_city_fall.1.c
		owner = {
			add_country_modifier = {
				modifier = country_improvement_carthaginian_fleet
				days = 3650
			}
		}
		remove_location_modifier = cr_carthage_walls_of_carthage
		hidden_effect = {
			remove_variable = cr_walls_of_carthage_exist
		}
		pa_imp_sacking_conqueror_effect = yes
		owner = {
			if = {
				limit = {
					ironman = yes
					is_ai = no
					NOT = {
						has_variable = ach_carthago_delenda_est_var
					}
				}
				set_variable = ach_carthago_delenda_est_var
			}
		}
		owner = {
			if = {
				limit = {
					culture = { has_culture_group = culture_group:nuragic_group }
				}
				custom_tooltip = nuragic_dhe_city_fall.1
				# random_list = { 33 = { current_ruler = { add_trait = submissive } } 6 = { current_ruler = { add_trait = weak_willed } } 61 = { } }
				#P:A TODO No EU5 equivalent: a Nuragic ruler became submissive (33%) or weak_willed (6%); EU5 has neither trait
			}
		}
	}

	option = {
		name = dhe_city_fall.1.d
		trigger = {
			owner = {
				tag = ROM
				# current_ruler = { has_trait = lunatic }
			}
			always = no #P:A TODO No EU5 equivalent: option only for a Roman ruler with the "lunatic" trait, which EU5 lacks
		}
		owner = {
			if = {
				limit = {
					ironman = yes
					is_ai = no
					NOT = {
						has_variable = ach_carthago_delenda_est_var
					}
				}
				set_variable = ach_carthago_delenda_est_var
			}
		}
		pa_imp_salt_earth_effect = yes
		custom_tooltip = lunatic_tooltip
	}
	option = {
		name = dhe_city_fall.1.e
		trigger = {
			owner = {
				culture = { has_culture_group = culture_group:nuragic_group }
			}
		}
		owner = {
			add_country_modifier = {
				modifier = country_improvement_carthaginian_fleet
				days = 3650
			}
			ruler ?= {
				add_trait = trait:cruel
				# add_trait = victorious
				#P:A TODO No EU5 equivalent: the ruler also gained the "victorious" trait
			}
		}
		remove_location_modifier = cr_carthage_walls_of_carthage
		add_location_modifier = {
			modifier = into_ashes
			days = -1
		}
		hidden_effect = {
			remove_variable = cr_walls_of_carthage_exist
		}
		pa_imp_mass_murder_effect = yes

		custom_tooltip = nuragic_dhe_city_fall.1
	}
}
