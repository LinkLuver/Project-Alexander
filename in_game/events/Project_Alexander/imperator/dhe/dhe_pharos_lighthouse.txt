# Building the Pharos Lighthouse at Alexandria (Imperator p:516 = location:alexandria, exact). Not tag-bound: any owner
# of Alexandria. In Imperator the decision construction_of_the_pharos (owns Alexandria, 1200 gold, stability > 50, not
# built yet) paid 1200 gold, set the country variables pharos_lighthouse_progress = 1 and pharos_lighthouse_quality = 1,
# fired dhe_pharos_lighthouse.10 at once and dhe_pharos_lighthouse.1 after 200-400 days, and started the great work.
# dhe_pharos_lighthouse.1 is the hidden progress loop: four more steps of 200-400 days, each with a random complication
# (.2 to .6) that raises the quality; at progress 5 the quality picks the ending (.11 / .12 / .13).
namespace = dhe_pharos_lighthouse

# Hidden: construction progress loop.
dhe_pharos_lighthouse.1 = {
	type = country_event
	hidden = yes

	immediate = {
		if = {
			limit = { # Check if the building is completed or under progress
				var:pharos_lighthouse_progress < 5
				owns = location:alexandria
			}
			random_list = { # Progress Events
				2 = { # Building progresses without any event
					change_variable = {
						name = pharos_lighthouse_quality
						add = 1
					}
				}
				2 = { # Complication - Drought
					trigger = {
						NOT = {
							location:alexandria = {
								has_location_modifier = province_diminishment_starvation_lighthouse
							}
						}
						any_character = {
							in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 cabinet membership
							is_ruler = no
						}
					}
					trigger_event_non_silently = {
						id = dhe_pharos_lighthouse.2
					}
				}
				2 = { # Complication - Materials
					trigger = {
						any_character = {
							in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 cabinet membership
							is_ruler = no
						}
					}
					trigger_event_non_silently = {
						id = dhe_pharos_lighthouse.3
					}
				}
				2 = { # Complication - Workers Leaving
					trigger_event_non_silently = {
						id = dhe_pharos_lighthouse.4
					}
				}
				2 = { # Complication - Earthquake
					trigger_event_non_silently = {
						id = dhe_pharos_lighthouse.5
					}
				}
				2 = { # Building Progress - Quality Materials
					trigger_event_non_silently = {
						id = dhe_pharos_lighthouse.6
					}
				}
			}
			change_variable = {
				name = pharos_lighthouse_progress
				add = 1
			}
			trigger_event_silently = { # Trigger next event
				id = dhe_pharos_lighthouse.1
				days = { 200 400 }
			}
		}
		else_if = {
			limit = {
				var:pharos_lighthouse_progress = 5
				owns = location:alexandria
			}
			if = {
				limit = {
					var:pharos_lighthouse_quality < 3
				}
				trigger_event_non_silently = {
					id = dhe_pharos_lighthouse.11
				}
			}
			else_if = {
				limit = {
					var:pharos_lighthouse_quality >= 3
					var:pharos_lighthouse_quality < 5
				}
				trigger_event_non_silently = {
					id = dhe_pharos_lighthouse.12
				}
			}
			else_if = {
				limit = {
					var:pharos_lighthouse_quality >= 5
				}
				trigger_event_non_silently = {
					id = dhe_pharos_lighthouse.13
				}
			}
		}
		else = {
			remove_variable = pharos_lighthouse_progress
			remove_variable = pharos_lighthouse_quality
		}
	}
}

# Drought complications.
dhe_pharos_lighthouse.2 = {
	type = country_event
	title = dhe_pharos_lighthouse.2.t
	desc = dhe_pharos_lighthouse.2.desc
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		owns = location:alexandria
		any_character = {
			in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 cabinet membership
			is_ruler = no
		}
	}

	immediate = {
		ruler ?= { save_scope_as = pharos_ruler }
		random_character = {
			limit = {
				in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 cabinet membership
				is_ruler = no
			}
			save_scope_as = random_advisor
		}
	}

	option = { # Fix it all
		name = dhe_pharos_lighthouse.2.a
		custom_tooltip = dhe_pharos_lighthouse.2.a.tt
		add_gold = {
			value = 0
			subtract = monthly_income_total
			multiply = 12
			floor = yes
			max = -400
			min = -1200
		}
	}

	option = { # Fix it for the lighthouse workers
		name = dhe_pharos_lighthouse.2.b
		custom_tooltip = dhe_pharos_lighthouse.2.a.tt
		add_gold = {
			value = 0
			subtract = monthly_income_total
			multiply = 8
			min = -900
			max = -250
		}
		location:alexandria = {
			add_location_modifier = {
				modifier = province_diminishment_starvation_lighthouse
				days = 3650
			}
		}
		hidden_effect = {
			change_variable = {
				name = pharos_lighthouse_quality
				add = 1
			}
		}
	}

	option = {
		name = dhe_pharos_lighthouse.2.c
		custom_tooltip = dhe_pharos_lighthouse.2.c.tt

			}
		}
		hidden_effect = {
			change_variable = {
				name = pharos_lighthouse_quality
				add = 1
			}
		}
	}
}
