# Hidden: a married ruler has an affair with a woman at court; the child's birth is handled by ip_monarchy.39.
#P:A Was an Imperator character event (character = the ruler). The character is always root's ruler, so root.ruler is used and saved as scope:pa_character.
ip_monarchy.38 = {
	type = country_event
	hidden = yes

	trigger = { # Married monarch ruler and suitable married woman to have affair with exist
		government_type = government_type:monarchy
		exists = capital
		exists = ruler
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = { this = root.ruler }
		}
		ruler = {
			is_ruler = yes
			is_married = yes
			is_female = no # Imperator is_male = yes
		}
		any_character = {
			NOR = {
				is_spouse_of = root.ruler
				is_child_of = root.ruler
				is_close_relative = root.ruler #P:A TODO Approximate: Imperator skipped this check under egyption_succession_law (sibling marriage); EU5 has no such law, so close relatives are always excluded
				is_ruler = yes
				# is_rival = root #P:A TODO No EU5 equivalent: character rivals
				is_pregnant = yes #P:A TODO Fixed Invictus bug: the source had is_pregnant = no inside this NOR, which required the woman to be pregnant already
				AND = {
					num_of_children >= 1
					any_child = { is_child_of = root.ruler }
				}
			}
			is_female = yes
			age_in_years >= 18
			age_in_years <= 40
		}
		any_neighbor_country = { count >= 1 }
		ruler = {
			NOR = { # Ruler is capable of sin
				has_trait = pious # Imperator devout
				# has_trait = chaste
				# has_trait = incapable
				# has_trait = shrewd
				# has_trait = self_controlled
				#P:A TODO No EU5 equivalent: Imperator traits chaste, incapable, shrewd, self_controlled
				has_variable = mon_38_cooldown
			}
		}
		NOR = {
			has_variable = ip_mon_38_flag
			has_variable = mon_38_cooldown
		}
	}

	immediate = {
		set_variable = { # Toggle
			name = mon_38_cooldown
			days = 18250 # 50 years
		}
		ruler = {
			set_variable = { # Toggle
				name = mon_38_cooldown
				days = 18250 # 50 years
			}
			save_scope_as = pa_character
			save_scope_as = bastard_ruler
		}
		random_character = { # Find best lover
			limit = {
				NOR = {
					is_spouse_of = scope:bastard_ruler
					is_child_of = scope:bastard_ruler
					is_close_relative = scope:bastard_ruler #P:A TODO Approximate: Imperator skipped this check under egyption_succession_law; EU5 has no such law
					is_ruler = yes
					# is_rival = root #P:A TODO No EU5 equivalent: character rivals
					is_pregnant = yes #P:A TODO Fixed Invictus bug: see trigger
					AND = {
						num_of_children >= 1
						any_child = { is_child_of = scope:bastard_ruler }
					}
				}
				is_female = yes
				age_in_years >= 18
				age_in_years <= 40
			}
			# order_by = prominence #P:A TODO Approximate: Imperator picked the most prominent woman (ordered_character by prominence); EU5 characters have no prominence, so a random one is picked
			save_scope_as = affair_strumpet
		}
		if = { # Save husband as cuckold
			limit = {
				scope:affair_strumpet = {
					is_married = yes
					any_spouse = {
						is_alive = yes
						employer = root
					}
				}
			}
			scope:affair_strumpet = {
				random_spouse = { # Imperator scope:affair_strumpet.spouse
					limit = {
						is_alive = yes
						employer = root
					}
					save_scope_as = affair_cuckold
				}
			}
		}
		scope:affair_strumpet = { # Impregnate lover
			set_variable = affair_pregnancy
			# make_pregnant = { father = scope:bastard_ruler known_bastard = yes number_of_children = 1 }
			#P:A TODO Approximate: EU5 has no make_pregnant; the child is created at once (age 0) below instead of being born after a pregnancy. ip_monarchy.39 still fires 300 days later and finds it as the lover's youngest child by the ruler
		}
		create_character = {
			culture = scope:bastard_ruler.culture
			religion = scope:bastard_ruler.religion
			father = scope:bastard_ruler
			mother = scope:affair_strumpet
			age = 0
			no_stats = yes
			trait_category = none
			# known_bastard: created without a dynasty; ip_monarchy.39 options b/c give it the ruler's dynasty
		}
		trigger_event_non_silently = { # Trigger next event for country after birth
			id = ip_monarchy.39
			days = 300
		}
	}
}

# The ruler's affair has produced a child (fired by ip_monarchy.38, and by Imperator on_giving_birth).
ip_monarchy.39 = {
	type = country_event
	title = ip_monarchy.39.t
	desc = ip_monarchy.39.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = { # Previous characters alive and relevant
		government_type = government_type:monarchy
		exists = capital
		exists = scope:bastard_ruler
		scope:bastard_ruler = {
			is_alive = yes
			is_ruler = yes
			employer = root
		}
		exists = scope:affair_strumpet
		scope:affair_strumpet = {
			is_alive = yes
			employer = root
			num_of_children >= 1
			any_child = {
				is_alive = yes
				is_child_of = scope:bastard_ruler
			}
		}
		any_neighbor_country = { count >= 1 }
	}

	immediate = {
		scope:affair_strumpet = {
			ordered_child = {
				limit = { is_child_of = root.ruler }
				order_by = {
					value = 0
					subtract = age_in_days
				}
				max = 1
				check_range_bounds = no
				save_scope_as = affair_child
			}
		}
		scope:bastard_ruler = {
			if = {
				limit = { is_married = yes }
				random_spouse = { save_scope_as = bastard_consort } # Imperator scope:bastard_ruler.spouse
			}
		}
		hidden_effect = {
			scope:affair_child = {
				save_scope_as = pa_character
				employer = {
					trigger_event_silently = pa_character_events.35
				}
			}
		}
	}

	option = { # This must never be revealed
		name = ip_monarchy.39.a
		custom_tooltip = ip_monarchy.39.at
		random_neighbor_country = { save_scope_as = ip_monarchy_36_neighbor_scope } # Pick neighbor for child

			random_list = { # Child dies or may return, merc banishment to hide death notice for ruler

						reason = vanished

					}
				}
				50 = {
					set_variable = {
						name = ip_mon_38_flag
						value = root.ruler
					}
					scope:affair_child = {
						set_variable = {
							name = ip_mon_38_child_flag
							value = root.ruler
						}
						move_country = scope:ip_monarchy_36_neighbor_scope # Imperator banish
						save_scope_as = pa_character
						employer = {
							trigger_event_silently = {
								id = ip_monarchy.70
								days = { 6570 8030 }
							}
						}
					}
				}
			}
		}
		# scope:affair_strumpet = { add_trait = mourning } #P:A TODO No EU5 equivalent: the mother gets the Imperator trait "mourning"
		# if = { limit = { scope:bastard_ruler = { is_married = yes } } scope:bastard_ruler.spouse = { add_loyalty = loyalty_consort_reassured } }
		#P:A TODO No EU5 equivalent: the consort gains loyalty (EU5 characters have no loyalty)
		add_gold = { # Payment for services rendered
			value = 0
			subtract = root.monthly_income_total
			multiply = 2
			min = -6000
		}

# Hidden: the royal bastard of ip_monarchy.36 has grown up; fires ip_monarchy.37 for the ruler's country and brings the child (and the mother) back to it.
#P:A Was an Imperator character event (character = the bastard child). Root is the child's employer when it fires (ip_monarchy.36 fires it there, 18-22 years later), the child is scope:pa_character.
ip_monarchy.69 = { # .37 trigger
	type = country_event
	hidden = yes

	trigger = {
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				is_alive = yes
				is_ruler = no
				has_variable = ip_mon_36_child_flag
				any_country = {
					has_variable = ip_mon_36_flag
					var:ip_mon_36_flag = prev.var:ip_mon_36_child_flag
				}
			}
		}
		trigger_else = {
			any_character = {
				is_alive = yes
				is_ruler = no
				has_variable = ip_mon_36_child_flag
				any_country = {
					has_variable = ip_mon_36_flag
					var:ip_mon_36_flag = prev.var:ip_mon_36_child_flag
				}
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					is_alive = yes
					is_ruler = no
					has_variable = ip_mon_36_child_flag
					any_country = {
						has_variable = ip_mon_36_flag
						var:ip_mon_36_flag = prev.var:ip_mon_36_child_flag
					}
				}
				save_scope_as = pa_character
			}
		}
		random_country = {
			limit = {
				has_variable = ip_mon_36_flag
				var:ip_mon_36_flag = scope:pa_character.var:ip_mon_36_child_flag
			}
			save_scope_as = ip_mon_69_ruler_country
			set_variable = {
				name = ip_mon_36_child_scope_flag
				value = scope:pa_character
				days = 30
			}
			if = {
				limit = { exists = scope:pa_character.mother }
				set_variable = {
					name = ip_mon_36_mother_scope_flag
					value = scope:pa_character.mother
					days = 30
				}
			}
			trigger_event_non_silently = ip_monarchy.37
		}
		hidden_effect = {
			#P:A TODO Approximate: Imperator moved the child and the mother to the employer of the old ruler (var:ip_mon_36_child_flag.employer); here they move to the country holding ip_mon_36_flag, which is that country and still exists if the old ruler has died
			if = { # Move child back to ruler country if needed
				limit = {
					exists = scope:ip_mon_69_ruler_country
					scope:pa_character = {
						NOT = { employer = scope:ip_mon_69_ruler_country }
					}
				}
				scope:pa_character = {
					pa_imp_remove_all_positions = yes
					move_country = scope:ip_mon_69_ruler_country
					remove_character_modifier = foreign_citizen
					# set_as_minor_character = this #P:A TODO No EU5 equivalent: the child is made a minor character (outside the ruler's family); in EU5 it simply has no dynasty yet
					# add_loyalty = loyalty_banished_affair #P:A TODO No EU5 equivalent: the child loses loyalty (EU5 characters have no loyalty)
				}
			}
			if = { # Move mother back to ruler country if needed
				limit = {
					exists = scope:ip_mon_69_ruler_country
					exists = scope:pa_character.mother
					scope:pa_character.mother = {
						is_alive = yes #P:A TODO Fixed Invictus bug: the source had is_alive = yes inside the NOR below, which only moved dead mothers
						NOR = {
							employer = scope:ip_mon_69_ruler_country
							is_ruler = yes
							is_married = yes
						}
					}
				}
				scope:pa_character.mother = {
					pa_imp_remove_all_positions = yes
					move_country = scope:ip_mon_69_ruler_country
					remove_character_modifier = foreign_citizen
					# set_as_minor_character = this #P:A TODO No EU5 equivalent: the mother is made a minor character
					# add_loyalty = loyalty_banished_affair #P:A TODO No EU5 equivalent: the mother loses loyalty (EU5 characters have no loyalty)
				}
			}
		}
	}
}
