# Imperial power for very large countries. imperial_events.1 was in Indomita's monthly_country_pulse; imperial_events.2
# was fired by the scripted GUI button change_imperial_power_button (no EU5 equivalent; the orchestrator decides).
namespace = imperial_events

# A great empire chooses its imperial focus.
imperial_events.1 = {
	type = country_event
	title = imperial_events.1.t
	desc = imperial_events.1.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		num_locations >= 600 #P:A TODO Approximate: Imperator num_of_cities >= 600 (territories); EU5 owned locations, which are smaller and more numerous
		NOT = { has_variable = imperial_power }
	}

	immediate = {
		ruler ?= { save_scope_as = imperial_ruler }
	}

	option = {
		# political
		name = imperial_events.1.a
		set_variable = {
			name = imperial_power
			value = flag:political
		}
		show_as_tooltip = {
			add_country_modifier = {
				modifier = imperial_political_mod_5
				years = -1
			}
		}
	}
	option = {
		# warfare
		name = imperial_events.1.b
		set_variable = {
			name = imperial_power
			value = flag:warfare
		}
		show_as_tooltip = {
			add_country_modifier = {
				modifier = imperial_military_mod_5
				years = -1
			}
		}
	}
	option = {
		# economic
		name = imperial_events.1.c
		set_variable = {
			name = imperial_power
			value = flag:economic
		}
		show_as_tooltip = {
			add_country_modifier = {
				modifier = imperial_economic_mod_5
				years = -1
			}
		}
	}
	after = {
		pa_imp_set_imperial_power = yes
	}
}
