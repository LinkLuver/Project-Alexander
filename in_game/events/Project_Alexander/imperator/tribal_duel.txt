tribal_duel.1 = {
	type = country_event
	title = tribal_duel.1.t
	desc = tribal_duel.1.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		exists = scope:target
		exists = scope:champion
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
	}

	option = {
		name = tribal_duel.1.a
		# Ruler
		trigger_event_non_silently = {
			id = tribal_duel.3
			days = 3
		}
		ruler = {
			set_variable = {
				name = i_want_to_fight
				days = 10
			}
			# family = { add_prestige = 50 } #P:A TODO No EU5 equivalent: +50 prestige for the ruler's family (EU5 dynasties have no prestige)
		}
	}

	option = {
		name = tribal_duel.1.b
		# Champion
		trigger_event_non_silently = {
			id = tribal_duel.3
			days = 3
		}
		# ruler = { family = { add_prestige = -50 } } #P:A TODO No EU5 equivalent: -50 prestige for the ruler's family (EU5 dynasties have no prestige)
		scope:champion = {
			set_variable = {

				days = 10
			}
			add_character_modifier = {
				modifier = duel_champion
				days = 730
			}
		}
	}
}

							NOT = { has_variable = maimed_tooltip }
						}
						show_as_tooltip = {
							add_trait = trait:maimed
						}
						set_variable = {
							name = maimed_tooltip
							days = 180
						}
					}
				}
				else_if = {
					limit = { has_trait = one_eyed }
					if = {
						limit = {
							NOT = { has_variable = eyes_tooltip }
						}
						show_as_tooltip = {
							add_trait = trait:one_eyed
						}
						set_variable = {
							name = eyes_tooltip
							days = 180
						}
					}
				}
				else_if = {
					limit = { has_trait = blind }
					if = {
						limit = {
							NOT = { has_variable = blind_tooltip }
						}
						show_as_tooltip = {
							add_trait = trait:blind
						}
						set_variable = {
							name = blind_tooltip
							days = 180
						}
					}
				}
			}
		}
		scope:fighter_of_duel = {
			# add_popularity = -10 #P:A TODO No EU5 equivalent: -10 character popularity
			employer ?= {
				add_stability = -5
			}
		}
	}

	option = { #scope:fighter_of_duel is dead
		name = tribal_duel.2.c
		trigger = {
			scope:fighter_of_duel = {
				is_alive = no
			}
			scope:target = {
				is_alive = yes
			}
		}
		add_stability = -5 # Imperator: scope:fighter_of_duel.employer (the dead fighter served root)
	}

	option = { #scope:target is dead
		name = tribal_duel.2.d
		trigger = {
			scope:fighter_of_duel = {
				is_alive = yes
			}
			scope:target = {
				is_alive = no
			}
		}
		pa_imp_tribal_duel_rewards_effect = yes
	}

	after = {
		hidden_effect = {
			pa_imp_tribal_duel_description_distribution = yes #Don't give two descriptions in a row
		}
		remove_variable = roll
	}
