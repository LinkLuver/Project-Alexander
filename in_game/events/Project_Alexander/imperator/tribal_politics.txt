# The tribes are decentralizing because of instability
tribal_pol.1 = {
	type = country_event
	title = tribal_pol.1.t
	desc = tribal_pol.1.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		government_type = government_type:tribe
		NOT = { has_variable = tribal_pol_1_cooldown }
		societal_value:centralization_vs_decentralization <= 0 #P:A TODO Approximate: Imperator centralization >= 0 (-100..100); EU5 centralization is the left side of the centralization/decentralization societal value
		stability <= 50
		# any_character = { has_office = office_warchief }
		# any_character = { has_office = office_magistrate }
		any_cabinet_character = { #P:A TODO Approximate: Imperator required a warchief and a magistrate office holder; EU5 has no offices, two cabinet members stand in
			count >= 2
		}
		num_locations >= 1 # Imperator num_of_cities
	}

	immediate = {
		set_variable = {
			name = tribal_pol_1_cooldown
			years = 15
		}
		random_owned_location = {
			save_scope_as = randomized_country_province
		}
		ruler ?= {
			save_scope_as = pa_ruler
		}
		random_cabinet_character = { #P:A TODO Approximate: Imperator office_warchief holder; a cabinet member stands in
			save_scope_as = warchief_official
		}
		random_cabinet_character = { #P:A TODO Approximate: Imperator office_magistrate holder; another cabinet member stands in
			limit = {
				NOT = { this = scope:warchief_official }
			}
			save_scope_as = magistrate_official
		}
	}

	option = { # Use Military Power to force people to centralize
		name = tribal_pol.1.a
		change_societal_value = { #P:A TODO Approximate: Imperator add_centralization = 1 (centralization_large); EU5 moves the centralization/decentralization societal value towards centralization
			type = centralization_vs_decentralization
			value = -1
		}
		# add_tyranny = tyranny_medium #P:A TODO No EU5 equivalent: +3 tyranny (EU5 has no tyranny)
	}

	option = { # Use Oratory Power to force people to centralize
		name = tribal_pol.1.b
		trigger = {
			NOT = {
				ruler ?= {
					has_trait = silver_tongue
				}
			}
		}
		# add_political_influence = subtract_influence_medium #P:A TODO No EU5 equivalent: -15 political influence (EU5 has no political influence)
		change_societal_value = { #P:A TODO Approximate: Imperator add_centralization = 1 (centralization_large); EU5 societal value moved towards centralization
			type = centralization_vs_decentralization
			value = -1
		}
	}

	option = { # More effective use of Oratory Power to force people to centralize
		name = tribal_pol.1.d
		# highlight = yes (Imperator UI highlight; no EU5 field)
		trigger = {
			ruler ?= {
				has_trait = silver_tongue
			}
		}
		custom_tooltip = silver_tongued_tooltip
		# add_political_influence = subtract_influence_small #P:A TODO No EU5 equivalent: -10 political influence (EU5 has no political influence)
		change_societal_value = { #P:A TODO Approximate: Imperator add_centralization = 2 (centralization_huge); EU5 societal value moved towards centralization
			type = centralization_vs_decentralization
			value = -2
		}
	}

	option = { # Let people decentralize
		name = tribal_pol.1.c
		change_societal_value = { #P:A TODO Approximate: Imperator add_centralization = -1 (subtract_centralization_large); EU5 societal value moved towards decentralization
			type = centralization_vs_decentralization
			value = 1
		}
	}
}
