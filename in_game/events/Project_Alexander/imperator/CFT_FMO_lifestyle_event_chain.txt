#P:A Imported from Indomita: events/CFT_FMO_lifestyle_event_chain.txt
# Ruler lifestyle event chains of Indomita's character focus tree (CFT). EU5 has no focus trees:
# - "<focus>_focus_tree_enabled" (the ruler picked that focus; Indomita gives AI rulers the focus of their best skill)
#   is approximated by the ruler's best EU5 skill (civic/finesse -> ADM, martial -> MIL, oratory/charisma -> DIP).
# - focus experience and focus perks (add_focus_experience / add_focus_perk) have no EU5 equivalent.
# - character health has no EU5 equivalent.
namespace = lifestyle_event_chains

# A New Hobby: the ruler considers taking charge of the mints
lifestyle_event_chains.1 = {
	type = country_event
	title = lifestyle_event_chains.1.t
	desc = lifestyle_event_chains.1.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		exists = capital # Imperator has_land = yes
		ruler ?= {
			# has_variable = civic_focus_tree_enabled
			adm >= { value = mil } #P:A TODO Approximate: Indomita civic focus tree enabled; EU5 has no focus trees, the ruler's best skill must be ADM
			adm >= { value = dip }
			# character_experience > 60 #P:A TODO No EU5 equivalent: more than 60 focus-tree experience (Indomita character focus system)
			# health > 40 #P:A TODO No EU5 equivalent: character health above 40
			NOR = {
				has_character_modifier = master_minter_mod
				has_character_modifier = good_minter_mod
				has_character_modifier = learning_to_mint
			}
		}
		OR = {
			government_type = government_type:monarchy
			government_type = government_type:tribe
			# has_law = princeps_civitatis #P:A TODO No EU5 equivalent: Imperator republic laws (princeps civitatis, dictatorial elections) also allowed the chain
			# has_law = republican_election_4
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
	}

	option = {
		name = lifestyle_event_chains.1.a
		add_gold = {
			value = monthly_income_total # Imperator has_monthly_income
			multiply = -3
		}
		hidden_effect = {
			trigger_event_non_silently = {
				id = lifestyle_event_chains.2
				days = { 30 45 }
			}
		}
	}

	option = {
		name = lifestyle_event_chains.1.b
		custom_tooltip = no_effect_tooltip
	}
}

# The ruler summons a master minter
lifestyle_event_chains.2 = {
	type = country_event
	title = lifestyle_event_chains.2.t
	desc = lifestyle_event_chains.2.desc
	outcome = positive
	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		hidden_effect = {
			create_character = {
				culture = root.culture
				religion = root.religion
				age = 35
				save_scope_as = master_minter
			}
			scope:master_minter = {
				pa_imp_bad_martial_effect = yes
				pa_imp_bad_charisma_effect = yes
				pa_imp_good_finesse_effect = yes
				pa_imp_bad_zeal_effect = yes
				add_character_modifier = {
					modifier = master_minter_mod
					days = -1
				}
			}
		}
	}

	option = {
		name = lifestyle_event_chains.2.a
		add_gold = {
			value = monthly_income_total # Imperator has_monthly_income
			multiply = -3
		}
		ruler ?= {
			add_character_modifier = {
				modifier = learning_to_mint
				days = -1
			}
		}
		hidden_effect = {
			trigger_event_non_silently = {
				id = lifestyle_event_chains.3
				days = { 60 80 }
			}
		}
	}
}

# Competing hypotheses: heliocentrism or geocentrism
lifestyle_event_chains.15 = {
	type = country_event
	title = lifestyle_event_chains.15.t
	desc = lifestyle_event_chains.15.desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler
		}
		# Imperator picked research-office holders (has_tech_office) by their best finesse, zeal and charisma, falling back to any
		# non-ruler character (alternative_limit). EU5 has no offices, no zeal skill and no alternative_limit.
		ordered_character = { #P:A TODO Approximate: Imperator research-office holder with the highest finesse (else any non-ruler); EU5 non-ruler character with the highest ADM
			limit = {
				is_ruler = no
			}
			order_by = adm
			save_scope_as = finesse_thinker
		}
		random_character = { #P:A TODO Approximate: Imperator research-office holder with the highest zeal (else any non-ruler); EU5 has no zeal, a non-ruler character is picked, preferring clergy-estate and pious or zealous characters
			limit = {
				is_ruler = no
				NOT = { this = scope:finesse_thinker }
			}
			weight = {
				factor = 1
				modifier = {
					factor = 10
					OR = {
						has_estate = estate_type:clergy_estate
						pa_imp_char_is_religious_trigger = yes
					}
				}
			}
			save_scope_as = zeal_thinker
		}
		ordered_character = { #P:A TODO Approximate: Imperator research-office holder with the highest charisma (else any non-ruler); EU5 non-ruler character with the highest DIP
			limit = {
				is_ruler = no
				NOT = { this = scope:finesse_thinker }
				NOT = { this = scope:zeal_thinker }
			}
			order_by = dip
			save_scope_as = charisma_thinker
		}
	}

	option = { #heliocentric
		name = lifestyle_event_chains.15.a
		# add_research = { technology = religious_tech value = 15 } #P:A TODO No EU5 equivalent: +15 research in Imperator's religious technology track
		random_list = {
			50 = {
				ruler ?= { remove_trait = trait:zealot } # Imperator zealous
			}
			20 = {
				ruler ?= { add_trait = trait:obsessive_perfectionist } #P:A TODO Approximate: Imperator trait obsessive mapped to EU5 obsessive_perfectionist
