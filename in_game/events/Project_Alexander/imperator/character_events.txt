# Namespace renamed character_events -> pa_character_events (EU5 already has a character_events namespace).
# Imperator character events are converted to country events: root = the character's employer, the character is scope:pa_character.
namespace = pa_character_events

# Hidden: a wealthy character who is not head of his family dies; his fortune is contested (pa_character_events.2).
# Imperator fired this from on_character_death on the dead character. Written for EU5 on_character_death
# (root = country, scope:target = the dead character).
pa_character_events.1 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = scope:target
		scope:target = {
			employer = root # Imperator: exists = employer, country = employer
			# prisoner = no #P:A Removed: prisoner = no is always true in EU5
			# wealth > 1000 #P:A TODO No EU5 equivalent: the dead character's personal wealth is above 1000 (EU5 characters have no wealth)
			is_dynasty_head = no
		}
		NOT = {
			has_variable = character_events_1_flag
		}
		always = no #P:A TODO No EU5 equivalent: the event is about who inherits the dead character's personal fortune (wealth > 1000), and EU5 characters have no wealth
	}

	weight_multiplier = {
		base = 1
		# modifier = { factor = 3 wealth > 2000 } #P:A TODO No EU5 equivalent: 3x as likely if the fortune is above 2000 (EU5 characters have no wealth)
	}

	immediate = {
		scope:target = {
			save_scope_as = target
		}
		set_variable = {
			name = character_events_1_flag
			value = 1
		}
		trigger_event_non_silently = {
			id = pa_character_events.2
			days = 90
		}
	}
}

# Hidden: a character who moves to (or is captured by) another country becomes a foreign citizen there.
# Imperator fired this from on_being_captured and on_move_country on the character. EU5 has no captured characters.
# Meant for EU5 on_character_moved_country (root = character, scope:target = new country, scope:owner = old country):
# the on_action has to save the character as pa_character and fire this event on scope:target.
# No random-character fallback: the event only makes sense for the character who moved.
pa_character_events.4 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = scope:pa_character
	}

	immediate = {
		scope:pa_character = {
			if = {
				limit = {
					# is_governor = no #P:A Removed: EU5 has no governors, so this is always true
					is_mercenary_leader = no # Imperator is_mercenary
					is_alive = yes
					# trigger_if = { limit = { has_variable = came_from } NOT = { employer = { civil_war_with = root.var:came_from } } }
					#P:A TODO No EU5 equivalent: Imperator exempted a character who had joined the other side of a civil war against his former country (var:came_from was set by the Invictus on_move_country; EU5 has no country-vs-country civil war trigger)
					NOT = {
						has_character_modifier = foreign_citizen
					}
				}
				add_character_modifier = {
					modifier = foreign_citizen
					days = -1
				}
			}
		}
	}
}

# Hidden: a child born to foreign-citizen parents is a foreign citizen too.
# Imperator fired this from on_giving_birth (root = mother, scope:newborn). Written for EU5 on_character_birth
# (root = country, scope:character = the newborn); the mother is read from the newborn.
pa_character_events.5 = {
	type = country_event
	hidden = yes

	trigger = {
		exists = scope:character
		scope:character = {
			exists = mother
			mother = {
				OR = {
					any_spouse = { # spouse is a foreign citizen at the same court
						is_alive = yes
						employer = root # Imperator: employer = root.employer (the mother's employer)
						has_character_modifier = foreign_citizen
					}
					AND = { # spouse is dead, abroad or a citizen, and the mother is a foreign citizen
						any_spouse = {
							NOR = {
								is_alive = yes
								employer = root
								has_character_modifier = foreign_citizen
							}
						}
						has_character_modifier = foreign_citizen
					}
					AND = { # unmarried foreign-citizen mother
						num_of_spouses = 0 # Imperator: NOT = { exists = spouse }
						has_character_modifier = foreign_citizen
					}
				}
			}
		}
	}

	immediate = {
		scope:character = {
			add_character_modifier = {
				modifier = foreign_citizen
				days = -1
			}
		}
	}
}

# Hidden: an ostracized exile seizes the throne of the minor AI monarchy or tribe that took him in, and his home country hears of it (pa_character_events.52).
# Imperator fired this 1-2 years after ostracize_char (20% chance) on the exile (variable ostracized_from = the country that ostracized him).
# Written as a country event on the exile's new employer with scope:pa_character = the exile; without a caller it finds an exile itself.
pa_character_events.51 = {
	type = country_event
	hidden = yes

	trigger = {
		is_ai = yes
		OR = {
			government_type = government_type:monarchy # Imperator is_monarchy
			government_type = government_type:tribe # Imperator is_tribal
		}
		num_locations <= 100 # Imperator num_of_cities
		NOT = {
			pa_imp_is_diadochi_trigger = yes
		}
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				employer = root
				is_alive = yes
				has_variable = ostracized_from
				var:ostracized_from = {
					exists = capital # Imperator has_land
				}
			}
		}
		trigger_else = {
			any_character = {
				is_alive = yes
				has_variable = ostracized_from
				var:ostracized_from = {
					exists = capital
				}
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					is_alive = yes
					has_variable = ostracized_from
					var:ostracized_from = {
						exists = capital
					}
				}
				save_scope_as = pa_character
			}
		}
		scope:pa_character = {
			save_scope_as = coup_leader
			# set_as_minor_character = this #P:A TODO No EU5 equivalent: the exile stops being a minor character (EU5 has no minor characters)
		}
		set_new_ruler = scope:coup_leader # Imperator set_as_ruler
		scope:coup_leader = { #P:A TODO Approximate: Imperator create_family made the exile the head of a new noble family in this country; EU5 founds a new dynasty for him only if he has none
			if = {
				limit = {
					NOT = { exists = dynasty }
				}
				found_dynasty = random # as vanilla DHE/flavor_DLH.txt
			}
		}
		save_scope_as = coup_target
		scope:coup_leader.var:ostracized_from = {
			save_scope_as = from_over_here
		}
		scope:from_over_here = {
			trigger_event_non_silently = pa_character_events.52
		}
	}
}
