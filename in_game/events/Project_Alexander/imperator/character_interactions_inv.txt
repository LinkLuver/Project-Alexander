character_interactions_inv.3 = {
	type = country_event
	title = character_interactions_inv_3_title
	desc = character_interactions_inv_3_desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		exists = scope:target
	}

	immediate = {
		ruler ?= { save_scope_as = pa_ruler }
	}

	option = {	#Scope:target will be forced to retire
		name = character_interactions_inv_3.a
		# show_as_tooltip = { pay_price = price_force_retirement }
		show_as_tooltip = {
			add_gold = -50 # gold part of price_force_retirement
		}
		#P:A TODO No EU5 equivalent: price_force_retirement also cost 30 political influence
		scope:target = {
			custom_tooltip = character_interactions_inv_3_A_tooltip
			show_as_tooltip = {
				# add_loyalty = forced_to_retire
				add_character_modifier = {
					modifier = forced_into_retirement
					days = -1
				}
				# if = { limit = { exists = scope:recipient } scope:target = { add_holding = scope:recipient } }
			}
			custom_tooltip = character_interactions_inv_3_B_tooltip
			# show_as_tooltip = { add_loyalty = refused_retirement }
			#P:A TODO No EU5 equivalent: the accept/refuse tooltips listed loyalty changes (EU5 characters have no loyalty)
			hidden_effect = {
				random_list = {
					90 = {
						set_variable = accepted_retirement
						# add_loyalty = forced_to_retire
						#P:A TODO No EU5 equivalent: forced_to_retire loyalty change (EU5 characters have no loyalty)
						add_character_modifier = {
							modifier = forced_into_retirement
							days = -1
						}
						# if = { limit = { exists = scope:recipient } scope:target = { add_holding = scope:recipient } }
						#P:A TODO No EU5 equivalent: the retiree received a holding in scope:recipient's lands (EU5 has no holdings)
						if = {
							limit = {
								scope:target = {
									OR = {
										is_general = yes
										is_admiral = yes
									}
								}
							}
							remove_commander = scope:target
						}
						# else_if = { limit = { scope:target = { is_governor = yes } } remove_as_governor = yes }
						# else_if = { limit = { scope:target = { OR = { has_tech_office = yes has_any_office = yes } } } remove_all_offices = yes }
						#P:A TODO No EU5 equivalent: a retiring governor or office holder lost the governorship/offices (EU5 has no governors or offices)
					}
					15 = {
						set_variable = rejected_retirement
						# add_loyalty = refused_retirement
						#P:A TODO No EU5 equivalent: refused_retirement loyalty change (EU5 characters have no loyalty)
					}
				}
			}
			# add_gold = 50
			#P:A TODO No EU5 equivalent: +50 personal wealth for the target (EU5 characters have no wealth)
		}
		trigger_event_non_silently = {
			id = character_interactions_inv.4
			days = 5
		}
	}
	option = {	#On second thought this isn't a good idea
		name = character_interactions_inv_3.b
	}
}

# Hidden: the ruler of a supported client state changed (Imperator on_ruler_change; EU5 on_new_ruler:
# root = country, scope:old_ruler, scope:new_ruler). Imperator's on_action only fired it for countries with
# accepted_vassalage_1 and saved scope:supported_country / scope:dead_supported_ruler; done here instead.
character_interactions_inv.8 = {
	type = country_event
	hidden = yes

	trigger = {
		has_variable = accepted_vassalage_1
	}

	immediate = {
		save_scope_as = supported_country
		scope:old_ruler ?= { save_scope_as = dead_supported_ruler }
		if = {
			limit = {
				is_subject = yes
				NOT = {
					ruler ?= {
						has_character_modifier = ruler_being_supported
					}
				}
			}
			overlord = { trigger_event_non_silently = character_interactions_inv.7 }
		}
		else = {
			remove_variable = accepted_vassalage_1
		}
	}
}

ambitions_inv.1 = { #Martial
	type = country_event
	title = ambitions_inv_1_title
	desc = ambitions_inv_1_desc
	outcome = neutral
	illustration_tags = {
		10 = interior
		10 = happy
	}

	trigger = {
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = { is_married = yes }
			any_character = {
				mil >= scope:pa_character.mil
				is_adult = yes
				is_ruler = no
				#P:A Removed: prisoner = no is always true in EU5
			}
		}
		trigger_else = {
			ruler ?= { is_married = yes }
			any_character = {
				mil >= root.ruler.mil
				is_adult = yes
				is_ruler = no
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			ruler = { save_scope_as = pa_character }
		}
		random_character = {
			limit = {
				mil >= scope:pa_character.mil
				is_adult = yes
				is_ruler = no
			}
			save_scope_as = scheme_target
		}
		random_owned_location = {
			limit = { is_capital = no }
			save_scope_as = your_province_scope
		}
	}

	option = {
		name = ambitions_inv_1.a
		# add_gold = -50
		#P:A TODO No EU5 equivalent: the ruler paid 50 from personal wealth (EU5 characters have no wealth)
		# add_popularity = 5
		#P:A TODO No EU5 equivalent: +5 ruler popularity
		# scope:scheme_target = { add_loyalty = more_drinks }
		#P:A TODO No EU5 equivalent: more_drinks loyalty bonus for the host (EU5 characters have no loyalty)
	}
	option = {
		name = ambitions_inv_1.b
		# add_popularity = 5
		#P:A TODO No EU5 equivalent: +5 ruler popularity
		# scope:scheme_target = { add_loyalty = dishonored_by_ruler }
		#P:A TODO No EU5 equivalent: dishonored_by_ruler loyalty malus for the host
	}
	option = {	#Fight to the death
		name = ambitions_inv_1.c
		trigger = {
			scope:pa_character = { mil > 45 } # Imperator martial > 9
		}
		custom_tooltip = ambitions_inv_1_tt

			random_list = {
				1 = {
					kill_character = {
						target = scope:pa_character
						reason = combat # Imperator death_battle
					}
				}
				1 = {
					kill_character = {
						target = scope:scheme_target
						reason = combat # Imperator death_battle
					}
