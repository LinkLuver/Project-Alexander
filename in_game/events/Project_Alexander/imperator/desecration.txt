# The owner of a sacked shrine learns of the desecration.
desecration.1 = {
	type = location_event
	title = desecration.1.t
	desc = desecration.1.desc
	outcome = negative
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	immediate = {
		show_as_tooltip = {
			scope:shrine_location = {
				# remove_province_deity = yes #P:A TODO No EU5 equivalent: removes the location's patron deity (EU5 locations have no deities)
				add_location_modifier = {
					modifier = upset_local_priests
					days = 365
				}
			}
		}
	}

	option = {
		name = desecration.1.a
		owner = {
			add_opinion = {
				target = scope:sacker
				modifier = sacked_our_holy_site
			}
		}
	}
}
