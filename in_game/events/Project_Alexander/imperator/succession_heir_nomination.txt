# Heir nomination ("anoint heir") events. In Imperator heir_events.5/10/11 were fired by the scripted GUI anoint_heir
# on the ruler's country after it made a character heir apparent: scope:nominee = the new heir, scope:lost_heir = the
# previous heir, scope:nominator = the ruler. The caller must save scope:nominee (and scope:lost_heir when there was a
# previous heir) before firing them; scope:nominator falls back to the ruler.
# Not imported (bound to specific tags or dynasties): heir_events.1/2/3/12/13/14/15 (regnal names of the Seleukid,
# Lagid, Mithridatid, Zipoeteid, Attalid and Teukrid dynasties) and heir_events.6/7/8/9 (EGY, SEL, Antigonid, MAC/EPI
# coronations).
namespace = heir_events

# An heir apparent is crowned over the siblings: a sibling is jealous.
heir_events.5 = {
	type = country_event
	title = heir_events.5.t
	desc = heir_events.5.desc
	illustration_tags = {
		10 = interior
		10 = happy
	}

	trigger = {
		exists = scope:nominee
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:nominator } }
			ruler = { save_scope_as = nominator }
		}
		hidden_effect = {
			random_character = {
				limit = {
					is_eligible_heir = root #P:A TODO Approximate: Imperator random_successor (a character in the line of succession); EU5 picks any eligible heir of the country
					NOT = { has_character_modifier = nominated_heir_modifier }
					NOT = { this = scope:nominee } # in Imperator the nominee always carried nominated_heir_modifier
					trigger_if = {
						limit = { exists = scope:lost_heir }
						NOT = { this = scope:lost_heir }
					}
					age_in_years > 5
				}
				save_scope_as = angry_sibling
			}
			# random_character = { limit = { is_head_of_family = yes is_ruler = no loyalty < 55 } save_scope_as = conspiring_hof }
			#P:A TODO No EU5 equivalent: picked a disloyal family head (loyalty < 55) as scope:conspiring_hof; only used for add_friend below
		}
	}

	option = {
		name = heir_events.5.a
		# if = {
		# 	limit = { exists = scope:angry_sibling }
		# 	scope:angry_sibling = {
		# 		add_loyalty = jealous_sibling
		# 		if = { limit = { exists = scope:conspiring_hof } add_friend = scope:conspiring_hof }
		# 	}
		# }
		#P:A TODO No EU5 equivalent: the jealous sibling lost loyalty (jealous_sibling) and befriended a disloyal family head (EU5 characters have no loyalty or friends)
		# scope:lost_heir = { add_loyalty = jealous_sibling }
		#P:A TODO No EU5 equivalent: the previous heir lost loyalty (jealous_sibling; EU5 characters have no loyalty)
	}
}
