# Wild Accusations: an opposition leader accuses the corrupt ruler of election fraud.
ip_republic.1 = {
	type = country_event
	title = ip_republic.1.t
	desc = ip_republic.1.desc
	outcome = negative
	illustration_tags = {
		10 = interior
		10 = angry
	}

	trigger = {
		government_type = government_type:republic
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				is_ruler = yes
				# corruption >= 5
				NOT = { has_trait = honest } #P:A TODO Approximate: Imperator corruption >= 5 (EU5 characters have no corruption); "not honest" stands in
				NOT = {
					has_variable = under_fraud_scrutiny
				}
			}
		}
		trigger_else = {
			any_character = {
				is_ruler = yes
				NOT = { has_trait = honest } #P:A TODO Approximate: Imperator corruption >= 5 (EU5 characters have no corruption); "not honest" stands in
				NOT = {
					has_variable = under_fraud_scrutiny
				}
			}
		}
		# employer = { any_character = { is_party_leader = yes NOT = { is_same_party_as = ROOT } } }
		any_character = { #P:A TODO Approximate: Imperator "a leader of another party"; EU5 has no parties, an adult character of another estate than the ruler stands in
			is_adult = yes
			is_ruler = no
			NOT = { has_estate = root.ruler.estate_type }
		}
	}

	weight_multiplier = {
		base = 1
		modifier = {
			factor = 3
			# corruption >= 20
			ruler = { has_trait = corrupt } #P:A TODO Approximate: Imperator corruption >= 20; the corrupt trait stands in
		}
		# modifier = { factor = 3 any_rival = { is_party_leader = yes NOT = { is_same_party_as = ROOT } } }
		#P:A TODO No EU5 equivalent: x3 if the ruler has a rival who leads another party (EU5 has no character rivals or parties)
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					is_ruler = yes
					NOT = { has_trait = honest }
					NOT = {
						has_variable = under_fraud_scrutiny
					}
				}
				save_scope_as = pa_character
			}
		}
		random_character = { #P:A TODO Approximate: Imperator picked a leader of another party, weighted by office importance (title_importance_modifier); EU5 has no parties or offices
			limit = {
				is_adult = yes
				is_ruler = no
				NOT = { has_estate = scope:pa_character.estate_type }
			}
			save_scope_as = ip_republic_1_accuser
		}
		scope:pa_character = {
			set_variable = {
				name = under_fraud_scrutiny
				value = 1
			}
		}
	}

	option = {
		name = ip_republic.1.a
		custom_tooltip = increased_scrutiny.tooltip
		# scope:pa_character = { add_popularity = -5 } # subtract_popularity_small
		#P:A TODO No EU5 equivalent: -5 popularity for the ruler (EU5 characters have no popularity)
	}

	option = {
		name = ip_republic.1.b
		custom_tooltip = ip_republic.1.b.tooltip
		# scope:pa_character = { add_popularity = -5 } # subtract_popularity_small
		#P:A TODO No EU5 equivalent: -5 popularity for the ruler (EU5 characters have no popularity)
		add_gold = {
			value = monthly_income_total
			multiply = pa_imp_corruption_svalue_halved
			max = 5000
		}
		hidden_effect = {
			scope:pa_character = {
				remove_variable = under_fraud_scrutiny
			}
		}
	}

	option = {
		name = ip_republic.1.c
		# add_tyranny = 10
		#P:A TODO No EU5 equivalent: +10 tyranny (EU5 has no tyranny)
		# scope:ip_republic_1_accuser = { add_loyalty = loyalty_demanded_resignation add_prominence = 20 } # prominence_huge
		#P:A TODO No EU5 equivalent: the accuser gains the "demanded resignation" loyalty modifier and +20 prominence (EU5 characters have neither)
		hidden_effect = {
			scope:pa_character = {
				remove_variable = under_fraud_scrutiny
			}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }

		}

				base = 1

# Hidden: a wealthy office holder is caught embezzling; the ruler decides in ip_republic.5.
ip_republic.4 = {
	type = country_event
	hidden = yes

	trigger = {
		government_type = government_type:republic
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 has no offices, a cabinet seat stands in
				# wealth >= 1000
				OR = { #P:A TODO Approximate: Imperator personal wealth >= 1000 (EU5 characters have no wealth); a corrupt, greedy or embezzler trait stands in
					has_trait = corrupt
					has_trait = greedy
					has_trait = embezzler
				}
				is_ruler = no
				NOT = {
					has_variable = ip_republic_4_embezzler
				}
			}
		}
		trigger_else = {
			any_character = {
				in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 has no offices, a cabinet seat stands in
				OR = { #P:A TODO Approximate: Imperator personal wealth >= 1000 (EU5 characters have no wealth); a corrupt, greedy or embezzler trait stands in
					has_trait = corrupt
					has_trait = greedy
					has_trait = embezzler
				}
				is_ruler = no
				NOT = {
					has_variable = ip_republic_4_embezzler
				}
			}
		}
		exists = ruler
	}

	weight_multiplier = {
		base = 1
		# modifier = { factor = 2 has_trait = gluttonous }
		#P:A TODO No EU5 equivalent: x2 if the character is gluttonous (EU5 has no such trait)
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					in_cabinet = yes
					OR = {
						has_trait = corrupt
						has_trait = greedy
						has_trait = embezzler
					}
					is_ruler = no
					NOT = {
						has_variable = ip_republic_4_embezzler
					}
				}
				save_scope_as = pa_character
			}
		}
		scope:pa_character = {
			save_scope_as = ip_republic_5_antagonist
			set_variable = {
				name = ip_republic_4_embezzler
				days = 3600
			}
		}
		# Imperator: employer = { current_ruler = { trigger_event = { id = ip_republic.5 days = 0 } } }
		ruler = {
			save_scope_as = pa_character
		}
		trigger_event_non_silently = {
			id = ip_republic.5
			days = 0
		}
	}
}
