# Fugitives from a conquered country reach us (first banishment destination of annexation.3, option b).
# Fired by annexation.3 on the receiving country; scope:annexer, scope:annexee and scope:banish_first/second/third
# travel with it.
annexation.1 = {
	type = country_event
	title = annexation.1.t
	desc = annexation.1.desc
	illustration_tags = {
		10 = exterior
		10 = armed
	}

	immediate = {
		ruler ?= {
			save_scope_as = current_ruler # left_portrait = current_ruler
		}
	}

	option = {
		name = annexation.1.a
		# scope:banish_first = { add_loyalty = loyalty_offered_refuge set_home_country = ROOT set_as_minor_character = THIS }
		# scope:banish_second = { add_loyalty = loyalty_offered_refuge set_home_country = ROOT set_as_minor_character = THIS }
		# scope:banish_third = { add_loyalty = loyalty_offered_refuge set_home_country = ROOT set_as_minor_character = THIS }
		#P:A TODO No EU5 equivalent: the refugees gained loyalty (loyalty_offered_refuge), had their home country set to us and became minor characters (EU5 has no character loyalty, home country or minor characters); annexation.3 already moved them to our country
		ai_chance = {
			base = 20
		}
	}

	option = {
		name = annexation.1.b
		trigger = {
			NAND = {
				scope:banish_first ?= {
					dynasty ?= root.ruler.dynasty # has_same_family = root.current_ruler
				}
				scope:banish_second ?= {
					dynasty ?= root.ruler.dynasty
				}
				scope:banish_third ?= {
					dynasty ?= root.ruler.dynasty
				}
			}
		}
		if = {
			limit = {
				scope:banish_first ?= {
					is_alive = yes
					NOT = { dynasty ?= root.ruler.dynasty }
				}
			}
			kill_character = { #P:A TODO Approximate: Imperator death reason "crucified"; EU5 execution used
				target = scope:banish_first
				reason = execution
			}
		}
		if = {
			limit = {
				scope:banish_second ?= {
					is_alive = yes
					NOT = { dynasty ?= root.ruler.dynasty }
				}
			}
			kill_character = { #P:A TODO Approximate: Imperator death reason "crucified"; EU5 execution used
				target = scope:banish_second
				reason = execution
			}
		}
		if = {
			limit = {
				scope:banish_third ?= {
					is_alive = yes
					NOT = { dynasty ?= root.ruler.dynasty }
				}
			}
			kill_character = { #P:A TODO Approximate: Imperator death reason "crucified"; EU5 execution used
				target = scope:banish_third
				reason = execution
			}
		}
		scope:annexer ?= {
			add_opinion = {
				modifier = killed_fugitives
				target = root
			}
		}
		ai_chance = {
			base = 80
		}
	}
}

# The elite of a country we conquered awaits our judgment.
# Imperator fired it from on_annexation_event_effect (annexation by conquest, i.e. the annexee was not our subject).
# Meant for on_military_annex: root = annexing country, scope:target = annexed country (fired before the annexation).
#P:A TODO Approximate: Imperator death reasons are mapped in this event as slaughtered_during_conquest -> country_annexed, obscurity / mysteriously_retired -> vanished, sacrifice -> sacrifice; EU5 kill_character has no silent mode (the kills sit in hidden_effect)
annexation.3 = {
	type = country_event
	title = annexation.3.t
	desc = annexation.3.desc
	illustration_tags = {
		10 = exterior
		10 = angry
	}

	trigger = {
		OR = {
			exists = scope:annexee
			exists = scope:target
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:annexer } }
			save_scope_as = annexer
		}
		if = {
			limit = { NOT = { exists = scope:annexee } }
			scope:target = {
				save_scope_as = annexee
			}
		}
		# scope:annexee = { every_country_treasure = { add_to_list = annexed_treasures } }
		#P:A TODO No EU5 equivalent: the annexed country's treasures were listed here and handed to the annexer at the end of immediate (EU5 has no treasures)
		hidden_effect = {
			# scope:annexer = {
			# 	every_character = {
			# 		limit = { prisoner = yes has_variable = came_from var:came_from = scope:annexee }
			# 		set_as_minor_character = THIS
			# 		add_to_list = imprisonment_list
			# 	}
			# }
			#P:A TODO No EU5 equivalent: prisoners from the annexed country held by the annexer joined the imprisonment list (EU5 has no prisoners)
			# every_living_character = { limit = { has_variable = came_from var:came_from = scope:annexee } set_as_minor_character = THIS }
			#P:A TODO No EU5 equivalent: characters who came from the annexed country became minor characters (EU5 has no minor characters)

			scope:annexee = {

				#Much of this logic is replicated in the on_diplo_annexation_event_effect on_action, please don't change this without also changing that.
				#Comments should also be kept similar.

				#This is where we send foreigners home:
				# if = {
				# 	limit = { any_character = { has_variable = came_from var:came_from = scope:annexer prisoner = yes } }
				# 	every_character = {
				# 		limit = { has_variable = came_from var:came_from = scope:annexer prisoner = yes }
				# 		scope:annexee = { release_prisoner = { target = prev } }
				# 		remove_all_positions = yes
				# 		move_country_with_message = scope:annexer
				# 		if = { limit = { has_character_modifier = foreign_citizen } remove_character_modifier = foreign_citizen }
				# 	}
				# }
				#P:A TODO No EU5 equivalent: our citizens held prisoner by the annexed country were released and sent home (EU5 has no prisoners)

				#Mercenaries also go Home:
				# if = {
				# 	limit = { any_character = { is_mercenary = yes OR = { is_general = yes is_admiral = yes } } }
				# 	every_character = { limit = { is_mercenary = yes OR = { is_general = yes is_admiral = yes } } move_country = c:MER }
				# }
				#P:A TODO No EU5 equivalent: mercenary commanders returned to the Imperator mercenary country MER (EU5 mercenaries are companies, not characters of a country)

				#This is finding where banished people go:
				if = {
					limit = {
						scope:annexer = {
							any_country_in_diplomatic_range = { # in_diplomatic_range
								NOR = {
									this = scope:annexer
									this = scope:annexee
								}
							}
						}
					}
					scope:annexer = {
						random_country_in_diplomatic_range = {
							limit = {
								NOR = {
									this = scope:annexer
									this = scope:annexee
								}
							}
							weight = {
								base = 1
								modifier = {
									factor = 50
									has_mutual_scripted_relation = { # alliance_with = scope:annexee
										type = relation_type:alliance
										target = scope:annexee
									}
								}
								modifier = {
									factor = 100
									opinion = {
										target = scope:annexee
										value > 80
									}
								}
							}
							save_scope_as = banishment_country
						}
					}
				}

				#This is finding the second country for banished people to go:
				if = {
					limit = {
						exists = scope:banishment_country
						scope:annexer = {
							any_country_in_diplomatic_range = {
								NOR = {
									this = scope:banishment_country
									this = scope:annexer
									this = scope:annexee
								}
							}
						}
					}
					scope:annexer = {
						random_country_in_diplomatic_range = {
							limit = {
								NOR = {
									this = scope:banishment_country
									this = scope:annexer
									this = scope:annexee
								}
							}
							weight = {
								base = 1
								modifier = {
									factor = 50
									has_mutual_scripted_relation = {
										type = relation_type:alliance
										target = scope:annexee
									}
								}
								modifier = {
									factor = 100
									opinion = {
										target = scope:annexee
										value > 80
									}
								}
							}
							save_scope_as = banishment_country_2
						}
					}
				}

				#Banishment Logic:
				#First these characters are sent to the Annexer - Depending on choices they are later killed or Banished.

				#First Destination:
				if = {
					limit = {
						exists = scope:banishment_country #First Banishment Destination
					}
					ordered_character = { #Destined for the first Banishment Destination
						limit = {
							is_alive = yes
							NOT = { is_in_list = banishment_list }
						}
						order_by = pa_imp_banishment_importance_1
						check_range_bounds = no
						max = 3
						if = {
							limit = { NOT = { exists = scope:banish_first } }
							save_scope_as = banish_first
						}
						else_if = {
							limit = { NOT = { exists = scope:banish_second } }
							save_scope_as = banish_second
						}
						else_if = {
							limit = { NOT = { exists = scope:banish_third } }
							save_scope_as = banish_third
						}
						add_to_list = banishment_list
					}
				}

				#Second Destination:
				if = {
					limit = {
						exists = scope:banishment_country_2 #Second Banishment Destination
					}
					ordered_character = { #Destined for the second Banishment Destination
						limit = {
							is_alive = yes
							NOT = { is_in_list = banishment_list }
						}
						order_by = pa_imp_banishment_importance_2
						check_range_bounds = no
						max = 3
						if = {
							limit = { NOT = { exists = scope:banish_2_first } }
							save_scope_as = banish_2_first
						}
						else_if = {
							limit = { NOT = { exists = scope:banish_2_second } }
							save_scope_as = banish_2_second
						}
						else_if = {
							limit = { NOT = { exists = scope:banish_2_third } }
							save_scope_as = banish_2_third
						}
						add_to_list = banishment_list

				#These are people we can imprison:
				ordered_character = { #Prison Candidates
					limit = {
						is_alive = yes
						is_ruler = no
						NOT = { is_in_list = imprisonment_list }
					}
					check_range_bounds = no
					order_by = pa_imp_importance_value
					max = 2
					add_to_list = imprisonment_list
				}
				if = { #Ruler also goes to Prison
					limit = {
						exists = ruler
					}
					ruler = {
						save_scope_as = ruler_target_annex
						if = {
							limit = {
								OR = {
									is_general = yes
									is_admiral = yes
								}
							}
							remove_commander = scope:ruler_target_annex # remove_command = yes
						}
						# if = { limit = { scope:annexer = { exists = current_ruler } is_rival = scope:annexer.current_ruler } remove_rival = scope:annexer.current_ruler }
						#P:A TODO No EU5 equivalent: ended a rivalry between the conquered ruler and our ruler (EU5 has no character rivals)
						if = {
							limit = {
								is_alive = yes
								NOT = { is_in_list = imprisonment_list }
							}
							add_to_list = imprisonment_list
						}
					}
				}

				#Families!

				#This is where we find our candidate families to accept:
				ordered_dynasty = {
					limit = {
						any_character_in_dynasty = {
							is_alive = yes
							employer = scope:annexee
						}
					}
					order_by = total_dynastic_power #P:A TODO Approximate: Imperator ordered families by family prestige; EU5 dynasties have no prestige, total dynastic power used
					max = 4
					check_range_bounds = no
					if = {
						limit = { NOT = { exists = scope:first_family } }
						save_scope_as = first_family
					}
					else_if = {
						limit = { NOT = { exists = scope:second_family } }
						save_scope_as = second_family
					}
					else_if = {
						limit = { NOT = { exists = scope:third_family } }
						save_scope_as = third_family
					}
					else_if = {
						limit = { NOT = { exists = scope:fourth_family } }
						save_scope_as = fourth_family
					}
				}

				#These are members of the Annexed country's great families that we may be accepting:
				# EU5 dynasties span countries (Imperator families belong to one country), so every pick below is limited to
				# characters of the annexed country.
				#P:A TODO Approximate: when the EU5 dynasty head serves another country, the dynasty's most important member in the annexed country stands in as the family head

							limit = {
								dynasty_head ?= {
									is_alive = yes
									employer = scope:annexee
								}

								save_scope_as = first_family_head
							}
						}
						else = {
							ordered_character_in_dynasty = {
								limit = { employer = scope:annexee }
								order_by = pa_imp_importance_value
								max = 1
								check_range_bounds = no
								save_scope_as = first_family_head
							}
						}
						scope:first_family_head ?= {
							add_to_list = first_annex_family_list

								limit = { employer = scope:annexee }
								add_to_list = first_annex_family_list
							}

							check_range_bounds = no

								value = adm
								add = dip
								add = mil
							}
							add_to_list = first_annex_family_list
							every_spouse = {
								limit = { employer = scope:annexee }
								add_to_list = first_annex_family_list
							}
						}
						ordered_character_in_dynasty = {
							limit = {
								employer = scope:annexee
								is_female = yes
								age_in_years < 40
							}
							order_by = {
								value = adm
								add = dip
								add = mil
								if = {

										age_in_years < 35

									}

								}
							}
							check_range_bounds = no
							max = 2
							add_to_list = first_annex_family_list
						}
					}

								dynasty_head ?= {

					any_character_in_dynasty = {
						is_alive = yes
						employer = scope:annexee
					}
				}

		}

			exists = scope:target

				any_dynasty = {
					any_character_in_dynasty = {

					}

								add = dip
								add = mil
								if = {
									limit = {
										age_in_years < 35
										is_married = no
									}
									add = 1000
								}
								if = {
									limit = { pa_imp_has_bloodline_trigger = yes }
									add = 1000
								}
								if = {
									limit = { age_in_years > 12 }
									add = 500
								}
							}
							check_range_bounds = no
							max = 2
							add_to_list = third_annex_family_list
						}
					}
				}
				#Fourth Family:
				if = {
					limit = {
						exists = scope:fourth_family
					}
					scope:fourth_family = {
						if = {
							limit = {
								dynasty_head ?= {
									is_alive = yes
									employer = scope:annexee
								}
							}
							dynasty_head = {
								save_scope_as = fourth_family_head
							}
						}
						else = {
							ordered_character_in_dynasty = {
								limit = { employer = scope:annexee }
								order_by = pa_imp_importance_value
								max = 1
								check_range_bounds = no
								save_scope_as = fourth_family_head
							}
						}
						scope:fourth_family_head ?= {
							add_to_list = fourth_annex_family_list
							every_spouse = {
								limit = { employer = scope:annexee }
								add_to_list = fourth_annex_family_list
							}
						}
						ordered_character_in_dynasty = {
							limit = {
								employer = scope:annexee
								NOT = { is_in_list = fourth_annex_family_list }
							}
							max = 2
							check_range_bounds = no
							order_by = {
								value = adm
								add = dip
								add = mil
							}
							add_to_list = fourth_annex_family_list
							every_spouse = {
								limit = { employer = scope:annexee }
								add_to_list = fourth_annex_family_list
							}
						}
						ordered_character_in_dynasty = {
							limit = {
								employer = scope:annexee
								is_female = yes
								age_in_years < 40
							}
							order_by = {
								value = adm
								add = dip
								add = mil
								if = {
									limit = {
										age_in_years < 35
										is_married = no
									}
									add = 1000
								}
								if = {
									limit = { pa_imp_has_bloodline_trigger = yes }
									add = 1000
								}
								if = {
									limit = { age_in_years > 12 }
									add = 500
								}
							}
							check_range_bounds = no
							max = 2
							add_to_list = fourth_annex_family_list
						}
						save_scope_as = family_4
					}
				}

				every_character = { #Now we move everyone of interest:
					limit = {
						OR = {
							is_in_list = first_annex_family_list
							is_in_list = second_annex_family_list
							is_in_list = third_annex_family_list
							is_in_list = fourth_annex_family_list
						}
					}
					pa_imp_remove_all_positions = yes
					move_country = scope:annexer
					# set_as_minor_character = THIS
					# set_home_country = scope:annexer
					#P:A TODO No EU5 equivalent: the moved characters became minor characters with the annexer as home country (EU5 has no minor characters or home country)
				}

				every_character = { #The rest vanish into obscurity.
					limit = {
						NOR = {
							is_in_list = first_annex_family_list
							is_in_list = second_annex_family_list
							is_in_list = third_annex_family_list
							is_in_list = fourth_annex_family_list
						}
					}
					kill_character = {
						target = this
						reason = vanished # death_obscurity
					}
				}
				every_character = {
					kill_character = {
						target = this
						reason = vanished # death_obscurity
					}
				}
			}
		}
		# every_in_list = { list = annexed_treasures transfer_treasure_to_country = ROOT }
		#P:A TODO No EU5 equivalent: treasures of the annexed country go to us (EU5 has no treasures)
	}

					limit = { has_character_modifier = foreign_citizen }
					remove_character_modifier = foreign_citizen

				pa_imp_annexation_character_cleanup = yes #This cleans up all the variables so we don't track them down and kill them by mistake later.
			}

				limit = {
					any_character = {

							is_in_list = second_annex_family_list

							is_in_list = fourth_annex_family_list
						}
					}
				}
				set_variable = integration_event_check
				trigger_event_non_silently = annexation.4
