#P:A Imported from Invictus: events/befriend.txt
# The Invictus "befriend" chain: a ruler tries to befriend a character (Imperator character interaction, scripted effect befriend_effect).
# befriend.1-6 were drawn at random by the custom on_action befriend_pulse (fallback befriend_fail -> befriend.8),
# befriend.7/8 by befriend_resolution (fallback befriend_end -> befriend.9); befriend.10-12 are country events.
# Imperator character events are country events here; the character is scope:pa_character.
#P:A TODO EU5 characters have no friends/rivals, no loyalty and no befriend interaction; see the TODOs per line.
#P:A TODO goto_location has no EU5 equivalent; dropped from every event in this file
namespace = befriend

# Imperator note: each of befriend.1-6 is one step of a befriend attempt (3 steps per attempt); the options add progress through
# pa_imp_set_friend_progress_*_effect and every option fires pa_imp_fire_next_friend_event_effect. befriend.7 fires on success, befriend.8 on failure,
# befriend.9 cleans up. In Imperator the chain started from the befriend character interaction (befriend_effect), which set the
# character variable currently_befriending = <target>; EU5 has no such interaction yet.

# Political Ambitions: the target wants an office
befriend.1 = {
	type = country_event
	title = befriend.1.t
	desc = befriend.1.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		trigger_if = {
			limit = { exists = scope:pa_character }
			scope:pa_character = {
				has_variable = currently_befriending
				exists = var:currently_befriending
				NOT = {
					has_variable = befriend_toggle_1
				}
				# can_get_friends = yes #P:A TODO No EU5 equivalent: room for another friend (EU5 characters have no friends)
				# NOT = { is_rival = var:currently_befriending } #P:A TODO No EU5 equivalent: character rivals
				var:currently_befriending = {
					employer = root
					in_cabinet = no #P:A TODO Approximate: Imperator has_any_office = no -> not in the cabinet
					# is_governor = no #P:A Removed: EU5 has no governors, so this is always true
					is_general = no
					is_admiral = no
				}
			}
		}
		trigger_else = {
			any_character = {
				has_variable = currently_befriending
				exists = var:currently_befriending
				NOT = {
					has_variable = befriend_toggle_1
				}
				var:currently_befriending = {
					employer = root
					in_cabinet = no
					is_general = no
					is_admiral = no
				}
			}
		}
	}

	immediate = {
		if = {
			limit = { NOT = { exists = scope:pa_character } }
			random_character = {
				limit = {
					has_variable = currently_befriending
					exists = var:currently_befriending
					NOT = {
						has_variable = befriend_toggle_1
					}
					var:currently_befriending = {
						employer = root
						in_cabinet = no
						is_general = no
						is_admiral = no
					}
				}
				save_scope_as = pa_character
			}
		}
		scope:pa_character = {
			set_variable = {
				name = befriend_toggle_1
				days = 720
			}
			var:currently_befriending = {
				save_scope_as = friend_target
				save_scope_as = demand_office_event_scope
			}
			pa_imp_befriend_event_counter_variable_effect = yes
		}
		pa_imp_demand_office_skill_effect = yes # Picks which office schemer wants based on their best skill
		pa_imp_demand_office_master_scope_effect = yes # Sets scope of disputed office's current holder (if they exist) (Imperator: on the employer, i.e. root)
	}

	option = {
		name = befriend.1.a
		scope:pa_character = {
			pa_imp_set_friend_progress_large_effect = yes
		}
		pa_imp_demand_office_transfer_effect = yes # Imperator: employer = { ... }
		# if = {
		# 	limit = { exists = scope:demand_office_old_officer_scope }
		# 	scope:demand_office_old_officer_scope = { add_loyalty = loyalty_removed_office }
		# }
		# else = {
		# 	add_popularity = subtract_popularity_medium
		# }
		#P:A TODO No EU5 equivalent: the displaced office holder gets the "removed from office" loyalty modifier, or with no holder the befriending character loses 10 popularity (EU5 characters have no loyalty or popularity)
		# ROOT.employer = {
		# 	add_political_influence = subtract_influence_small
		# 	add_tyranny = 5.0
		# }
		#P:A TODO No EU5 equivalent: -10 political influence and +5 tyranny
		scope:pa_character = {
			pa_imp_fire_next_friend_event_effect = yes
		}

		ai_chance = {
			modifier = {
				factor = 1000
				scope:pa_character = { pa_imp_ruler_befriending_disloyal_powerful_character = yes }
				# root.employer = {
				# 	political_influence >= influence_medium # Must be 5 PI left for e.g. bribing
				# 	tyranny < 25
				# }
				#P:A TODO No EU5 equivalent: at least 15 political influence and tyranny below 25
			}
		}
	}

	option = {
		name = befriend.1.b
		scope:pa_character = {
			pa_imp_fire_next_friend_event_effect = yes
		}
	}
}
