# Some people of another faith convert in one of our locations; an advisor proposes converting them all.
rel_flavor_eastern.1 = {
	type = country_event
	title = rel_flavor_eastern.1.t
	desc = rel_flavor_eastern.1.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		pa_imp_country_eastern_religion_trigger = yes
		NOT = { has_variable = rel_flavor_eastern_1_cooldown }
		any_character = {
			is_ruler = no
			in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 has no offices, cabinet membership stands in
		}
		any_owned_location = {
			NOT = { has_location_modifier = province_improvement_eastern_religion_peaceful_relations }
			dominant_religion = root.religion # Imperator dominant_province_religion
			any_pop = {
				count > 2 #P:A TODO Approximate: Imperator counted more than 2 pops; EU5 counts pop groups
				NOR = {
					religion = root.religion
					religion = religion:armenian_religion
					religion = religion:mesopotamian_pantheon # Imperator mesopotamian_religion
					religion = religion:caucasian_pantheon # Imperator caucasian_religion
				}
			}
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler # Imperator left_portrait = current_ruler
		}
		set_variable = {
			name = rel_flavor_eastern_1_cooldown
			days = 3650
		}
		random_owned_location = {
			limit = {
				NOT = { has_location_modifier = province_improvement_eastern_religion_peaceful_relations }
				dominant_religion = root.religion
				any_pop = {
					count > 2 #P:A TODO Approximate: Imperator counted more than 2 pops; EU5 counts pop groups
					NOR = {
						religion = root.religion
						religion = religion:armenian_religion
						religion = religion:mesopotamian_pantheon
						religion = religion:caucasian_pantheon
					}
				}
			}
			save_scope_as = country_province
		}
		random_character = {
			limit = {
				is_ruler = no
				in_cabinet = yes #P:A TODO Approximate: Imperator has_any_office; EU5 cabinet membership
			}
			save_scope_as = random_advisor
		}
	}

	option = { # Convert them all
		name = rel_flavor_eastern.1.a
		# add_tyranny = tyranny_large #P:A TODO No EU5 equivalent: +5 tyranny
		scope:country_province = {
			every_pop = { # Imperator: ordered_pops_in_province with max = total_population (every pop)
				limit = {
					NOR = {
						religion = root.religion
						religion = religion:armenian_religion
						religion = religion:mesopotamian_pantheon
						religion = religion:caucasian_pantheon
					}
				}
				change_pop_religion = root.religion
			}
		}
	}

	option = { # Convert some of them peacefully
		name = rel_flavor_eastern.1.b
		scope:country_province = {
			ordered_pop = { #P:A TODO Approximate: Imperator converted the happiest such pop; EU5 converts size 1 of the most satisfied one (one Imperator pop converted 1:1 to EU5 size)
				limit = {
					NOR = {
						religion = root.religion
						religion = religion:armenian_religion
						religion = religion:mesopotamian_pantheon
						religion = religion:caucasian_pantheon
					}
				}
				order_by = pop_satisfaction # Imperator pop_hapiness
				max = 1
				check_range_bounds = no
				if = {
					limit = { pop_size > 1 }
					split_pop = {
						size = 1
						religion = root.religion
					}
				}
				else = {
					change_pop_religion = root.religion
				}
			}
		}
	}

	option = { # No conversion
		name = rel_flavor_eastern.1.c
		scope:country_province = {
			add_location_modifier = {
				modifier = province_improvement_eastern_religion_peaceful_relations
				days = 3650
			}
		}
	}
}
