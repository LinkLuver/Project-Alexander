	title = rel_flavor_kemetic.48.t
	desc = rel_flavor_kemetic.48.desc
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	trigger = {
		religion = religion:egyptian_pantheon
		any_owned_location = {
			dominant_religion = religion:egyptian_pantheon # dominant_province_religion
			NOT = { location_rank = location_rank:rural_settlement } # has_city_status = yes
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		random_owned_location = {
			limit = {
				dominant_religion = religion:egyptian_pantheon
				NOT = { location_rank = location_rank:rural_settlement }
			}
			save_scope_as = randomized_country_province
		}
	}

	option = {
		name = rel_flavor_kemetic.48.a
		add_gold = {
			value = root.monthly_income_total
			multiply = 6
			max = 5000
		}
		scope:randomized_country_province = {
			add_location_modifier = {
				modifier = province_improvement_touched_by_commerce_god
				days = 1825
			}
		}
	}

	option = {
		name = rel_flavor_kemetic.48.b
		add_gold = {
			value = 0
			subtract = root.monthly_income_total
			multiply = 4
			min = -6000
		}
		add_country_modifier = {
			modifier = country_improvement_commerce_blessing
			days = 3650
		}
		scope:randomized_country_province = {
			add_location_modifier = {
				modifier = province_improvement_touched_by_commerce_god
				days = 3650
			}
		}
	}
}

rel_flavor_kemetic.49 = { #Hellenics/Arabians turning to our faith
	type = country_event
	title = rel_flavor_kemetic.49.t
	desc = rel_flavor_kemetic.49.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		religion = religion:egyptian_pantheon
		NOT = { has_variable = rel_flavor_kemetic_49_cooldown }
		any_owned_location = {
			#P:A TODO Approximate: Imperator roman_pantheon (Greeks and Romans) and arabian_pantheon; the mod splits them into hellenic_pantheon/roman_pantheon and the arabian_religion group
			OR = { # dominant_province_religion
				dominant_religion = religion:hellenic_pantheon
				dominant_religion = religion:roman_pantheon
				dominant_religion.group = religion_group:arabian_religion
			}
			#P:A TODO Approximate: Imperator "more than 5 pops of these religions"; EU5 checks more than 5 (thousand) followers of one of them (pop count converted 1:1 to EU5 size)
			OR = {
				religion_population = { religion = religion:hellenic_pantheon value > 5 }
				religion_population = { religion = religion:roman_pantheon value > 5 }
				religion_population = { religion = religion:nabatean_pantheon value > 5 }
				religion_population = { religion = religion:west_arabian_pantheon value > 5 }
				religion_population = { religion = religion:south_arabian_pantheon value > 5 }
				religion_population = { religion = religion:east_arabian_pantheon value > 5 }
			}
		}
		any_character = {
			pa_imp_has_priest_office_trigger = yes
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = rel_flavor_kemetic_49_cooldown
			days = 3650
		}
		random_owned_location = {
			limit = {
				OR = {
					dominant_religion = religion:hellenic_pantheon
					dominant_religion = religion:roman_pantheon
					dominant_religion.group = religion_group:arabian_religion
				}
				OR = {
					religion_population = { religion = religion:hellenic_pantheon value > 5 }
					religion_population = { religion = religion:roman_pantheon value > 5 }
					religion_population = { religion = religion:nabatean_pantheon value > 5 }
					religion_population = { religion = religion:west_arabian_pantheon value > 5 }
					religion_population = { religion = religion:south_arabian_pantheon value > 5 }
					religion_population = { religion = religion:east_arabian_pantheon value > 5 }
				}
			}
			save_scope_as = country_province
		}
		random_character = {
			limit = {
				pa_imp_has_priest_office_trigger = yes
			}
			save_scope_as = random_priest
		}
	}

	option = {
		name = rel_flavor_kemetic.49.a
		# add_tyranny = tyranny_medium #P:A TODO No EU5 equivalent: +3 tyranny
		scope:country_province = {
			ordered_pop = {
				#P:A TODO Approximate: Imperator converted up to 6 of the happiest matching pops (about 6,000 people); EU5 splits 6 (thousand) people off the most satisfied matching pop
				limit = {
					OR = {
						religion = religion:hellenic_pantheon
						religion = religion:roman_pantheon
						religion.group = religion_group:arabian_religion
					}
				}
				order_by = pop_satisfaction # Imperator pop_hapiness
				max = 1
				check_range_bounds = no
				split_pop = {
					size = 6
					religion = root.religion
				}
			}
		}
		# scope:random_priest = { add_loyalty = loyalty_priest_allowed } #P:A TODO No EU5 equivalent: the priest gains 5 loyalty for 5 years (EU5 characters have no loyalty)
	}

	option = {
		name = rel_flavor_kemetic.49.b
		scope:country_province = {
			ordered_pop = {
				#P:A TODO Approximate: Imperator converted up to 2 of the happiest matching pops (about 2,000 people); EU5 splits 2 (thousand) people off the most satisfied matching pop
				limit = {
					OR = {
						religion = religion:hellenic_pantheon
						religion = religion:roman_pantheon
						religion.group = religion_group:arabian_religion
					}
				}
				order_by = pop_satisfaction # Imperator pop_hapiness
				max = 1
				check_range_bounds = no
				split_pop = {
					size = 2
					religion = root.religion
				}
			}
		}
		# scope:random_priest = { add_loyalty = loyalty_priest_refused } #P:A TODO No EU5 equivalent: the priest loses 10 loyalty for 5 years (EU5 characters have no loyalty)
	}
}

rel_flavor_kemetic.50 = { #Religious leader inspired to become better
	type = country_event
	title = rel_flavor_kemetic.50.t
	desc = rel_flavor_kemetic.50.desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	trigger = {
		religion = religion:egyptian_pantheon
		NOT = { has_variable = rel_flavor_kemetic_50_cooldown }
		exists = ruler
		ruler = {
			pa_imp_char_is_religious_trigger = yes
			num_of_traits_of_category = { type = ruler value < 4 } #P:A TODO Approximate: Imperator number_of_personality_traits < max_number_of_traits_svalue (4); EU5 counts ruler traits
		}
		any_character = {
			religion = religion:egyptian_pantheon # has_religion
			pa_imp_has_priest_office_trigger = yes
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = rel_flavor_kemetic_50_cooldown
			days = 10950
		}
		random_character = {
			limit = {
				is_ruler = no
				religion = religion:egyptian_pantheon
				pa_imp_has_priest_office_trigger = yes
			}
			save_scope_as = random_advisor
		}
		hidden_effect = {
			ruler = {
				random_list = {
					1 = {
						trigger = {
							NOT = { has_trait = tolerant }
						}
						set_variable = {
							name = tolerant_trait_flag
						}
					}

						trigger = {
							NOT = { has_trait = kind_hearted } #P:A TODO Approximate: Imperator trait "merciful"; EU5 kind_hearted
						}
						set_variable = {
							name = merciful_trait_flag

				trigger = has_variable

				# founder_trait_flag = { add_trait = founder }
				# benefactor_trait_flag = { add_trait = benefactor }

rel_flavor_kemetic.51 = { # Hellenics convinced by charismatic character to become our religion
	type = country_event
	title = rel_flavor_kemetic.51.t
	desc = rel_flavor_kemetic.51.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		religion = religion:egyptian_pantheon
		NOT = { has_variable = rel_flavor_kemetic_51_cooldown }
		any_owned_location = {
			dominant_religion = root.religion # dominant_province_religion
			#P:A TODO Approximate: Imperator "more than 2 pops of roman_pantheon" (Greeks and Romans); EU5 checks more than 2 (thousand) followers of hellenic_pantheon or roman_pantheon (pop count converted 1:1 to EU5 size)
			OR = {
				religion_population = { religion = religion:hellenic_pantheon value > 2 }
				religion_population = { religion = religion:roman_pantheon value > 2 }
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = rel_flavor_kemetic_51_cooldown
			days = 7300
		}
		random_owned_location = {
			limit = {
				dominant_religion = root.religion
				OR = {
					religion_population = { religion = religion:hellenic_pantheon value > 2 }
					religion_population = { religion = religion:roman_pantheon value > 2 }
				}
			}
			weight = {
				base = 1
				modifier = {
					factor = 10
					OR = {
						religion_population = { religion = religion:hellenic_pantheon value > 10 }
						religion_population = { religion = religion:roman_pantheon value > 10 }
					}
				}
				modifier = {
					factor = 10
					OR = {
						religion_population = { religion = religion:hellenic_pantheon value > 5 }
						religion_population = { religion = religion:roman_pantheon value > 5 }
					}
				}
			}
			save_scope_as = country_province
		}
	}

	option = {
		name = rel_flavor_kemetic.51.a
		scope:country_province = {
			ordered_pop = {
				#P:A TODO Approximate: Imperator converted the happiest matching pop (about 1,000 people); EU5 splits 1 (thousand) people off the most satisfied matching pop
				limit = {
					OR = {
						religion = religion:hellenic_pantheon
						religion = religion:roman_pantheon
					}
				}
				order_by = pop_satisfaction # Imperator pop_hapiness
				max = 1
				check_range_bounds = no
				split_pop = {
					size = 1
					religion = root.religion
				}
			}
		}
	}

	option = {
		name = rel_flavor_kemetic.51.b
		pa_imp_remove_stability_effect = yes
		add_gold = {
			value = 0
			subtract = root.monthly_income_total
			multiply = 2
			min = -6000
		}
		scope:country_province = {
			every_pop = {
				limit = {
					OR = {
						religion = religion:hellenic_pantheon
						religion = religion:roman_pantheon
					}
				}
				change_pop_religion = root.religion
			}
		}
		create_character = {
			culture = root.ruler.culture
			religion = root.religion
			age = 30
			female = no
			save_scope_as = community_ambassador
		}
		hidden_effect = {
			scope:community_ambassador = {
				while = {
					count = 3
					if = {
						limit = {
							num_of_traits_of_category = { type = ruler value < 4 } #P:A TODO Approximate: Imperator number_of_personality_traits < max_number_of_traits_svalue (4); EU5 counts ruler traits
						}
						random_list = {
							10 = {
								trigger = {
									NOT = { has_trait = silver_tongue }
								}
								add_trait = trait:silver_tongue # Imperator silver_tongued
							}
							10 = {
								trigger = {
									NOT = { has_trait = humble }
								}
								add_trait = trait:humble #P:A TODO Approximate: EU5 humble is an artist trait
							}
							# 10 = { trigger = { NOT = { has_trait = intelligent } } add_trait = intelligent }
							# 10 = { trigger = { NOT = { has_trait = loving } } add_trait = loving }
							#P:A TODO No EU5 equivalent: Imperator traits "intelligent" and "loving"
						}
					}
				}
			}
		}
	}
}
