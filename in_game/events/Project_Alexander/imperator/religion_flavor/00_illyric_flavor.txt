#P:A Imported from Invictus: events/religion_flavor/00_illyric_flavor.txt
# Illyrian cults. In Invictus the ruler of an Illyrian-pantheon country picks a cult (decision cult_choice_illyric, 25 political
# influence, fires illyric_flavor.1) and gets a cultist trait; the pulse events then check that trait.
#P:A TODO No EU5 equivalent: the cultist traits (sun_cultist, serpent_cultist, horseman_cultist, wolf_cultist, pirate_cultist) do not exist in EU5.
#   illyric_flavor.1 sets a ruler variable illyric_<cult>_cultist instead (the trait bonuses are lost).
#   The mod splits Imperator illyrian_pantheon into sun_illyrian_pantheon and snake_illyrian_pantheon, so the sun and serpent
#   events also fire for countries of that religion (Approximate); the Thracian Horseman events need the ruler variable.
#   The decision is not imported, so nothing fires illyric_flavor.1 yet.
namespace = illyric_flavor

# Cult Choice Event
illyric_flavor.1 = {
	type = country_event
	title = illyric_flavor_1_title
	desc = illyric_flavor_1_desc
	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
			set_variable = cult_chosen
		}
	}

	option = {
		name = illyric_flavor_1.a
		## Cult of the Sun - Religious Boost ##
		ruler = {
			# force_add_trait = sun_cultist
			set_variable = illyric_sun_cultist #P:A TODO Approximate: stands in for the Invictus trait sun_cultist (its bonuses are lost; no such EU5 trait)
		}
	}

	option = {
		name = illyric_flavor_1.b
		## Cult of the Serpent - Economic Boost ##
		ruler = {
			# force_add_trait = serpent_cultist
			set_variable = illyric_serpent_cultist #P:A TODO Approximate: stands in for the Invictus trait serpent_cultist (its bonuses are lost; no such EU5 trait)
		}
	}

	option = {
		name = illyric_flavor_1.c
		## Cult of the Thracian Horseman - Horsey Boost ##
		ruler = {
			# force_add_trait = horseman_cultist
			set_variable = illyric_horseman_cultist #P:A TODO Approximate: stands in for the Invictus trait horseman_cultist (its bonuses are lost; no such EU5 trait)
		}
	}

	option = {
		name = illyric_flavor_1.d
		trigger = {
			has_variable = wolf_cult_var # set by the Invictus Messapia mission events (not imported)
		}
		## Cult of the Wolf - Messapian Special ##
		ruler = {
			# force_add_trait = wolf_cultist
			set_variable = illyric_wolf_cultist #P:A TODO Approximate: stands in for the Invictus trait wolf_cultist (its bonuses are lost; no such EU5 trait)
		}
	}

	option = {
		name = illyric_flavor_1.e
		trigger = {
			has_variable = pirate_cult_var # set by the Invictus Adriatic pirate missions (not imported)
		}
		## Cult of the Skull and Bones - Pirate Illyria Special ##
		ruler = {
			# force_add_trait = pirate_cultist
			set_variable = illyric_pirate_cultist #P:A TODO Approximate: stands in for the Invictus trait pirate_cultist (its bonuses are lost; no such EU5 trait)
		}
	}
}

# Bright Day
illyric_flavor.2 = {
	type = country_event
	title = illyric_flavor_2_title
	desc = illyric_flavor_2_desc
	interface_lock = no
	outcome = positive
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	trigger = {
		# religion = illyrian_pantheon
		# current_ruler = { has_trait = sun_cultist }
		OR = { #P:A TODO Approximate: Imperator required an Illyrian-pantheon ruler with the sun_cultist trait; here a Sun Illyrian country, or an Illyrian one whose ruler chose the sun cult in illyric_flavor.1
			religion = religion:sun_illyrian_pantheon
			AND = {
				religion = religion:snake_illyrian_pantheon
				ruler ?= { has_variable = illyric_sun_cultist }
			}
		}
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
	}

	option = {
		name = illyric_flavor_2.a
		add_stability = 3
	}

	option = {
		name = illyric_flavor_2.b
		add_legitimacy = 2 # legitimacy_small
		# current_ruler = { add_popularity = popularity_small } #P:A TODO No EU5 equivalent: +5 ruler popularity (EU5 characters have no popularity)
	}
}
