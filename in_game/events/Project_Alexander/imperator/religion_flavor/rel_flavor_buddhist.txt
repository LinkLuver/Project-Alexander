# A Buddhist country sends teachers from the capital to a city with no Buddhists.
rel_flavor_buddhist.1 = {
	type = country_event
	title = rel_flavor_buddhist.1.t
	desc = rel_flavor_buddhist.1.desc
	illustration_tags = {
		10 = exterior
		10 = regular
	}

	trigger = {
		religion = religion:buddhism
		NOT = { has_variable = rel_flavor_buddhist_1_cooldown }
		any_owned_location = {
			is_capital = no
			NOR = {
				dominant_religion = religion:buddhism # Imperator dominant_province_religion
				any_pop = {
					religion = religion:buddhism
				}
			}
			population > 3 #P:A TODO Approximate: Imperator total_population > 3 counted pops; EU5 population is in thousands (1 pop = 1 thousand)
			NOT = { location_rank = location_rank:rural_settlement } # Imperator has_city_status = yes
		}
		capital = {
			dominant_religion = religion:buddhism
		}
	}

	immediate = {
		ruler ?= {
			save_scope_as = pa_ruler # Imperator left_portrait = current_ruler
		}
		set_variable = {
			name = rel_flavor_buddhist_1_cooldown
			days = 3650
		}
		random_owned_location = {
			limit = {
				is_capital = no
				NOR = {
					dominant_religion = root.religion
					any_pop = {
						religion = religion:buddhism
					}
				}
				population > 3 #P:A TODO Approximate: Imperator total_population > 3 counted pops; EU5 population is in thousands
				NOT = { location_rank = location_rank:rural_settlement }
			}
			save_scope_as = country_province
		}
	}

	option = {
		name = rel_flavor_buddhist.1.a
		# add_political_influence = subtract_influence_medium #P:A TODO No EU5 equivalent: -15 political influence
		capital = {
			random_pop = { #P:A TODO Approximate: Imperator move_pop moved one whole Buddhist pop to the city; EU5 splits off size 1 and moves it (one Imperator pop converted 1:1 to EU5 size)
				limit = {
					religion = religion:buddhism
				}
				weight = {
					base = 1
					modifier = {
						factor = 10
						OR = {
							pop_type = pop_type:burghers # Imperator citizen
							pop_type = pop_type:peasants # Imperator freemen
						}
					}
				}
				split_pop = {
					size = 1
					location = scope:country_province
				}
			}
			random_pop = { #P:A TODO Approximate: second moved pop, as above
				limit = {
					religion = religion:buddhism
				}
				weight = {
					base = 1
					modifier = {
						factor = 10
						OR = {
							pop_type = pop_type:burghers
							pop_type = pop_type:peasants
						}
					}
				}
				split_pop = {
					size = 1
					location = scope:country_province
				}
			}
		}
		scope:country_province = {
			ordered_pop = { #P:A TODO Approximate: Imperator converted the 3 happiest non-Buddhist pops; EU5 converts size 1 of each of the 3 most satisfied (one Imperator pop converted 1:1 to EU5 size)
				limit = {
					NOT = { religion = religion:buddhism }
				}
				order_by = pop_satisfaction # Imperator pop_hapiness
				max = 3
				check_range_bounds = no
				if = {
					limit = { pop_size > 1 }
					split_pop = {
						size = 1
						religion = religion:buddhism
					}
				}
				else = {
					change_pop_religion = religion:buddhism
				}
			}
		}
	}

	option = {
		name = rel_flavor_buddhist.1.b
		custom_tooltip = no_effect_tooltip
	}

	hidden = yes

	}

	immediate = {
		if = {
			limit = {
				scope:reborn_heir = { age_in_years <= 15 }
			}
			random_list = {
				20 = {
					trigger = {
						# scope:reborn_heir = { number_of_personality_traits < max_number_of_traits_svalue OR = { NOT = { has_trait = original_thinker } NOT = { has_trait = confident } NOT = { has_trait = disciplined } } }

						id = rel_flavor_buddhist.6
					}

						scope:reborn_heir = {
