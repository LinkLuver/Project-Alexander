# Jain festivals with a shared 20-year cooldown (var:festival_cooldown). Invictus has no caller for these three events, so they are dormant until hooked up (a yearly country pulse would suit them).
namespace = jain_flavor

# Paryushana
# One of the most holy festivals, this is a day about coming together with the community and focusing on the Jain principles and teachings. Tied to this festival are Kshamavani and Samvatsari which are days of forgiveness which are about forgiving of the evil and bad things one has done in the year.
jain_flavor.8 = {
	type = country_event
	title = jain_flavor.8.t
	desc = jain_flavor.8.desc
	outcome = positive
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	trigger = {
		religion = religion:jain
		NOT = { has_variable = festival_cooldown }
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}
		set_variable = {
			name = festival_cooldown
			days = 7300
		}
	}

	option = { # It is time to fast!
		name = jain_flavor.8.a
		add_country_modifier = {
			modifier = paryushana_festival_mod
			days = 730
		}
		ruler = {
			add_character_modifier = {
				modifier = fasting_for_paryushana
				days = 730
			}
		}
	}

	option = { # A Grand Festival will be held!
		name = jain_flavor.8.b
		add_gold = {
			value = monthly_income_total # Imperator has_monthly_income
			multiply = -3 # subtract_monthly_income_small
			max = 5000
		}
		ordered_owned_location = {
			limit = {
				NOT = { dominant_religion = religion:jain }
			}
			order_by = population # Imperator gave no order
			max = 3
			pa_imp_spread_religion_in_province = {
				RELIGION = jain
			}
		}
		add_country_modifier = {
			modifier = paryushana_festival_mod
			days = 730
		}
	}

	option = { # Now is not the time.
		name = jain_flavor.8.c
		# current_ruler = { add_popularity = -10 } #P:A TODO No EU5 equivalent: -10 ruler popularity (EU5 characters have no popularity)
		add_legitimacy = -2 # subtract_legitimacy_small
	}
}

# Akshaya Tritiya
# For jains This festival commemorates the first tirthankara, Rishabhanatha, ending his one-year asceticism by consuming sugarcane juice poured into his cupped hands. Celebrated with fasts and asceticism and pilgrimages to Palitana in Gujarat. For the event, this should probably increase commerce and trade briefly for Outer Saurashtra to reflect pilgrims going there.
jain_flavor.10 = {
	type = country_event
	title = jain_flavor.10.t
	desc = jain_flavor.10.desc
	outcome = positive
	illustration_tags = {
		10 = exterior
		10 = happy
	}

	trigger = {
		religion = religion:jain
		NOT = { has_variable = festival_cooldown }
	}

	immediate = {
		ruler = {
			save_scope_as = pa_ruler
		}

		}
	}

	option = {
		name = jain_flavor.10.a

		capital = {
			add_location_modifier = {

			}
		}
	}
