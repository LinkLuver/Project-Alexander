holy_war.1 = {
	type = country_event
	title = holy_war.1.title
	desc = holy_war.1.desc
	outcome = neutral

	illustration_tags = {
		10 = exterior
		10 = armed
	}

	trigger = {
		exists = scope:preach_io
		NOT = { is_member_of_international_organization = scope:preach_io }
		country_can_join_international_organization = scope:preach_io
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
	}

	option = {
		name = holy_war.1.a
		ai_chance = {
			factor = 1
			modifier = {
				factor = {
					value = "capital.distance_to_squared(scope:preach_io.international_organization_target.capital)"
					divide = 2000000
					multiply = -1
					add = 1
					min = 0.05
					max = 1
				}
			}
			modifier = {
				factor = 0
				NOT = { knows_country = scope:preach_io.international_organization_target }
			}
			modifier = {
				factor = 0.05
				scope:preach_io = { holy_war_overwhelms_target = yes }
			}
			modifier = {
				factor = 3
				is_rival_of = scope:preach_io.international_organization_target
			}
			modifier = {
				factor = 1.25
				opinion = { target = scope:preach_io.international_organization_target value < 0 }
			}
			modifier = {
				factor = 1.5
				exists = scope:preach_actor
				opinion = { target = scope:preach_actor value >= 50 }
			}
			modifier = {
				factor = 0.25
				exists = scope:preach_actor
				is_rival_of = scope:preach_actor
			}
			modifier = {
				factor = 0.5
				exists = scope:preach_actor
				opinion = { target = scope:preach_actor value < 0 }
			}
			modifier = {
				factor = 0.5
				at_war = yes
			}
			modifier = {
				factor = 0.5
				war_exhaustion >= 5
			}
			modifier = {
				factor = 0.5
				num_loans > 0
			}
			modifier = {
				factor = 0.1
				is_during_bankruptcy = yes
			}
		}
		
		scope:preach_io = {
			add_country_to_international_organization = root
		}
		if = {
			limit = { exists = scope:preach_actor }
			scope:preach_actor = {
				add_opinion = { target = root modifier = opinion_improve_relations_member }
				trigger_event_non_silently = { id = holy_war.7 }
			}
		}
	}
	option = {
		name = holy_war.1.b
		ai_chance = {
			factor = 1
		}
		if = {
			limit = { exists = scope:preach_actor }
			scope:preach_actor = {
				add_opinion = { target = root modifier = opinion_refused_holy_war_call }
				trigger_event_non_silently = { id = holy_war.8 }
			}
		}
	}
}
