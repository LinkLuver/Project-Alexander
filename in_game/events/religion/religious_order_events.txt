religious_orders.1 = {	#Dying benefactor leaves wealth to order
	type = country_event
	title = religious_orders.1.title
	desc = religious_orders.1.desc
	outcome = positive

	illustration_tags = {
		10 = happy
		10 = interior
	}

	trigger = {
		any_religious_order_in_country = { count > 0 }
	}
	
	immediate = {
		event_illustration_estate_background_effect = { background = estate_type:clergy_estate }
		random_religious_order_in_country = { save_scope_as = target_order }
		ruler_or_regent = { save_scope_as = target_character }
	}

	option = {
		name = religious_orders.1.a

		scope:target_order = {

			ordered_location_in_religious_order = {
				order_by = {
					value = population
					multiply = proximity
				}
				max = root.country_rank_level
				check_range_bounds = no
				add_location_modifier = {
					modifier = benefactor_wealth_modifier
					mode = replace
					years = modifier_duration_years_very_short
				}
			}
		}
	}

	option = {
		name = religious_orders.1.b

		change_gold_effect = { scale = 3 }
		scope:target_order = {
			change_religious_order_zeal = religious_order_mild_penalty
		}
	}
}

religious_orders.2 = {	#Religious Order requests new holding
	type = country_event
	title = religious_orders.2.title
	desc = religious_orders.2.desc
	outcome = positive

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		any_religious_order_in_country = { 
			root = {
				num_order_holdings_of_order = {
					religious_order = prev
					value < 3
				}
			}
			count > 0
		}
		num_provinces > 1
	}

	immediate = {
		event_illustration_estate_background_effect = { background = estate_type:clergy_estate }
		random_religious_order_in_country = {
			limit = {
				root = {
					num_order_holdings_of_order = {
						religious_order = prev
						value < 3
					}
				}
			}
			save_scope_as = target_order 
		}
		ordered_province = {
			limit = {
				NOT = {
					any_location_in_province = { is_capital = yes }
				}
				any_location_in_province = { is_holding_of_any_religious_order = no }
			}
			order_by = province_capital.proximity
			max = 4
			check_range_bounds = no
			add_to_list = religious_order_2_list
		}

		random_in_list = {
			list = religious_order_2_list
			save_scope_as = target_province

			random_location_in_province = {
				limit = { is_holding_of_any_religious_order = no }

				save_scope_as = target_location
			}
		}
		ruler_or_regent = { save_scope_as = target_character }
	}

	option = {
		name = religious_orders.2.a

		scope:target_location = { attach_to_religious_order = scope:target_order }

		random_list = {
			1 = {
				show_chance = no
				desc = locals_embrace_the_order_desc

				scope:target_province = {
					add_province_modifier = {
						modifier = religious_order_local_ties_modifier
						years = modifier_duration_years_very_short
						mode = replace
					}
				}
			}
			1 = {
				show_chance = no
				desc = order_works_to_bolster_local_wealth_desc

				scope:target_province = {
					add_province_modifier = {
						modifier = pious_wealth_modifier
						years = modifier_duration_years_very_short
						mode = replace
					}
				}
			}

			1 = {
				show_chance = no
				desc = order_projects_central_authority_desc

				scope:target_province = {
					add_province_modifier = {
						modifier = propagation_of_central_authority_modifier
						years = modifier_duration_years_very_short
						mode = replace
					}
				}
			}

			1 = {
				show_chance = no
				desc = order_appreciates_new_holding_desc

				scope:target_order = {
					change_religious_order_zeal = religious_order_low_bonus
				}
			}
		}
	}

	option = {
		name = religious_orders.2.b

		add_gold = {
			value = monthly_income_trade_and_tax
			multiply = -3
		}
		add_gold_to_estate = {
			estate_type = estate_type:clergy_estate
			value = {
				add = root.monthly_income_trade_and_tax
				multiply = 3
			}
		}
		add_estate_satisfaction = {
			type = estate_type:clergy_estate
			value = estate_satisfaction_mild_bonus
		}
	}

	option = {
		name = religious_orders.2.c

		add_political_influence = political_influence_extreme_bonus
		scope:target_order = {
			change_religious_order_zeal = religious_order_low_penalty
		}
	}
}

religious_orders.3 = {	#Religious Order Projects Zeal
	type = country_event
	title = religious_orders.3.title
	desc = religious_orders.3.desc
	outcome = positive

	image = "gfx/interface/illustrations/disaster/religious_turmoil.dds"

	trigger = {
		any_religious_order_in_country = {
			count > 0
		}
	}

	immediate = {
		random_religious_order_in_country = { save_scope_as = target_order }

		if = {
			limit = {
				any_province = { "religion_percentage(root.religion)" < 0.8 }
			}
			ordered_province = {
				limit = { "religion_percentage(root.religion)" < 0.8 }
				order_by = {
					value = population
					multiply = {
						value = 1
						subtract = "religion_percentage(root.religion)"
					}
				}
				max = 1
				save_scope_as = target_province
			}
		}
	}

	option = {
		name = religious_orders.3.a

		scope:target_order = {
			change_religious_order_zeal = religious_order_mild_bonus
		}
	}

	option = {
		name = religious_orders.3.b

		trigger = {
			"estate(estate_type:burghers_estate)" = {
				estate_gold >= {
					value = root.monthly_income_trade_and_tax
					multiply = 2
				}
			}
		}

		transfer_gold_from_estate = {
			estate_type = estate_type:burghers_estate
			value = {
				value = root.monthly_income_trade_and_tax
				multiply = 2
			}
		}
	}

	option = {
		name = religious_orders.3.c

		trigger = { exists = scope:target_province }

		scope:target_province = {
			add_province_modifier = {
				modifier = harvest_souls_modifier
				years = modifier_duration_years_very_short
				mode = replace
			}
		}
	}
}
