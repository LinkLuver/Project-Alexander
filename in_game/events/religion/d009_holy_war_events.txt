holy_war.12 = {
	type = country_event
	title = holy_war.12.title
	desc = holy_war.12.desc
	outcome = negative

	illustration_tags = {
		10 = interior
		10 = regular
	}

	immediate = {
		root = { save_scope_as = parias_payer }
		international_organization:crusade = { save_scope_as = threatening_crusade }
		international_organization:crusade = {
			random_international_organization_member = {
				limit = {
					has_special_status_in_international_organization = {
						type = special_status:crusade_temporal_leader
						international_organization = international_organization:crusade
					}
				}
				save_scope_as = parias_demander
			}
		}
		if = {
			limit = { NOT = { country_exists = scope:parias_demander } }
			scope:threatening_crusade = {
				leader_country ?= {
					save_scope_as = parias_demander
				}
			}
		}
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
	}

	option = {
		name = holy_war.12.a
		ai_chance = {
			factor = 1
		}
		holy_war_scaled_gold_transfer_effect = {
			payer = root
			payee = scope:parias_demander
			scale = 6
		}
		custom_tooltip = {
			text = holy_war.12.a.tt
			international_organization:crusade = {
				change_variable = { name = preparation add = -30 }
				clamp_variable = { name = preparation max = 100 min = 0 }
				change_variable = { name = religious_zeal add = -10 }
				clamp_variable = { name = religious_zeal max = 100 min = 0 }
			}
		}
		scope:parias_demander = { trigger_event_non_silently = { id = holy_war.13 } }
	}
	option = {
		name = holy_war.12.b
		ai_chance = {
			factor = 1
		}
		scope:parias_demander = {
			add_opinion = { target = root modifier = opinion_refused_holy_war_call }
		}
		custom_tooltip = {
			text = holy_war.12.b.tt
			international_organization:crusade = {
				change_variable = { name = religious_zeal add = 10 }
				clamp_variable = { name = religious_zeal max = 100 min = 0 }
			}
		}
		scope:parias_demander = { trigger_event_non_silently = { id = holy_war.14 } }
	}
}

			factor = 1
		}

				text = d009_order_recent_service_tt

					set_variable = { name = recent_order_service value = yes days = 1825 }
