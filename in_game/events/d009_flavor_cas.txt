flavor_cas.143 = { #The Spanish Road
	type = country_event
	title = flavor_cas.143.title
	desc = flavor_cas.143.desc
	outcome = neutral

	fire_only_once = yes

	dynamic_historical_event = {
		tag = SPA
		from = 1567.1.1
		to = 1648.1.1
		monthly_chance = 3
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		hidden_trigger = { has_global_variable = battle_for_the_strait_winner_country }
		custom_description = {
			text = battle_for_the_strait_is_winner_text
			this = global_var:battle_for_the_strait_winner_country
		}
		has_advance = paved_road_advance
		region:italy_region = {
			ROOT = { owns_or_has_subject_in = prev }
		}
		region:north_german_region = {
			any_area_in_region = {
				OR = {
					this = area:brabant_area
					this = area:flanders_area
					this = area:friesland_area
					this = area:holland_area
					this = area:wallonia_area
				}
				ROOT = { owns_or_has_subject_in = prev }
			}
		}
	}

	illustration_tags = {
        10 = regular
        10 = exterior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
		random_owned_location = {
			limit = {
				region = region:italy_region
			}
			province = {
				save_scope_as = target_province1
			}
		}
		random_owned_location = {
			limit = {
				OR = {
					area = area:brabant_area
					area = area:flanders_area
					area = area:friesland_area
					area = area:holland_area
					area = area:wallonia_area
				}
			}
			province = {
				save_scope_as = target_province2
			}
		}
	}
	
	option = {
		name = flavor_cas.143.a
		historical_option = yes
		custom_tooltip = {
			text = flavor_cas.143.a_tt
			scope:target_province1 = {
				every_location_in_province = {
					save_scope_as = fro_location
					every_neighbor_location = {
						limit = {
							province = scope:target_province1
						}
						save_scope_as = to_location
						root = {
							construct_road = {
								from = scope:fro_location
								to  = scope:to_location
								road_type = road_type:paved_road
							}
						}
					}
				}
			}
			scope:target_province2 = {
				every_location_in_province = {
					save_scope_as = fro_location
					every_neighbor_location = {
						limit = {
							province = scope:target_province2
						}
						save_scope_as = to_location
						root = {
							construct_road = {
								from = scope:fro_location
								to  = scope:to_location
								road_type = road_type:paved_road
							}
						}
					}
				}
			}
		}
		change_societal_value = {

flavor_cas.148 = { #Empresa de China
	type = country_event
	title = flavor_cas.148.title
	desc = flavor_cas.148.desc
	outcome = neutral
	
	fire_only_once = yes

	historical_info = flavor_cas.148.historical_info

	dynamic_historical_event = {
		tag = CAS
		from = 1550.1.1
		to = 1650.1.1
		monthly_chance = 1
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		hidden_trigger = { has_global_variable = battle_for_the_strait_winner_country }
		custom_description = {
			text = battle_for_the_strait_is_winner_text
			this = global_var:battle_for_the_strait_winner_country
		}
		has_presence_in = region:indonesia_region
		sub_continent:east_asia = {
			any_country_with_capital_in_geography = {
				is_subject = no
				within_diplomatic_range = root
				NOT = { is_allied_with = { target = root } }
				has_ports = yes
			}
		}
	}
	
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		sub_continent:east_asia = {
			ordered_country_with_capital_in_geography = {
				order_by = total_population
				limit = {
					is_subject = no
					within_diplomatic_range = root
					NOT = { is_allied_with = { target = root } }
					has_ports = yes
				}
				position = 0
				save_scope_as = target_country

				random_owned_location = {
					limit = {
						sub_continent = sub_continent:east_asia
						is_coastal = yes
					}
					save_scope_as = target_location
				}
			}
		}
	}
	
	option = {
		name = flavor_cas.148.a
		historical_option = yes
		scope:target_country = {
			root = { add_spy_network = { target = prev value = 50 } }
		}
		add_casus_belli = {
			target = scope:target_country
			type = casus_belli:cb_conquer_province
			province = scope:target_location.province
		}
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_left
		}
	}
	
	option = {
		name = flavor_cas.148.b
		change_societal_value = {
			type = belligerent_vs_conciliatory
			value = societal_value_move_to_right
		}
	}
}
