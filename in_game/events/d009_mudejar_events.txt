d009_mudejar.2 = { #The Mudéjar Quarter Expands
	type = country_event
	title = d009_mudejar.2.title
	desc = d009_mudejar.2.desc
	outcome = neutral

	trigger = {
		has_dlc = "d009_across_the_pillars"
		religion = religion:catholic
		has_estate_privilege = estate_privilege:mudejar_aljamas

		NOT = {
			has_variable = d009_mudejar_2_exhausted
		}

		"religion_population_in_country(religion:sunni)" > 0
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }

		ordered_owned_location = {
			limit = {
				any_pop = {
					religion = religion:sunni
					owner ?= root
				}
			}

			order_by = {
				value = 0
				if = {
					limit = {
						NOT = {
							has_town_rights = town_rights_type:iberian_aljama_charter
						}
					}
					add = 1000
				}
			}

			max = 1
			check_range_bounds = no
			save_scope_as = target_location
		}

		if = {
			limit = {
				scope:target_location = {
					has_town_rights = town_rights_type:iberian_aljama_charter
				}
			}
			set_variable = {
				name = d009_mudejar_2_exhausted
				years = 20
			}
		}
	}

	option = {
		name = d009_mudejar.2.a
		historical_option = yes

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = capital_economy_vs_traditional_economy
			value = societal_value_move_to_left
		}

		scope:target_location = {
			grant_town_rights = town_rights_type:iberian_aljama_charter

			add_location_modifier = {
				modifier = mudejar_quarter_prosperity
				mode = add
			}
		}
	}

	option = {
		name = d009_mudejar.2.b

		add_stability = stability_mild_bonus

		change_societal_value = {
			type = capital_economy_vs_traditional_economy
			value = societal_value_move_to_right
		}
	}
}
