flavor_ara.37 = { #Expanding the Fortifications in Italy
	type = country_event
	title = flavor_ara.37.title
	desc = flavor_ara.37.desc
	outcome = neutral

	dynamic_historical_event = {
		tag = ARA
		from = 1400.1.1
		to = 1600.1.1
		monthly_chance = 3
	}

	fire_only_once = yes
	
	illustration_tags = {
		10 = regular
		10 = exterior
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		has_presence_in = region:italy_region

		has_policy = ara_italian_dual_court

		total_effective_building_levels:italian_court_building > 0

		any_border_location = {
			is_capital = no
			region = region:italy_region
			has_fort = no
		}
	}
	
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }

		random_owned_non_rural_location = {
			limit = {
				has_building = building_type:italian_court_building
			}
			save_scope_as = italian_court
		}

		ruler_or_regent ?= {
			save_scope_as = target_character
		}

		ordered_border_location = {
			order_by = population
			limit = {
				is_capital = no
				region = region:italy_region
				has_fort = no
			}
			check_range_bounds = no
			max = 1
			save_scope_as = target_location
		}
	}

	option = {
		name = flavor_ara.37.a

		scope:target_location = {
			construct_latest_fort_building = {
				cost_multiplier = 0.5
			}
		}

		scope:italian_court = {
			change_prosperity = prosperity_mild_bonus
		}

		add_all_estate_satisfaction = {
			value = estate_satisfaction_mild_bonus
		}

		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}
	}

	option = {
		name = flavor_ara.37.b

		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}

		capital ?= {
			change_prosperity = prosperity_mild_bonus
		}
	}
}
