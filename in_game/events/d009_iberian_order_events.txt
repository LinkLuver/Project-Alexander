d009_iberian_order_events.1 = {
	type = country_event
	title = d009_iberian_order_events.1.title
	desc = d009_iberian_order_events.1.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = CAS
		from = 1474.1.1
		to = 1494.1.1
		monthly_chance = 15
	}

	illustration_tags = {
		10 = angry
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }

		ruler_or_regent = {
			save_scope_as = target_ruler
		}
		random_religious_order_in_country = {
			limit = {
				gfx_culture_applicable = santiago_gfx
			}
			save_scope_as = target_order
		}

		d009_crown_appointee_create_effect = yes
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		any_religious_order_in_country = {
			gfx_culture_applicable = santiago_gfx
		}
	}

	option = {
		name = d009_iberian_order_events.1.a
		historical_option = yes

		add_political_influence = political_influence_severe_bonus
		add_prestige = prestige_mild_bonus
		d009_crown_appoints_order_master_effect = yes
	}

	option = {
		name = d009_iberian_order_events.1.b

		add_political_influence = political_influence_mild_penalty
		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		scope:target_order = {
			change_religious_order_zeal = religious_order_mild_bonus
			custom_tooltip = {
				text = d009_order_recent_service_tt
				every_location_in_religious_order = {
					set_variable = { name = recent_order_service value = yes days = 1825 }
				}
			}
		}
	}

	option = {
		name = d009_iberian_order_events.1.c

		add_religious_influence = religious_influence_mild_penalty
		scope:target_order = {
			change_religious_order_zeal = religious_order_low_bonus
		}
	}
}

d009_iberian_order_events.2 = {
	type = country_event
	title = d009_iberian_order_events.2.title
	desc = d009_iberian_order_events.2.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = CAS
		from = 1400.1.1
		to = 1500.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }

		random_religious_order_in_country = {
			limit = {
				gfx_culture_applicable = santiago_gfx
			}
			save_scope_as = target_order
		}
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		any_religious_order_in_country = {
			gfx_culture_applicable = santiago_gfx
			any_location_in_religious_order = {
				this = location:ucles
			}
		}
		location:ucles = {
			owner ?= root
		}
	}

	option = {
		name = d009_iberian_order_events.2.a

		change_gold_effect = { scale = 3 }
		scope:target_order = {
			change_religious_order_zeal = religious_order_moderate_penalty
		}
	}

	option = {
		name = d009_iberian_order_events.2.b
		historical_option = yes

		change_gold_effect = { scale = -2 }
		location:ucles = {
			if = {
				limit = {
					NOT = {
						has_building = building_type:monastery
					}
				}
				construct_building = {
					building_type = building_type:monastery
				}
			}
		}
		scope:target_order = {
			change_religious_order_zeal = religious_order_mild_bonus
			custom_tooltip = {
				text = d009_order_recent_service_tt
				every_location_in_religious_order = {
					set_variable = { name = recent_order_service value = yes days = 1825 }
				}
			}
		}
	}

	option = {
		name = d009_iberian_order_events.2.c

		change_gold_effect = { scale = 1 }
		scope:target_order = {
			change_religious_order_zeal = religious_order_low_penalty
		}
	}
}
