namespace = d009_flavor_religious_orders

d009_flavor_religious_orders.1 = {
	type = country_event
	title = d009_flavor_religious_orders.1.title
	desc = d009_flavor_religious_orders.1.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = CAS
		from = 1337.1.1
		to = 1340.1.1
		monthly_chance = 50
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }

		character:cas_rodrigo_perez_alcantara = {
			save_scope_as = old_head_of_order
			add_trait = trait:sickly
		}
		scope:old_head_of_order = {
			religious_order = {
				save_scope_as = target_order
			}
		}
		create_character = {
			first_name = name_gonzalo
			last_name = Martinez_Oviedo
			culture = culture:leonese
			religion = religion:catholic
			birth_date = 1310.1.1 #unkown
			female = no
			birth_location = location:oviedo
			estate = estate_type:nobles_estate
			adm = { 60 90 }
			dip = { 30 60 }
			mil = { 70 99 }
			save_scope_as = target_character
			script = cas_gonzalo_martinez_oviedo
			create_in_limbo = yes
		}
		ruler = {
			save_scope_as = target_ruler
		}
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		current_month >= 7
		character:cas_alfonso_xi_burgundy = {
			is_alive = yes
			is_ruler = yes
		}
		character:cas_rodrigo_perez_alcantara = {
			is_alive = yes
		}
	}

	option = {
		name = d009_flavor_religious_orders.1.a
		historical_option = yes

		kill_character = {
			target = scope:old_head_of_order
			reason = sudden_illness
		}

		scope:target_character = {
			move_country = root
		}
		scope:target_order = {
			set_head_character = scope:target_character
			change_religious_order_zeal = religious_order_mild_penalty
		}
	}

	option = {
		name = d009_flavor_religious_orders.1.b

		kill_character = {
			target = scope:old_head_of_order
			reason = sudden_illness
		}

		scope:target_order = {
			change_religious_order_zeal = religious_order_mild_bonus
		}
	}
}

	illustration_tags = {

	}

			save_scope_as = target_character
		}

	}

		name = d009_flavor_religious_orders.2.b
		
		add_religious_influence = religious_influence_severe_bonus

		}
	}
}

d009_flavor_religious_orders.3 = {
	type = country_event
	title = d009_flavor_religious_orders.3.title
	desc = d009_flavor_religious_orders.3.desc
	outcome = neutral

	fire_only_once = yes
	dynamic_historical_event = {
		tag = CAS
		from = 1340.6.1
		to = 1343.1.1
		monthly_chance = 10
	}

	illustration_tags = {
		10 = angry
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }

		ruler = {
			save_scope_as = target_ruler
		}
		character:cas_gonzalo_martinez_oviedo = {
			save_scope_as = target_character
		}
		scope:target_character = {
			religious_order = {
				save_scope_as = target_order
			}
		}
		religion:catholic = {
			random_country_in_religion = {
				save_scope_as = exile_country
			}
		}
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"
		dynamic_historical_event_fired = d009_flavor_religious_orders.2
		character:cas_alfonso_xi_burgundy = {
			is_alive = yes
			is_ruler = yes
		}
		character:cas_gonzalo_martinez_oviedo ?= {
			is_alive = yes
			is_religious_order_member = yes
		}
	}

	option = {
		name = d009_flavor_religious_orders.3.a
		historical_option = yes

		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_bonus
		}
		scope:target_order = {
			change_religious_order_zeal = religious_order_severe_penalty
		}
		kill_character = {
			target = scope:target_character
			reason = execution
			killer = scope:target_ruler
		}
	}

	option = {
		name = d009_flavor_religious_orders.3.b

		add_estate_satisfaction = {
			type = estate_type:nobles_estate
			value = estate_satisfaction_mild_penalty
		}
		scope:target_order = {
			random_character_in_order = {
				limit = {
					NOT = { this = scope:target_character }
					has_trait_category = general
				}
				save_scope_as = new_head
			}
			if = {
				limit = { NOT = { exists = scope:new_head } }
				random_character_in_order = {
					limit = { NOT = { this = scope:target_character } }
					save_scope_as = new_head
				}
			}
			if = {
				limit = { exists = scope:new_head }
				set_head_character = scope:new_head
			}
		}
		scope:target_character = {
			move_country = scope:exile_country
		}
	}
}
