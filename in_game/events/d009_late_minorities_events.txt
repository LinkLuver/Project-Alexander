namespace = d009_late_minorities

d009_late_minorities.1 = { #The Sabbatean Messianic Movement
	type = country_event
	title = d009_late_minorities.1.title
	desc = d009_late_minorities.1.desc
	outcome = neutral
	historical_info = d009_late_minorities.1.historical_info

	fire_only_once = yes

	dynamic_historical_event = {
		tag = TUR
		from = 1665.1.1
		to = 1667.1.1
		monthly_chance = 100
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"

		"religion_population_in_country(religion:judaism)" > 0
		"culture_population_in_country(culture:sephardi)" > 0
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:burghers_estate }

		create_character = {
			first_name = name_sabbatai
			last_name = Zevi
			adm = { 40 60 }
			dip = { 60 80 }
			mil = { 5 20 }
			culture = culture:romanyoti
			religion = religion:judaism
			female = no
			min_age = 35
			estate = estate_type:clergy_estate
			create_in_limbo = yes
			save_scope_as = target_character
		}

		set_variable = { name = d009_sabbatai_zevi value = scope:target_character }
	}

	option = {
		name = d009_late_minorities.1.a
		historical_option = yes

		add_stability = stability_mild_penalty

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}

		set_variable = d009_sabbatean_movement_allowed
	}

	option = {
		name = d009_late_minorities.1.b

		add_stability = stability_mild_bonus

		custom_tooltip = {
			text = d009_late_minorities.1.b.tt
			every_pop = {
				limit = {
					religion = religion:judaism
					culture = culture:sephardi
				}

				add_pop_satisfaction = pop_satisfaction_mild_penalty
			}
		}

		hidden_effect = {
			kill_character_silently = scope:target_character
		}
	}
}
