namespace = pa_senate

# Each Senate bloc can force a constitutional showdown, but only after a year
# of sustained low approval while holding a substantial share of the chamber.
# The leader and one non-capital location defect to the resulting rebel side.

scripted_effect pa_senate_start_optimates_civil_war = {
	random_list = {
		70 = { create_rebel = { category = estate estate = nobles_estate name = pa_senate_optimates_rebels save_scope_as = pa_senate_rebels } }
		20 = { create_rebel = { category = estate estate = burghers_estate name = pa_senate_optimates_rebels save_scope_as = pa_senate_rebels } }
		10 = { create_rebel = { category = estate estate = peasants_estate name = pa_senate_optimates_rebels save_scope_as = pa_senate_rebels } }
	}
	var:pa_senate_optimates_leader ?= { if = { limit = { this != root.ruler } change_character_allegiance = scope:pa_senate_rebels } }
	var:pa_senate_optimates_leader.dynasty ?= { every_character_in_dynasty = { limit = { owner = root is_alive = yes is_adult = yes this != root.ruler } change_character_allegiance = scope:pa_senate_rebels } }
	random_owned_location = {
		limit = { is_capital = no }
		every_pop = { limit = { owner = root } change_pop_allegiance = scope:pa_senate_rebels }
	}
	scope:pa_senate_rebels = { start_civil_war = {} }
}

pa_senate.1 = {
	type = country_event
	category = disaster_event
	title = pa_senate.1.title
	desc = pa_senate.1.desc
	outcome = negative
	trigger = { tag = ROM has_variable = pa_senate_crisis_pending }
	option = {
		name = pa_senate.1.a
		trigger = { gold >= 100 }
		add_gold = -100
		set_variable = { name = pa_senate_optimates_approval value = { value = var:pa_senate_optimates_approval add = 30 max = 100 } }
		set_variable = { name = pa_senate_optimates_crisis value = 0 }
		remove_variable = pa_senate_crisis_pending
	}
	option = {
		name = pa_senate.1.b
		high_risk_option = yes
		remove_variable = pa_senate_crisis_pending
		pa_senate_start_optimates_civil_war = yes
	}
}

# Hard cap for a Roman Senate session. The normal duration modifier ends most
# sessions sooner. If still open, resolve with the actual native backing.
pa_senate.41 = {
	type = country_event
	hidden = yes
	immediate = {
		if = {
			limit = { tag = ROM has_variable = pa_senate_country_session_guard is_active_parliament = yes has_ongoing_parliament_debate = yes }
			if = { limit = { parliament_issue_support > 0.5 } end_parliament = yes }
			else = { end_parliament = no }
		}
	}
}

pa_senate.40 = {
	type = country_event
	title = pa_senate.40.title
	hidden = yes
	immediate = {
		if = {
			limit = { tag = ROM exists = international_organization:pa_roman_senate }
			international_organization:pa_roman_senate = {
				if = {
					limit = { has_variable = pa_senate_session_guard is_active_parliament = yes has_ongoing_parliament_debate = yes }
					if = { limit = { parliament_issue_support > 0.5 } end_parliament = yes }
					else = { end_parliament = no }
				}
			}
		}
	}
}
