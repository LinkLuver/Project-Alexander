namespace = d009_convivencia

d009_convivencia.1 = { #The Three Communities
	type = country_event
	title = d009_convivencia.1.title
	desc = d009_convivencia.1.desc
	outcome = neutral
	historical_info = d009_convivencia.1.historical_info

	trigger = {
		has_dlc = "d009_across_the_pillars"
		religion = religion:catholic

		culture = {
			has_culture_group = culture_group:iberian_group
		}

		NOT = {
			any_in_global_list = {
				variable = d009_convivencia_intro_locations
				owner = root
			}
		}

		has_presence_in = region:iberia_region

		"religion_population_in_country(religion:sunni)" > 0
		"religion_population_in_country(religion:judaism)" > 0
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:clergy_estate }

		region:iberia_region = {
			ordered_ownable_location_in_region = {
				limit = {
					owner = root
				}

				order_by = {
					add = "religion_percentage(religion:sunni)"
					if = {
						limit = { "religion_percentage(religion:judaism)" > 0 }
						multiply = 1000
					}
				}

				max = 1
				check_range_bounds = no
				save_scope_as = target_location
			}
		}

		add_to_global_variable_list = { name = d009_convivencia_intro_locations target = scope:target_location }

		if = {
			limit = {
				owns = location:toledo
			}

			trigger_event_non_silently = {
				id = d009_convivencia.2
				months = { 12 60 }
			}
		}

		set_variable = {
			name = d009_convivencia_courts_delay
			years = { 10 15 }
		}
	}

	option = {
		name = d009_convivencia.1.a

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_right
		}

		if = {
			limit = {
				is_situation_active = situation:golden_century
				has_global_variable = battle_for_the_strait_winner_country
				this = global_var:battle_for_the_strait_winner_country
			}
			gc_change_european_dominance = { value = gc_dominance_weak_bonus }
		}

		if = {
			limit = {
				is_situation_active = situation:battle_for_the_strait
				country_exists = c:GRA

				OR = {
					has_or_had_tag = CAS
					has_or_had_tag = MOR
				}
			}
			change_influence_over_granada_effect = { value = -2 }
		}

		scope:target_location = {
			custom_tooltip = {
				text = d009_convivencia.1.a.tt

				every_pop = {
					limit = {
						owner ?= root

						OR = {
							religion = religion:sunni
							religion = religion:judaism
						}
					}

					add_pop_satisfaction = pop_satisfaction_mild_bonus
				}
			}
		}
	}

	option = {
		name = d009_convivencia.1.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = centralization_vs_decentralization
			value = societal_value_move_to_left
		}

		if = {
			limit = {
				is_situation_active = situation:golden_century
				has_global_variable = battle_for_the_strait_winner_country
				this = global_var:battle_for_the_strait_winner_country
			}
			gc_change_european_dominance = { value = gc_dominance_weak_penalty }
		}

		if = {
			limit = {
				is_situation_active = situation:battle_for_the_strait
				country_exists = c:GRA

				OR = {
					has_or_had_tag = CAS
					has_or_had_tag = MOR
				}
			}
			change_influence_over_granada_effect = { value = 2 }
		}

		scope:target_location = {
			custom_tooltip = {
				text = d009_convivencia.1.b.tt

				every_pop = {
					limit = {
						owner ?= root

						OR = {
							religion = religion:sunni
							religion = religion:judaism
						}
					}

					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}
}

d009_convivencia.2 = { #The Tránsito Synagogue
	type = country_event
	title = d009_convivencia.2.title
	desc = d009_convivencia.2.desc
	outcome = neutral
	historical_info = d009_convivencia.2.historical_info

	fire_only_once = yes

	trigger = {
		has_dlc = "d009_across_the_pillars"
		owns = location:toledo

		custom_tooltip = {
			text = d009_convivencia_2_judaism_population_tt

			location:toledo = {
				any_pop = {
					religion = religion:judaism
				}
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		if = {
			limit = {
				character:cas_samuel_halevi_abulafia = {
					is_alive = yes
				}
			}
			character:cas_samuel_halevi_abulafia = {
				if = {
					limit = { exists = owner }
					owner = {
						save_scope_as = target_character_previous_owner
					}
				}
				save_scope_as = target_character
			}
		}
		else = {
			create_character = {
				adm = { 60 80 }
				dip = { 50 70 }
				mil = { 5 20 }
				culture = culture:sephardi
				religion = religion:judaism
				female = no
				min_age = 30
				estate = estate_type:burghers_estate
				create_in_limbo = yes
				save_scope_as = target_character
			}
		}

		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:burghers_estate }
	}

	option = {
		name = d009_convivencia.2.a

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_right
		}

		location:toledo = {
			change_prosperity = prosperity_severe_bonus
		}

		scope:target_character = {
			if = {
				limit = {
					NOT = { employer = root }
				}

				move_country = root
			}
		}
	}

	option = {
		name = d009_convivencia.2.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = spiritualist_vs_humanist
			value = societal_value_move_to_left
		}

		custom_tooltip = {
			text = d009_convivencia.2.b.tt
			location:toledo = {
				every_pop = {
					limit = {
						owner ?= root
						religion = religion:judaism
					}

					add_pop_satisfaction = pop_satisfaction_mild_penalty
				}
			}
		}
	}
}

d009_convivencia.5 = { #Jewish Financiers and the Crown
	type = country_event
	title = d009_convivencia.5.title
	desc = d009_convivencia.5.desc
	outcome = neutral
	historical_info = d009_convivencia.5.historical_info

	trigger = {
		has_dlc = "d009_across_the_pillars"
		religion = religion:catholic

		culture = {
			has_culture_group = culture_group:iberian_group
		}

		current_year < 1390

		NOT = {
			has_variable = d009_convivencia_5_fired
		}

		"religion_population_in_country(religion:judaism)" > 0
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		if = {
			limit = { is_ai = no }

			ordered_pop = {
				limit = {
					culture = culture:sephardi
					religion = religion:judaism
					pop_type = pop_type:burghers
				}
				max = 1
				order_by = pop_size
				order_by = select_important_pop_value
				save_scope_as = target_pop
			}

			if = {
				limit = { NOT = { exists = scope:target_pop } }
				ordered_pop = {
					limit = { religion = religion:judaism }
					max = 1
					order_by = pop_size
					order_by = {
						value = 0
						add = {
							if = {
								limit = { pop_type = pop_type:burghers }
								value = 1000
							}
						}
						add = {
							if = {
								limit = { culture = culture:sephardi }
								value = 100
							}
						}
					}
					save_scope_as = target_pop
				}
			}

			ordered_pop = {
				limit = { culture = culture:castilian }
				max = 1
				order_by = pop_size
				order_by = select_important_pop_value
				save_scope_as = target_pop_bckg
			}
		}

		set_variable = { name = d009_convivencia_5_fired years = 20 }
	}

	option = {
		name = d009_convivencia.5.a

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_penalty }
		change_gold_effect = { scale = 1 }

		change_societal_value = {
			type = capital_economy_vs_traditional_economy
			value = societal_value_move_to_left
		}
	}

	option = {
		name = d009_convivencia.5.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }
		change_gold_effect = { scale = -1 }

		change_societal_value = {
			type = capital_economy_vs_traditional_economy
			value = societal_value_move_to_right
		}
	}
}
