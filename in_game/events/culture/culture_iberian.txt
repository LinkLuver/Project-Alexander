iberian_culture.3 = { #Settlement of Granada
	type = country_event
	title = iberian_culture.3.title
	desc = iberian_culture.3.desc
	outcome = neutral
	
    trigger = {
        culture = { has_culture_group = culture_group:iberian_group }
		religion.group = religion_group:christian
        own_entire_area = area:granada_area
		capital ?= {
			area = {
				any_location_in_area = {
					owner ?= root
					dominant_culture = culture:castilian
				}
			}
		}
		area:granada_area = {
			any_location_in_area = {
				owner ?= root
				dominant_culture = culture:andalusi
			}
		}
	}

	illustration_tags = {
        10 = regular
        10 = exterior
    }

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		capital = {
			area = {
				random_location_in_area = {
					limit = {
						owner ?= root
						dominant_culture = culture:castilian
					}
					province_definition = {
						save_scope_as = origin_province
					}
				}
			}
		}
		area:granada_area = {
			random_location_in_area = {
				limit = {
					owner ?= root
					dominant_culture = culture:andalusi
				}
				province_definition = {
					save_scope_as = target_province
				}
			}
		}
	}

	option = {
		name = iberian_culture.3.a
		add_migration = {
			owner ?= root
			from = scope:origin_province
			to = scope:target_province
			culture = culture:castilian
			amount = 0.1
			months = 60
		}
	}
	
	option = {
		name = iberian_culture.3.b
		add_migration = {
			owner ?= root
			from = scope:origin_province
			to = scope:target_province
			culture = culture:castilian
			amount = 0.05
			months = 60
		}
	}
}
