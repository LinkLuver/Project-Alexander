maghrebi_culture.1 = { # <Jewish> Community of <Location>
    type = country_event
    title = maghrebi_culture.1.title
    desc = maghrebi_culture.1.desc
    outcome = neutral

    trigger = {
        OR = {
            culture = { has_culture_group = culture_group:amazigh_group }
            culture = { has_culture_group = culture_group:maghrebi_group }
        }
        religion.group = religion_group:muslim
        "religion_percentage_in_country(religion:judaism)" > 0.001
        NOT = { has_variable = maghreb_jewish_community_cooldown }
        root.capital.region = {
            any_location_in_region = {
                owner ?= root
                any_pop = {
                    owner = root
                    OR = {
                        culture = culture:mustaarabi
                        culture = culture:sephardi
                    }
                    religion = religion:judaism
                }
            }
        }
    }

    illustration_tags = {
        10 = regular
        10 = interior
    }

    immediate = {
        event_illustration_estate_effect = { foreground = estate_type:clergy_estate background = estate_type:clergy_estate }
          root.capital.region = {
            random_location_in_region = {
                limit = {
                    owner ?= root
                    any_pop = {
                        owner = root
                        OR = {
                            culture = culture:mustaarabi
                            culture = culture:sephardi
                        }
                        religion = religion:judaism
                    }
                }
                save_scope_as = target_location
            }
        }
        set_variable = {
            name = maghreb_jewish_community_cooldown
            value = 1
            years = 20
        }  
    }	


    option = { # Grant special privileges to the local jewish pops
        name = maghrebi_culture.1.a
        add_government_power = government_power_extreme_bonus
        scope:target_location = {
            change_development = development_mild_bonus
            change_prosperity = prosperity_mild_bonus
        }
        add_estate_satisfaction = {
            type = estate_type:dhimmi_estate
            value = estate_satisfaction_mild_bonus
        }
        add_estate_satisfaction = {
            type = estate_type:clergy_estate
            value = estate_satisfaction_mild_penalty
        }
    }

    option = { # Force them to convert
        name = maghrebi_culture.1.b
        add_stability = stability_weak_penalty
        add_political_influence = political_influence_mild_bonus
        add_estate_satisfaction = {
            type = estate_type:dhimmi_estate
            value = estate_satisfaction_extreme_penalty
        }
        add_estate_satisfaction = {
            type = estate_type:clergy_estate
            value = estate_satisfaction_mild_bonus
        }
        scope:target_location = {
            every_pop = {
                limit = {
                    OR = {
                        culture = culture:mustaarabi
                        culture = culture:sephardi
                    }
                    religion = religion:judaism
                }
                split_pop = {
                    fraction = 0.33
                    religion = root.religion
                }
            }
        }
    }
}

scripted_effect cache_locations_in_maghreb_for_andalusian_migration = {
	region:maghreb_region = {
		every_ownable_location_in_region = {
			limit = {
				has_owner = yes
				location_rank != location_rank:rural_settlement
				owner.religion = religion:sunni
			}
			set_local_variable = {
				name = rating_for_this
				value = andalusi_migration_town_rating
			}
			add_to_local_variable_map = {
				name = maghreb_towns
				key = this
				value = local_var:rating_for_this
			}
		}
	}
}

scripted_effect select_target_location_for_andalusian_migration = {
	random_key_in_local_variable_map = {
		variable = maghreb_towns
		weight = {
			base = 1
			modifier = {
				factor = "local_variable_map(maghreb_towns|this)"
			}
		}
		save_scope_as = $scope_name$
	}
}

maghrebi_cultural_change_events.1000 = {
	type = country_event
	title = maghrebi_cultural_change_events.1000.title
	desc = maghrebi_cultural_change_events.1000.desc
	hidden = yes

	trigger = {
		has_presence_in = region:iberia_region
		religion.group = religion_group:christian
		culture = {
			has_culture_group = culture_group:iberian_group
		}
		"religion_population_in_country(religion:sunni)" > 0
	}

	immediate = {
		set_local_variable = {
			name = percentage_to_flee
			value = {
				value = 0.0025 # What is the % of pops we expect to flee?
				multiply = 4 # years delay between events to approximate
			}
		}
	
		change_local_variable = {
			name = percentage_to_flee # A bit of a random spread
			multiply = { 0.67 1.5 }
		}

		cache_locations_in_maghreb_for_andalusian_migration = yes

		every_owned_location = {
			limit = {
				"religion_population(religion:sunni)" > 0
			}
			save_scope_as = target_source
			select_target_location_for_andalusian_migration = { scope_name = target_destination }
			save_scope_value_as = {
				name = target_amount
				value = {
					value = "religion_population(religion:sunni)"
					multiply = local_var:percentage_to_flee
					min = { 0.1 0.3 }
					max = { 6 12 }
				}
			}
			root = {
				trigger_event_non_silently = {
					id = maghrebi_cultural_change_events.1001
					months = { 0 48 }
				}
			}
		}
	}

	option = {
		name = maghrebi_cultural_change_events.1000.a
		
	}
}
