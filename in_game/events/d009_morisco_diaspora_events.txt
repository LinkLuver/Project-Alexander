d009_morisco_diaspora.1 = { #The Moriscos Arrive
	type = country_event
	title = d009_morisco_diaspora.1.title
	desc = d009_morisco_diaspora.1.desc
	outcome = neutral
	historical_info = d009_morisco_diaspora.1.historical_info

	trigger = {
		has_dlc = "d009_across_the_pillars"

		culture = {
			has_culture_group = culture_group:maghrebi_group
		}

		NOT = {
			has_variable = d009_morisco_diaspora_1_fired
		}

		"religion_population_in_country(religion:sunni)" > 0
		"culture_population_in_country(culture:andalusi)" > 0
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:peasants_estate }

		set_variable = d009_morisco_diaspora_1_fired

		random_owned_location = {
			limit = {
				any_pop = {
					owner ?= root

					OR = {
						religion = religion:sunni
						culture = culture:andalusi
					}
				}
			}

			save_scope_as = target_location
		}
	}

	option = {
		name = d009_morisco_diaspora.1.a
		historical_option = yes

		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_right
		}

		every_owned_location = {
			limit = {
				any_pop = {
					owner ?= root

					OR = {
						religion = religion:sunni
						culture = culture:andalusi
					}
				}
			}

			add_location_modifier = {
				modifier = d009_morisco_community
				mode = add
			}

			add_to_global_variable_list = { name = d009_morisco_community_locations target = this }
			set_variable = d009_morisco_community_ever
		}

		set_variable = { name = d009_morisco_community_recent years = 30 }
	}

	option = {
		name = d009_morisco_diaspora.1.b

		add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_mild_bonus }

		change_societal_value = {
			type = outward_vs_inward
			value = societal_value_move_to_left
		}

		every_owned_location = {
			limit = {
				any_pop = {
					owner ?= root

					OR = {
						religion = religion:sunni
						culture = culture:andalusi
					}
				}
			}

			add_location_modifier = {
				modifier = d009_morisco_community
				mode = add
			}

			add_to_global_variable_list = { name = d009_morisco_community_locations target = this }
			set_variable = d009_morisco_community_ever
		}

		set_variable = { name = d009_morisco_community_recent years = 30 }
	}
}

d009_morisco_diaspora.9 = { #Two Communities, One Road to Morocco
	type = country_event
	title = d009_morisco_diaspora.9.title
	desc = d009_morisco_diaspora.9.desc
	outcome = neutral
	historical_info = d009_morisco_diaspora.9.historical_info

	fire_only_once = yes

	dynamic_historical_event = {
		tag = MOR
		from = 1492.1.1
		to = 1700.1.1
		monthly_chance = 5
	}

	trigger = {
		has_dlc = "d009_across_the_pillars"

		OR = {
			has_or_had_tag = MOR
			AND = {
				has_ruler = yes
				ruler = {
					has_dynasty = yes
					dynasty = dynasty:wattasid_dynasty
				}
			}
		}

		custom_tooltip = {
			text = d009_morisco_diaspora_9_sephardi_population_tt
			"culture_population_in_country(culture:sephardi)" > 0
		}

		custom_tooltip = {
			text = d009_morisco_diaspora_9_morisco_community_tt
			any_in_global_list = {
				variable = d009_morisco_community_locations
				owner = root
			}
		}
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }
	}

	option = {
		name = d009_morisco_diaspora.9.a
		historical_option = yes

		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_move_to_left
		}

		if = {
			limit = {
				NOT = {
					has_estate_privilege = estate_privilege:mellah_charter
				}
			}

			grant_estate_privilege = estate_privilege:mellah_charter
		}
	}

	option = {
		name = d009_morisco_diaspora.9.b

		change_societal_value = {
			type = individualism_vs_communalism
			value = societal_value_move_to_right
		}

		add_stability = stability_mild_bonus
	}
}
